157template <typename SpT> void printExecSpaceConfiguration(std::string name, const bool detail = false) {
158 if (!Kokkos::is_space<SpT>::value) {
159 std::string msg("SpT is not Kokkos execution space");
160 fprintf(stderr, ">> Error in file %s, line %d\n", __FILE__, __LINE__);
161 fprintf(stderr, " %s\n", msg.c_str());
162 throw std::logic_error(msg.c_str());
163 }
164 bool is_printed(false);
165#if defined(KOKKOS_ENABLE_SERIAL)
166 if (std::is_same<SpT, Kokkos::Serial>::value) {
167 is_printed = true;
168 std::cout << std::setw(16) << name << ":: Serial \n";
169 }
170#endif
171#if defined(KOKKOS_ENABLE_OPENMP)
172 if (std::is_same<SpT, Kokkos::OpenMP>::value) {
173 is_printed = true;
174 std::cout << std::setw(16) << name << ":: OpenMP \n";
175 }
176#endif
177#if defined(KOKKOS_ENABLE_CUDA)
178 if (std::is_same<SpT, Kokkos::Cuda>::value) {
179 is_printed = true;
180 std::cout << std::setw(16) << name << ":: Cuda \n";
181 }
182#endif
183#if defined(KOKKOS_ENABLE_HIP)
184 if (std::is_same<SpT, Kokkos::HIP>::value) {
185 is_printed = true;
186 std::cout << std::setw(16) << name << ":: HIP \n";
187 }
188#endif
189 if (!is_printed) {
190 std::cout << std::setw(16) << name << ":: not supported Kokkos execution space\n";
191 SpT().print_configuration(std::cout, detail);
192 throw std::logic_error("Error: not supported Kokkos execution space");
193 }
194 if (detail)
195 SpT().print_configuration(std::cout, true);
196}
