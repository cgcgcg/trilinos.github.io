21 template<class SC,class LO,class GO,class NO>
22 SubdomainSolver<SC,LO,GO,NO>::SubdomainSolver(ConstXMatrixPtr k,
23 ParameterListPtr parameterList,
24 string description,
25 GOVecPtr blockCoarseSize) :
26 K_ (k),
27 ParameterList_ (parameterList),
28 Description_ (description),
29 IsInitialized_ (false),
30 IsComputed_ (false)
31 {
32 FROSCH_TIMER_START_SUBDOMAINSOLVER(subdomainSolverTime,"SubdomainSolver::SubdomainSolver");
33 FROSCH_ASSERT(!K_.is_null(),"FROSch::SubdomainSolver: K_ is null.");
34 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos2")) {
35 if (K_->getRowMap()->lib()==UseTpetra) {
36 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*K_);
37 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
38 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
39 TEUCHOS_TEST_FOR_EXCEPT(tpetraMat.is_null());
40
41 TMultiVectorPtr xTmp;
42 TMultiVectorPtr bTmp;
43
44 Amesos2SolverTpetra_ = Amesos2::create<TCrsMatrix,TMultiVector>(ParameterList_->get("Solver","Klu"),tpetraMat,xTmp,bTmp);
45 ParameterListPtr parameterList = sublist(ParameterList_,"Amesos2");
46 parameterList->setName("Amesos2");
47 Amesos2SolverTpetra_->setParameters(parameterList);
48 } else {
49 FROSCH_ASSERT(false, "This can't happen. Use Tpetra linear algebra stack.");
50 }
51 } else if (!ParameterList_->get("SolverType","Amesos2").compare("MueLu")) {
52#ifdef HAVE_SHYLU_DDFROSCH_MUELU
53 MueLuFactory_ = rcp(new MueLu::ParameterListInterpreter<SC,LO,GO,NO>(parameterList->sublist("MueLu").sublist("MueLu Parameter")));
54 RCP<XMultiVector> nullspace;
55
56 if (!ParameterList_->sublist("MueLu").get("NullSpace","Laplace").compare("Laplace")) {
57 nullspace = XMultiVectorFactory::Build(K_->getRowMap(), 1);
58 nullspace->putScalar(1.);
59 }
60 else if (!ParameterList_->sublist("MueLu").get("NullSpace","Laplace").compare("SPP")) { // Hier matrix zu block matrix konvertieren
61 FROSCH_ASSERT(blockCoarseSize.size()==2,"Wrong size of blockCoarseSize for MueLu nullspace...");
62 unsigned dofs = (unsigned) ParameterList_->sublist("MueLu").get("Dimension",2);
63 nullspace = XMultiVectorFactory::Build(K_->getRowMap(), dofs+1);
64 ConstXMapPtr nullspaceMap = nullspace->getMap();
65 //nullspace of upper part
66 for (unsigned j=0; j<nullspace->getLocalLength(); j++) {
67 GO globIndex = nullspaceMap->getGlobalElement(j);
68 if (globIndex<=(GO)(dofs*blockCoarseSize[0]-1)) {
69 unsigned vecIndex = (globIndex)%dofs;
70 nullspace->getDataNonConst(vecIndex)[j] = 1.;
71 } else {
72 nullspace->getDataNonConst(dofs)[j] = 1.;
73 }
74 }
75 }
76 MueLuHierarchy_ = MueLuFactory_->CreateHierarchy(); // Das vor den if block
77 MueLuHierarchy_->GetLevel(0)->Set("A", Teuchos::rcp_const_cast<XMatrix>(K_)); // Das in den if block
78 MueLuHierarchy_->GetLevel(0)->Set("Nullspace", nullspace);
79#else
80 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "MueLu");
81#endif
82 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Ifpack2")) {
83#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
84 // Convert matrix to Tpetra
85 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*K_);
86 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
87 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
88 TEUCHOS_TEST_FOR_EXCEPT(tpetraMat.is_null());
89
90 Ifpack2::Details::OneLevelFactory<TRowMatrix> ifpack2Factory;
91 Ifpack2Preconditioner_ = ifpack2Factory.create(ParameterList_->get("Solver","FILU"),tpetraMat);
92
93 ParameterListPtr parameterList = sublist(ParameterList_,"Ifpack2");
94 parameterList->setName("Ifpack2");
95 Ifpack2Preconditioner_->setParameters(*parameterList);
96#else
97 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "Ifpack2");
98#endif
99 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Belos")) {
100#ifdef HAVE_SHYLU_DDFROSCH_BELOS
101 RCP<XMultiVector> xSolution;// = FROSch::ConvertToXpetra<SC, LO, GO, NO>(UseTpetra,*this->solution_,TeuchosComm);
102 RCP<XMultiVector> xRightHandSide;// = FROSch::ConvertToXpetra<SC, LO, GO, NO>(UseTpetra,*residualVec_,TeuchosComm);//hier residualVec. Bei linProb rhs_
103
104 RCP<const Belos::OperatorT<XMultiVector> > OpK = rcp_dynamic_cast<const Belos::XpetraOp<SC,LO,GO,NO> >(K_);
105 TEUCHOS_TEST_FOR_EXCEPT(OpK.is_null());
106
107 BelosLinearProblem_.reset(new Belos::LinearProblem<SC,XMultiVector,Belos::OperatorT<XMultiVector> >(OpK,xSolution,xRightHandSide));
108
109 Belos::SolverFactory<SC,XMultiVector,Belos::OperatorT<XMultiVector> > belosFactory;
110 ParameterListPtr solverParameterList = sublist(ParameterList_,"Belos");
111
112 BelosSolverManager_ = belosFactory.create(solverParameterList->get("Solver","GMRES"),sublist(solverParameterList,solverParameterList->get("Solver","GMRES")));
113
114 BelosSolverManager_->setProblem(BelosLinearProblem_);
115#else
116 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "Belos");
117#endif
118 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Thyra")) {
119#ifdef HAVE_SHYLU_DDFROSCH_THYRA
120 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*K_);
121 RCP<const Thyra::LinearOpBase<SC> > thyraOp = ThyraUtils<SC,LO,GO,NO>::toThyra(crsOp.getCrsMatrix());
122
123 Stratimikos::LinearSolverBuilder<SC> linearSolverBuilder;
124#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
125 linearSolverBuilder.setPreconditioningStrategyFactory(abstractFactoryStd<Thyra::PreconditionerFactoryBase<SC>,Thyra::Ifpack2PreconditionerFactory<TCrsMatrix> >(),"Ifpack2");
126#endif
127
128 ParameterListPtr parameterList = sublist(ParameterList_,"Thyra");
129 linearSolverBuilder.setParameterList(parameterList);
130
131 RCP<Thyra::LinearOpWithSolveFactoryBase<SC> > lOWSFactory = linearSolverBuilder.createLinearSolveStrategy("");
132
133 // AH 04/02/2020: Is this necessary?
134 // RCP<FancyOStream> out = VerboseObjectBase::getDefaultOStream();
135 // lOWSFactory->setOStream(out);
136 // lOWSFactory->setVerbLevel(VERB_HIGH);
137
138 LOWS_ = Thyra::linearOpWithSolve(*lOWSFactory,thyraOp);
139
140#else
141 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "Thyra");
142#endif
143 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelBlockPreconditioner")) {
144
145 Teuchos::RCP< const Teuchos::Comm< int > > TC = K_->getMap()->getComm();
146 Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > > RepeatedMaps(1);
147 UNVecPtr dofsPerNodeVector;
148 ConstXMultiVectorPtrVecPtr nullSpaceBasisVec(1);
149 Teuchos::ArrayRCP<DofOrdering> dofOrderings;
150 Teuchos::ArrayRCP<Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > > > dofsMapsVec = Teuchos::null;
151 //Teuchos::ArrayRCP<Teuchos::RCP<Xpetra::Map<LO,GO,NO> > > MainCoarseMapVector = Teuchos::null;
152
153 FROSCH_ASSERT(ParameterList_->isParameter("Repeated Map Vector"),"Currently TwoLevelBlockPreconditioner cannot be constructed without Repeated Maps Vector ");
154 FROSCH_ASSERT(ParameterList_->isParameter("DofsPerNode Vector"),"Currently, TwoLevelBlockPreconditioner cannot be constructed without DofsPerNode Vector.");
155 FROSCH_ASSERT(ParameterList_->isParameter("DofOrdering Vector"),"Currently, TwoLevelBlockPreconditioner cannot be constructed without DofOrdering Vector.");
156
157 if (ParameterList_->isParameter("Repeated Map Vector")) {
158 RepeatedMaps = ExtractVectorFromParameterList<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > >(*ParameterList_,"Repeated Map Vector");
159 }
160 if (ParameterList_->isParameter("DofsPerNode Vector")) {
161 dofsPerNodeVector = ExtractVectorFromParameterList<UN>(*ParameterList_,"DofsPerNode Vector");
162 }
163 if (ParameterList_->isParameter("DofOrdering Vector")) {
164 dofOrderings = ExtractVectorFromParameterList<DofOrdering>(*ParameterList_,"DofOrdering Vector");
165 }
166 if (ParameterList_->isParameter("Dofs Maps Vector")) {
167 dofsMapsVec = ExtractVectorFromParameterList<Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > >>(*ParameterList_,"Dofs Maps Vector");
168 }
169
170 FROSCH_ASSERT(RepeatedMaps.size()==dofsPerNodeVector.size(),"RepeatedMaps.size()!=dofsPerNodeVector.size()");
171 FROSCH_ASSERT(RepeatedMaps.size()==dofOrderings.size(),"RepeatedMaps.size()!=dofOrderings.size()");
172 TLBP = Teuchos::rcp(new TwoLevelBlockPreconditioner<SC,LO,GO,NO>(K_,ParameterList_));
173 TLBP->initialize(ParameterList_->get("Dimension",3),
174 dofsPerNodeVector,dofOrderings,
175 ParameterList_->get("Overlap",1),
176 RepeatedMaps,
177 nullSpaceBasisVec);
178
179 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelPreconditioner")) {
180
181 Teuchos::RCP< const Teuchos::Comm< int > > TC = K_->getMap()->getComm();
182 Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > > RepeatedMaps(1);
183 Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > > NodesMaps(1);
184
185 UNVecPtr dofsPerNodeVector(1);
186 Teuchos::ArrayRCP<DofOrdering> dofOrderings(1);
187 Teuchos::ArrayRCP<Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > > > dofsMapsVec = Teuchos::null;
188 Teuchos::ArrayRCP<Teuchos::RCP<Xpetra::Map<LO,GO,NO> > > MainCoarseMapVector = Teuchos::null;
189
190 //FROSCH_ASSERT(ParameterList_->isParameter("Repeated Map Vector"),"Currently TwoLevelBlockPreconditioner cannot be constructed without Repeated Maps Vector ");
191 if (ParameterList_->isParameter("Repeated Map Vector")) {
192 RepeatedMaps = ExtractVectorFromParameterList<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > >(*ParameterList_,"Repeated Map Vector");
193 }
194
195 if (ParameterList_->isParameter("Nodes Map Vector")) {
196 NodesMaps = ExtractVectorFromParameterList<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > >(*ParameterList_,"Nodes Map Vector");
197 }
198
199 if (ParameterList_->isParameter("DofsPerNode Vector")) {
200 dofsPerNodeVector = ExtractVectorFromParameterList<UN>(*ParameterList_,"DofsPerNode Vector");
201 }
202
203 if (ParameterList_->isParameter("DofOrdering Vector")) {
204 dofOrderings = ExtractVectorFromParameterList<DofOrdering>(*ParameterList_,"DofOrdering Vector");
205 }
206
207 if (ParameterList_->isParameter("Main Map Vector")) {
208 MainCoarseMapVector = ExtractVectorFromParameterList<Teuchos::RCP<Xpetra::Map<LO,GO,NO> > >(*ParameterList_,"Main Map Vector");
209 }
210 if (ParameterList_->isParameter("Dofs Maps Vector")) {
211 dofsMapsVec = ExtractVectorFromParameterList<Teuchos::ArrayRCP<Teuchos::RCP<const Xpetra::Map<LO,GO,NO> > >>(*ParameterList_,"Dofs Maps Vector");
212 }
213
214 ConstXMultiVectorPtr nodeList = null;
215 GOVecPtr dirichletBoundaryDofs = null;
216 ConstXMultiVectorPtr nullSpaceBasisVec = null;
217
218 TLP = Teuchos::rcp(new TwoLevelPreconditioner<SC,LO,GO,NO>(K_,ParameterList_));
219 TLP->initialize(ParameterList_->get("Dimension",3),
220 dofsPerNodeVector[0],
221 ParameterList_->get("Overlap",1),
222 nullSpaceBasisVec,
223 nodeList,
224 dofOrderings[0],
225 RepeatedMaps[0],
226 dofsMapsVec[0],
227 dirichletBoundaryDofs);
228 } else {
229 FROSCH_ASSERT(false,"SolverType unknown...");
230 }
231 }

258 template<class SC,class LO,class GO,class NO>
259 int SubdomainSolver<SC,LO,GO,NO>::initialize()
260 {
261 FROSCH_TIMER_START_SUBDOMAINSOLVER(initializeTime,"SubdomainSolver::initialize");
262#ifdef HAVE_SHYLU_DDFROSCH_AMESOS
263 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos")) {
264 IsInitialized_ = true;
265 IsComputed_ = false;
266 AMESOS_CHK_ERR(AmesosSolver_->SymbolicFactorization());
267 } else
268#endif
269 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos2")) {
270 {
271 IsInitialized_ = true;
272 IsComputed_ = false;
273 Amesos2SolverTpetra_->symbolicFactorization();
274 }
275#ifdef HAVE_SHYLU_DDFROSCH_MUELU
276 } else if (!ParameterList_->get("SolverType","Amesos2").compare("MueLu")) {
277 IsInitialized_ = true;
278 IsComputed_ = false;
279#endif
280#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
281 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Ifpack2")) {
282 IsInitialized_ = true;
283 IsComputed_ = false;
284 Ifpack2Preconditioner_->initialize();
285#endif
286#ifdef HAVE_SHYLU_DDFROSCH_BELOS
287 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Belos")) {
288 IsInitialized_ = true;
289 IsComputed_ = false;
290#endif
291#ifdef HAVE_SHYLU_DDFROSCH_THYRA
292 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Thyra")) {
293 // TODO: In the current implementation initialize() and compute() are part of apply() for Thyra.
294 IsInitialized_ = true;
295 IsComputed_ = false;
296#endif
297 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelBlockPreconditioner")) {
298 IsInitialized_ = true;
299 IsComputed_ = false;
300 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelPreconditioner")) {
301 IsInitialized_ = true;
302 IsComputed_ = false;
303 } else {
304 FROSCH_ASSERT(false,"SolverType unknown...");
305 }
306 return 0;
307 }

309 template<class SC,class LO,class GO,class NO>
310 int SubdomainSolver<SC,LO,GO,NO>::compute()
311 {
312 FROSCH_TIMER_START_SUBDOMAINSOLVER(computeTime,"SubdomainSolver::compute");
313 FROSCH_ASSERT(this->IsInitialized_,"ERROR: SubdomainSolver has to be initialized before calling compute()");
314#ifdef HAVE_SHYLU_DDFROSCH_AMESOS
315 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos")) {
316 IsComputed_ = true;
317 AMESOS_CHK_ERR(AmesosSolver_->NumericFactorization());
318 } else
319#endif
320 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos2")) {
321 {
322 IsComputed_ = true;
323 Amesos2SolverTpetra_->numericFactorization();
324 }
325#ifdef HAVE_SHYLU_DDFROSCH_MUELU
326 } else if (!ParameterList_->get("SolverType","Amesos2").compare("MueLu")) {
327 MueLuFactory_->SetupHierarchy(*MueLuHierarchy_);
328 MueLuHierarchy_->IsPreconditioner(false);
329 IsComputed_ = true;
330#endif
331#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
332 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Ifpack2")) {
333 Ifpack2Preconditioner_->compute();
334 IsComputed_ = true;
335#endif
336#ifdef HAVE_SHYLU_DDFROSCH_BELOS
337 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Belos")) {
338 ParameterListPtr solverParameterList = sublist(ParameterList_,"Belos");
339 if (solverParameterList->get("OneLevelPreconditioner",false)) {
340
341 RCP<FROSch::OneLevelPreconditioner<SC,LO,GO,NO> > prec = rcp(new OneLevelPreconditioner<SC,LO,GO,NO> (K_, solverParameterList));
342
343 bool buildRepeatedMap = solverParameterList->get("Build Repeated Map",false);
344 prec->initialize(solverParameterList->get("Overlap",1),buildRepeatedMap);
345 prec->compute();
346 RCP<Belos::OperatorT<MultiVector<SC,LO,GO,NO> > > OpP = rcp(new Belos::XpetraOp<SC, LO, GO, NO>(prec));
347
348 if (!solverParameterList->get("PreconditionerPosition","left").compare("left")) {
349 BelosLinearProblem_->setLeftPrec(OpP);
350
351 } else if (!solverParameterList->get("PreconditionerPosition","left").compare("right")) {
352 BelosLinearProblem_->setRightPrec(OpP);
353
354 } else {
355 FROSCH_ASSERT(false,"PreconditionerPosition unknown...");
356 }
357 }
358 IsComputed_ = true;
359#endif
360#ifdef HAVE_SHYLU_DDFROSCH_THYRA
361 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Thyra")) {
362 IsComputed_ = true;
363#endif
364 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelBlockPreconditioner")) {
365 TLBP->compute();
366 IsComputed_ = true;
367 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelPreconditioner")) {
368 TLP->compute();
369 IsComputed_ = true;
370 } else {
371 FROSCH_ASSERT(false,"SolverType unknown...");
372 }
373 return 0;
374 }

376 // Y = alpha * A^mode * X + beta * Y
377 template<class SC,class LO,class GO,class NO>
378 void SubdomainSolver<SC,LO,GO,NO>::apply(const XMultiVector &x,
379 XMultiVector &y,
380 ETransp mode,
381 SC alpha,
382 SC beta) const
383 {
384 FROSCH_TIMER_START_SUBDOMAINSOLVER(applyTime,"SubdomainSolver::apply");
385 FROSCH_ASSERT(IsComputed_,"!IsComputed_.");
386
387 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos2")) {
388 {
389 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorX = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(&x);
390 TMultiVectorPtr tpetraMultiVectorX = xTpetraMultiVectorX->getTpetra_MultiVector();
391
392 if (YTmp_.is_null()) YTmp_ = XMultiVectorFactory::Build(y.getMap(),x.getNumVectors());
393 *YTmp_ = y;
394 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorY = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(YTmp_.get());
395 TMultiVectorPtr tpetraMultiVectorY = xTpetraMultiVectorY->getTpetra_MultiVector();
396
397 Amesos2SolverTpetra_->setX(tpetraMultiVectorY);
398 Amesos2SolverTpetra_->setB(tpetraMultiVectorX);
399
400 Amesos2SolverTpetra_->solve(); // Was ist, wenn man mit der transponierten Matrix lösen will
401 }
402#ifdef HAVE_SHYLU_DDFROSCH_MUELU
403 } else if (!ParameterList_->get("SolverType","Amesos2").compare("MueLu")) {
404 if (YTmp_.is_null()) YTmp_ = XMultiVectorFactory::Build(y.getMap(),x.getNumVectors());
405
406 int mgridSweeps = ParameterList_->sublist("MueLu").get("mgridSweeps",-1);
407 if (mgridSweeps>0) {
408 MueLuHierarchy_->Iterate(x,*YTmp_,mgridSweeps);
409 }
410 else{
411 typename ScalarTraits<SC>::magnitudeType tol = ParameterList_->sublist("MueLu").get("tol",1.e-6);
412 MueLuHierarchy_->Iterate(x,*YTmp_,tol);
413 }
414 y = *YTmp_;
415#endif
416#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
417 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Ifpack2")) {
418 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorX = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(&x);
419 TMultiVectorPtr tpetraMultiVectorX = xTpetraMultiVectorX->getTpetra_MultiVector();
420
421 if (YTmp_.is_null()) YTmp_ = XMultiVectorFactory::Build(y.getMap(),x.getNumVectors());
422 *YTmp_ = y;
423 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorY = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(YTmp_.get());
424 TMultiVectorPtr tpetraMultiVectorY = xTpetraMultiVectorY->getTpetra_MultiVector();
425
426 Ifpack2Preconditioner_->apply(*tpetraMultiVectorX,*tpetraMultiVectorY,mode,alpha,beta);
427#endif
428#ifdef HAVE_SHYLU_DDFROSCH_BELOS
429 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Belos")) {
430
431 ConstXMultiVectorPtr xPtr = rcpFromRef(x);
432 if (YTmp_.is_null()) YTmp_ = XMultiVectorFactory::Build(y.getMap(),x.getNumVectors());
433 BelosLinearProblem_->setProblem(YTmp_,xPtr);
434 BelosSolverManager_->solve();
435 y = *YTmp_;
436#endif
437#ifdef HAVE_SHYLU_DDFROSCH_THYRA
438 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Thyra")) {
439 ConstXMultiVectorPtr xPtr = rcpFromRef(x);
440 RCP<const Thyra::MultiVectorBase<SC> > thyraX = ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(xPtr);
441
442 // AH 04/02/2020: Is there an easier way to do this? This seems to be close to the miminum in order to have two connected Thyra and Xpetra Multivectors
443 if (ThyraYTmp_.is_null()) {
444 XMultiVectorPtr yTmp = XMultiVectorFactory::Build(y.getMap(),x.getNumVectors());
445 ThyraYTmp_ = rcp_const_cast<Thyra::MultiVectorBase<SC> >(ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(yTmp));
446 }
447 const RCP<Tpetra::MultiVector<SC,LO,GO,NO> > yTmpTpMultVec = Thyra::TpetraOperatorVectorExtraction<SC,LO,GO,NO>::getTpetraMultiVector(ThyraYTmp_);
448 TEUCHOS_TEST_FOR_EXCEPT(is_null(yTmpTpMultVec));
449 YTmp_ = rcp(new Xpetra::TpetraMultiVector<SC,LO,GO,NO>(yTmpTpMultVec));
450 TEUCHOS_TEST_FOR_EXCEPT(is_null(YTmp_));
451
452 Thyra::SolveStatus<SC> status = Thyra::solve<SC>(*LOWS_, Thyra::NOTRANS, *thyraX, ThyraYTmp_.ptr());
453 y = *YTmp_;
454
455 /*
456 const RCP<Tpetra::MultiVector<SC,LO,GO,NO> > yTpMultVec = Thyra::TpetraOperatorVectorExtraction<SC,LO,GO,NO>::getTpetraMultiVector(rcpFromPtr(Y_inout));
457 TEUCHOS_TEST_FOR_EXCEPT(is_null(yTpMultVec));
458 xY = rcp(new Xpetra::TpetraMultiVector<SC,LO,GO,NO>(yTpMultVec));
459 TEUCHOS_TEST_FOR_EXCEPT(is_null(xY));
460 */
461#endif
462 } else if (!ParameterList_->get("SolverType","Amesos2").compare("TwoLevelBlockPreconditioner")) {

516 {
517 FROSCH_TIMER_START_SUBDOMAINSOLVER(resetMatrixTime,"SubdomainSolver::resetMatrix");
518 K_ = k;
519 FROSCH_ASSERT(!K_.is_null(),"FROSch::SubdomainSolver: K_ is null.");
520 if (!ParameterList_->get("SolverType","Amesos2").compare("Amesos2")) {
521 if (K_->getRowMap()->lib()==UseTpetra) {
522 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*K_);
523 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
524 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
525 TEUCHOS_TEST_FOR_EXCEPT(tpetraMat.is_null());
526
527 if (reuseInitialize) {
528 Amesos2SolverTpetra_->setA(tpetraMat,Amesos2::SYMBFACT);
529 } else {
530 Amesos2SolverTpetra_->setA(tpetraMat,Amesos2::CLEAN);
531 }
533 } else {
534 FROSCH_ASSERT(false, "This can't happen. Use Tpetra linear algebra stack.");
535 }
536 } else if (!ParameterList_->get("SolverType","Amesos2").compare("MueLu")) {
537#ifdef HAVE_SHYLU_DDFROSCH_MUELU
538 FROSCH_ASSERT(false,"FROSch::SubdomainSolver: resetMatrix() is not implemented for MueLu yet.");
539#else
540 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "MueLu");
541#endif
542 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Ifpack2")) {
543#ifdef HAVE_SHYLU_DDFROSCH_IFPACK2
544 // Convert matrix to Tpetra
545 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*K_);
546 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
547 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
548 TEUCHOS_TEST_FOR_EXCEPT(tpetraMat.is_null());
549
550 FROSCH_ASSERT(false,"FROSch::SubdomainSolver: resetMatrix() is not implemented for Ifpack2 yet.");
551#else
552 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "Ifpack2");
553#endif
554 } else if (!ParameterList_->get("SolverType","Amesos2").compare("Belos")) {
555#ifdef HAVE_SHYLU_DDFROSCH_BELOS
556 FROSCH_ASSERT(false,"FROSch::SubdomainSolver: resetMatrix() is not implemented for Belos yet.");
557#else
558 ThrowErrorMissingPackage("FROSch::SubdomainSolver", "Belos");

563#else

567 FROSCH_ASSERT(false,"SolverType unknown...");
