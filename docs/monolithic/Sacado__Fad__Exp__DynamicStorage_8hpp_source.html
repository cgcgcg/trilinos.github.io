107 SACADO_INLINE_FUNCTION
108 DynamicStorage& operator=(const DynamicStorage& x) {
109 if (this != &x) {
110 val_ = x.val_;
111 if (sz_ != x.sz_) {
112 sz_ = x.sz_;
113 if (x.sz_ > len_) {
114 if (len_ != 0)
115 ds_array<U>::destroy_and_release(dx_, len_);
116 len_ = x.sz_;
117 dx_ = ds_array<U>::get_and_fill(x.dx_, sz_);
118 }
119 else
120 ds_array<U>::copy(x.dx_, dx_, sz_);
121 }
122 else
123 ds_array<U>::copy(x.dx_, dx_, sz_);
124 }
125 return *this;
126 }
