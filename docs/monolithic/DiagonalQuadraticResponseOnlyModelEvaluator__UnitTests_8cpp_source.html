84 TEST_EQUALITY_CONST(model->Ng(), 1);
85
86 RCP<const Thyra::VectorSpaceBase<Scalar> > p_space = model->get_p_space(0);
87 RCP<const Thyra::VectorSpaceBase<Scalar> > g_space = model->get_g_space(0);
88

91 out << "\nval = " << val << "\n";
92 Thyra::V_S(p_init.ptr(), val);
93
94 RCP<Thyra::VectorBase<Scalar> >
95 g = createMember(g_space),
96 g_grad = createMember(p_space);
97
98 eval_g_DgDp<Scalar>(*model, 0, *p_init, 0,
99 g.ptr(),derivativeGradient<Scalar>(g_grad) );
100
101 out << "\ng =\n" << *g;
102 out << "\ng_grad =\n" << *g_grad;
103
104 const Ordinal globalDim = p_space->dim();
105 out << "\nglobalDim = " << globalDim << "\n";
106
107 TEST_FLOATING_EQUALITY( get_ele<Scalar>(*g, 0),
108 as<Scalar>(0.5*globalDim)*val*val, as<ScalarMag>(g_tol_scale*ST::eps()/globalDim));
109
110 TEST_FLOATING_EQUALITY(
111 norm_2<Scalar>(*g_grad),
112 ST::magnitude(ST::squareroot(as<Scalar>(globalDim)*val*val)),
113 as<ScalarMag>(g_tol_scale*ST::eps()/globalDim));
114
115}

125TEUCHOS_UNIT_TEST_TEMPLATE_1_DECL( DiagonalQuadraticResponseOnlyModelEvaluator,
126 offsets, Scalar )
127{
128
129 typedef Teuchos::ScalarTraits<Scalar> ST;
130 typedef typename ST::magnitudeType ScalarMag;
131 typedef Thyra::Ordinal Ordinal;
132 //typedef Thyra::ModelEvaluatorBase MEB; // unused
133 using Thyra::VectorSpaceBase;
134 using Thyra::VectorBase;
135 using Thyra::derivativeGradient;
136 using Thyra::eval_g_DgDp;
137 using Thyra::get_mv;
138 using Thyra::get_ele;
139 using Thyra::norm_2;
140 using Thyra::V_S;
141
142 const RCP<Thyra::DiagonalQuadraticResponseOnlyModelEvaluator<Scalar> >
143 model = Thyra::diagonalQuadraticResponseOnlyModelEvaluator<Scalar>(g_localDim);
144
145 const RCP<const VectorSpaceBase<Scalar> > p_space = model->get_p_space(0);
146 const RCP<const VectorSpaceBase<Scalar> > g_space = model->get_g_space(0);
147
148 const Scalar p_soln_val = as<Scalar>(3.0);
149 {
150 const RCP<VectorBase<Scalar> > p_soln = createMember(p_space);
151 V_S(p_soln.ptr(), p_soln_val);
152 model->setSolutionVector(p_soln);
153 }
154
155 const Scalar diag_val = as<Scalar>(4.0);
156 {
157 const RCP<VectorBase<Scalar> > diag = createMember(p_space);
158 V_S(diag.ptr(), diag_val);
159 model->setDiagonalVector(diag);
160 }
161
162 const Scalar g_offset = as<Scalar>(5.0);
163 {
164 model->setScalarOffset(g_offset);
165 }
166
167 const Scalar nonlinearTermFactor = 1e-3;
168 {
169 model->setNonlinearTermFactor(nonlinearTermFactor);
170 }
171
172 const Scalar p_val = as<Scalar>(2.0);
173 const RCP<VectorBase<Scalar> > p_init = createMember(p_space);
174 V_S(p_init.ptr(), p_val);
175
176 RCP<VectorBase<Scalar> >
177 g = createMember(g_space),
178 g_grad = createMember(p_space);
179
180 eval_g_DgDp<Scalar>(*model, 0, *p_init, 0,
181 g.ptr(),derivativeGradient<Scalar>(g_grad) );
182
183 out << "\ng =\n" << *g;
184 out << "\ng_grad =\n" << *g_grad;
185
186 const Ordinal globalDim = p_space->dim();
187 out << "\nglobalDim = " << globalDim << "\n";
188
189 TEST_FLOATING_EQUALITY(
190 get_ele<Scalar>(*g, 0),
191 as<Scalar>(
192 0.5 * globalDim
193 * (diag_val * sqr(p_val - p_soln_val)
194 + nonlinearTermFactor * cube(p_val - p_soln_val)
195 )
196 + g_offset
197 ),
198 as<ScalarMag>(g_tol_scale*ST::eps()/globalDim)
199 );
200
201 TEST_FLOATING_EQUALITY(
202 sum(*g_grad),
203 as<Scalar>(
204 globalDim
205 * ( diag_val * (p_val - p_soln_val)
206 + 1.5 * nonlinearTermFactor * sqr(p_val - p_soln_val) )
207 ),
208 as<ScalarMag>(g_tol_scale*ST::eps()/globalDim)
209 );
210
211}
