22 template <class SC,class LO,class GO,class NO>
23 TwoLevelPreconditioner<SC,LO,GO,NO>::TwoLevelPreconditioner(ConstXMatrixPtr k,
24 ParameterListPtr parameterList) :
25 OneLevelPreconditioner<SC,LO,GO,NO> (k,parameterList)
26 {
27 FROSCH_DETAILTIMER_START_LEVELID(twoLevelPreconditionerTime,"TwoLevelPreconditioner::TwoLevelPreconditioner::");
28 if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("IPOUHarmonicCoarseOperator")) {
29 // Set the LevelID in the sublist
30 parameterList->sublist("IPOUHarmonicCoarseOperator").set("Level ID",this->LevelID_);
31 CoarseOperator_ = IPOUHarmonicCoarseOperatorPtr(new IPOUHarmonicCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"IPOUHarmonicCoarseOperator")));
32 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("GDSWCoarseOperator")) {
33 // Set the LevelID in the sublist
34 parameterList->sublist("GDSWCoarseOperator").set("Level ID",this->LevelID_);
35 CoarseOperator_ = GDSWCoarseOperatorPtr(new GDSWCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"GDSWCoarseOperator")));
36 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("RGDSWCoarseOperator")) {
37 // Set the LevelID in the sublist
38 parameterList->sublist("RGDSWCoarseOperator").set("Level ID",this->LevelID_);
39 CoarseOperator_ = RGDSWCoarseOperatorPtr(new RGDSWCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"RGDSWCoarseOperator")));
40 } else {
41 FROSCH_ASSERT(false,"CoarseOperator Type unkown.");
42 } // TODO: Add ability to disable individual levels
43 if (this->UseMultiplicative_) {
44 this->MultiplicativeOperator_->addOperator(CoarseOperator_);
45 }
46 else{
47 this->SumOperator_->addOperator(CoarseOperator_);
48 }
49 }

81 template <class SC,class LO,class GO,class NO>
82 int TwoLevelPreconditioner<SC,LO,GO,NO>::initialize(UN dimension,
83 UN dofsPerNode,
84 int overlap,
85 ConstXMultiVectorPtr nullSpaceBasis,
86 ConstXMultiVectorPtr nodeList,
87 DofOrdering dofOrdering,
88 ConstXMapPtr repeatedMap,
89 ConstXMapPtrVecPtr dofsMaps,
90 GOVecPtr dirichletBoundaryDofs)
91 {
92 FROSCH_TIMER_START_LEVELID(initializeTime,"TwoLevelPreconditioner::initialize");
94 // Checks //
96 FROSCH_ASSERT(dofOrdering == NodeWise || dofOrdering == DimensionWise || dofOrdering == Custom,"ERROR: Specify a valid DofOrdering.");
97 int ret = 0;
98
100 // Maps //
102 if (repeatedMap.is_null()) {
103 FROSCH_DETAILTIMER_START_LEVELID(buildRepeatedMapTime,"BuildRepeatedMap");
104 repeatedMap = BuildRepeatedMap(this->K_->getCrsGraph()); // Todo: Achtung, die UniqueMap könnte unsinnig verteilt sein. Falls es eine repeatedMap gibt, sollte dann die uniqueMap neu gebaut werden können. In diesem Fall, sollte man das aber basierend auf der repeatedNodesMap tun
105 }
106 // Build dofsMaps and repeatedNodesMap
107 ConstXMapPtr repeatedNodesMap;
108 if (dofsMaps.is_null()) {
109 FROSCH_DETAILTIMER_START_LEVELID(buildDofMapsTime,"BuildDofMaps");
110 if (0>BuildDofMaps(repeatedMap,dofsPerNode,dofOrdering,repeatedNodesMap,dofsMaps)) ret -= 100; // Todo: Rückgabewerte
111 } else {
112 FROSCH_ASSERT(dofsMaps.size()==dofsPerNode,"dofsMaps.size()!=dofsPerNode");
113 for (UN i=0; i<dofsMaps.size(); i++) {
114 FROSCH_ASSERT(!dofsMaps[i].is_null(),"dofsMaps[i].is_null()");
115 }
116 if (repeatedNodesMap.is_null()) {
117 repeatedNodesMap = dofsMaps[0];
118 }
119 }
120
122 // Communicate nodeList //
124 if (!nodeList.is_null()) {
125 FROSCH_DETAILTIMER_START_LEVELID(communicateNodeListTime,"Communicate Node List");
126 ConstXMapPtr nodeListMap = nodeList->getMap();
127 if (!nodeListMap->isSameAs(*repeatedNodesMap)) {
128 RCP<MultiVector<SC,LO,GO,NO> > tmpNodeList = MultiVectorFactory<SC,LO,GO,NO>::Build(repeatedNodesMap,nodeList->getNumVectors());
129 RCP<Import<LO,GO,NO> > scatter = ImportFactory<LO,GO,NO>::Build(nodeListMap,repeatedNodesMap);
130 tmpNodeList->doImport(*nodeList,*scatter,INSERT);
131 nodeList = tmpNodeList.getConst();
132 }
133 }
134
136 // Determine dirichletBoundaryDofs //
138 if (dirichletBoundaryDofs.is_null()) {
139 FROSCH_DETAILTIMER_START_LEVELID(determineDirichletRowsTime,"Determine Dirichlet Rows");
140#ifdef FindOneEntryOnlyRowsGlobal_Matrix
141 dirichletBoundaryDofs = FindOneEntryOnlyRowsGlobal(this->K_.getConst(),repeatedMap);
142#else
143 dirichletBoundaryDofs = FindOneEntryOnlyRowsGlobal(this->K_->getCrsGraph(),repeatedMap);
144#endif
145 }
146
148 // Initialize OverlappingOperator //
150 if (!this->ParameterList_->get("OverlappingOperator Type","AlgebraicOverlappingOperator").compare("AlgebraicOverlappingOperator")) {
151 AlgebraicOverlappingOperatorPtr algebraicOverlappigOperator = rcp_static_cast<AlgebraicOverlappingOperator<SC,LO,GO,NO> >(this->OverlappingOperator_);
152 if (0>algebraicOverlappigOperator->initialize(overlap,repeatedMap)) ret -= 1;
153 } else {
154 FROSCH_ASSERT(false,"OverlappingOperator Type unkown.");
155 }
156
158 // Initialize CoarseOperator //
160 if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("IPOUHarmonicCoarseOperator")) {
161 // Build Null Space
162 if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Laplace")) {
163 nullSpaceBasis = BuildNullSpace<SC,LO,GO,NO>(dimension,NullSpaceType::Laplace,repeatedMap,dofsPerNode,dofsMaps);
164 } else if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Linear Elasticity")) {
165 nullSpaceBasis = BuildNullSpace(dimension,NullSpaceType::Elasticity,repeatedMap,dofsPerNode,dofsMaps,nodeList);
166 } else if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Input")) {
167 FROSCH_ASSERT(!nullSpaceBasis.is_null(),"Null Space Type is 'Input', but nullSpaceBasis.is_null().");
168 ConstXMapPtr nullSpaceBasisMap = nullSpaceBasis->getMap();
169 if (!nullSpaceBasisMap->isSameAs(*repeatedMap)) {
170 FROSCH_DETAILTIMER_START_LEVELID(communicateNullSpaceBasis,"Communicate Null Space");
171 RCP<MultiVector<SC,LO,GO,NO> > tmpNullSpaceBasis = MultiVectorFactory<SC,LO,GO,NO>::Build(repeatedMap,nullSpaceBasis->getNumVectors());
172 RCP<Import<LO,GO,NO> > scatter = ImportFactory<LO,GO,NO>::Build(nullSpaceBasisMap,repeatedMap);
173 tmpNullSpaceBasis->doImport(*nullSpaceBasis,*scatter,INSERT);
174 nullSpaceBasis = tmpNullSpaceBasis.getConst();
175 }
176 } else {
177 FROSCH_ASSERT(false,"Null Space Type unknown.");
178 }
179 IPOUHarmonicCoarseOperatorPtr iPOUHarmonicCoarseOperator = rcp_static_cast<IPOUHarmonicCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
180 if (0>iPOUHarmonicCoarseOperator->initialize(dimension,dofsPerNode,repeatedNodesMap,dofsMaps,nullSpaceBasis,nodeList,dirichletBoundaryDofs)) ret -=10;
181 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("GDSWCoarseOperator")) {
182 GDSWCoarseOperatorPtr gDSWCoarseOperator = rcp_static_cast<GDSWCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
183 if (0>gDSWCoarseOperator->initialize(dimension,dofsPerNode,repeatedNodesMap,dofsMaps,dirichletBoundaryDofs,nodeList)) ret -=10;
184 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("RGDSWCoarseOperator")) {
185 RGDSWCoarseOperatorPtr rGDSWCoarseOperator = rcp_static_cast<RGDSWCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
186 if (0>rGDSWCoarseOperator->initialize(dimension,dofsPerNode,repeatedNodesMap,dofsMaps,dirichletBoundaryDofs,nodeList)) ret -=10;
187 } else {
188 FROSCH_ASSERT(false,"CoarseOperator Type unkown.");
189 }
190 return ret;
191 }
