152 BASKER_INLINE
153 void basic_convert
154 (
155 Int _m,
156 Int *_perm,
157 Int _nblks,
158 Int _parts,
159 Int *_row_tabs,
160 Int *_col_tabs,
161 Int *_treetab
162 )
163 {
164 nblks = _nblks;
165 //For now defaulting parts to 2
166 //nparts = _parts;
167 nparts = 2;
168 //For now do nothing with perm
169 //Note: cannt use init_value because circle depend basker
170 BASKER_ASSERT((nblks+1)>0, "struct basic convert");
171 MALLOC_INT_1DARRAY(row_tabs, nblks+1);
172 for(Int i =0; i < nblks+1; i++)
173 {
174 row_tabs[i] = _row_tabs[i];
175 }
176 MALLOC_INT_1DARRAY(col_tabs, nblks+1);
177 for(Int i=0; i < nblks+1; i++)
178 {
179 col_tabs[i] = _col_tabs[i];
180 }
181 MALLOC_INT_1DARRAY(treetab, nblks+1);
182 for(Int i=0; i < nblks+1; i++)
183 {
184 treetab[i] = _treetab[i];
185 }
186 }//end basic_convert

286 FREE_INT_1DARRAY(left_most);

288 if(row_counts_flg > 0)

292 if(col_counts_flg > 0)

294 FREE_INT_1DARRAY(col_counts);
295 }
296 if(WS_flg > 0)
297 {
298 FREE_INT_1DARRAY(WS);
299 }
300 if(S_col_flg > 0)
301 {
302 FREE_INT_1DARRAY(S_col);
303 }
304 if(S_row_flg > 0)
305 {
306 FREE_INT_1DARRAY(S_row);
307 }
308 if(L_row_counts_flg > 0)
309 {
310 FREE_INT_1DARRAY(L_row_counts);

312 if(L_col_counts_flg > 0)
313 {
314 FREE_INT_1DARRAY(L_col_counts);

329 {
330 FREE_INT_1DARRAY(S_row_counts);
331 }
332
333 }//end Finalize()

335 BASKER_INLINE
336 void init_parent(Int size)
337 {
338 if(size <=0)
339 //{return;} JDB: Changed because of kokkos
340 {size = 1;} //defaul to 1
341 if(parent_flg == 0)
342 {
343 BASKER_ASSERT(size > 0, "struct parent size");
344 MALLOC_INT_1DARRAY(parent, size);
345 parent_flg =size;
346 }
347 else if(size > parent_flg)
348 {
349 BASKER_ASSERT(size > 0, "struct parent size2");
350 REALLOC_1DARRAY(parent, parent_flg, size);
351 parent_flg = size;
352 }
353 //BV::init_value(parent, parent_flg, 0);
354 for(Int j = 0; j < parent_flg; j++)
355 {
356 parent(j) = (Int) 0;
357 }
358
359 }//end init_parent
