83 typename IPOUHarmonicCoarseOperator<SC,LO,GO,NO>::XMapPtr IPOUHarmonicCoarseOperator<SC,LO,GO,NO>::BuildRepeatedMapCoarseLevel(ConstXMapPtr &nodesMap,
84 UN dofsPerNode,
85 ConstXMapPtrVecPtr dofsMaps,
86 UN partition)
87 {
88 //FROSCH_ASSERT(numVert+numEdg+numFac != nodesMap->getGlobalNumElements(),"ERROR: Map does not match number of Entities");
89 const GO INVALID = Teuchos::OrdinalTraits<GO>::invalid();
90 UN blockId = 1; //This is not implemented for Block Variant yet
91 stringstream blockIdStringstream;
92 blockIdStringstream << blockId+1;
93 string blockIdString = blockIdStringstream.str();
94 RCP<ParameterList> coarseSpaceList = sublist(sublist(this->ParameterList_,"Blocks"),blockIdString.c_str());
95
96 Teuchos::Array<GO> nodeEle = nodesMap->getLocalElementList();
97
98 Teuchos::Array<GO> dofEle(nodeEle.size()*dofsPerNode);
99 Teuchos::Array<GO> dmapEle(nodeEle.size());
100 //GDSW Type CoarseOperator
101 if (partition == 0) {
102 for (unsigned j = 0; j<dofsPerNode; j++) {
103 for ( unsigned i = 0; i<nodeEle.size(); i++) {
104 //vertices
105 if (nodeEle[i]<NumEnt_[0]) {
106 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[0];
107 dmapEle[i] = nodeEle[i]+j*NumEnt_[0];
108 }
109 //shortEdges
110 else if (nodeEle[i]>=NumEnt_[0] && nodeEle[i]<NumEnt_[1]+NumEnt_[0]) {
111 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[1]+(dofsPerNode-1)*NumEnt_[0];
112 dmapEle[i] = nodeEle[i]+j*NumEnt_[1]+(dofsPerNode-1)*NumEnt_[0];
113 }
114 //straightEdges
115 else if (nodeEle[i]<NumEnt_[2]+NumEnt_[0]+NumEnt_[1] && nodeEle[i]>=NumEnt_[1]+NumEnt_[0]) {
116 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[2]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1));
117 dmapEle[i] = nodeEle[i]+j*NumEnt_[2]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1));
118 }
119 //edges
120 else if (nodeEle[i]<NumEnt_[3]+NumEnt_[2]+NumEnt_[0]+NumEnt_[1] && nodeEle[i]>=NumEnt_[1]+NumEnt_[0]+NumEnt_[2]) {
121 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[3]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1))+(NumEnt_[2]*(dofsPerNode-1));
122 dmapEle[i] = nodeEle[i]+j*NumEnt_[3]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1))+(NumEnt_[2]*(dofsPerNode-1));
123 }
124 //faces
125 else if (nodeEle[i]<NumEnt_[3]+NumEnt_[2]+NumEnt_[0]+NumEnt_[1]+NumEnt_[4] && nodeEle[i]>=NumEnt_[1]+NumEnt_[0]+NumEnt_[2]+NumEnt_[3]) {
126 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[4]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1))+(NumEnt_[2]*(dofsPerNode-1))+(NumEnt_[3]*(dofsPerNode-1));
127 dmapEle[i] = nodeEle[i]+j*NumEnt_[4]+(NumEnt_[0]*(dofsPerNode-1))+(NumEnt_[1]*(dofsPerNode-1))+(NumEnt_[2]*(dofsPerNode-1))+(NumEnt_[3]*(dofsPerNode-1));
128 } else {
129 if (nodesMap->getComm()->getRank() == 0) std::cout<<"This should never happen\n";
130 }
131 }
132 dofsMaps[j] = Xpetra::MapFactory<LO,GO,NO>::Build(Xpetra::UseTpetra,INVALID,dmapEle,0,nodesMap->getComm());
133 }
134 } //RGDSW type CoarseOperator
135 else if (partition == 1) {
136 for (unsigned j = 0;j<dofsPerNode;j++) {
137 for (unsigned i = 0; i<nodeEle.size(); i++) {
138 //roots
139 if (nodeEle[i]<NumEnt_[5]) {
140 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[5];
141 dmapEle[i] = nodeEle[i]+j*NumEnt_[5];
142 }
143 else if (NumEnt_[6] !=-1 &&nodeEle[i]>=NumEnt_[5] && nodeEle[i]<NumEnt_[6]+NumEnt_[5]) {
144 dofEle[i*dofsPerNode+j] = nodeEle[i]+j*NumEnt_[6]+(dofsPerNode-1)*NumEnt_[5];
145 dmapEle[i] = nodeEle[i]+j*NumEnt_[6]+(dofsPerNode-1)*NumEnt_[5];
146 }
147 }
148 dofsMaps[j] = Xpetra::MapFactory<LO,GO,NO>::Build(Xpetra::UseTpetra,INVALID,dmapEle,0,nodesMap->getComm());
149 }
150 } else if (partition == 2) {
151 FROSCH_ASSERT(false,"GDSWStar is not implemented yet!");
152 }
153 Teuchos::RCP<Xpetra::Map<LO,GO,NO> > tmpMap = Xpetra::MapFactory<LO,GO,NO>::Build(Xpetra::UseTpetra,INVALID,dofEle,0,nodesMap->getComm());
154 return tmpMap;
155 }

179 ConstXMultiVectorPtrVecPtr nullSpaceBasisVec,
180 GOVecPtr2D dirichletBoundaryDofsVec,
181 ConstXMultiVectorPtrVecPtr nodeListVec)
182 {
183 FROSCH_DETAILTIMER_START_LEVELID(buildCoarseSpaceTime,"IPOUHarmonicCoarseOperator::buildCoarseSpace");
184
185 UN TotalNumberOfBlocks = repeatedNodesMapVec.size();
186
187 FROSCH_ASSERT(dofsPerNodeVec.size()==TotalNumberOfBlocks,"dofsPerNodeVec.size()!=TotalNumberOfBlocks");

192
193 // Todo: Move this to a function in HarmonicCoarseOperator at some point
194 for (UN i=0; i<TotalNumberOfBlocks; i++) {
195 FROSCH_ASSERT(!repeatedNodesMapVec[i].is_null(),"repeatedNodesMapVec[i].is_null()");
196 FROSCH_ASSERT(!repeatedDofMapsVec[i].is_null(),"repeatedDofMapsVec[i].is_null()");
197 FROSCH_ASSERT(!nullSpaceBasisVec[i].is_null(),"nullSpaceBasisVec[i].is_null()");
198

206 nodeListVec[i]);
207 }

212 template <class SC,class LO,class GO,class NO>
213 int IPOUHarmonicCoarseOperator<SC,LO,GO,NO>::resetCoarseSpaceBlock(UN blockId,
214 UN dimension,
215 UN dofsPerNode,
216 ConstXMapPtr nodesMap,
217 ConstXMapPtrVecPtr dofsMaps,
218 ConstXMultiVectorPtr nullSpaceBasis,
219 GOVecPtr dirichletBoundaryDofs,
220 ConstXMultiVectorPtr nodeList)
221 {
222 FROSCH_DETAILTIMER_START_LEVELID(resetCoarseSpaceBlockTime,"IPOUHarmonicCoarseOperator::resetCoarseSpaceBlock");
223 FROSCH_ASSERT(dofsMaps.size()==dofsPerNode,"dofsMaps.size()!=dofsPerNode");
224 FROSCH_ASSERT(blockId<=this->NumberOfBlocks_,"Block does not exist yet and can therefore not be reset ("+to_string(blockId)+", "+to_string(this->NumberOfBlocks_)+").");
225
226 // Process the parameter list
227 stringstream blockIdStringstream;
228 blockIdStringstream << blockId+1;
229 string blockIdString = blockIdStringstream.str();
230 RCP<ParameterList> coarseSpaceList = sublist(sublist(this->ParameterList_,"Blocks"),blockIdString.c_str());
231
232 Verbosity verbosity = All;
233 if (!coarseSpaceList->get("Verbosity","All").compare("None")) {
234 verbosity = None;
235 } else if (!coarseSpaceList->get("Verbosity","All").compare("All")) {
236 verbosity = All;
237 } else {
238 FROSCH_ASSERT(false,"FROSch::IPOUHarmonicCoarseOperator: Specify a valid verbosity level.");

240
241 bool useForCoarseSpace = coarseSpaceList->get("Use For Coarse Space",true);
242
243 if (useForCoarseSpace) {
244 this->GammaDofs_.resize(this->GammaDofs_.size()+1);
245 this->IDofs_.resize(this->IDofs_.size()+1);
246 this->InterfaceCoarseSpaces_.resize(this->InterfaceCoarseSpaces_.size()+1);
247 this->DofsMaps_.resize(this->DofsMaps_.size()+1);
248 this->DofsPerNode_.resize(this->DofsPerNode_.size()+1);
249 this->NumberOfBlocks_++;
250
251 if (this->Verbose_) {
252 cout
253 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
254 << setw(89) << "-----------------------------------------------------------------------------------------"
255 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "

266 << "| " << left << setw(41) << "Spatial dimensions" << right
267 << " | " << setw(41) << dimension
268 << " |"
269 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
270 << "| " << left << setw(41) << "Number of degrees of freedom per node" << right
271 << " | " << setw(41) << dofsPerNode
272 << " |"
273 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
274 << "| " << left << setw(41) << "Interface partition of unity type" << right
275 << " | " << setw(41) << coarseSpaceList->sublist("InterfacePartitionOfUnity").get("Type","GDSW")
276 << " |"

283 << endl;

286 this->DofsMaps_[blockId] = dofsMaps;
287 this->DofsPerNode_[blockId] = dofsPerNode;
288
289 // Compute Interface Partition of Unity
290 // AH: Can we get rid of the PartitionType_?
291 InterfacePartitionOfUnityPtr interfacePartitionOfUnity;
292 if (!coarseSpaceList->sublist("InterfacePartitionOfUnity").get("Type","GDSW").compare("GDSW")) {
293 coarseSpaceList->sublist("InterfacePartitionOfUnity").sublist("GDSW").set("Test Unconnected Interface",this->ParameterList_->get("Test Unconnected Interface",true));
294 interfacePartitionOfUnity = InterfacePartitionOfUnityPtr(new GDSWInterfacePartitionOfUnity<SC,LO,GO,NO>(this->MpiComm_,this->SerialComm_,dimension,this->DofsPerNode_[blockId],nodesMap,this->DofsMaps_[blockId],sublist(sublist(coarseSpaceList,"InterfacePartitionOfUnity"),"GDSW"),verbosity,this->LevelID_));
295 this->PartitionType_ = 0;
296 } else if (!coarseSpaceList->sublist("InterfacePartitionOfUnity").get("Type","GDSW").compare("GDSWStar")) {
297 coarseSpaceList->sublist("InterfacePartitionOfUnity").sublist("GDSWStar").set("Test Unconnected Interface",this->ParameterList_->get("Test Unconnected Interface",true));
298 interfacePartitionOfUnity = InterfacePartitionOfUnityPtr(new GDSWStarInterfacePartitionOfUnity<SC,LO,GO,NO>(this->MpiComm_,this->SerialComm_,dimension,this->DofsPerNode_[blockId],nodesMap,this->DofsMaps_[blockId],sublist(sublist(coarseSpaceList,"InterfacePartitionOfUnity"),"GDSWStar"),verbosity,this->LevelID_));
299 this->PartitionType_ = 2;
300 } else if (!coarseSpaceList->sublist("InterfacePartitionOfUnity").get("Type","GDSW").compare("RGDSW")) {
301 coarseSpaceList->sublist("InterfacePartitionOfUnity").sublist("RGDSW").set("Test Unconnected Interface",this->ParameterList_->get("Test Unconnected Interface",true));
302 interfacePartitionOfUnity = InterfacePartitionOfUnityPtr(new RGDSWInterfacePartitionOfUnity<SC,LO,GO,NO>(this->MpiComm_,this->SerialComm_,dimension,this->DofsPerNode_[blockId],nodesMap,this->DofsMaps_[blockId],sublist(sublist(coarseSpaceList,"InterfacePartitionOfUnity"),"RGDSW"),verbosity,this->LevelID_));
303 this->PartitionType_ = 1;
304 } else {
305 FROSCH_ASSERT(false,"InterfacePartitionOfUnity Type is unknown.");
306 }
307
308 // Extract the interface and the interior from the DDInterface stored in the Interface Partition of Unity object
309 InterfaceEntityPtr interface = interfacePartitionOfUnity->getDDInterface()->getInterface()->getEntity(0);
310 InterfaceEntityPtr interior = interfacePartitionOfUnity->getDDInterface()->getInterior()->getEntity(0);

313 if (interface->getNumNodes()==0) {

315 coarseSpaceList->sublist("InterfacePartitionOfUnity").sublist("RGDSW").set("Test Unconnected Interface",this->ParameterList_->get("Test Unconnected Interface",true));

317 this->SerialComm_,

324 this->LevelID_,
325 interfacePartitionOfUnity->getDDInterfaceNonConst()));
326

329 interface = interior;

348 for (UN k=0; k<this->DofsPerNode_[blockId]; k++) {

376

378 this->CoarseDofsPerNode_ = nullSpaceBasis->getNumVectors();
379
380 this->KRowMap_ = PartitionOfUnity_->getAssembledPartitionOfUnityMap();
381 NumEnt_ = interfacePartitionOfUnity->getDDInterface()->getNumEnt();
382
383 if (!this->DistributionList_->get("Type","linear").compare("ZoltanDual")) {
384 FROSCH_ASSERT(this->NumberOfBlocks_==1,"Distribution Type ZoltanDual only works for one Block");
385 Teuchos::RCP<DDInterface<SC,LO,GO,NO> > theInterface =Teuchos::rcp_const_cast<DDInterface<SC,LO,GO,NO> >(interfacePartitionOfUnity->getDDInterface());
386 this->buildGlobalGraph(theInterface);
387 int dim = dimension;
388 sublist(this->ParameterList_,"CoarseSolver")->set("Dimension",dim);
389 }
390
391 // Build local basis
392 LocalPartitionOfUnityBasis_ = LocalPartitionOfUnityBasisPtr(new LocalPartitionOfUnityBasis<SC,LO,GO,NO>(this->MpiComm_,this->SerialComm_,this->DofsPerNode_[blockId],sublist(coarseSpaceList,"LocalPartitionOfUnityBasis"),interfaceNullspaceBasis.getConst(),PartitionOfUnity_->getLocalPartitionOfUnity(),PartitionOfUnity_->getPartitionOfUnityMaps())); // sublist(coarseSpaceList,"LocalPartitionOfUnityBasis") testen

395
396 this->InterfaceCoarseSpaces_[blockId] = LocalPartitionOfUnityBasis_->getLocalPartitionOfUnitySpace();
397 FROSCH_NOTIFICATION("FROSch::IPOUHarmonicCoarseOperator",this->Verbose_,"Need to build block coarse sizes for use in MueLu nullspace. This is not performed here yet.");
398 //if (this->Verbose_) {RCP<FancyOStream> fancy = fancyOStream(rcpFromRef(cout)); this->MVPhiGamma_[blockId]->describe(*fancy,VERB_EXTREME);}
399 }
