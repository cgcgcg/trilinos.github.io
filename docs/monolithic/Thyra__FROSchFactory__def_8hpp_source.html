60 //Main Function to use FROSch as Prec
61 template<class SC, class LO , class GO, class NO>
62 void FROSchFactory<SC,LO,GO,NO>::initializePrec(const ConstLinearOpSourceBasePtr& fwdOpSrc,
63 PreconditionerBase<SC>* prec,
64 const ESupportSolveUse supportSolveUse) const
65 {
66 //PreCheck
67 TEUCHOS_ASSERT(nonnull(fwdOpSrc));
68 //TEUCHOS_ASSERT(this->isCompatible(*fwdOpSrc));
69 TEUCHOS_ASSERT(prec);
70
71 // Retrieve wrapped concrete Xpetra matrix from FwdOp
72 const ConstLinearOpBasePtr fwdOp = fwdOpSrc->getOp();
73 TEUCHOS_TEST_FOR_EXCEPT(is_null(fwdOp));
74
75 // Check whether it is Epetra/Tpetra
76 bool bIsTpetra = ThyraUtils<SC,LO,GO,NO>::isTpetra(fwdOp);
77 bool bIsBlocked = ThyraUtils<SC,LO,GO,NO>::isBlockedOperator(fwdOp);
78 TEUCHOS_TEST_FOR_EXCEPT((bIsTpetra == false) && bIsBlocked == false);
79 TEUCHOS_TEST_FOR_EXCEPT((bIsTpetra == true) && bIsBlocked == true);
80
81 // Check whether to use HalfPrecision
82#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
83 const bool useHalfPrecision = paramList_->get<bool>("Use Half Precision", false);
84#endif
85
86 // Retrieve Matrix
87 ConstXMatrixPtr A = ThyraUtils<SC,LO,GO,NO>::toXpetra(fwdOp);
88 TEUCHOS_TEST_FOR_EXCEPT(is_null(A));
89
90 CommPtr comm = A->getMap()->getComm();
91 UnderlyingLib underlyingLib = A->getMap()->lib();
92
93 // Retrieve concrete preconditioner object
94 const Ptr<DefaultPreconditioner<SC> > defaultPrec = ptr(dynamic_cast<DefaultPreconditioner<SC> *>(prec));
95 TEUCHOS_TEST_FOR_EXCEPT(is_null(defaultPrec));
96
97 // extract preconditioner operator
98 auto precOp = defaultPrec->getNonconstUnspecifiedPrecOp();
99
100 // Abstract SchwarzPreconditioner
101 RCP<SchwarzPreconditioner<SC,LO,GO,NO> > schwarzPreconditioner = null;
102#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
103 RCP<SchwarzPreconditioner<HalfSC,LO,GO,NO> > halfSchwarzPreconditioner = null;
104#endif
105 const bool startingOver = (precOp.is_null() || !paramList_->isParameter("Recycling") || !paramList_->get("Recycling",true));
106
107 if (startingOver) {
108 FROSCH_ASSERT(paramList_->isParameter("FROSch Preconditioner Type"),"FROSch Preconditioner Type is not defined!");
109
110 if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("AlgebraicOverlappingPreconditioner")) {
111 // Extract the repeated map
112 ConstXMapPtr repeatedMap = extractRepeatedMap(comm,underlyingLib);
113
114 RCP<AlgebraicOverlappingPreconditioner<SC,LO,GO,NO> > AOP(new AlgebraicOverlappingPreconditioner<SC,LO,GO,NO>(A,paramList_));
115
116 AOP->initialize(paramList_->get("Overlap",1),
117 repeatedMap);
118
119 schwarzPreconditioner = AOP;
120 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("GDSWPreconditioner")) {
121 // Extract the repeated map
122 ConstXMapPtr repeatedMap = extractRepeatedMap(comm,underlyingLib);
123
124 // Extract the coordinate list
125 ConstXMultiVectorPtr coordinatesList = extractCoordinatesList(comm,underlyingLib);
126
127 // Extract the dof ordering
128 DofOrdering dofOrdering = NodeWise;
129 if (!paramList_->get("DofOrdering","NodeWise").compare("NodeWise")) {
130 dofOrdering = NodeWise;
131 } else if (!paramList_->get("DofOrdering","NodeWise").compare("DimensionWise")) {
132 dofOrdering = DimensionWise;
133 } else if (!paramList_->get("DofOrdering","NodeWise").compare("Custom")) {
134 dofOrdering = Custom;
135 } else {
136 FROSCH_ASSERT(false,"ERROR: Specify a valid DofOrdering.");
137 }
138
139 RCP<GDSWPreconditioner<SC,LO,GO,NO> > GP(new GDSWPreconditioner<SC,LO,GO,NO>(A,paramList_));
140
141 GP->initialize(paramList_->get("Dimension",3),
142 paramList_->get("DofsPerNode",1),
143 dofOrdering,
144 paramList_->get("Overlap",1),
145 repeatedMap,
146 coordinatesList);
147
148 schwarzPreconditioner = GP;
149 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("RGDSWPreconditioner")) {
150 // Extract the repeated map
151 ConstXMapPtr repeatedMap = extractRepeatedMap(comm,underlyingLib);
152
153 // Extract the coordinate list
154 ConstXMultiVectorPtr coordinatesList = extractCoordinatesList(comm,underlyingLib);
155
156 // Extract the dof ordering
157 DofOrdering dofOrdering = NodeWise;
158 if (!paramList_->get("DofOrdering","NodeWise").compare("NodeWise")) {
159 dofOrdering = NodeWise;
160 } else if (!paramList_->get("DofOrdering","NodeWise").compare("DimensionWise")) {
161 dofOrdering = DimensionWise;
162 } else if (!paramList_->get("DofOrdering","NodeWise").compare("Custom")) {
163 dofOrdering = Custom;
164 } else {
165 FROSCH_ASSERT(false,"ERROR: Specify a valid DofOrdering.");
166 }
167#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
168 if (useHalfPrecision)
169 {
170 auto ANonConst = rcp_const_cast<XMatrix>(A);
171 auto halfA = Xpetra::convertToHalfPrecision(ANonConst);
172
173 RCP<RGDSWPreconditioner<HalfSC,LO,GO,NO> > halfRGP(new RGDSWPreconditioner<HalfSC,LO,GO,NO>(halfA,paramList_));
174
175 XMultiVectorPtr coordinatesListNonConst = (coordinatesList == null ? null : rcp_const_cast<XMultiVector>(coordinatesList));
176 auto halfCoordinatesList = rcp_dynamic_cast<const HalfPrecMultiVector> (Xpetra::convertToHalfPrecision(coordinatesListNonConst));
177 halfRGP->initialize(paramList_->get("Dimension",3),
178 paramList_->get("DofsPerNode",1),
179 dofOrdering,
180 paramList_->get("Overlap",1),
181 repeatedMap,
182 halfCoordinatesList);
183
184 halfSchwarzPreconditioner = halfRGP;
185 } else
186#endif
187 {
188 RCP<RGDSWPreconditioner<SC,LO,GO,NO> > RGP(new RGDSWPreconditioner<SC,LO,GO,NO>(A,paramList_));
189
190 RGP->initialize(paramList_->get("Dimension",3),
191 paramList_->get("DofsPerNode",1),
192 dofOrdering,
193 paramList_->get("Overlap",1),
194 repeatedMap,
195 coordinatesList);
196
197 schwarzPreconditioner = RGP;
198 }
199 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("OneLevelPreconditioner")) {
200 // Extract the repeated map
201 ConstXMapPtr repeatedMap = extractRepeatedMap(comm,underlyingLib);
202
203 RCP<OneLevelPreconditioner<SC,LO,GO,NO> > OLP(new OneLevelPreconditioner<SC,LO,GO,NO>(A,paramList_));
204
205 OLP->initialize(paramList_->get("Overlap",1),
206 repeatedMap);
207
208 schwarzPreconditioner = OLP;
209 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("TwoLevelPreconditioner")) {
210 // Extract the repeated map
211 ConstXMapPtr repeatedMap = extractRepeatedMap(comm,underlyingLib);
212
213 // Extract the null space
214 ConstXMultiVectorPtr nullSpaceBasis = extractNullSpace(comm,underlyingLib);
215
216 // Extract the coordinate list
217 ConstXMultiVectorPtr coordinatesList = extractCoordinatesList(comm,underlyingLib);
218
219 // Extract the dof ordering
220 DofOrdering dofOrdering = NodeWise;
221 if (!paramList_->get("DofOrdering","NodeWise").compare("NodeWise")) {
222 dofOrdering = NodeWise;
223 } else if (!paramList_->get("DofOrdering","NodeWise").compare("DimensionWise")) {
224 dofOrdering = DimensionWise;
225 } else if (!paramList_->get("DofOrdering","NodeWise").compare("Custom")) {
226 dofOrdering = Custom;
227 } else {
228 FROSCH_ASSERT(false,"ERROR: Specify a valid DofOrdering.");
229 }
230#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
231 if (useHalfPrecision)
232 {
233 auto ANonConst = rcp_const_cast<XMatrix>(A);
234 auto halfA = Xpetra::convertToHalfPrecision(ANonConst);
235
236 RCP<TwoLevelPreconditioner<HalfSC,LO,GO,NO> > halfTLP(new TwoLevelPreconditioner<HalfSC,LO,GO,NO>(halfA,paramList_));
237
238 XMultiVectorPtr coordinatesListNonConst = rcp_const_cast<XMultiVector>(coordinatesList);
239 XMultiVectorPtr nullSpaceBasisNonConst = rcp_const_cast<XMultiVector>(nullSpaceBasis);
240 auto halfCoordinatesList = (coordinatesListNonConst == null ? null : rcp_dynamic_cast<const HalfPrecMultiVector> (Xpetra::convertToHalfPrecision(coordinatesListNonConst)));
241 auto halfNullSpaceBasis = (nullSpaceBasisNonConst == null ? null : rcp_dynamic_cast<const HalfPrecMultiVector> (Xpetra::convertToHalfPrecision(nullSpaceBasisNonConst)));
242 halfTLP->initialize(paramList_->get("Dimension",3),
243 paramList_->get("DofsPerNode",1),
244 paramList_->get("Overlap",1),
245 halfNullSpaceBasis,
246 halfCoordinatesList,
247 dofOrdering,
248 repeatedMap);
249
250 halfSchwarzPreconditioner = halfTLP;
251 } else
252#endif
253 {
254 RCP<TwoLevelPreconditioner<SC,LO,GO,NO> > TLP(new TwoLevelPreconditioner<SC,LO,GO,NO>(A,paramList_));
255
256 TLP->initialize(paramList_->get("Dimension",3),
257 paramList_->get("DofsPerNode",1),
258 paramList_->get("Overlap",1),
259 nullSpaceBasis,
260 coordinatesList,
261 dofOrdering,
262 repeatedMap);
263
264 schwarzPreconditioner = TLP;
265 }
266 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("TwoLevelBlockPreconditioner")) {
267 ConstXMapPtrVecPtr repeatedMaps = null;
268 ConstXMultiVectorPtrVecPtr coordinatesList = null;
269 UNVecPtr dofsPerNodeVector;
270 DofOrderingVecPtr dofOrderings;
271
272 FROSCH_ASSERT(paramList_->isParameter("DofsPerNode Vector"),"Currently, TwoLevelBlockPreconditioner cannot be constructed without DofsPerNode Vector.");
273 FROSCH_ASSERT(paramList_->isParameter("DofOrdering Vector"),"Currently, TwoLevelBlockPreconditioner cannot be constructed without DofOrdering Vector.");
274 // Extract the repeated map vector
275 if (paramList_->isParameter("Repeated Map Vector")) {
276 XMapPtrVecPtr repeatedMapsTmp = ExtractVectorFromParameterList<XMapPtr>(*paramList_,"Repeated Map Vector");
277 XMultiVectorPtrVecPtr nodeListVecTmp = ExtractVectorFromParameterList<XMultiVectorPtr>(*paramList_,"Coordinates List Vector");
278 if (!repeatedMapsTmp.is_null()) {
279 repeatedMaps.resize(repeatedMapsTmp.size());
280 for (unsigned i=0; i<repeatedMaps.size(); i++) {
281 repeatedMaps[i] = repeatedMapsTmp[i].getConst();
282 }
283 }
284 // Extract the nodeList map vector
285 if (!nodeListVecTmp.is_null()) {
286 coordinatesList.resize(nodeListVecTmp.size());
287 for (unsigned i = 0; i<coordinatesList.size();i++) {
288 coordinatesList[i] = nodeListVecTmp[i].getConst();
289 }
290 }
291
292 FROSCH_ASSERT(!repeatedMaps.is_null(),"FROSch::FROSchFactory: repeatedMaps.is_null()");
293 // Extract the DofsPerNode vector
294 dofsPerNodeVector = ExtractVectorFromParameterList<UN>(*paramList_,"DofsPerNode Vector");
295 // Extract the DofOrdering vector
296 dofOrderings = ExtractVectorFromParameterList<DofOrdering>(*paramList_,"DofOrdering Vector");
297 } else {
298 FROSCH_ASSERT(false,"Currently, TwoLevelBlockPreconditioner cannot be constructed without Repeated Maps.");
299 }
300
301 FROSCH_ASSERT(repeatedMaps.size()==dofsPerNodeVector.size(),"RepeatedMaps.size()!=dofsPerNodeVector.size()");
302 FROSCH_ASSERT(repeatedMaps.size()==dofOrderings.size(),"RepeatedMaps.size()!=dofOrderings.size()");
303
304 RCP<TwoLevelBlockPreconditioner<SC,LO,GO,NO> > TLBP(new TwoLevelBlockPreconditioner<SC,LO,GO,NO>(A,paramList_));
305
306 TLBP->initialize(paramList_->get("Dimension",3),
307 dofsPerNodeVector,
308 dofOrderings,
309 paramList_->get("Overlap",1),
310 coordinatesList,
311 repeatedMaps);
312
313 schwarzPreconditioner = TLBP;
314 } else {
315 FROSCH_ASSERT(false,"Thyra::FROSchFactory: Preconditioner Type is unknown.");
316 }
317
318 //-----------------------------------------------
319
320 LinearOpBasePtr thyraPrecOp = null;
321
322 if (bIsTpetra) {
323 RCP<Tpetra::Operator<SC,LO,GO,NO> > tpOp;
324#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
325 if (halfSchwarzPreconditioner != null)
326 {
327 halfSchwarzPreconditioner->compute();
328 auto halfSchwarzOperator = rcp (new HalfPrecOp (halfSchwarzPreconditioner));
329 RCP<TpetraPreconditioner<SC,LO,GO,NO> > TPreconditioner (new TpetraPreconditioner<SC,LO,GO,NO>(halfSchwarzOperator));
330 tpOp = rcp_dynamic_cast<Tpetra::Operator<SC,LO,GO,NO> >(TPreconditioner);
331 } else
332#endif
333 {
334 schwarzPreconditioner->compute();
335 auto schwarzOperator = rcp_dynamic_cast<Xpetra::Operator<SC,LO,GO,NO> >(schwarzPreconditioner);
336 RCP<TpetraPreconditioner<SC,LO,GO,NO> > TPreconditioner (new TpetraPreconditioner<SC,LO,GO,NO>(schwarzOperator));
337 tpOp = rcp_dynamic_cast<Tpetra::Operator<SC,LO,GO,NO> >(TPreconditioner);
338 }
339 thyraPrecOp = Thyra::createLinearOp(tpOp);
340 } else
341 {
342 RCP<Operator<SC,LO,GO,NO> > xpOp;
343#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
344 if (halfSchwarzPreconditioner != null) {
345 halfSchwarzPreconditioner->compute();
346 xpOp = rcp_dynamic_cast<Operator<SC,LO,GO,NO> >(rcp (new HalfPrecOp (halfSchwarzPreconditioner)));
347 } else
348#endif
349 {
350 schwarzPreconditioner->compute();
351 xpOp = rcp_dynamic_cast<Operator<SC,LO,GO,NO> >(schwarzPreconditioner);
352 }
353
354 ConstVectorSpaceBasePtr thyraRangeSpace = ThyraUtils<SC,LO,GO,NO>::toThyra(schwarzPreconditioner->getRangeMap());
355 ConstVectorSpaceBasePtr thyraDomainSpace = ThyraUtils<SC,LO,GO,NO>::toThyra(schwarzPreconditioner->getDomainMap());
356 thyraPrecOp = fROSchLinearOp<SC,LO,GO,NO>(thyraRangeSpace,thyraDomainSpace,xpOp,bIsTpetra);
357 }
358 TEUCHOS_TEST_FOR_EXCEPT(is_null(thyraPrecOp));
359
360 //Set SchwarzPreconditioner
361 defaultPrec->initializeUnspecified(thyraPrecOp);
362 } else {
363 // cast to SchwarzPreconditioner
364 RCP<Operator<SC,LO,GO,NO> > xpetraOp;
365 if (bIsTpetra) {
366 RCP<Thyra::TpetraLinearOp<SC,LO,GO,NO> > thyra_precOp = rcp_dynamic_cast<Thyra::TpetraLinearOp<SC,LO,GO,NO>>(precOp, true);
367 RCP<Tpetra::Operator<SC,LO,GO,NO> > tpOp = thyra_precOp->getTpetraOperator();
368 RCP<TpetraPreconditioner<SC,LO,GO,NO> > tpetra_precOp = rcp_dynamic_cast<TpetraPreconditioner<SC,LO,GO,NO>>(tpOp, true);
369 xpetraOp = rcp_dynamic_cast<Operator<SC,LO,GO,NO> >(tpetra_precOp->getSchwarzPreconditioner());
370 } else
371 {
372 RCP<FROSchLinearOp<SC,LO,GO,NO> > fROSch_LinearOp = rcp_dynamic_cast<FROSchLinearOp<SC,LO,GO,NO> >(precOp,true);
373 xpetraOp = fROSch_LinearOp->getXpetraOperator();
374 }
375
376#if defined(HAVE_TPETRA_INST_DOUBLE) && defined(HAVE_TPETRA_INST_FLOAT)
377 if (useHalfPrecision)
378 {
379 if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("RGDSWPreconditioner") ||
380 !paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("TwoLevelPreconditioner")) {
381 auto ANonConst = rcp_const_cast<XMatrix>(A);
382 auto halfA = rcp_const_cast<const HalfPrecMatrix>(Xpetra::convertToHalfPrecision(ANonConst));
383 auto halfPrecOp = rcp_dynamic_cast<HalfPrecOp>(xpetraOp);
384 auto halfXpetraOp = rcp_dynamic_cast<Operator<HalfSC,LO,GO,NO>>(halfPrecOp->GetHalfPrecisionOperator());
385 RCP<OneLevelPreconditioner<HalfSC,LO,GO,NO> > TLP = rcp_dynamic_cast<OneLevelPreconditioner<HalfSC,LO,GO,NO> >(halfXpetraOp, true);
386 TLP->resetMatrix(halfA);
387
388 halfSchwarzPreconditioner = rcp_dynamic_cast<SchwarzPreconditioner<HalfSC,LO,GO,NO> >(halfXpetraOp, true);
389 halfSchwarzPreconditioner->compute();
390 }
391 } else
392#endif
393 {
394 if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("AlgebraicOverlappingPreconditioner")) {
395 RCP<AlgebraicOverlappingPreconditioner<SC,LO,GO,NO> > AOP = rcp_dynamic_cast<AlgebraicOverlappingPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
396 AOP->resetMatrix(A);
397 schwarzPreconditioner = AOP;
398 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("GDSWPreconditioner")) {
399 RCP<AlgebraicOverlappingPreconditioner<SC,LO,GO,NO> > GP = rcp_dynamic_cast<AlgebraicOverlappingPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
400 GP->resetMatrix(A);
401 schwarzPreconditioner = GP;
402 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("RGDSWPreconditioner")) {
403 RCP<OneLevelPreconditioner<SC,LO,GO,NO> > RGP = rcp_dynamic_cast<OneLevelPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
404 RGP->resetMatrix(A);
405 schwarzPreconditioner = RGP;
406 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("OneLevelPreconditioner")) {
407 RCP<OneLevelPreconditioner<SC,LO,GO,NO> > OLP = rcp_dynamic_cast<OneLevelPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
408 OLP->resetMatrix(A);
409 schwarzPreconditioner = OLP;
410 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("TwoLevelPreconditioner")) {
411 RCP<OneLevelPreconditioner<SC,LO,GO,NO> > TLP = rcp_dynamic_cast<OneLevelPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
412 TLP->resetMatrix(A);
413 schwarzPreconditioner = TLP;
414 } else if (!paramList_->get("FROSch Preconditioner Type","TwoLevelPreconditioner").compare("TwoLevelBlockPreconditioner")) {
415 RCP<OneLevelPreconditioner<SC,LO,GO,NO> > TLBP = rcp_dynamic_cast<OneLevelPreconditioner<SC,LO,GO,NO> >(xpetraOp, true);
416 TLBP->resetMatrix(A);
417 schwarzPreconditioner = TLBP;
418 } else {
419 FROSCH_ASSERT(false,"Thyra::FROSchFactory: Preconditioner Type is unknown.");
420 }
421 // recompute SchwarzPreconditioner
422 schwarzPreconditioner->compute();
423 }
424 }
425 }

498 template <class SC, class LO, class GO, class NO>
499 typename FROSchFactory<SC,LO,GO,NO>::ConstXMapPtr FROSchFactory<SC,LO,GO,NO>::extractRepeatedMap(CommPtr comm,
500 UnderlyingLib lib) const
501 {
502 ConstXMapPtr repeatedMap = null;
503 if (paramList_->isParameter("Repeated Map")) {
504 repeatedMap = ExtractPtrFromParameterList<XMap>(*paramList_,"Repeated Map").getConst();
505 if (repeatedMap.is_null()) {
506 if (lib==UseTpetra) { // If coordinatesList.is_null(), we look for Tpetra/Epetra RCPs
507 RCP<const Tpetra::Map<LO,GO,NO> > repeatedMapTmp = ExtractPtrFromParameterList<const Tpetra::Map<LO,GO,NO> >(*paramList_,"Repeated Map");
508
509 RCP<const TpetraMap<LO,GO,NO> > xTpetraRepeatedMap(new const TpetraMap<LO,GO,NO>(repeatedMapTmp));
510 repeatedMap = rcp_dynamic_cast<ConstXMap>(xTpetraRepeatedMap);
511 }
512 }
513 FROSCH_ASSERT(!repeatedMap.is_null(),"FROSch::FROSchFactory: repeatedMap.is_null()");
514 }
515 return repeatedMap;
516 }

518 template <class SC, class LO, class GO, class NO>
519 typename FROSchFactory<SC,LO,GO,NO>::ConstXMultiVectorPtr FROSchFactory<SC,LO,GO,NO>::extractCoordinatesList(CommPtr comm,
