27template<class HandleType, class ZViewType, class RHSViewType>
28inline
29void forward(HandleType& ahandle, ZViewType& Z, RHSViewType& RHS)
30{
31 using value_type = typename ZViewType::value_type ;
32 using execution_space = typename ZViewType::device_type::execution_space ;
33 using memory_space = typename ZViewType::device_type::memory_space ;
34 using ViewMatrixType = Kokkos::View<value_type**, Kokkos::LayoutLeft, memory_space>;
35#ifdef ADELUS_HOST_PINNED_MEM_MPI
36 #if defined(KOKKOS_ENABLE_CUDA)
37 using ViewMatrixHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace>;//CudaHostPinnedSpace
38 #elif defined(KOKKOS_ENABLE_HIP)
39 using ViewMatrixHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace>;//HIPHostPinnedSpace
40 #endif
41#endif
42
43 MPI_Comm row_comm = ahandle.get_row_comm();
44 MPI_Comm col_comm = ahandle.get_col_comm();
45 int myrow = ahandle.get_myrow();
46 int nprocs_row = ahandle.get_nprocs_row();
47 int nprocs_col = ahandle.get_nprocs_col();
48 int nrows_matrix = ahandle.get_nrows_matrix();
49 int my_rows = ahandle.get_my_rows();
50
51 int k_row; // torus-wrap row corresponding to kth global row
52 int k_col; // torus-wrap column corresponding to kth global col
53 int istart; // Starting row index for pivot column
54 int count_row; // dummy index
55 value_type d_one = static_cast<value_type>( 1.0);
56 value_type d_min_one = static_cast<value_type>(-1.0);
57
58 ViewMatrixType piv_col( "piv_col", my_rows, 1 ); // portion of pivot column I am sending
59 ViewMatrixType ck( "ck", 1, RHS.extent(1) ); // rhs corresponding to current column of the backsubstitution
60#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
61 ViewMatrixHostPinnType h_piv_col( "h_piv_col", my_rows, 1 );
62 ViewMatrixHostPinnType h_ck( "h_ck", 1, RHS.extent(1) );
63#endif
64
65#ifdef PRINT_STATUS
66 printf("Rank %i -- forward() Begin forward solve with myrow %d, nprocs_row %d, nprocs_col %d, nrows_matrix %d, ncols_matrix %d, my_rows %d, my_cols %d, my_rhs %d, nrhs %d, value_type %s, execution_space %s, memory_space %s\n", ahandle.get_myrank(), myrow, nprocs_row, nprocs_col, nrows_matrix, ahandle.get_ncols_matrix(), my_rows, ahandle.get_my_cols(), ahandle.get_my_rhs(), ahandle.get_nrhs(), typeid(value_type).name(), typeid(execution_space).name(), typeid(memory_space).name());
67#endif
68
69#ifdef GET_TIMING
70 double t1, fwdsolvetime;
71 t1 = MPI_Wtime();
72#endif
73
74 // Perform the Forward Substitution
75 for (int k=0; k<= nrows_matrix-2; k++) {
76 k_row=k%nprocs_col;
77 k_col=k%nprocs_row;
78 istart = (k+1-myrow)/nprocs_col;
79 if (istart * nprocs_col < k+1-myrow) istart++;
80
81 if (istart < my_rows) {
82 Kokkos::deep_copy( subview(piv_col, Kokkos::make_pair(0, my_rows - istart), 0),
83 subview(Z, Kokkos::make_pair(istart, my_rows), k/nprocs_row) );
84 }
85 count_row = my_rows - istart;
86
87 //Note: replace MPI_Send/MPI_Irecv with MPI_Bcast
88 // Rank k_col broadcasts the pivot_col to all
89 // other ranks in the row_comm
90#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
91 Kokkos::deep_copy(h_piv_col,piv_col);
92 MPI_Bcast(reinterpret_cast<char *>(h_piv_col.data()), count_row*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_col, row_comm);
93 Kokkos::deep_copy(piv_col,h_piv_col);
94#else //GPU-aware MPI
95 MPI_Bcast(reinterpret_cast<char *>(piv_col.data()), count_row*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_col, row_comm);
96#endif
97
98 if (ahandle.get_my_rhs() > 0) {
99 //ck = RHS(k/nprocs_col,0);
100 //MPI_Bcast((char *)(&ck),sizeof(ADELUS_DATA_TYPE),MPI_CHAR,k_row,col_comm);
101 //count_row=0;
102 //printf("Point 2: k %d, istart %d, my_rows %d\n", k, istart, my_rows);
103 //for (int i=istart;i<=my_rows-1;i++) {
104 // RHS(i,0) = RHS(i,0) - piv_col(count_row) * ck;
105 // count_row++;
106 //}
107 int curr_lrid = k/nprocs_col;//note: nprocs_col (global var) cannot be read in a device function
108 if (curr_lrid < static_cast<int>(RHS.extent(0))) { //note: to avoid out-of-bounds access on the RHS
109 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
110 ck(0,i) = RHS(curr_lrid,i);
111 });
112 }
113
114#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
115 Kokkos::deep_copy(h_ck,ck);
116 MPI_Bcast(reinterpret_cast<char *>(h_ck.data()), RHS.extent(1)*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_row, col_comm);
117 Kokkos::deep_copy(ck,h_ck);
118#else //GPU-aware MPI
119 Kokkos::fence();
120 MPI_Bcast(reinterpret_cast<char *>(ck.data()), RHS.extent(1)*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_row, col_comm);
121#endif
122
123 auto sub_pivot_col = subview(piv_col, Kokkos::make_pair(0, my_rows - istart), Kokkos::ALL());
124 auto sub_rhs = subview(RHS, Kokkos::make_pair(istart, my_rows), Kokkos::ALL());
125 if (istart < my_rows) {
126 KokkosBlas::gemm("N", "N", d_min_one, sub_pivot_col, ck, d_one, sub_rhs);
127 }
128 }
129 MPI_Barrier(ahandle.get_comm());
130 }// end of for (k=0; k<= nrows_matrix-2; k++)
131
132#ifdef GET_TIMING
133 fwdsolvetime = MPI_Wtime() - t1;
134 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
135 "Total time in forward solve", &fwdsolvetime);
136#endif
137}
