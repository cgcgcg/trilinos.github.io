72template<typename Traits>
73void PHX::DagManager<Traits>::
74registerEvaluator(const Teuchos::RCP<PHX::Evaluator<Traits> >& p)
75{
76#ifdef PHX_TEUCHOS_TIME_MONITOR
77 // Add counter to name so that all timers have unique names
78 static int count=0;
79 std::stringstream uniqueName;
80 uniqueName << "Phalanx: Evaluator " << count++ <<": [" << evaluation_type_name_ << "] ";
81 evalTimers.push_back(
82 Teuchos::TimeMonitor::getNewTimer(uniqueName.str() + p->getName()));
83#endif
84
85 // insert evaluated fields into map, check for multiple evaluators
86 // that provide the same field.
87 nodes_.push_back(PHX::DagNode<Traits>(static_cast<int>(nodes_.size()),p));
88 const std::vector<Teuchos::RCP<PHX::FieldTag>>& evaluatedFields =
89 p->evaluatedFields();
90 for (auto i=evaluatedFields.cbegin(); i != evaluatedFields.cend(); ++i) {
91 auto check = field_to_node_index_.insert(std::make_pair((*i)->identifier(), static_cast<int>(nodes_.size()-1)));
92 if (!allow_multiple_evaluators_for_same_field_) {
93 TEUCHOS_TEST_FOR_EXCEPTION(check.second == false,
94 PHX::multiple_evaluator_for_field_exception,
95 *this
96 << "\n\nError: PHX::DagManager::registerEvaluator() - The field \""
97 << (*i)->identifier()
98 << "\" that is evaluated by the evaluator named \""
99 << p->getName()
100 << "\" is already evaluated by another registered evaluator named \""
101 << (nodes_[field_to_node_index_[(*i)->identifier()]]).get()->getName()
102 << "\"."
103 // << " Printing evaluators:\n" << *p << "\n"
104 // << (evaluators_[field_to_index_[(*i)->identifier()]])
105 << std::endl);
106 }
107 }
108
109 // Insert contributed fields into a separate node list
110 const auto& contributedFields = p->contributedFields();
111 for (auto i=contributedFields.cbegin(); i != contributedFields.cend(); ++i) {
112 auto& set = contributed_field_to_node_index_[(*i)->identifier()];
113 set.insert(static_cast<int>(nodes_.size()-1));
114 }
115
116}

135template<typename Traits>
136void PHX::DagManager<Traits>::
137sortAndOrderEvaluators()
138{
139#ifdef PHX_TEUCHOS_TIME_MONITOR
140 Teuchos::TimeMonitor(*Teuchos::TimeMonitor::getNewTimer("Phalanx::SortAndOrderEvaluators"));
141#endif
142
143 // *************************
144 // Color all nodes white, reset the discovery and final times
145 // *************************
146 for (auto& n : nodes_)
147 n.resetDfsParams(PHX::Color::WHITE);
148
149 topoSortEvalIndex.clear();
150
151 // *************************
152 // Insert contributed fields into the field_to_node_index_ if there
153 // is no evaluator already assigned. Here we support two cases - (1)
154 // there is an evaluator that "evalautes" the field and is already
155 // assigned but other evaluators declared as "contributor" for the
156 // same field are present and (2) there no evaluators that
157 // "evaluate" the field, only "contributors". In the second case, we
158 // need to pick one of the contributors and insert it into
159 // field_to_node_index_ for the dfs algortihm to work. We also need
160 // to remove this from the contributed_field_to_node_index_ so that
161 // it does not have a cyclic dependency with itself.
162 // *************************
163 // for (const auto& contrib_field : contributed_field_to_node_index_) {
164 // const auto& identifier = contrib_field.first;
165 // auto node_index_it = field_to_node_index_.find(identifier);
166 // if (node_index_it == field_to_node_index_.end()) {
167 // field_to_node_index_.insert(std::make_pair(identifier, *contrib_field.second.begin()));
168 // contributed_field_to_node_index_[identifier].erase(field_to_node_index_[identifier]);
169 // }
170 // }
171
172 // *************************
173 // Loop over required fields
174 // *************************
175 int time = 0;
176 for (const auto& req_field : required_fields_) {
177
178 // Look in evaluated fields first
179 auto node_index_it = field_to_node_index_.find(req_field->identifier());
180 if (node_index_it != field_to_node_index_.end()) {
181 auto& node = nodes_[node_index_it->second];
182 if (node.color() == PHX::Color::WHITE)
183 dfsVisit(node,time);
184 }
185
186 // Add in contributed fields for this tag. There could be multiple
187 // contrib or we may not have found any evaluated fields for this
188 // tag if they are all contributed.
189 auto contrib_field_search = contributed_field_to_node_index_.find(req_field->identifier());
190 if (contrib_field_search != contributed_field_to_node_index_.end()) {
191 const auto& node_list_to_add = contrib_field_search->second;
192 for (auto node_index : node_list_to_add) {
193 auto& node = nodes_[node_index];
194 if (node.color() == PHX::Color::WHITE)
195 dfsVisit(node,time);
196 }
197 }
198
199 // If no evaluated or contrib found, error out.
200 if ( (node_index_it == field_to_node_index_.end()) &&
201 (contrib_field_search == contributed_field_to_node_index_.end()) ) {
202
203 if (write_graphviz_file_on_error_)
204 this->writeGraphvizFileNew(graphviz_filename_for_errors_, true, true);
205
206 TEUCHOS_TEST_FOR_EXCEPTION(node_index_it == field_to_node_index_.end(),
207 PHX::missing_evaluator_exception,
208 *this
209 << "\n\nERROR: The required field \""
210 << req_field->identifier()
211 << "\" does not have an evaluator. Current "
212 << "list of Evaluators are printed above this "
213 << "error message.\n");
214 }
215 }
216
217 // Create a list of fields to allocate
218 fields_.clear();
219 for (std::size_t i = 0; i < topoSortEvalIndex.size(); i++) {
220 const auto& fields = (nodes_[topoSortEvalIndex[i]]).get()->evaluatedFields();
221 fields_.insert(fields_.end(),fields.cbegin(),fields.cend());
222 }
223
224 // Contributed fields: If a field is only evaluated with
225 // "contributed" fields, then it is not in the list of
226 // evaluatedFields and therefore not in the fields to allocate
227 // vector above. We need to find such fields and add them to the
228 // list of fields to allocate.
229 for (std::size_t i = 0; i < topoSortEvalIndex.size(); i++) {
230 const auto& contrib_fields = (nodes_[topoSortEvalIndex[i]]).get()->contributedFields();
231 for (auto& cfield : contrib_fields) {
232 const auto& check_it = std::find_if(fields_.begin(), fields_.end(),[&cfield](const Teuchos::RCP<const PHX::FieldTag>& f)
233 {
234 if (f->identifier() == cfield->identifier())
235 return true;
236 return false;
237 }
238 );
239 if (check_it == fields_.end())
240 fields_.push_back(cfield);
241 }
242 }
243
244 this->createEvaluatorBindingFieldMap();
245
246 sorting_called_ = true;
247}

250template<typename Traits>
251void PHX::DagManager<Traits>::
252dfsVisit(PHX::DagNode<Traits>& node, int& time)
253{
254 node.setColor(PHX::Color::GREY);
255 time += 1;
256 node.setDiscoveryTime(time);
257
258 // Add the adjacencies.
259 // NOTE: we could do this for all nodes before entering the DFS
260 // algorithm, but then the safety check forces users to satisfy
261 // dependencies for nodes that may potentially NOT be in the final
262 // graph. So we have to do it here when we know we actually will
263 // use the node.
264 {
265 const auto& req_fields = node.get()->dependentFields();
266 for (const auto& field : req_fields) {
267
268 // Look in evalauted fields
269 auto node_index_it = field_to_node_index_.find(field->identifier());
270 if (node_index_it == field_to_node_index_.end()) {
271
272 // If failed to find, look in contributed fields
273 auto contrib_field_search = contributed_field_to_node_index_.find(field->identifier());
274 if (contrib_field_search == contributed_field_to_node_index_.end()) {
275
276 if (write_graphviz_file_on_error_)
277 this->writeGraphvizFileNew(graphviz_filename_for_errors_, true, true);
278
279 TEUCHOS_TEST_FOR_EXCEPTION(node_index_it == field_to_node_index_.end(),
280 PHX::missing_evaluator_exception,
281 *this
282 << "\n\nERROR: The required field \""
283 << field->identifier()
284 << "\" does not have an evaluator. Current "
285 << "list of Evaluators are printed above this "
286 << "error message.\n\n"
287 << "\nPlease inspect the DagManager output above, or \n"
288 << "visually inspect the error graph that was dumped by \n"
289 << "running the graphviz dot program on the file\n"
290 << graphviz_filename_for_errors_ << ": \n\n"
291 << "dot -Tjpg -o error.jpg "
292 << graphviz_filename_for_errors_ << "\n\n"
293 << "The above command generates a jpg file, \"error.jpg\"\n"
294 << "that you can view in any web browser/graphics program.\n");
295 }
296 }
297
298 if (node_index_it != field_to_node_index_.end())
299 node.addAdjacency(node_index_it->second);
300
301 // For required contributed fields, add the extra evaluators as adjacencies too.
302 {
303 auto contrib_field_search = contributed_field_to_node_index_.find(field->identifier());
304 if (contrib_field_search != contributed_field_to_node_index_.end()) {
305 const auto& node_list_to_add = contrib_field_search->second;
306 for (auto node_to_add : node_list_to_add)
307 node.addAdjacency(node_to_add);
308 }
309 }
310 }
311
312 // For "contributed" fields, if an evaluator exists that also
313 // "evalautes" this field, then we assume that the evaluator that
314 // "evalautes" the field performs the initialization of the field
315 // for the contributions. So the contributed fields must have a
316 // dependency on the evalauted field evaluator.
317 const auto& contrib_fields = node.get()->contributedFields();
318 for (const auto& cfield : contrib_fields) {
319 const auto& evaluated_field_search = field_to_node_index_.find(cfield->identifier());
320 if (evaluated_field_search != field_to_node_index_.end()) {
321 node.addAdjacency(evaluated_field_search->second);
322 }
323 }
324
325 }
326
327 for (auto& adj_node_index : node.adjacencies()) {
328
329 auto& adj_node = nodes_[adj_node_index];
330
331 if (adj_node.color() == PHX::Color::WHITE) {
332 dfsVisit(adj_node,time);
333 }
334 else if (adj_node.color() == PHX::Color::GREY) {
335
336 std::ostringstream os_adj_node;
337 this->printEvaluator(*(adj_node.get()),os_adj_node);
338 std::ostringstream os_node;
339 this->printEvaluator(*(node.get()),os_node);
340
341 if (write_graphviz_file_on_error_)
342 this->writeGraphvizFileNew(graphviz_filename_for_errors_, true, true);
343
344 TEUCHOS_TEST_FOR_EXCEPTION(adj_node.color() == PHX::Color::GREY,
345 PHX::circular_dag_exception,
346 *this
347 << "\n\nERROR: In constructing the directed acyclic graph from \n"
348 << "the node dependencies, a circular dependency has been \n"
349 << "identified. The dependence is injected in going from node:\n\n"
350 << os_adj_node.str()
351 << "\n back to node\n\n"
352 << os_node.str()
353 << "\nPlease inspect the DagManager output above, or \n"
354 << "visually inspect the error graph that was dumped by \n"
355 << "running the graphviz dot program on the file\n"
356 << graphviz_filename_for_errors_ << ": \n\n"
357 << "dot -Tjpg -o error.jpg "
358 << graphviz_filename_for_errors_<< "\n\n"
359 << "The above command generates a jpg file, \"error.jpg\"\n"
360 << "that you can view in any web browser/graphics program.\n");
361 }
362 }
363 node.setColor(PHX::Color::BLACK);
364 time += 1;
365 node.setFinalTime(time);
366 topoSortEvalIndex.push_back(node.index()); // for topo sort
367}

370template<typename Traits>
371void PHX::DagManager<Traits>::
372printEvaluator(const PHX::Evaluator<Traits>& e, std::ostream& os) const
373{
374 os << e.getName() << "\n";
375 if (e.evaluatedFields().size() > 0) {
376 os << " *Evaluated Fields:\n";
377 for (const auto& f : e.evaluatedFields())
378 os << " " << f->identifier() << "\n";
379 }
380 if (e.contributedFields().size() > 0) {
381 os << " *Contributed Fields:\n";
382 for (const auto& f : e.contributedFields())
383 os << " " << f->identifier() << "\n";
384 }
385 if (e.dependentFields().size() > 0) {
386 os << " *Dependent Fields:\n";
387 for (const auto& f : e.dependentFields())
388 os << " " << f->identifier() << "\n";
389 }
390 if (e.unsharedFields().size() > 0) {
391 os << " *Unshared Fields:\n";
392 for (const auto& f : e.unsharedFields())
393 os << " " << f->identifier() << "\n";
394 }

398template<typename Traits>
399void PHX::DagManager<Traits>::
400postRegistrationSetup(typename Traits::SetupData d,
401 PHX::FieldManager<Traits>& vm,
402 const bool& buildDeviceDAG)
403{
404 // Call each evaluators' post registration setup

409 if (build_device_dag_) {
410 device_evaluators_ = Kokkos::View<PHX::DeviceEvaluatorPtr<Traits>*,PHX::Device>("device_evaluators_",topoSortEvalIndex.size());

412 device_evaluators_(n).ptr = nodes_[topoSortEvalIndex[n]].getNonConst()->createDeviceEvaluator();
413 }

417template<typename Traits>
418void PHX::DagManager<Traits>::
419evaluateFields(typename Traits::EvalData d)
420{
421 for (std::size_t n = 0; n < topoSortEvalIndex.size(); ++n) {
422
423#ifdef PHX_TEUCHOS_TIME_MONITOR
424 Teuchos::TimeMonitor Time(*evalTimers[topoSortEvalIndex[n]]);
425#endif
426
427#ifdef PHX_DEBUG
428 if (nonnull(start_stop_debug_ostream_)) {
429 *start_stop_debug_ostream_ << "Phalanx::DagManager: Starting node: "
430 << nodes_[topoSortEvalIndex[n]].get()->getName()
431 << std::endl;
432 }
433
434 auto print_fields = std::getenv("PHX_PRINT_FIELDS");
435 if (print_fields) {
436 std::cout << "************************************************\n"
437 << "Printing fields BEFORE executing evaluator: "
438 << nodes_[topoSortEvalIndex[n]].get()->getName()
439 << "\n Evaluated Fields:\n";
440 auto& eft = nodes_[topoSortEvalIndex[n]].get()->evaluatedFields();
441 for (const auto& t : eft)
442 std::cout << " " << t->identifier() << std::endl;
443 std::cout << " Contributed Fields:\n";
444 auto& cft = nodes_[topoSortEvalIndex[n]].get()->contributedFields();
445 for (const auto& t : cft)
446 std::cout << " " << t->identifier() << std::endl;
447 std::cout << " Dependent Fields:\n";
448 auto& dft = nodes_[topoSortEvalIndex[n]].get()->dependentFields();
449 for (const auto& t : dft)
450 std::cout << " " << t->identifier() << std::endl;
451 std::cout << "************************************************\n";
452 nodes_[topoSortEvalIndex[n]].get()->printFieldValues(std::cout);
453 }
454#endif
455
456 using clock = std::chrono::steady_clock;
457 std::chrono::time_point<clock> start = clock::now();
458
459 typename PHX::Device().fence(); // temporary fence until UVM in evaluateFields fixed
460
461 nodes_[topoSortEvalIndex[n]].getNonConst()->evaluateFields(d);
462
463 nodes_[topoSortEvalIndex[n]].sumIntoExecutionTime(clock::now()-start);
464
465#ifdef PHX_DEBUG
466 if (print_fields) {
467 std::cout << "************************************************\n"
468 << "Printing fields AFTER executing evaluator: "
469 << nodes_[topoSortEvalIndex[n]].get()->getName()
470 << "\n Evaluated Fields:\n";
471 auto& eft = nodes_[topoSortEvalIndex[n]].get()->evaluatedFields();
472 for (const auto& t : eft)
473 std::cout << " " << t->identifier() << std::endl;
474 std::cout << " Contributed Fields:\n";
475 auto& cft = nodes_[topoSortEvalIndex[n]].get()->contributedFields();
476 for (const auto& t : cft)
477 std::cout << " " << t->identifier() << std::endl;
478 std::cout << " Dependent Fields:\n";
479 auto& dft = nodes_[topoSortEvalIndex[n]].get()->dependentFields();
480 for (const auto& t : dft)
481 std::cout << " " << t->identifier() << std::endl;
482 std::cout << "************************************************\n";
483 nodes_[topoSortEvalIndex[n]].get()->printFieldValues(std::cout);
484 }
485
486 if (nonnull(start_stop_debug_ostream_)) {
487 *start_stop_debug_ostream_ << "Phalanx::DagManager: Completed node: "
488 << nodes_[topoSortEvalIndex[n]].getNonConst()->getName()
489 << std::endl;
490 }
491#endif
492
493 }

500 template<typename Traits>
501 struct RunDeviceDag {
502
503 Kokkos::View<PHX::DeviceEvaluatorPtr<Traits>*,PHX::Device> evaluators_;
504
505 // The EvalData may be pass by reference. Remove the reference so
506 // that we copy by value to device.
507 const typename std::remove_reference<typename Traits::EvalData>::type data_;
508
509 RunDeviceDag(const Kokkos::View<PHX::DeviceEvaluatorPtr<Traits>*,PHX::Device>& evaluators,
510 typename Traits::EvalData data) :
511 evaluators_(evaluators),
512 data_(data) {}
513
514 KOKKOS_INLINE_FUNCTION
515 void operator()(const Kokkos::TeamPolicy<PHX::exec_space>::member_type& team) const
516 {
517 const int num_evaluators = static_cast<int>(evaluators_.extent(0));
518 for (int e=0; e < num_evaluators; ++e) {
519 evaluators_(e).ptr->prepareForRecompute(team,data_);
520 evaluators_(e).ptr->evaluate(team,data_);
521 team.team_barrier();
522 }
523 }
524
525 };
526
527}

547 TEUCHOS_TEST_FOR_EXCEPTION(true,std::runtime_error,

549#endif
550}

553#ifdef PHX_ENABLE_KOKKOS_AMT
554template<typename Traits>
555void PHX::DagManager<Traits>::
556evaluateFieldsTaskParallel(const int& work_size,
557 typename Traits::EvalData d)
558{
559 using execution_space = PHX::exec_space;
560 using memory_space = PHX::Device::memory_space;
561 using policy_type = Kokkos::TaskScheduler<execution_space>;
562
563 // Requested the ability to query policy for required sizes of calls
564 // to spawn and wait_all. For now hard code to something
565 // reasonable!?!
566 const unsigned required_memory = 1000000;
567 for (std::size_t n = 0; n < topoSortEvalIndex.size(); ++n) {
568 // const auto& node = nodes_[topoSortEvalIndex[n]];
569 // const auto& adjacencies = node.adjacencies();
570 }
571
572 policy_type policy(memory_space(),required_memory);
573
574 // Issue in reusing vector. The assign doesn't like the change of policy.
575 //node_futures_.resize(nodes_.size());
576 std::vector<Kokkos::Future<void,PHX::exec_space>> node_futures_(nodes_.size());
577
578 for (std::size_t n = 0; n < topoSortEvalIndex.size(); ++n) {
579
580 auto& node = nodes_[topoSortEvalIndex[n]];
581 const auto& adjacencies = node.adjacencies();
582
583 // Since this is registered in the order of the topological sort,
584 // we know all dependent futures of a node are already
585 // constructed.
586 std::vector<Kokkos::Future<void,execution_space>> dependent_futures(adjacencies.size());
587 auto adj_iterator = adjacencies.cbegin();
588 for (std::size_t i=0; i < adjacencies.size(); ++i,++adj_iterator)
589 dependent_futures[i] = node_futures_[*adj_iterator];
590
591 auto future = node.getNonConst()->createTask(policy,work_size,dependent_futures,d);
592 TEUCHOS_TEST_FOR_EXCEPTION(future.is_null(), std::logic_error,
593 "Error in PHX::DagManager<Traits>::evaluateFieldsTaskParallel():\n"
594 << "The policy is out of memory. Increase the memory pool size!\n");
595 node_futures_[topoSortEvalIndex[n]] = future;
596 }
597
598 Kokkos::wait(policy);
599}
600#endif

652//=======================================================================
653template<typename Traits>
654void PHX::DagManager<Traits>::
655writeGraphviz(std::ostream& ofs,
656 bool writeEvaluatedFields,
657 bool writeDependentFields) const
658{
659 ofs << "digraph G {\n";
660
661 // This can be called from inside a DFS during an error, so we can't
662 // change the DFS node objects when starting a new search. Need to
663 // copy the node vector.
664 std::vector<PHX::DagNode<Traits>> nodes_copy;
665 nodes_copy.insert(nodes_copy.end(),nodes_.cbegin(),nodes_.cend());
666
667 for (auto& n : nodes_copy)
668 n.resetDfsParams(PHX::Color::WHITE);
669
670 // Loop over required fields
671 int missing_node_index = nodes_.size();
672 for (const auto& req_field : required_fields_) {
673 // Look in evaluated fields
674 auto node_index_it = field_to_node_index_.find(req_field->identifier());
675 if (node_index_it != field_to_node_index_.end()) {
676 auto& node = nodes_copy[node_index_it->second];
677 if (node.color() == PHX::Color::WHITE)
678 writeGraphvizDfsVisit(node,
679 nodes_copy,
680 ofs,
681 writeEvaluatedFields,
682 writeDependentFields);
683 }
684
685 // Check for contributed fields
686 auto contrib_field_search = contributed_field_to_node_index_.find(req_field->identifier());
687 if (contrib_field_search != contributed_field_to_node_index_.end()) {
688 const auto& node_list_to_add = contrib_field_search->second;
689 for (auto node_index : node_list_to_add) {
690 auto& node = nodes_copy[node_index];
691 if (node.color() == PHX::Color::WHITE)
692 writeGraphvizDfsVisit(node,
693 nodes_copy,
694 ofs,
695 writeEvaluatedFields,
696 writeDependentFields);
697 }
698 }
699
700 if ( (node_index_it == field_to_node_index_.end()) &&
701 (contrib_field_search == contributed_field_to_node_index_.end()) ) {
702 ofs << missing_node_index
703 << " [" << "fontcolor=\"red\"" << ", label=\" ** MISSING EVALUATOR **\\n "
704 << req_field->identifier() << " **** MISSING ****\"]\n";
705 missing_node_index += 1;
706 }
707 }
708
709 ofs << "}";
710}

712//=======================================================================
713template<typename Traits>
714void PHX::DagManager<Traits>::
715writeGraphvizDfsVisit(PHX::DagNode<Traits>& node,
716 std::vector<PHX::DagNode<Traits>>& nodes_copy,
717 std::ostream& ofs,
718 const bool writeEvaluatedFields,
719 const bool writeDependentFields) const
720{
721 node.setColor(PHX::Color::GREY);
722
723 // Add valid adjacencies, write node
724 {
725 std::string font_color = "";
726 std::vector<std::string> dependent_field_labels;
727
728 // Dependent field adjacencies
729 const auto& req_fields = node.get()->dependentFields();
730 for (const auto& field : req_fields) {
731 // Look in evaluated fields
732 auto node_index_it = field_to_node_index_.find(field->identifier());
733 if (node_index_it == field_to_node_index_.end()) {
734 // If failed to find, look in contributed fields
735 auto contrib_field_search = contributed_field_to_node_index_.find(field->identifier());
736 if (contrib_field_search == contributed_field_to_node_index_.end()) {
737 font_color = "red";
738 std::string dependent_field_label = field->identifier() + " **** MISSING ****";
739 dependent_field_labels.emplace(dependent_field_labels.end(),dependent_field_label);
740 }
741 else
742 dependent_field_labels.push_back(field->identifier());
743 }
744 else {
745 dependent_field_labels.push_back(field->identifier());
746 if (node_index_it != field_to_node_index_.end()) {
747 node.addAdjacency(node_index_it->second);
748 }
749 }
750
751 // Add contributed field dependencies
752 const auto& contrib_node_index_set_it = contributed_field_to_node_index_.find(field->identifier());
753 if (contrib_node_index_set_it != contributed_field_to_node_index_.end()) {
754 for (const auto& cnode_index : (*contrib_node_index_set_it).second) {
755 node.addAdjacency(cnode_index);
756 }
757 }
758 }
759
760 // For "contributed" fields, if an evaluator exists that also
761 // "evalautes" this field, then we assume that the evaluator that
762 // "evalautes" the field performs the initialization of the field
763 // for the contributions. So the contributed fields must have a
764 // dependency on the evalauted field evaluator.
765 const auto& contrib_fields = node.get()->contributedFields();
766 for (const auto& cfield : contrib_fields) {
767 const auto& evaluated_field_search = field_to_node_index_.find(cfield->identifier());
768 if (evaluated_field_search != field_to_node_index_.end()) {
769 node.addAdjacency(evaluated_field_search->second);
770 }
771 }
772
773 // Write the node
774 ofs << node.index()
775 << " [fontcolor=\"" << font_color
776 << "\", label=\"" << node.get()->getName();
777 if (writeEvaluatedFields) {
778 ofs << "\\n Evaluates:";
779 const auto& eval_fields = node.get()->evaluatedFields();
780 for (const auto& field : eval_fields)
781 ofs << "\\n " << field->identifier();
782 }
783 if (writeEvaluatedFields) {
784 ofs << "\\n Contributes:";
785 for (const auto& field : contrib_fields)
786 ofs << "\\n " << field->identifier();
787 }
788 if (writeDependentFields) {
789 ofs << "\\n Dependencies:";
790 for(const auto& field : dependent_field_labels)
791 ofs << "\\n " << field;
792 }
793 ofs << "\"]\n";
794 }
795
796 // Write edges and trace adjacencies
797 for (auto& adj_node_index : node.adjacencies()) {
798
799 auto& adj_node = nodes_copy[adj_node_index];
800
801 if (adj_node.color() == PHX::Color::WHITE) {
802 ofs << node.index() << "->" << adj_node.index() << "\n";
803 writeGraphvizDfsVisit(adj_node,
804 nodes_copy,
805 ofs,
806 writeEvaluatedFields,
807 writeDependentFields);
808 }
809 else if (adj_node.color() == PHX::Color::GREY) {
810 ofs << node.index() << "->" << adj_node.index() << " [color=red]\n";
811 }
812 else { // BLACK node
813 ofs << node.index() << "->" << adj_node.index() << "\n";
814 }
815 }
816
817 node.setColor(PHX::Color::BLACK);
818}

890//=======================================================================
891template<typename Traits>
892void PHX::DagManager<Traits>::
893analyzeGraph(double& speedup, double& parallelizability) const
894{
895 using std::vector;
896 using duration = std::chrono::duration<double>;
897
898 duration T_1 = duration(0.0);
899 for (std::size_t n = 0; n < topoSortEvalIndex.size(); ++n) {
900
901 const auto& node = nodes_[topoSortEvalIndex[n]];
902 T_1 += node.executionTime();
903
904 duration t_data_ready(0.0);
905 const auto& adjacencies = node.adjacencies();
906 for (const auto& adj_node_index : adjacencies) {
907 t_data_ready = std::max(t_data_ready, nodes_[adj_node_index].finishTime());
908 }
909 const_cast<PHX::DagNode<Traits>&>(node).setStartTime(t_data_ready);
910 const_cast<PHX::DagNode<Traits>&>(node).setFinishTime(t_data_ready + node.executionTime());
911
912 }
913
914 duration T_inf = duration(0.0);
915 for (std::size_t n = 0; n < topoSortEvalIndex.size(); ++n) {
916 const auto& node = nodes_[topoSortEvalIndex[n]];
917 T_inf = std::max(T_inf,node.finishTime());
918 }
919
920 speedup = T_1.count() / T_inf.count();
921 parallelizability = ( 1.0 - (1.0/speedup) )
922 / ( 1.0 - (1.0 / static_cast<double>(topoSortEvalIndex.size())) );
923
924}

935template<typename Traits>
936const std::unordered_map<std::string,std::pair<int,int>>&
937PHX::DagManager<Traits>::getFieldUseRange()
938{
939 if (field_use_range_evaluated_)
940 return field_use_range_;
941
942 // Initialize the fields just outside valid range for debugging
943 for (const auto& f : fields_)
944 field_use_range_[f->identifier()] = std::make_pair(topoSortEvalIndex.size(),-1);
945
946 for (int idx=0; idx < static_cast<int>(topoSortEvalIndex.size()); ++idx) {
947 const auto& evaluator = nodes_[topoSortEvalIndex[idx]].getNonConst();
948
949 // Evaluated fields
950 for (const auto& f : evaluator->evaluatedFields()) {
951 auto& range = field_use_range_[f->identifier()];
952 range.first = std::min(range.first,idx);
953 range.second = std::max(range.first,idx);
954 }
955 // Contributed fields
956 for (const auto& f : evaluator->contributedFields()) {
957 auto& range = field_use_range_[f->identifier()];
958 range.first = std::min(range.first,idx);
959 range.second = std::max(range.first,idx);
960 }
961 // Dependent fields
962 for (const auto& f : evaluator->dependentFields()) {
963 auto& range = field_use_range_[f->identifier()];
964 range.first = std::min(range.first,idx);
965 range.second = std::max(range.first,idx);
966 }
967 }
968
969 // Required field values must exist to the end of the DAG evalaution
970 // since users will access the data after the DAG is finished
971 // running.
972 for (const auto& f : required_fields_)
973 field_use_range_[f->identifier()].second = topoSortEvalIndex.size()-1;
974
975#ifdef PHX_DEBUG
976 for (const auto& f : field_use_range_) {
977 TEUCHOS_TEST_FOR_EXCEPTION(f.second.first > f.second.first,
978 std::logic_error,
979 "ERROR - PHX::DagManager::getFieldUseRange() - the field "
980 << f.first << " has an invalid use range of [" << f.second.first << ","
981 << f.second.second << "].");
982 }
983#endif
984
985 field_use_range_evaluated_ = true;
986 return field_use_range_;
987}
