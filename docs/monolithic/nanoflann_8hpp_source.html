78 template <typename DistanceType, typename IndexType = size_t, typename CountType = size_t>
79 class KNNResultSet
80 {
81 IndexType * indices;
82 DistanceType* dists;
83 CountType capacity;
84 CountType count;
85
86 public:
87 inline KNNResultSet(CountType capacity_) : indices(0), dists(0), capacity(capacity_), count(0)
88 {
89 }
90
91 inline void init(IndexType* indices_, DistanceType* dists_)
92 {
93 indices = indices_;
94 dists = dists_;
95 count = 0;
96 if (capacity)
97 dists[capacity-1] = (std::numeric_limits<DistanceType>::max)();
98 }
99
100 inline CountType size() const
101 {
102 return count;
103 }
104
105 inline bool full() const
106 {
107 return count == capacity;
108 }
109
110
115 inline bool addPoint(DistanceType dist, IndexType index)
116 {
117 CountType i;
118 for (i = count; i > 0; --i) {
119#ifdef NANOFLANN_FIRST_MATCH // If defined and two points have the same distance, the one with the lowest-index will be returned first.
120 if ( (dists[i-1] > dist) || ((dist == dists[i-1]) && (indices[i-1] > index)) ) {
121#else
122 if (dists[i-1] > dist) {
123#endif
124 if (i < capacity) {
125 dists[i] = dists[i-1];
126 indices[i] = indices[i-1];
127 }
128 }
129 else break;
130 }
131 if (i < capacity) {
132 dists[i] = dist;
133 indices[i] = index;
134 }
135 if (count < capacity) count++;
136
137 // tell caller that the search shall continue
138 return true;
139 }
140
141 inline DistanceType worstDist() const
142 {
143 return dists[capacity-1];
144 }
145 };

275 inline DistanceType evalMetric(const T* a, const size_t b_idx, size_t size, DistanceType worst_dist = -1) const
276 {
277 DistanceType result = DistanceType();
278 const T* last = a + size;
279 const T* lastgroup = last - 3;
280 size_t d = 0;
281
282 /* Process 4 items with each loop for efficiency. */
283 while (a < lastgroup) {
284 const DistanceType diff0 = std::abs(a[0] - data_source.kdtree_get_pt(b_idx,d++));
285 const DistanceType diff1 = std::abs(a[1] - data_source.kdtree_get_pt(b_idx,d++));
286 const DistanceType diff2 = std::abs(a[2] - data_source.kdtree_get_pt(b_idx,d++));
287 const DistanceType diff3 = std::abs(a[3] - data_source.kdtree_get_pt(b_idx,d++));
288 result += diff0 + diff1 + diff2 + diff3;
289 a += 4;
290 if ((worst_dist > 0) && (result > worst_dist)) {
291 return result;
292 }
293 }
294 /* Process last 0-3 components. Not needed for standard vector lengths. */
295 while (a < last) {
296 result += std::abs( *a++ - data_source.kdtree_get_pt(b_idx, d++) );
297 }
298 return result;
299 }

555 const size_t BLOCKSIZE = 8192;

557 class PooledAllocator
558 {
559 /* We maintain memory alignment to word boundaries by requiring that all
560 allocations be in multiples of the machine wordsize. */
561 /* Size of machine word in bytes. Must be power of 2. */
562 /* Minimum number of bytes requested at a time from the system. Must be multiple of WORDSIZE. */
563
564
565 size_t remaining; /* Number of bytes left in current block of storage. */
566 void* base; /* Pointer to base of current block of storage. */
567 void* loc; /* Current location in block to next allocate memory. */
568
569 void internal_init()
570 {
571 remaining = 0;
572 base = NULL;
573 usedMemory = 0;
574 wastedMemory = 0;
575 }
576
577 public:
578 size_t usedMemory;
579 size_t wastedMemory;
580
584 PooledAllocator() {
585 internal_init();
586 }
587
591 ~PooledAllocator() {
592 free_all();
593 }
594
596 void free_all()
597 {
598 while (base != NULL) {
599 void *prev = *(static_cast<void**>( base)); /* Get pointer to prev block. */
600 ::free(base);
601 base = prev;
602 }
603 internal_init();
604 }
605
610 void* malloc(const size_t req_size)
611 {
612 /* Round size up to a multiple of wordsize. The following expression
613 only works for WORDSIZE that is a power of 2, by masking last bits of
614 incremented size to zero.
615 */
616 const size_t size = (req_size + (WORDSIZE - 1)) & ~(WORDSIZE - 1);
617
618 /* Check whether a new block must be allocated. Note that the first word
619 of a block is reserved for a pointer to the previous block.
620 */
621 if (size > remaining) {
622
623 wastedMemory += remaining;
624
625 /* Allocate new storage. */
626 const size_t blocksize = (size + sizeof(void*) + (WORDSIZE - 1) > BLOCKSIZE) ?
627 size + sizeof(void*) + (WORDSIZE - 1) : BLOCKSIZE;
628
629 // use the standard C malloc to allocate memory
630 void* m = ::malloc(blocksize);
631 if (!m) {
632 fprintf(stderr, "Failed to allocate memory.\n");
633 return NULL;
634 }
635
636 /* Fill first word of new block with pointer to previous block. */
637 static_cast<void**>(m)[0] = base;
638 base = m;
639
640 size_t shift = 0;
641 //int size_t = (WORDSIZE - ( (((size_t)m) + sizeof(void*)) & (WORDSIZE-1))) & (WORDSIZE-1);
642
643 remaining = blocksize - sizeof(void*) - shift;
644 loc = (static_cast<char*>(m) + sizeof(void*) + shift);
645 }
646 void* rloc = loc;
647 loc = static_cast<char*>(loc) + size;
648 remaining -= size;
649
650 usedMemory += size;
651
652 return rloc;
653 }

931 for (int i = 0; i < (DIM > 0 ? DIM : obj.dim); ++i) {
932 bbox[i].low = dataset_get(obj, obj.vind[left], i);
933 bbox[i].high = dataset_get(obj, obj.vind[left], i);
934 }
935 for (IndexType k = left + 1; k < right; ++k) {
936 for (int i = 0; i < (DIM > 0 ? DIM : obj.dim); ++i) {
937 if (bbox[i].low > dataset_get(obj, obj.vind[k], i)) bbox[i].low = dataset_get(obj, obj.vind[k], i);
938 if (bbox[i].high < dataset_get(obj, obj.vind[k], i)) bbox[i].high = dataset_get(obj, obj.vind[k], i);
939 }
940 }
941 }
942 else {
943 IndexType idx;
944 int cutfeat;
945 DistanceType cutval;
946 middleSplit_(obj, &obj.vind[0] + left, right - left, idx, cutfeat, cutval, bbox);
947
948 node->node_type.sub.divfeat = cutfeat;
949
950 BoundingBox left_bbox(bbox);
951 left_bbox[cutfeat].high = cutval;
952 node->child1 = divideTree(obj, left, left + idx, left_bbox);
953
954 BoundingBox right_bbox(bbox);
955 right_bbox[cutfeat].low = cutval;
956 node->child2 = divideTree(obj, left + idx, right, right_bbox);
957
958 node->node_type.sub.divlow = left_bbox[cutfeat].high;
959 node->node_type.sub.divhigh = right_bbox[cutfeat].low;
960
961 for (int i = 0; i < (DIM > 0 ? DIM : obj.dim); ++i) {
962 bbox[i].low = std::min(left_bbox[i].low, right_bbox[i].low);
963 bbox[i].high = std::max(left_bbox[i].high, right_bbox[i].high);
964 }
965 }

967 return node;

970 void middleSplit_(Derived &obj, IndexType* ind, IndexType count, IndexType& index, int& cutfeat, DistanceType& cutval, const BoundingBox& bbox)
971 {
972 const DistanceType EPS = static_cast<DistanceType>(0.00001);
973 ElementType max_span = bbox[0].high-bbox[0].low;
974 for (int i = 1; i < (DIM > 0 ? DIM : obj.dim); ++i) {
975 ElementType span = bbox[i].high - bbox[i].low;
976 if (span > max_span) {
977 max_span = span;
978 }
979 }
980 ElementType max_spread = -1;
981 cutfeat = 0;
982 for (int i = 0; i < (DIM > 0 ? DIM : obj.dim); ++i) {
983 ElementType span = bbox[i].high-bbox[i].low;
984 if (span > (1 - EPS) * max_span) {
985 ElementType min_elem, max_elem;
986 computeMinMax(obj, ind, count, i, min_elem, max_elem);
987 ElementType spread = max_elem - min_elem;;
988 if (spread > max_spread) {
989 cutfeat = i;
990 max_spread = spread;
991 }
992 }
993 }
994 // split in the middle
995 DistanceType split_val = (bbox[cutfeat].low + bbox[cutfeat].high) / 2;
996 ElementType min_elem, max_elem;
997 computeMinMax(obj, ind, count, cutfeat, min_elem, max_elem);
998
999 if (split_val < min_elem) cutval = min_elem;
1000 else if (split_val > max_elem) cutval = max_elem;
1001 else cutval = split_val;
1002
1003 IndexType lim1, lim2;
1004 planeSplit(obj, ind, count, cutfeat, cutval, lim1, lim2);
1005
1006 if (lim1 > count / 2) index = lim1;
1007 else if (lim2 < count / 2) index = lim2;
1008 else index = count/2;
1009 }

1020 void planeSplit(Derived &obj, IndexType* ind, const IndexType count, int cutfeat, DistanceType &cutval, IndexType& lim1, IndexType& lim2)
1021 {
1022 /* Move vector indices for left subtree to front of list. */
1023 IndexType left = 0;
1024 IndexType right = count-1;
1025 for (;; ) {
1026 while (left <= right && dataset_get(obj, ind[left], cutfeat) < cutval) ++left;
1027 while (right && left <= right && dataset_get(obj, ind[right], cutfeat) >= cutval) --right;
1028 if (left > right || !right) break; // "!right" was added to support unsigned Index types
1029 std::swap(ind[left], ind[right]);
1030 ++left;
1031 --right;
1032 }
1033 /* If either list is empty, it means that all remaining features
1034 * are identical. Split in the middle to maintain a balanced tree.
1035 */
1036 lim1 = left;
1037 right = count-1;
1038 for (;; ) {
1039 while (left <= right && dataset_get(obj, ind[left], cutfeat) <= cutval) ++left;
1040 while (right && left <= right && dataset_get(obj, ind[right], cutfeat) > cutval) --right;
1041 if (left > right || !right) break; // "!right" was added to support unsigned Index types
1042 std::swap(ind[left], ind[right]);
1043 ++left;
1044 --right;
1045 }
1046 lim2 = left;
1047 }

1355 template <class RESULTSET>
1356 bool searchLevel(RESULTSET& result_set, const ElementType* vec, const NodePtr node, DistanceType mindistsq,
1357 distance_vector_t& dists, const float epsError) const
1358 {
1359 /* If this is a leaf node, then do check and return. */
1360 if ((node->child1 == NULL) && (node->child2 == NULL)) {
1361 //count_leaf += (node->lr.right-node->lr.left); // Removed since was neither used nor returned to the user.
1362 DistanceType worst_dist = result_set.worstDist();
1363 for (IndexType i = node->node_type.lr.left; i<node->node_type.lr.right; ++i) {
1364 const IndexType index = BaseClassRef::vind[i];// reorder... : i;
1365 DistanceType dist = distance.evalMetric(vec, index, (DIM > 0 ? DIM : BaseClassRef::dim));
1366 if (dist < worst_dist) {
1367 if(!result_set.addPoint(dist, BaseClassRef::vind[i])) {
1368 // the resultset doesn't want to receive any more points, we're done searching!
1369 return false;
1370 }
1371 }
1372 }
1373 return true;
1374 }
1375
1376 /* Which child branch should be taken first? */
1377 int idx = node->node_type.sub.divfeat;
1378 ElementType val = vec[idx];
1379 DistanceType diff1 = val - node->node_type.sub.divlow;
1380 DistanceType diff2 = val - node->node_type.sub.divhigh;
1381
1382 NodePtr bestChild;
1383 NodePtr otherChild;
1384 DistanceType cut_dist;
1385 if ((diff1 + diff2) < 0) {
1386 bestChild = node->child1;
1387 otherChild = node->child2;
1388 cut_dist = distance.accum_dist(val, node->node_type.sub.divhigh, idx);
1389 }
1390 else {
1391 bestChild = node->child2;
1392 otherChild = node->child1;
1393 cut_dist = distance.accum_dist( val, node->node_type.sub.divlow, idx);
1394 }
1395
1396 /* Call recursively to search next level down. */
1397 if(!searchLevel(result_set, vec, bestChild, mindistsq, dists, epsError)) {
1398 // the resultset doesn't want to receive any more points, we're done searching!
1399 return false;
1400 }
1401
1402 DistanceType dst = dists[idx];
1403 mindistsq = mindistsq + cut_dist - dst;
1404 dists[idx] = cut_dist;
1405 if (mindistsq*epsError <= result_set.worstDist()) {
1406 if(!searchLevel(result_set, vec, otherChild, mindistsq, dists, epsError)) {
1407 // the resultset doesn't want to receive any more points, we're done searching!
1408 return false;
1409 }
1410 }
1411 dists[idx] = dst;
1412 return true;
1413 }
