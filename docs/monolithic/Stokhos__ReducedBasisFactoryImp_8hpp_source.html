27template <typename ordinal_type, typename value_type>
28Teuchos::RCP<Stokhos::ReducedPCEBasis<ordinal_type, value_type> >
29Stokhos::ReducedBasisFactory<ordinal_type, value_type>::
30createReducedBasis(
31 ordinal_type p,
32 const Teuchos::Array< Stokhos::OrthogPolyApprox<ordinal_type, value_type> >& pce,
33 const Teuchos::RCP<const Stokhos::Quadrature<ordinal_type, value_type> >& quad,
34 const Teuchos::RCP< const Stokhos::Sparse3Tensor<ordinal_type, value_type> >& Cijk) const
35{
36 // Compute reduced basis
37 Teuchos::RCP<Stokhos::ReducedPCEBasis<ordinal_type, value_type> > red_basis;
38
39 if (reduction_method == "Monomial Proj Gram-Schmidt")
40 red_basis = Teuchos::rcp(new Stokhos::MonomialProjGramSchmidtPCEBasis<ordinal_type,value_type>(p, pce, quad, params));
41
42 else if (reduction_method == "Monomial Proj Gram-Schmidt2")
43 red_basis = Teuchos::rcp(new Stokhos::MonomialProjGramSchmidtPCEBasis2<ordinal_type,value_type>(p, pce, quad, params));
44
45 else if (reduction_method == "Monomial Gram-Schmidt")
46 red_basis = Teuchos::rcp(new Stokhos::MonomialGramSchmidtPCEBasis<ordinal_type,value_type>(p, pce, quad, params));
47
48 else if (reduction_method == "Product Lanczos")
49 red_basis = Teuchos::rcp(new Stokhos::ProductLanczosPCEBasis<ordinal_type,value_type>(p, pce, quad, Cijk, params));
50
51 else if (reduction_method == "Product Lanczos Gram-Schmidt")
52 red_basis = Teuchos::rcp(new Stokhos::ProductLanczosGramSchmidtPCEBasis<ordinal_type,value_type>(p, pce, quad, Cijk, params));
53
54 else
55 TEUCHOS_TEST_FOR_EXCEPTION(
56 true, std::logic_error,
57 "Invalid reduced basis method " << reduction_method);
58
59 return red_basis;
60}
