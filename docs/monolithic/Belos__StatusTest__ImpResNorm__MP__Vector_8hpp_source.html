476template <class StorageType, class MV, class OP>
477StatusType StatusTestImpResNorm<Sacado::MP::Vector<StorageType>,MV,OP>::
478checkStatus (Iteration<ScalarType,MV,OP>* iSolver)
479{
480 //std::cout << "check status start" << std::endl;
481 using Teuchos::as;
482 using Teuchos::RCP;
483
484 const MagnitudeType zero = STM::zero ();
485 const MagnitudeType one = STM::one ();
486 const LinearProblem<ScalarType,MV,OP>& lp = iSolver->getProblem ();
487
488 // Compute scaling term (done once for each block that's being solved)
489 if (firstcallCheckStatus_) {
490 StatusType status = firstCallCheckStatusSetup (iSolver);
491 if (status == Failed) {
492 status_ = Failed;
493 return status_;
494 }
495 }
496
497 // mfh 23 Apr 2012: I don't know exactly what this code does. It
498 // has something to do with picking the block of right-hand sides
499 // which we're currently checking.
500 if (curLSNum_ != lp.getLSNumber ()) {
501 //
502 // We have moved on to the next rhs block
503 //
504 curLSNum_ = lp.getLSNumber();
505 curLSIdx_ = lp.getLSIndex();
506 curBlksz_ = (int)curLSIdx_.size();
507 int validLS = 0;
508 for (int i=0; i<curBlksz_; ++i) {
509 if (curLSIdx_[i] > -1 && curLSIdx_[i] < numrhs_)
510 validLS++;
511 }
512 curNumRHS_ = validLS;
513 curSoln_ = Teuchos::null;
514 } else {
515 //
516 // We are in the same rhs block, return if we are converged
517 //
518 if (status_ == Passed) {
519 //std::cout << "check status end 2" << std::endl;
520 return status_;
521 }
522 }
523
524 //
525 // Get the "native" residual norms from the solver for this block of
526 // right-hand sides. If the solver's getNativeResiduals() method
527 // actually returns a multivector, compute the norms of the columns
528 // of the multivector explicitly. Otherwise, we assume that
529 // resvector_ contains the norms.
530 //
531 // Note that "compute the norms explicitly" doesn't necessarily mean
532 // the "explicit" residual norms (in the sense discussed in this
533 // class' documentation). These are just some vectors returned by
534 // the solver. Some Krylov methods, like CG, compute a residual
535 // vector "recursively." This is an "implicit residual" in the
536 // sense of this class' documentation. It equals the explicit
537 // residual in exact arithmetic, but due to rounding error, it is
538 // usually different than the explicit residual.
539 //
540 // FIXME (mfh 23 Apr 2012) This method does _not_ respect the
541 // OrthoManager used by the solver.
542 //
543 std::vector<MagnitudeType> tmp_resvector( curBlksz_ );
544 RCP<const MV> residMV = iSolver->getNativeResiduals (&tmp_resvector);
545 if (! residMV.is_null ()) {
546 // We got a multivector back. Compute the norms explicitly.
547 tmp_resvector.resize (MVT::GetNumberVecs (*residMV));
548 MVT::MvNorm (*residMV, tmp_resvector, resnormtype_);
549 typename std::vector<int>::iterator p = curLSIdx_.begin();
550 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
551 // Check if this index is valid
552 if (*p != -1) {
553 resvector_[*p] = tmp_resvector[i];
554 }
555 }
556 } else {
557 typename std::vector<int>::iterator p = curLSIdx_.begin();
558 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
559 // Check if this index is valid
560 if (*p != -1) {
561 resvector_[*p] = tmp_resvector[i];
562 }
563 }
564 }
565 //
566 // Scale the unscaled residual norms we computed or obtained above.
567 //
568 if (scalevector_.size () > 0) {
569 // There are per-vector scaling factors to apply.
570 typename std::vector<int>::iterator p = curLSIdx_.begin();
571 for (; p<curLSIdx_.end(); ++p) {
572 // Check if this index is valid
573 if (*p != -1) {
574 // Scale the vector accordingly
575 testvector_[ *p ] = resvector_[ *p ] / scalevalue_;
576 mask_assign(scalevector_[ *p ]!= zero, testvector_[ *p ]) /= scalevector_[ *p ];
577 }
578 }
579 }
580 else { // There are no per-vector scaling factors.
581 typename std::vector<int>::iterator p = curLSIdx_.begin();
582 for (; p<curLSIdx_.end(); ++p) {
583 // Check if this index is valid
584 if (*p != -1) {
585 testvector_[ *p ] = resvector_[ *p ] / scalevalue_;
586 }
587 }
588 }
589
590 // Count how many scaled residual norms (in testvector_) pass, using
591 // the current tolerance (currTolerance_) rather than the original
592 // tolerance (tolerance_). If at least quorum_ of them pass, we
593 // have a quorum for the whole test to pass.
594 //
595 // We also check here whether any of the scaled residual norms is
596 // NaN, and throw an exception in that case.
597 int have = 0;
598 ind_.resize( curLSIdx_.size() );
599 std::vector<int> lclInd( curLSIdx_.size() );
600 typename std::vector<int>::iterator p = curLSIdx_.begin();
601 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
602 // Check if this index is valid
603 if (*p != -1) {
604 // Check if any of the residuals are larger than the tolerance.
605 const int ensemble_size = StorageType::static_size;
606 bool all_converged = true;
607 for (int ii=0; ii<ensemble_size; ++ii) {
608 if (testvector_[ *p ].coeff(ii) > currTolerance_.coeff(ii)) {
609 ++ensemble_iterations[ii];
610 all_converged = false;
611 }
612 else if (!(testvector_[ *p ].coeff(ii) <= currTolerance_.coeff(ii))) {
613 // Throw an std::exception if the current residual norm is
614 // NaN. We know that it's NaN because it is not less than,
615 // equal to, or greater than the current tolerance. This is
616 // only possible if either the residual norm or the current
617 // tolerance is NaN; we assume the former. We also mark the
618 // test as failed, in case you want to catch the exception.
619 status_ = Failed;
620 TEUCHOS_TEST_FOR_EXCEPTION(true, StatusTestError, "Belos::"
621 "StatusTestImpResNorm::checkStatus(): One or more of the current "
622 "implicit residual norms is NaN.");
623 }
624 }
625 if (all_converged) {
626 ind_[have] = *p;
627 have++;
628 }
629 }
630 }
631 // "have" is the number of residual norms that passed.
632 ind_.resize(have);
633 lclInd.resize(have);
634
635 // Now check the exact residuals
636 if (have) { // At least one residual norm has converged.
637 //
638 // Compute the explicit residual norm(s) from the current solution update.
639 //
640 RCP<MV> cur_update = iSolver->getCurrentUpdate ();
641 curSoln_ = lp.updateSolution (cur_update);
642 RCP<MV> cur_res = MVT::Clone (*curSoln_, MVT::GetNumberVecs (*curSoln_));
643 lp.computeCurrResVec (&*cur_res, &*curSoln_);
644 tmp_resvector.resize (MVT::GetNumberVecs (*cur_res));
645 std::vector<MagnitudeType> tmp_testvector (have);
646
647 MVT::MvNorm (*cur_res, tmp_resvector, resnormtype_);
648
649 // Scale the explicit residual norm(s), just like the implicit norm(s).
650 if ( scalevector_.size() > 0 ) {
651 for (int i=0; i<have; ++i) {
652 // Scale the vector accordingly
653 tmp_testvector[ i ] = tmp_resvector[ lclInd[i] ] / scalevalue_;
654 mask_assign(scalevector_[ ind_[i] ]!=zero,tmp_testvector[ i ]) /= scalevector_[ ind_[i] ];
655 }
656 }
657 else {
658 for (int i=0; i<have; ++i) {
659 tmp_testvector[ i ] = tmp_resvector[ lclInd[i] ] / scalevalue_;
660 }
661 }
662
663 //
664 // Check whether the explicit residual norms also pass the
665 // convergence test. If not, check whether we want to try
666 // iterating a little more to force both implicit and explicit
667 // residual norms to pass.
668 //
669 int have2 = 0;
670 for (int i = 0; i < have; ++i) {
671 // testvector_ contains the implicit (i.e., recursive, computed
672 // by the algorithm) (possibly scaled) residuals. All of these
673 // pass the convergence test.
674 //
675 // tmp_testvector contains the explicit (i.e., ||B-AX||)
676 // (possibly scaled) residuals. We're checking whether these
677 // pass as well. The explicit residual norms only have to meet
678 // the _original_ tolerance (tolerance_), not the current
679 // tolerance (currTolerance_).
680
681
682 // Absolute difference between the current explicit and
683 // implicit residual norm.
684 const MagnitudeType diff = STM::magnitude (testvector_[ind_[i]] - tmp_testvector[i]);
685
686 // Check if any of the residuals are larger than the tolerance.
687 const int ensemble_size = StorageType::static_size;
688 typedef typename StorageType::value_type value_type;
689 bool all_converged = true;
690 {
691 for (int ii=0; ii<ensemble_size; ++ii) {
692 if (!(tmp_testvector[i].coeff(ii) <= tolerance_.coeff(ii))) {
693 if (diff.coeff(ii) > currTolerance_.coeff(ii)) {
694 lossDetected_ = true;
695 }
696 else {
697 const value_type onePointFive = as<value_type>(3) / as<value_type> (2);
698 const value_type oneTenth = as<value_type>(1) / as<value_type> (10);
699
700 currTolerance_.fastAccessCoeff(ii) = currTolerance_.coeff(ii) - onePointFive * diff.coeff(ii);
701 while (currTolerance_.coeff(ii) < as<value_type>(0)) {
702 currTolerance_.fastAccessCoeff(ii) += oneTenth * diff.coeff(ii);
703 }
704 all_converged = false;
705 }
706 }
707 }
708 }
709 if (all_converged) {
710 ind_[have2] = ind_[i];
711 have2++; // This right-hand side has converged.
712 }
713 }
714 have = have2;
715 ind_.resize(have);
716 }
717
718 // Check whether we've met the quorum of vectors necessary for the
719 // whole test to pass.
720 int need = (quorum_ == -1) ? curNumRHS_: quorum_;
721 status_ = (have >= need) ? Passed : Failed;
722
723
724 //std::cout << "check status end default" << std::endl;
725 // Return the current status
726 return status_;
727}

731{
732 for (int j = 0; j < indent; j ++)

755 }

784template <class StorageType, class MV, class OP>
785StatusType StatusTestImpResNorm<Sacado::MP::Vector<StorageType>,MV,OP>::
786firstCallCheckStatusSetup (Iteration<ScalarType,MV,OP>* iSolver)
787{
788 int i;
789 const MagnitudeType zero = STM::zero ();
790 const MagnitudeType one = STM::one ();
791 const LinearProblem<ScalarType,MV,OP>& lp = iSolver->getProblem();
792 // Compute scaling term (done once for each block that's being solved)
793 if (firstcallCheckStatus_) {
794 //
795 // Get some current solver information.
796 //
797 firstcallCheckStatus_ = false;
798
799 if (scaletype_== NormOfRHS) {
800 Teuchos::RCP<const MV> rhs = lp.getRHS();
801 numrhs_ = MVT::GetNumberVecs( *rhs );
802 scalevector_.resize( numrhs_ );
803 MVT::MvNorm( *rhs, scalevector_, scalenormtype_ );
804 }
805 else if (scaletype_==NormOfInitRes) {
806 Teuchos::RCP<const MV> init_res = lp.getInitResVec();
807 numrhs_ = MVT::GetNumberVecs( *init_res );
808 scalevector_.resize( numrhs_ );
809 MVT::MvNorm( *init_res, scalevector_, scalenormtype_ );
810 }
811 else if (scaletype_==NormOfPrecInitRes) {
812 Teuchos::RCP<const MV> init_res = lp.getInitPrecResVec();
813 numrhs_ = MVT::GetNumberVecs( *init_res );
814 scalevector_.resize( numrhs_ );
815 MVT::MvNorm( *init_res, scalevector_, scalenormtype_ );
816 }
817 else {
818 numrhs_ = MVT::GetNumberVecs( *(lp.getRHS()) );
819 }
820
821 resvector_.resize( numrhs_ );
822 testvector_.resize( numrhs_ );
823
824 curLSNum_ = lp.getLSNumber();
825 curLSIdx_ = lp.getLSIndex();
826 curBlksz_ = (int)curLSIdx_.size();
827 int validLS = 0;
828 for (i=0; i<curBlksz_; ++i) {
829 if (curLSIdx_[i] > -1 && curLSIdx_[i] < numrhs_)
830 validLS++;
831 }
832 curNumRHS_ = validLS;
833 //
834 // Initialize the testvector.
835 for (i=0; i<numrhs_; i++) { testvector_[i] = one; }
836
837 // Return an error if the scaling is zero.
838 if (scalevalue_ == zero) {
839 return Failed;
840 }
841 }
842 return Undefined;
843}
