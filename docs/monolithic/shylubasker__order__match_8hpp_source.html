56 template <class Int, class Entry, class Exe_Space>
57 BASKER_INLINE
58 int Basker<Int,Entry, Exe_Space>::mc64(Int n_, Int nnz_, Int *colptr, Int *row_idx, Entry *val,
59 Int _job, Int *_perm, Entry *_scale_row, Entry *_scale_col)
60 {
61 //Note using primative types to match fortran
62 #ifdef BASKER_SUPERLUDIS_MC64
63 typedef int int_t;
64 #else
65 typedef Int int_t;
66 #endif
67 typedef double entry_t;
68
69 int_t liw, ldw, num;
70 int_t *iw;
71 #if defined(BASKER_MC64) || defined(BASKER_SUPERLUDIS_MC64)
72 int_t icntl[10], info[10];
73 #endif
74 int_t job = _job;
75 entry_t *dw;
76 entry_t *nzval_abs;
77 int_t n = n_;
78 int_t nnz = nnz_;
79
80 nzval_abs = (entry_t*)malloc(nnz*sizeof(entry_t));
81
82 liw = 5*n;
83 if(job == 3)
84 { liw = 10*n + n; }
85 iw = (int_t*) malloc(liw*sizeof(int_t));
86 ldw = 3*n+nnz;
87 dw = (entry_t*) malloc(ldw*sizeof(entry_t));
88
89 //Convert to 1 formatting
90 for(Int i = 0; i <= n; ++i)
91 colptr[i] = colptr[i]+1;
92 for(Int i = 0; i < nnz; ++i)
93 row_idx[i] = row_idx[i]+1;
94
95 //call init
96 #ifdef BASKER_MC64
97 mc64id_(icntl);
98 #elif defined(BASKER_SUPERLUDIS_MC64)
99 mc64id_dist(icntl);
100 #endif
101
102 for(Int i = 0; i < nnz; ++i)
103 { nzval_abs[i] = abs(val[i]); }
104
105 Int *perm;
106 perm = (Int*) malloc(n*sizeof(Int));
107
108 #ifdef BASKER_MC64
109 mc64ad_(&job, &n, &nnz, colptr, row_idx, nzval_abs,
110 &num, perm, &liw, iw, &ldw, dw, icntl, info);
111 #elif defined(BASKER_SUPERLUDIS_MC64)
112 mc64ad_dist(&job, &n, &nnz, colptr, row_idx, nzval_abs,
113 &num, perm, &liw, iw, &ldw, dw, icntl, info);
114 #endif
115
116 //debug
117
118 //convert indexing back
119 for(Int i=0; i <= n; ++i)
120 { colptr[i] = colptr[i]-1; }
121 for(Int i=0; i < nnz; ++i)
122 { row_idx[i] = row_idx[i]-1; }
123 for(Int i=0; i < n; ++i)
124 { _perm[i] = perm[i]-1; }
125 for(Int i =0; i < n; ++i)
126 { _scale_row[i] = exp (dw[i]); }
127 for(Int i =0; i < n; ++i)
128 { _scale_col[i] = exp (dw[i+n]); }
129
130 //add job 5 special
131
132 free(iw);
133 free(dw);
134 free(nzval_abs);
135
136 return 0;
137
138 }//end mc64
