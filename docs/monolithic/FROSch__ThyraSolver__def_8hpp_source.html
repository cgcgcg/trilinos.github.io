41 template<class SC,class LO,class GO,class NO>
42 void ThyraSolver<SC,LO,GO,NO>::apply(const XMultiVector &x,
43 XMultiVector &y,
44 ETransp mode,
45 SC alpha,
46 SC beta) const
47 {
48 FROSCH_TIMER_START_SOLVER(applyTime,"ThyraSolver::apply");
49 FROSCH_ASSERT(this->IsComputed_,"FROSch::ThyraSolver: !this->IsComputed_.");
50
51 ConstMultiVectorBasePtr xThyra = ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(rcpFromRef(x));
52 if (YX_.is_null()) YX_ = XMultiVectorFactory::Build(y.getMap(),y.getNumVectors());
53 if (YT_.is_null()) YT_ = rcp_const_cast<MultiVectorBase<SC> >(ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(YX_));
54
55 EOpTransp tMode = NOTRANS;
56 switch (mode) {
57 case Teuchos::NO_TRANS: tMode = Thyra::NOTRANS; break;
58 case Teuchos::TRANS: tMode = Thyra::TRANS; break;
59 case Teuchos::CONJ_TRANS: tMode = Thyra::CONJTRANS; break;
60 default: FROSCH_ASSERT(false,"FROSch::ThyraPreconditionerTpetra: mode unknown.");
61 }
62
63 SolveStatus<SC> status = solve<SC>(*ThyraSolver_,tMode,*xThyra,YT_.ptr());
64
65 // It seems that we have to convert the Thyra vector back to Xpetra. Is there a cheaper/more elegant way?
66 // Same for ThyraPreconditioner
67 XMultiVectorPtr yXpetra = ThyraUtils<SC,LO,GO,NO>::toXpetra(YT_,y.getMap()->getComm());
68 y = *yXpetra;
69
70 y.update(alpha,*YX_,beta);
71 }

86 FROSCH_TIMER_START_SOLVER(ThyraSolverTime,"ThyraSolver::ThyraSolver");

91
92 LinearSolverBuilder<SC> linearSolverBuilder;

94 linearSolverBuilder.setParameterList(sublist(this->ParameterList_,"ThyraSolver"));
95
96 LinearOpWithSolveFactoryBasePtr thyraFactory = linearSolverBuilder.createLinearSolveStrategy("");
