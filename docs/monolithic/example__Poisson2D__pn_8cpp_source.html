1299template<typename Scalar>
1300void exactSolutionGrad(Scalar gradExact[2], const Scalar& x, const Scalar& y) {
1301
1302 // To enable derivatives of the gradient (i.e., 2nd derivatives of the exact solution) need 2 levels of fad types
1303 Sacado::Fad::SFad<Scalar,2> fad_x = x;
1304 Sacado::Fad::SFad<Scalar,2> fad_y = y;
1305 Sacado::Fad::SFad<Scalar,2> u;
1306

1308 fad_x.diff(0,2);
1309 fad_y.diff(1,2);
1310

1317/************ Source Term (RHS) ****************/
1318template<typename Scalar>
1319const Scalar sourceTerm(Scalar& x, Scalar& y){
1320
1321 Scalar u;
1322 Scalar grad_u[2];
1323 Scalar flux[2] = {0.0, 0.0};
1324 Scalar material[2][2];
1325 Scalar f = 0.;
1326
1327 // Indicate the independent variables
1328 x.diff(0,2);
1329 y.diff(1,2);
1330
1331 // Get exact solution and its gradient
1332 u = exactSolution(x, y);
1333 exactSolutionGrad(grad_u, x, y);
1334
1335 // Get material tensor
1336 materialTensor<Scalar>(material, x, y);
1337
1338 // Compute total flux = (A.grad u)
1339 for(int i = 0; i < 2; i++){
1340
1341 // Add diffusive flux
1342 for(int j = 0; j < 2; j++){
1343 flux[i] += material[i][j]*grad_u[j];
1344 }
1345 }
1346
1347 // Compute source term (right hand side): f = -div(A.grad u)
1348 f = -(flux[0].dx(0) + flux[1].dx(1));
1349
1350 return f;
1351}
