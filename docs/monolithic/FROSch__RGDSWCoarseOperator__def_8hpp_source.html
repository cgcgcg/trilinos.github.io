41 FROSCH_ASSERT(blockId<=this->NumberOfBlocks_,"Block does not exist yet and can therefore not be reset.");
42
43 if (!this->DistributionList_->get("Type","linear").compare("ZoltanDual")) {
44 FROSCH_ASSERT(false,"RGDSWCoarseOperator:: Distribution Type ZoltanDual only works for IPOUHarmonicCoarseOperator");

55 communicationStrategy = CommCrsMatrix;
56 } else if (!coarseSpaceList->get("Interface Communication Strategy","CreateOneToOneMap").compare("CrsGraph")) {
57 communicationStrategy = CommCrsGraph;
58 } else if (!coarseSpaceList->get("Interface Communication Strategy","CreateOneToOneMap").compare("CreateOneToOneMap")) {
59 communicationStrategy = CreateOneToOneMap;
60 } else {
61 FROSCH_ASSERT(false,"FROSch::RGDSWCoarseOperator: Specify a valid communication strategy for the identification of the interface components.");
62 }
63
64 Verbosity verbosity = All;
65 if (!coarseSpaceList->get("Verbosity","All").compare("None")) {
66 verbosity = None;
67 } else if (!coarseSpaceList->get("Verbosity","All").compare("All")) {
68 verbosity = All;
69 } else {
70 FROSCH_ASSERT(false,"FROSch::RGDSWCoarseOperator: Specify a valid verbosity level.");
71 }
72
73 bool useForCoarseSpace = coarseSpaceList->get("Use For Coarse Space",false);
74 string option = coarseSpaceList->get("Option","1");
75 DistanceFunction distanceFunction = ConstantDistanceFunction;
76 if (!option.compare("1")) {
77
78 } else if (!option.compare("2.2")) {
79 distanceFunction = InverseEuclideanDistanceFunction;
80 } else {
81 FROSCH_ASSERT(false,"Option is unknown!");
82 }
83
84 bool useRotations = coarseSpaceList->get("Rotations",true);
85 if (useRotations && nodeList.is_null()) {
86 //FROSCH_ASSERT(option==1,"Only option 1 can be constructed without a valid node list.");
87 useRotations = false;
88 FROSCH_WARNING("FROSch::RGDSWCoarseOperator",this->Verbose_,"Rotations cannot be used since nodeList.is_null().");

90
91 if (useForCoarseSpace) {
92 this->NumberOfBlocks_++;
93
94 this->GammaDofs_.resize(this->GammaDofs_.size()+1);
95 this->IDofs_.resize(this->IDofs_.size()+1);
96 this->InterfaceCoarseSpaces_.resize(this->InterfaceCoarseSpaces_.size()+1);
97 this->DofsMaps_.resize(this->DofsMaps_.size()+1);
98 this->DofsPerNode_.resize(this->DofsPerNode_.size()+1);

101 this->DofsPerNode_[blockId] = dofsPerNode;
102
103 Array<GO> tmpDirichletBoundaryDofs(dirichletBoundaryDofs()); // Here, we do a copy. Maybe, this is not necessary
104 sortunique(tmpDirichletBoundaryDofs);
105
106 this->DDInterface_.reset(new DDInterface<SC,LO,GO,NO>(dimension,this->DofsPerNode_[blockId],nodesMap.getConst(),verbosity,this->LevelID_,communicationStrategy));
107 this->DDInterface_->resetGlobalDofs(dofsMaps);
108 this->DDInterface_->removeDirichletNodes(tmpDirichletBoundaryDofs);
109
110 EntitySetPtr interface = this->DDInterface_->getInterface();
111 EntitySetPtr interior = this->DDInterface_->getInterior();
112
113 if (this->Verbose_) {
114 cout

144 // Check for interface

162 if (this->ParameterList_->get("Test Unconnected Interface",true)) {
163 this->DDInterface_->divideUnconnectedEntities(this->K_);

230 template <class SC,class LO,class GO,class NO>
231 typename RGDSWCoarseOperator<SC,LO,GO,NO>::XMultiVectorPtrVecPtr RGDSWCoarseOperator<SC,LO,GO,NO>::computeTranslations(UN blockId,
232 EntitySetPtr Roots,
233 EntitySetPtrVecPtr entitySetVector,
234 DistanceFunction distanceFunction)
235 {
236 FROSCH_DETAILTIMER_START_LEVELID(computeTranslationsTime,"RGDSWCoarseOperator::computeTranslations");
237 XMultiVectorPtrVecPtr translations(this->DofsPerNode_[blockId]);
238 XMapPtr serialGammaMap = MapFactory<LO,GO,NO>::Build(this->K_->getRangeMap()->lib(),this->GammaDofs_[blockId].size(),0,this->SerialComm_);
239 for (UN i=0; i<this->DofsPerNode_[blockId]; i++) {
240 if (Roots->getNumEntities()>0) {
241 translations[i] = MultiVectorFactory<SC,LO,GO,NO>::Build(serialGammaMap,Roots->getNumEntities());
242 } else {
243 translations[i] = null;
244 }
245 }
246
247 // Loop over Dofs
248 for (UN k=0; k<this->DofsPerNode_[blockId]; k++) {
249 // Loop over entitySetVector
250 for (UN i=0; i<entitySetVector.size(); i++) {
251 // Loop over entities
252 for (UN j=0; j<entitySetVector[i]->getNumEntities(); j++) {
253 InterfaceEntityPtr tmpEntity = entitySetVector[i]->getEntity(j);
254 LO rootID = tmpEntity->getRootID();
255 UN numRoots = tmpEntity->getRoots()->getNumEntities();
256 if (rootID==-1) {
257 //if (numRoots==0) cout << rootID << " " << numRoots << " " << tmpEntity->getAncestors()->getNumEntities() << endl;
258 FROSCH_ASSERT(numRoots!=0,"rootID==-1 but numRoots==0!");
259 for (UN m=0; m<numRoots; m++) {
260 InterfaceEntityPtr tmpRoot = tmpEntity->getRoots()->getEntity(m);
261 LO index = tmpRoot->getRootID();
262 // Offspring: loop over nodes
263 for (UN l=0; l<tmpEntity->getNumNodes(); l++) {
264 SC value = tmpEntity->getDistanceToRoot(l,m)/tmpEntity->getDistanceToRoot(l,numRoots);
265 translations[k]->replaceLocalValue(tmpEntity->getGammaDofID(l,k),index,value);
266 }
267 }
268 } else {
269 // Coarse node: loop over nodes
270 for (UN l=0; l<entitySetVector[i]->getEntity(j)->getNumNodes(); l++) {
271 translations[k]->replaceLocalValue(tmpEntity->getGammaDofID(l,k),rootID,ScalarTraits<SC>::one());
272 }
273 }
274 }
275 }
276 }
277 return translations;
278 }

280 template <class SC,class LO,class GO,class NO>
281 typename RGDSWCoarseOperator<SC,LO,GO,NO>::XMultiVectorPtrVecPtr RGDSWCoarseOperator<SC,LO,GO,NO>::computeRotations(UN blockId,
282 UN dimension,
283 ConstXMultiVectorPtr nodeList,
284 EntitySetPtr Roots,
285 EntitySetPtrVecPtr entitySetVector,
286 DistanceFunction distanceFunction)
287 {
288 FROSCH_DETAILTIMER_START_LEVELID(computeRotationsTime,"RGDSWCoarseOperator::computeRotations");
289 FROSCH_ASSERT(nodeList->getNumVectors()==dimension,"dimension of the nodeList is wrong.");
290 FROSCH_ASSERT(dimension==this->DofsPerNode_[blockId],"dimension!=this->DofsPerNode_[blockId]");
291
292 UN rotationsPerEntity = 0;
293 switch (dimension) {
294 case 1:
295 return null;
296 case 2:
297 rotationsPerEntity = 1;
298 break;
299 case 3:
300 rotationsPerEntity = 3;
301 break;
302 default:
303 FROSCH_ASSERT(false,"The dimension is neither 2 nor 3!");
304 }
305
306 XMultiVectorPtrVecPtr rotations(rotationsPerEntity);
307 XMapPtr serialGammaMap = MapFactory<LO,GO,NO>::Build(this->K_->getRangeMap()->lib(),this->GammaDofs_[blockId].size(),0,this->SerialComm_);
308 for (UN i=0; i<rotationsPerEntity; i++) {
309 if (Roots->getNumEntities()>0) {
310 rotations[i] = MultiVectorFactory<SC,LO,GO,NO>::Build(serialGammaMap,Roots->getNumEntities());
311 } else {
312 rotations[i] = null;
313 }
314 }
315
316 SC x,y,z,rx,ry,rz;
317 // Loop over entitySetVector
318 for (UN i=0; i<entitySetVector.size(); i++) {
319 // Loop over entities
320 for (UN j=0; j<entitySetVector[i]->getNumEntities(); j++) {
321 InterfaceEntityPtr tmpEntity = entitySetVector[i]->getEntity(j);
322 LO rootID = tmpEntity->getRootID();
323 UN numRoots = tmpEntity->getRoots()->getNumEntities();
324 if (rootID==-1) {
325 FROSCH_ASSERT(numRoots!=0,"rootID==-1 but numRoots==0!");
326 for (UN m=0; m<numRoots; m++) {
327 InterfaceEntityPtr tmpRoot = tmpEntity->getRoots()->getEntity(m);
328 LO index = tmpRoot->getRootID();
329 // Offspring: loop over nodes
330 for (UN l=0; l<tmpEntity->getNumNodes(); l++) {
331 SC value = tmpEntity->getDistanceToRoot(l,m)/tmpEntity->getDistanceToRoot(l,numRoots);
332
333 // Rotations
334 x = nodeList->getData(0)[tmpEntity->getLocalNodeID(l)];
335 y = nodeList->getData(1)[tmpEntity->getLocalNodeID(l)];
336
337 // Rotation 1
338 rx = y;
339 ry = -x;
340 rz = 0;
341 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),index,value*rx);
342 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),index,value*ry);
343 if (dimension == 3) {
344 z = nodeList->getData(2)[tmpEntity->getLocalNodeID(l)];
345
346 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),index,value*rz);
347
348 // Rotation 2
349 rx = -z;
350 ry = 0;
351 rz = x;
352 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),index,value*rx);
353 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),index,value*ry);
354 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),index,value*rz);
355
356 // Rotation 3
357 rx = 0;
358 ry = z;
359 rz = -y;
360 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),index,value*rx);
361 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),index,value*ry);
362 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),index,value*rz);
363 }
364 }
365 }
366 } else {
367 // Coarse node: loop over nodes
368 for (UN l=0; l<entitySetVector[i]->getEntity(j)->getNumNodes(); l++) {
369 // Rotations
370 x = nodeList->getData(0)[tmpEntity->getLocalNodeID(l)];
371 y = nodeList->getData(1)[tmpEntity->getLocalNodeID(l)];
372
373 // Rotation 1
374 rx = y;
375 ry = -x;
376 rz = 0;
377 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),rootID,rx);
378 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),rootID,ry);
379 if (dimension == 3) {
380 z = nodeList->getData(2)[tmpEntity->getLocalNodeID(l)];
381
382 rotations[0]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),rootID,rz);
383
384 // Rotation 2
385 rx = -z;
386 ry = 0;
387 rz = x;
388 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),rootID,rx);
389 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),rootID,ry);
390 rotations[1]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),rootID,rz);
391
392 // Rotation 3
393 rx = 0;
394 ry = z;
395 rz = -y;
396 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,0),rootID,rx);
397 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,1),rootID,ry);
398 rotations[2]->replaceLocalValue(tmpEntity->getGammaDofID(l,2),rootID,rz);
399 }
400 }
401 }
402 }
403 }
404 return rotations;
