37template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
38PermutingSmoother<Scalar, LocalOrdinal, GlobalOrdinal, Node>::PermutingSmoother(const std::string& mapName, const RCP<const FactoryBase>& mapFact, const std::string& type, const Teuchos::ParameterList& paramList, const LO& overlap, RCP<FactoryBase> permFact)
39 : type_(type)
40 , overlap_(overlap)
41 , permQT_(Teuchos::null)
42 , permP_(Teuchos::null)
43 , diagScalingOp_(Teuchos::null) {
44 this->SetParameterList(paramList);
45
46 permFact_ = permFact;
47 if (permFact_ == Teuchos::null) {
48 RCP<PermutationFactory> newPermFact = Teuchos::rcp(new PermutationFactory());
49 newPermFact->SetParameter("PermutationRowMapName", Teuchos::ParameterEntry(mapName));
50 newPermFact->SetFactory("PermutationRowMapFactory", mapFact);
51 permFact_ = newPermFact;
52 }
53
54 // create internal smoother
55 if (type_ == "ILU") {
56 TEUCHOS_TEST_FOR_EXCEPTION(true, Exceptions::RuntimeError, "MueLu::PermutingSmoother requires Epetra and Ifpack.");
57 } else {
58 s_ = Teuchos::rcp(new TrilinosSmoother(type_, this->GetParameterList(), overlap_));
59 }
60 TEUCHOS_TEST_FOR_EXCEPTION(s_ == Teuchos::null, Exceptions::RuntimeError, "");
61
62 // Use permuted matrix A
63 s_->SetFactory("A", permFact_);
64}

95template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
96void PermutingSmoother<Scalar, LocalOrdinal, GlobalOrdinal, Node>::Apply(MultiVector& X, const MultiVector& B, bool InitialGuessIsZero) const {
97 TEUCHOS_TEST_FOR_EXCEPTION(SmootherPrototype::IsSetup() == false, Exceptions::RuntimeError, "MueLu::PermutingSmoother::Apply(): Setup() has not been called");
98
99 typedef Teuchos::ScalarTraits<Scalar> STS;
100
101 Teuchos::RCP<MultiVector> Xtemp = MultiVectorFactory::Build(X.getMap(), 1, true);
102 Xtemp->update(STS::one(), X, STS::zero());
103
104 // TODO: unify scaling and left permutation operator
105 Teuchos::RCP<MultiVector> Btemp = MultiVectorFactory::Build(B.getMap(), 1, true);
106 Teuchos::RCP<MultiVector> Btemp2 = MultiVectorFactory::Build(B.getMap(), 1, true);
107 permP_->apply(B, *Btemp, Teuchos::NO_TRANS); // apply permutation operator to rhs
108 diagScalingOp_->apply(*Btemp, *Btemp2, Teuchos::NO_TRANS); // apply scaling operator to rhs
109
110 // apply smoother to permuted linear system
111 s_->Apply(*Xtemp, *Btemp2, InitialGuessIsZero);
112
113 // retransform smooth solution
114 permQT_->apply(*Xtemp, X, Teuchos::NO_TRANS);
115}
