43template <typename ValueType>
44void block_chebyshev_polynomial_coefficients(
45 const ValueType rho,
46 cusp::array1d<ValueType,cusp::host_memory>& coefficients,
47 const ValueType lower_bound = 1.0/30.0,
48 const ValueType upper_bound = 1.1)
49{
50 const size_t degree = 3;
51
52 ValueType x0 = lower_bound * rho;
53 ValueType x1 = upper_bound * rho;
54
55 // Chebyshev roots for the interval [-1,1]
56 cusp::array1d<ValueType,cusp::host_memory> std_roots(degree);
57
58 for( size_t i=0; i<degree; i++ )
59 std_roots[i] = std::cos( M_PI * (ValueType(i) + 0.5)/ degree );
60
61 // Chebyshev roots for the interval [x0,x1]
62 for( size_t i=0; i<degree; i++ )
63 std_roots[i] = 0.5 * (x1-x0) * (1 + std_roots[i]) + x0;
64
65 // Compute monic polynomial coefficients of polynomial with scaled roots
66 // TODO: Implement convolution method for polynomial multiplication
67 coefficients.resize(degree+1);
68 ValueType a = std_roots[0];
69 ValueType b = std_roots[1];
70 ValueType c = std_roots[2];
71 coefficients[0] = 1.0;
72 coefficients[1] = -(a+b+c);
73 coefficients[2] = (a*b) + (b*c) + (c*a);
74 coefficients[3] = -(a*b*c);
75
76 // Scale coefficients to enforce C(0) = 1.0
77 ValueType scale_factor = 1.0/coefficients.back();
78 cusp::blas::scal(coefficients, scale_factor);
79}

134template <typename ValueType, typename MemorySpace, typename Orientation>
135template<typename MatrixType, typename VectorType1, typename VectorType2>
136void block_polynomial<ValueType,MemorySpace,Orientation>
137::presmooth(const MatrixType& A, const VectorType1& b, VectorType2& x)
138{
139 CUSP_PROFILE_SCOPED();
140 int N = A.num_rows;
141 int numRHS = x.num_cols;
142 residual.resize(N, numRHS);
143 y.resize(N, numRHS);
144 h.resize(N, numRHS);
145
146 // Ignore the initial x and use b as the residual
147 ValueType scale_factor = default_coefficients[0];
148 //x <- scale_factor*b
149 cusp::axpby_array(scale_factor, b, ValueType(0), x, x);
150 for( size_t i = 1; i<default_coefficients.size(); i++ )
151 {
152 scale_factor = default_coefficients[i];
153 cusp::MVmultiply(A, x, y);
154 //x <- y + scale_factor*b
155 cusp::axpby_array(scale_factor, b, ValueType(1.0), y, x);
156 }
157}

159template <typename ValueType, typename MemorySpace, typename Orientation>
160template<typename MatrixType, typename VectorType1, typename VectorType2>
161void block_polynomial<ValueType,MemorySpace,Orientation>
162::postsmooth(const MatrixType& A, const VectorType1& b, VectorType2& x)
163{
164 CUSP_PROFILE_SCOPED();
165
166 // compute residual <- b - A*x
167 cusp::MVmultiply(A, x, residual);
168 cusp::axpby_array(ValueType(1), b, ValueType(-1), residual, residual);
169 ValueType scale_factor = default_coefficients[0];
170 cusp::axpby_array(scale_factor, residual, ValueType(0), h, h);
171 for( size_t i = 1; i<default_coefficients.size(); i++ )
172 {
173 scale_factor = default_coefficients[i];
174
175 cusp::MVmultiply(A, h, y);
176 cusp::axpby_array(ValueType(1.0), y, scale_factor, residual, h);
177 }
178
179 cusp::axpby_array(ValueType(1.0), h, ValueType(1.0), x, x);
180}

183template <typename ValueType, typename MemorySpace, typename Orientation>
184template<typename MatrixType, typename VectorType1, typename VectorType2, typename VectorType3>
185void block_polynomial<ValueType,MemorySpace,Orientation>
186::operator()(const MatrixType& A, const VectorType1& b, VectorType2& x, VectorType3& coefficients)
187{

196 cusp::MVmultiply(A, x, residual);

198 ValueType(-1));
199 }
200
201 ValueType scale_factor = coefficients[0];
202 cusp::blas::axpby(residual.values, h.values, h.values, scale_factor,
203 ValueType(0));
204
205 for( size_t i = 1; i<coefficients.size(); i++ )
206 {
207 scale_factor = coefficients[i];
208
209 cusp::MVmultiply(A, h, y);
210 cusp::blas::axpby(y.values, residual.values, h.values, ValueType(1.0),
211 scale_factor);
212 }
213
214 cusp::blas::axpy(h.values, x.values, ValueType(1.0));
215}
