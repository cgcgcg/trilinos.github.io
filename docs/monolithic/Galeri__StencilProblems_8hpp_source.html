53template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
54Teuchos::RCP<Matrix> Laplace1DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
55 GlobalOrdinal nx = this->list_.get("nx", (GlobalOrdinal)-1);
56
57 if (nx == -1)
58 nx = this->Map_->getGlobalNumElements();
59
60 // The Kokkos code path does not work for Epetra.
61 // Once Epetra has been removed this logic should be simplified.
62#if defined(HAVE_GALERI_KOKKOS) && defined(HAVE_GALERI_KOKKOSKERNELS)
63 bool keepBCs = false;
64 using Node = typename Map::node_type;
65 using tpetra_map = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
66 if constexpr (std::is_same_v<Map, tpetra_map>) {
67 this->A_ = TriDiagKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, 2.0, -1.0, -1.0, this->DirichletBC_, keepBCs, "Laplace 1D");
68 } else if (this->Map_->lib() == ::Xpetra::UseTpetra) {
69 this->A_ = TriDiagKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, 2.0, -1.0, -1.0, this->DirichletBC_, keepBCs, "Laplace 1D");
70 } else
71#endif
72 {
73 this->A_ = TriDiag<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, 2.0, -1.0, -1.0);
74 }
75 this->A_->setObjectLabel(this->getObjectLabel());
76 return this->A_;
77}

112template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
113Teuchos::RCP<Matrix> Laplace2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
114 Teuchos::ParameterList list = this->list_;
115 GlobalOrdinal nx = -1;
116 GlobalOrdinal ny = -1;
117
118 if (list.isParameter("nx")) {
119 if (list.isType<int>("nx"))
120 nx = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
121 else
122 nx = list.get<GlobalOrdinal>("nx");
123 }
124 if (list.isParameter("ny")) {
125 if (list.isType<int>("ny"))
126 ny = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
127 else
128 ny = list.get<GlobalOrdinal>("ny");
129 }
130
131 double one = 1.0;
132 Scalar stretchx = (Scalar)this->list_.get("stretchx", one);
133 Scalar stretchy = (Scalar)this->list_.get("stretchy", one);
134
135 if (nx == -1 || ny == -1) {
136 GlobalOrdinal n = this->Map_->getGlobalNumElements();
137 nx = (GlobalOrdinal)sqrt((double)n);
138 ny = nx;
139 TEUCHOS_TEST_FOR_EXCEPTION(nx * ny != n, std::logic_error, "You need to specify nx and ny.");
140 }
141 bool keepBCs = this->list_.get("keepBCs", false);
142
143 Scalar east = (Scalar)-one / (stretchx * stretchx);
144 Scalar west = (Scalar)-one / (stretchx * stretchx);
145 Scalar north = (Scalar)-one / (stretchy * stretchy);
146 Scalar south = (Scalar)-one / (stretchy * stretchy);
147 Scalar center = -(east + west + north + south);
148
149 // The Kokkos code path does not work for Epetra.
150 // Once Epetra has been removed this logic should be simplified.
151#if defined(HAVE_GALERI_KOKKOS) && defined(HAVE_GALERI_KOKKOSKERNELS)
152 using Node = typename Map::node_type;
153 using tpetra_map = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
154 if constexpr (std::is_same_v<Map, tpetra_map>) {
155 this->A_ = Cross2DKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, center, west, east, south, north, this->DirichletBC_, keepBCs, "Laplace 2D");
156 } else if (this->Map_->lib() == ::Xpetra::UseTpetra) {
157 this->A_ = Cross2DKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, center, west, east, south, north, this->DirichletBC_, keepBCs, "Laplace 2D");
158 } else
159#endif
160 {
161 this->A_ = Cross2D<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, center, west, east, south, north, this->DirichletBC_, keepBCs);
162 }
163 this->A_->setObjectLabel(this->getObjectLabel());
164 return this->A_;
165}

176class AnisotropicDiffusion2DProblem : public ScalarProblem<Map, Matrix, MultiVector> {
177 public:
178 AnisotropicDiffusion2DProblem(Teuchos::ParameterList& list, const Teuchos::RCP<const Map>& map)

183template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
184Teuchos::RCP<Matrix> AnisotropicDiffusion2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
185 /*
186 2D triangle mesh (meshType == tri)
187
188 *-*-*
189 |/|/|
190 *-*-*
191 |/|/|
192 *-*-*
193
194 or 2D triangle mesh (meshType == quad)
195
196 *-*-*
197 | | |
198 *-*-*
199 | | |
200 *-*-*
201
202 using piecewise linear continuous elements
203
204 PDE
205
206 -\nabla K \nabla u + 1/dt * u = f
207
208 for some constant symmetric 2x2 matrix
209
210 K = [[Kxx, Kxy],
211 [Kxy, Kyy]]
212 */
213
214 Teuchos::ParameterList list = this->list_;
215 GlobalOrdinal nx = -1;
216 GlobalOrdinal ny = -1;
217 using MT = typename Teuchos::ScalarTraits<Scalar>::magnitudeType;
218 if (list.isParameter("nx")) {
219 if (list.isType<int>("nx"))
220 nx = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
221 else
222 nx = list.get<GlobalOrdinal>("nx");
223 }
224 if (list.isParameter("ny")) {
225 if (list.isType<int>("ny"))
226 ny = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
227 else
228 ny = list.get<GlobalOrdinal>("ny");
229 }
230
231 Scalar zero = 0.0;
232 Scalar one = 1.0;
233 Scalar two = one + one;
234
235 MT one_MT = 1.0;
236 MT two_MT = one_MT + one_MT;
237
238 if (nx == -1 || ny == -1) {
239 GlobalOrdinal n = this->Map_->getGlobalNumElements();
240 nx = (GlobalOrdinal)sqrt((double)n);
241 ny = nx;
242 TEUCHOS_TEST_FOR_EXCEPTION(nx * ny != n, std::logic_error, "You need to specify nx and ny.");
243 }
244 bool keepBCs = this->list_.get("keepBCs", false);
245
246 MT dtInv = one_MT / this->list_.get("dt", one_MT);
247 Scalar Kxx = (Scalar)this->list_.get("Kxx", one);
248 Scalar Kxy = (Scalar)this->list_.get("Kxy", zero);
249 Scalar Kyy = (Scalar)this->list_.get("Kyy", one);
250
251 std::string meshType = this->list_.get("meshType", "tri");
252 Scalar a, b, c, d, e, z1, z2, z3, z4;
253
254 // stencil
255 // z3 e z4
256 // b a c
257 // z1 d z2
258
259 if (meshType == "tri") {
260 MT massDiag_MT = one_MT / two_MT / (Teuchos::as<MT>((nx + 1) * (ny + 1)));
261 Scalar massDiag = massDiag_MT;
262 Scalar massOffDiag = massDiag_MT / 6.0;
263
264 c = -Kxx + Kxy + dtInv * massOffDiag;
265 b = -Kxx + Kxy + dtInv * massOffDiag;
266 e = -Kyy + Kxy + dtInv * massOffDiag;
267 d = -Kyy + Kxy + dtInv * massOffDiag;
268 a = Kxx * two + Kyy * two - Kxy * two + dtInv * massDiag;
269 z1 = -Kxy + dtInv * massOffDiag;
270 z2 = zero;
271 z3 = zero;
272 z4 = -Kxy + dtInv * massOffDiag;
273 } else if (meshType == "quad") {
274 Scalar mass = one / ((Scalar)((nx + 1) * (ny + 1))) / (Scalar)36.0;
275 Scalar four_thirds = (Scalar)(4.0 / 3.0);
276 Scalar third = (Scalar)(1.0 / 3.0);
277 Scalar two_thirds = (Scalar)(2.0 / 3.0);
278 Scalar half = (Scalar)(1.0 / 2.0);
279 Scalar sixth = (Scalar)(1.0 / 6.0);
280
281 a = four_thirds * Kxx + four_thirds * Kyy + dtInv * mass * (Scalar)16.0;
282
283 c = -two_thirds * Kxx + third * Kyy + dtInv * mass * (Scalar)4.0;
284 b = -two_thirds * Kxx + third * Kyy + dtInv * mass * (Scalar)4.0;
285 e = third * Kxx - two_thirds * Kyy + dtInv * mass * (Scalar)4.0;
286 d = third * Kxx - two_thirds * Kyy + dtInv * mass * (Scalar)4.0;
287
288 z1 = -sixth * Kxx - sixth * Kyy - half * Kxy + dtInv * mass;
289 z2 = -sixth * Kxx - sixth * Kyy + half * Kxy + dtInv * mass;
290 z3 = -sixth * Kxx - sixth * Kyy + half * Kxy + dtInv * mass;
291 z4 = -sixth * Kxx - sixth * Kyy - half * Kxy + dtInv * mass;
292 } else
293 TEUCHOS_TEST_FOR_EXCEPTION(true, std::logic_error, "You need to specify meshType=\"tri\" or \"quad\".");
294
295 this->A_ = Star2D<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, a, b, c, d, e, z1, z2, z3, z4, this->DirichletBC_, keepBCs);
296 this->A_->setObjectLabel(this->getObjectLabel());
297 return this->A_;
298}

312template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
313Teuchos::RCP<Matrix> Laplace3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
314 Teuchos::ParameterList list = this->list_;
315 GlobalOrdinal nx = -1;
316 GlobalOrdinal ny = -1;
317 GlobalOrdinal nz = -1;
318
319 if (list.isParameter("nx")) {
320 if (list.isType<int>("nx"))
321 nx = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
322 else
323 nx = list.get<GlobalOrdinal>("nx");
324 }
325 if (list.isParameter("ny")) {
326 if (list.isType<int>("ny"))
327 ny = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
328 else
329 ny = list.get<GlobalOrdinal>("ny");
330 }
331 if (list.isParameter("nz")) {
332 if (list.isType<int>("nz"))
333 nz = Teuchos::as<GlobalOrdinal>(list.get<int>("nz"));
334 else
335 nz = list.get<GlobalOrdinal>("nz");
336 }
337 double one = 1.0;
338 Scalar stretchx = (Scalar)this->list_.get("stretchx", one);
339 Scalar stretchy = (Scalar)this->list_.get("stretchy", one);
340 Scalar stretchz = (Scalar)this->list_.get("stretchz", one);
341
342 if (nx == -1 || ny == -1 || nz == -1) {
343 GlobalOrdinal n = this->Map_->getGlobalNumElements();
344 nx = (GlobalOrdinal)Teuchos::ScalarTraits<double>::pow(n, 0.33334);
345 ny = nx;
346 nz = nx;
347 TEUCHOS_TEST_FOR_EXCEPTION(nx * ny * nz != n, std::logic_error, "You need to specify nx, ny, and nz");
348 }
349 bool keepBCs = this->list_.get("keepBCs", false);
350
351 Scalar right = (Scalar)-one / (stretchx * stretchx);
352 Scalar left = (Scalar)-one / (stretchx * stretchx);
353 Scalar front = (Scalar)-one / (stretchy * stretchy);
354 Scalar back = (Scalar)-one / (stretchy * stretchy);
355 Scalar up = (Scalar)-one / (stretchz * stretchz);
356 Scalar down = (Scalar)-one / (stretchz * stretchz);
357 Scalar center = -(right + left + front + back + up + down);
358
359 // The Kokkos code path does not work for Epetra.
360 // Once Epetra has been removed this logic should be simplified.
361#if defined(HAVE_GALERI_KOKKOS) && defined(HAVE_GALERI_KOKKOSKERNELS)
362 using Node = typename Map::node_type;
363 using tpetra_map = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
364 if constexpr (std::is_same_v<Map, tpetra_map>) {
365 this->A_ = Cross3DKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, nz, center, left, right, front, back, down, up, this->DirichletBC_, keepBCs, "Laplace 3D");
366 } else if (this->Map_->lib() == ::Xpetra::UseTpetra) {
367 this->A_ = Cross3DKokkos<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, nz, center, left, right, front, back, down, up, this->DirichletBC_, keepBCs, "Laplace 3D");
368 } else
369#endif
370 {
371 this->A_ = Cross3D<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, nz, center, left, right, front, back, down, up, this->DirichletBC_, keepBCs);
372 }
373
374 this->A_->setObjectLabel(this->getObjectLabel());
375 return this->A_;
376}

630template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
631Teuchos::RCP<Matrix> Recirc2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
632 Teuchos::ParameterList list = this->list_;
633 GlobalOrdinal nx = -1;
634 GlobalOrdinal ny = -1;
635
636 // TODO FIXME need to check that all parameters are provided, otherwise error out
637 if (list.isParameter("nx")) {
638 if (list.isType<int>("nx"))
639 nx = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
640 else
641 nx = list.get<GlobalOrdinal>("nx");
642 }
643 if (list.isParameter("ny")) {
644 if (list.isType<int>("ny"))
645 ny = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
646 else
647 ny = list.get<GlobalOrdinal>("ny");
648 }
649
650 double lx = 1.;
651 if (list.isParameter("lx")) {
652 lx = list.get<double>("lx");
653 }
654 double ly = 1.;
655 if (list.isParameter("ly")) {
656 ly = list.get<double>("ly");
657 }
658
659 double conv = 1.;
660 if (list.isParameter("convection")) {
661 conv = list.get<double>("convection");
662 }
663
664 double diff = 1.;
665 if (list.isParameter("diffusion")) {
666 diff = list.get<double>("diffusion");
667 }
668
669 auto& map = this->Map_;
670 LocalOrdinal numMyElements = map->getLocalNumElements();
671 Teuchos::ArrayView<const GlobalOrdinal> myGlobalElements = map->getLocalElementList();
672
673 auto A = MultiVectorTraits<Map, MultiVector>::Build(map, 1);
674 auto B = MultiVectorTraits<Map, MultiVector>::Build(map, 1);
675 auto C = MultiVectorTraits<Map, MultiVector>::Build(map, 1);
676 auto D = MultiVectorTraits<Map, MultiVector>::Build(map, 1);
677 auto E = MultiVectorTraits<Map, MultiVector>::Build(map, 1);
678
679 double zero = Teuchos::ScalarTraits<double>::zero();
680
681 A->putScalar(zero);
682 B->putScalar(zero);
683 C->putScalar(zero);
684 D->putScalar(zero);
685 E->putScalar(zero);
686
687 auto Adata = A->getDataNonConst(0);
688 auto Bdata = B->getDataNonConst(0);
689 auto Cdata = C->getDataNonConst(0);
690 auto Ddata = D->getDataNonConst(0);
691 auto Edata = E->getDataNonConst(0);
692
693 double hx = lx / (nx + 1);
694 double hy = ly / (ny + 1);
695
696 for (int i = 0; i < numMyElements; i++) {
697 int ix, iy;
698 ix = (myGlobalElements[i]) % nx;
699 iy = (myGlobalElements[i] - ix) / nx;
700 double x = hx * (ix + 1);
701 double y = hy * (iy + 1);
702 double ConvX = conv * 4 * x * (x - 1.) * (1. - 2 * y) / hx;
703 double ConvY = -conv * 4 * y * (y - 1.) * (1. - 2 * x) / hy;
704
705 // convection part
706
707 if (ConvX < zero) {
708 Cdata[i] += ConvX;
709 Adata[i] -= ConvX;
710 } else {
711 Bdata[i] -= ConvX;
712 Adata[i] += ConvX;
713 }
714
715 if (ConvY < zero) {
716 Edata[i] += ConvY;
717 Adata[i] -= ConvY;
718 } else {
719 Ddata[i] -= ConvY;
720 Adata[i] += ConvY;
721 }
722
723 // add diffusion part
724 Adata[i] += diff * 2. / (hx * hx) + diff * 2. / (hy * hy);
725 Bdata[i] -= diff / (hx * hx);
726 Cdata[i] -= diff / (hx * hx);
727 Ddata[i] -= diff / (hy * hy);
728 Edata[i] -= diff / (hy * hy);
729 }
730 Adata = Bdata = Cdata = Ddata = Edata = Teuchos::null;
731
732 this->A_ = Cross2D<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, A, B, C, D, E);
733 this->A_->setObjectLabel(this->getObjectLabel());
734 return this->A_;
735} // Recirc2DProblem
