190 track_alloc(_superpanel_buf.span() * sizeof(value_type));
191

193 track_alloc(_piv.span() * sizeof(ordinal_type));
194
195 _diag = value_type_array("diag", 2 * _m);
196 track_alloc(_diag.span() * sizeof(value_type));
