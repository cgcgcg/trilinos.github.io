158 template <typename CrsMatBaseType, typename GraphToolType> SymbolicTools(CrsMatBaseType &A, GraphToolType &G) {
159 _m = A.NumRows();
160
161 _ap = Kokkos::create_mirror_view(host_memory_space(), A.RowPtr());
162 _aj = Kokkos::create_mirror_view(host_memory_space(), A.Cols());
163 _perm = Kokkos::create_mirror_view(host_memory_space(), G.PermVector());
164 _peri = Kokkos::create_mirror_view(host_memory_space(), G.InvPermVector());
165
166 Kokkos::deep_copy(_ap, A.RowPtr());
167 Kokkos::deep_copy(_aj, A.Cols());
168 Kokkos::deep_copy(_perm, G.PermVector());
169 Kokkos::deep_copy(_peri, G.InvPermVector());
170 }
