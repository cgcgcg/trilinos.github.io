21// Variables INPUT
22// comm --- communicator that Adelus is running on
23// nprocsr --- number of processors assigned to a row
24// ncols --- number of columns(=rows) for the matrix
25// nrhs --- number of right hand sides
26//
27// OUTPUT
28// my_rows --- number of rows of the total matrix I own
29// my_cols --- number of columns of the total matrix I own
30// my_first_row --- global number of my first row
31// my_first_col --- global number of my first column
32// my_rhs --- number of right hand sides that I own
33// my_row --- my subblock of the matrix
34// my_col --- my subblock of the matrix
35
36#include <mpi.h>
37#include "Adelus_distribute.hpp"
38
39namespace Adelus {
40
41void distmat_( MPI_Comm comm,
42 const int nprocsr,
43 const int ncols,
44 const int nrhs,
45 int& my_rows,
46 int& my_cols,
47 int& my_first_row,
48 int& my_first_col,
49 int& my_rhs,
50 int& my_row,
51 int& my_col )
52{
53
54 int rank, nprocs, nprocs_col, nprocs_row, nrows;
55
56 // Determine who I am and the number of processors that are being used
57 MPI_Comm_rank(comm, &rank) ;
58
59 MPI_Comm_size(comm, &nprocs);
60
61 nrows = ncols;
62
63 nprocs_row = nprocsr;
64
65 nprocs_col = nprocs/nprocsr;
66
67 // Distribute the rows and columns
68
69 my_row = rank / nprocsr;
70 my_col = rank % nprocsr;
71
72 //
73 my_rows = nrows / nprocs_col;
74
75 my_first_row = my_row * my_rows + 1;
76
77 my_first_row = (my_row > (nrows%nprocs_col)) ? my_first_row + (nrows%nprocs_col) :
78 my_first_row + my_row;
79
80 if (my_row < (nrows%nprocs_col)) ++my_rows;
81
82 //
83 my_cols = ncols / nprocs_row;
84
85 my_first_col = my_col * my_cols + 1;
86
87 my_first_col = (my_col > (ncols%nprocs_row)) ? my_first_col + (ncols%nprocs_row) :
88 my_first_col + my_col;
89
90 if (my_col < (ncols%nprocs_row)) ++my_cols;
91
92 // Distribute the RHS per processor
93
94 my_rhs = nrhs / nprocs_row;
95 if (my_col < (nrhs%nprocs_row)) ++my_rhs;
96
97}

99}//namespace Adelus
