48 template<typename Traits,typename Derived>
49 struct TaskBase : public PHX::EvaluatorWithBaseImpl<Traits> {
50
51 virtual Kokkos::Future<void,PHX::exec_space>
52 createTask(Kokkos::TaskScheduler<PHX::exec_space>& policy,
53 const int& work_size,
54 const std::vector<Kokkos::Future<void,PHX::exec_space>>& dependent_futures,
55 typename Traits::EvalData ) override
56 {
57 if (dependent_futures.size() == 0)
58 return policy.host_spawn(PHX::TaskWrap<PHX::exec_space,Derived>(work_size,*dynamic_cast<Derived*>(this)),Kokkos::TaskTeam);
59 else if (dependent_futures.size() == 1)
60 return policy.host_spawn(PHX::TaskWrap<PHX::exec_space,Derived>(work_size,*dynamic_cast<Derived*>(this)),Kokkos::TaskTeam,dependent_futures[0]);
61
62 auto aggregate_future = policy.when_all(dependent_futures.size(),dependent_futures.data());
63 return policy.host_spawn(PHX::TaskWrap<PHX::exec_space,Derived>(work_size,*dynamic_cast<Derived*>(this)),Kokkos::TaskTeam,aggregate_future);
64 }
65
67 unsigned taskSize() const override
68 {
69 return sizeof(PHX::TaskWrap<PHX::exec_space,Derived>);
70 }
71 };
