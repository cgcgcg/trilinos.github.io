22 // Create a flattened map for a map representing a distribution for an
23 // embedded scalar type
24 template <typename LocalOrdinal, typename GlobalOrdinal, typename Node>
25 Teuchos::RCP< Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node> >
26 create_flat_map(const Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node>& map,
27 const LocalOrdinal block_size) {
28 using Tpetra::global_size_t;
29 using Teuchos::ArrayView;
30 using Teuchos::Array;
31 using Teuchos::RCP;
32 using Teuchos::rcp;
33
34 typedef Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node> Map;
35
36 // Get map info
37 const global_size_t num_global_entries = map.getGlobalNumElements();
38 const size_t num_local_entries = map.getLocalNumElements();
39 const GlobalOrdinal index_base = map.getIndexBase();
40 ArrayView<const GlobalOrdinal> element_list =
41 map.getLocalElementList();
42
43 // Create new elements
44 const global_size_t flat_num_global_entries = num_global_entries*block_size;
45 const size_t flat_num_local_entries = num_local_entries * block_size;
46 const GlobalOrdinal flat_index_base = index_base;
47 Array<GlobalOrdinal> flat_element_list(flat_num_local_entries);
48 for (size_t i=0; i<num_local_entries; ++i)
49 for (LocalOrdinal j=0; j<block_size; ++j)
50 flat_element_list[i*block_size+j] = element_list[i]*block_size+j;
51
52 // Create new map
53 RCP<Map> flat_map =
54 rcp(new Map(flat_num_global_entries, flat_element_list(),
55 flat_index_base, map.getComm()));
56
57 return flat_map;
58 }

60 // Create a flattened graph for a graph from a matrix with the
61 // MP::Vector scalar type (each block is an identity matrix)
62 // If flat_domain_map and/or flat_range_map are null, they will be computed,
63 // otherwise they will be used as-is.
64 template <typename LocalOrdinal, typename GlobalOrdinal, typename Node>
65 Teuchos::RCP< Tpetra::CrsGraph<LocalOrdinal,GlobalOrdinal,Node > >
66 create_flat_mp_graph(
67 const Tpetra::CrsGraph<LocalOrdinal,GlobalOrdinal,Node >& graph,
68 Teuchos::RCP<const Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node > >& flat_domain_map,
69 Teuchos::RCP<const Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node > >& flat_range_map,
70 const LocalOrdinal block_size) {
71 using Teuchos::ArrayRCP;
72 using Teuchos::RCP;
73 using Teuchos::rcp;
74
75 typedef Tpetra::Map<LocalOrdinal,GlobalOrdinal,Node> Map;
76 typedef Tpetra::CrsGraph<LocalOrdinal,GlobalOrdinal,Node> Graph;
77 typedef typename Graph::local_graph_device_type::row_map_type::non_const_type RowPtrs;
78 typedef typename Graph::local_graph_device_type::entries_type::non_const_type LocalIndices;
79
80 // Build domain map if necessary
81 if (flat_domain_map == Teuchos::null)
82 flat_domain_map = create_flat_map(*(graph.getDomainMap()), block_size);
83
84 // Build range map if necessary
85 if (flat_range_map == Teuchos::null)
86 flat_range_map = create_flat_map(*(graph.getRangeMap()), block_size);
87
88 // Build column map
89 RCP<const Map> flat_col_map =
90 create_flat_map(*(graph.getColMap()), block_size);
91
92 // Build row map if necessary
93 // Check if range_map == row_map, then we can use flat_range_map
94 // as the flattened row map
95 RCP<const Map> flat_row_map;
96 if (graph.getRangeMap() == graph.getRowMap())
97 flat_row_map = flat_range_map;
98 else
99 flat_row_map = create_flat_map(*(graph.getRowMap()), block_size);
100
101 // Build flattened row offsets and column indices
102 auto row_offsets = graph.getLocalRowPtrsHost();
103 auto col_indices = graph.getLocalIndicesHost();
104 const size_t num_row = graph.getLocalNumRows();
105 const size_t num_col_indices = col_indices.size();
106 RowPtrs flat_row_offsets("row_ptrs", num_row*block_size+1);
107 LocalIndices flat_col_indices("col_indices", num_col_indices * block_size);
108 auto flat_row_offsets_host = Kokkos::create_mirror_view(flat_row_offsets);
109 auto flat_col_indices_host = Kokkos::create_mirror_view(flat_col_indices);
110 for (size_t row=0; row<num_row; ++row) {
111 const size_t row_beg = row_offsets[row];
112 const size_t row_end = row_offsets[row+1];
113 const size_t num_col = row_end - row_beg;
114 for (LocalOrdinal j=0; j<block_size; ++j) {
115 const size_t flat_row = row*block_size + j;
116 const size_t flat_row_beg = row_beg*block_size + j*num_col;
117 flat_row_offsets_host[flat_row] = flat_row_beg;
118 for (size_t entry=0; entry<num_col; ++entry) {
119 const LocalOrdinal col = col_indices[row_beg+entry];
120 const LocalOrdinal flat_col = col*block_size + j;
121 flat_col_indices_host[flat_row_beg+entry] = flat_col;
122 }
123 }
124 }
125 flat_row_offsets_host[num_row*block_size] = num_col_indices*block_size;
126 Kokkos::deep_copy(flat_row_offsets, flat_row_offsets_host);
127 Kokkos::deep_copy(flat_col_indices, flat_col_indices_host);
128
129 // Build flattened graph
130 RCP<Graph> flat_graph =
131 rcp(new Graph(flat_row_map, flat_col_map,
132 flat_row_offsets, flat_col_indices));
133 flat_graph->fillComplete(flat_domain_map, flat_range_map);
134
135 return flat_graph;
136 }

138 // Create a flattened vector by unrolling the MP::Vector scalar type. The
139 // returned vector is a view of the original
140 template <typename Storage, typename LocalOrdinal, typename GlobalOrdinal, typename Node>
141 Teuchos::RCP<Tpetra::MultiVector<typename Storage::value_type, LocalOrdinal, GlobalOrdinal, Node>>
142 create_flat_vector_view(
143 const Tpetra::MultiVector<Sacado::MP::Vector<Storage>, LocalOrdinal, GlobalOrdinal, Node>& vec,
144 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& flat_map
145 ) {
146 using BaseScalar = typename Storage::value_type;
147 using FlatVector = Tpetra::MultiVector<BaseScalar, LocalOrdinal, GlobalOrdinal, Node>;
148
149 // Create flattenend view using reshaping conversion copy constructor
150#ifdef KOKKOS_ENABLE_IMPL_VIEW_LEGACY
151 typename FlatVector::wrapped_dual_view_type flat_vals(vec.getWrappedDualView());
152#else
153 using flat_dual_view_t = typename FlatVector::wrapped_dual_view_type::DVT;
154 auto w_d_v = vec.getWrappedDualView();
155 typename FlatVector::wrapped_dual_view_type flat_vals(//vec.getWrappedDualView()
156 flat_dual_view_t(Stokhos::reinterpret_as_unmanaged_scalar_dual_view_of_same_rank(w_d_v.implGetDualView())),
157 flat_dual_view_t(Stokhos::reinterpret_as_unmanaged_scalar_dual_view_of_same_rank(w_d_v.implGetOriginalDualView())));
158#endif
159
160 // Create flat vector
161 return Teuchos::make_rcp<FlatVector>(flat_map, flat_vals);
162 }

205 // Create a flattened matrix by unrolling the MP::Vector scalar type. The
206 // returned matrix is NOT a view of the original (and can't be)
207 template <typename Storage, typename LocalOrdinal, typename GlobalOrdinal,
208 typename Node>
209 Teuchos::RCP< Tpetra::CrsMatrix<typename Storage::value_type,
210 LocalOrdinal,GlobalOrdinal,Node> >
211 create_flat_matrix(
212 const Tpetra::CrsMatrix<Sacado::MP::Vector<Storage>,
213 LocalOrdinal,GlobalOrdinal,Node>& mat,
214 const Teuchos::RCP<const Tpetra::CrsGraph<LocalOrdinal,GlobalOrdinal,Node> >& flat_graph,
215 const LocalOrdinal block_size) {
216 using Teuchos::ArrayView;
217 using Teuchos::Array;
218 using Teuchos::RCP;
219 using Teuchos::rcp;
220
221 typedef Sacado::MP::Vector<Storage> Scalar;
222 typedef typename Storage::value_type BaseScalar;
223 typedef Tpetra::CrsMatrix<Scalar,LocalOrdinal,GlobalOrdinal,Node> Matrix;
224 typedef Tpetra::CrsMatrix<BaseScalar,LocalOrdinal,GlobalOrdinal,Node> FlatMatrix;
225
226 // Create flat matrix
227 RCP<FlatMatrix> flat_mat = rcp(new FlatMatrix(flat_graph));
228
229 // Set values
230 const size_t num_rows = mat.getLocalNumRows();
231 const size_t max_cols = mat.getLocalMaxNumRowEntries();
232 typename Matrix::local_inds_host_view_type indices, flat_indices;
233 typename Matrix::values_host_view_type values;
234 Array<BaseScalar> flat_values(max_cols);
235 for (size_t row=0; row<num_rows; ++row) {
236 mat.getLocalRowView(row, indices, values);
237 const size_t num_col = mat.getNumEntriesInLocalRow(row);
238 for (LocalOrdinal i=0; i<block_size; ++i) {
239 const LocalOrdinal flat_row = row*block_size + i;
240 for (size_t j=0; j<num_col; ++j)
241 flat_values[j] = values[j].coeff(i);
242 flat_graph->getLocalRowView(flat_row, flat_indices);
243 flat_mat->replaceLocalValues(flat_row, Kokkos::Compat::getConstArrayView(flat_indices),
244 flat_values(0, num_col));
245 }
246 }
247 flat_mat->fillComplete(flat_graph->getDomainMap(),
248 flat_graph->getRangeMap());
249
250 return flat_mat;
251 }
