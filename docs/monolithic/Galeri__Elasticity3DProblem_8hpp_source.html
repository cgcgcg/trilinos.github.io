24template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
25class Elasticity3DProblem : public Problem<Map, Matrix, MultiVector> {
26 public:
27 using RealValuedMultiVector = typename Problem<Map, Matrix, MultiVector>::RealValuedMultiVector;
28 Elasticity3DProblem(Teuchos::ParameterList& list, const Teuchos::RCP<const Map>& map)
29 : Problem<Map, Matrix, MultiVector>(list, map) {
30 E = list.get("E", Teuchos::as<typename Teuchos::ScalarTraits<Scalar>::magnitudeType>(1e9));
31 nu = list.get("nu", Teuchos::as<typename Teuchos::ScalarTraits<Scalar>::magnitudeType>(0.25));
32
33 nx_ = -1;
34 ny_ = -1;
35 nz_ = -1;
36
37 if (list.isParameter("nx")) {
38 if (list.isType<int>("nx"))
39 nx_ = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
40 else
41 nx_ = list.get<GlobalOrdinal>("nx");
42 }
43 if (list.isParameter("ny")) {
44 if (list.isType<int>("ny"))
45 ny_ = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
46 else
47 ny_ = list.get<GlobalOrdinal>("ny");
48 }
49 if (list.isParameter("nz")) {
50 if (list.isType<int>("nz"))
51 nz_ = Teuchos::as<GlobalOrdinal>(list.get<int>("nz"));
52 else
53 nz_ = list.get<GlobalOrdinal>("nz");
54 }
55
56 nDim_ = 3;
57 double one = 1.0;
58 stretch.push_back(list.get("stretchx", one));
59 stretch.push_back(list.get("stretchy", one));
60 stretch.push_back(list.get("stretchz", one));
61
62 // NOTE: -1 is because galeri counts points, not elements
63 dims.push_back(nx_ - 1);
64 dims.push_back(ny_ - 1);
65 dims.push_back(nz_ - 1);
66
67 TEUCHOS_TEST_FOR_EXCEPTION(nx_ <= 0 || ny_ <= 0 || nz_ <= 0, std::logic_error, "nx, ny and nz must be positive");
68 }
69
70 Teuchos::RCP<Matrix> BuildMatrix();
71 Teuchos::RCP<MultiVector> BuildNullspace();
72 Teuchos::RCP<RealValuedMultiVector> BuildCoords();
73
74 private:
75 typedef Scalar SC;
76 typedef LocalOrdinal LO;
77 typedef GlobalOrdinal GO;
78
79 struct Point {
80 SC x, y, z;
81
82 Point() { z = Teuchos::ScalarTraits<SC>::zero(); }
83 Point(SC x_, SC y_, SC z_ = Teuchos::ScalarTraits<SC>::zero())
84 : x(x_)
85 , y(y_)
86 , z(z_) {}
87 };
88
89 GlobalOrdinal nx_, ny_, nz_;
90 size_t nDim_;
91 std::vector<GO> dims;
92 // NOTE: nodes correspond to a local subdomain nodes. I have to construct overlapped subdomains because
93 // InsertGlobalValues in Epetra does not support inserting into rows owned by other processor
94 std::vector<Point> nodes_;
95 std::vector<std::vector<LO> > elements_;
96 std::vector<GO> local2Global_;
97
98 std::vector<char> dirichlet_;
99
100 typename Teuchos::ScalarTraits<Scalar>::magnitudeType E, nu;
101 std::vector<Scalar> stretch;
102 std::string mode_;
103
104#if KOKKOS_VERSION >= 40799
105 using impl_scalar_type = typename KokkosKernels::ArithTraits<SC>::val_type;
106#else
107 using impl_scalar_type = typename Kokkos::ArithTraits<SC>::val_type;
108#endif
109#if KOKKOS_VERSION >= 40799
110 using KAT = KokkosKernels::ArithTraits<impl_scalar_type>;
111#else
112 using KAT = Kokkos::ArithTraits<impl_scalar_type>;
113#endif
114 using Memory2D = Kokkos::View<impl_scalar_type**, Kokkos::HostSpace>;
115
116 void EvalD(const std::vector<Point>& refPoints, Point& gaussPoint, Memory2D S);
117
118 void BuildMesh();
119 void BuildMaterialMatrix(Memory2D& D);
120 void BuildReferencePoints(size_t& numRefPoints, std::vector<Point>& refPoints, size_t& numGaussPoints, std::vector<Point>& gaussPoints);
121};

123template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
124Teuchos::RCP<Matrix> Elasticity3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
125 BuildMesh();
126
127 const size_t numDofPerNode = 3;
128 const size_t numNodesPerElem = 8;
129 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
130
131 TEUCHOS_TEST_FOR_EXCEPTION(elements_[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
132
133 // Material constant
134 SC t = 1;
135
136 // Material matrix
137 RCP<Memory2D> D(new Memory2D);
138 BuildMaterialMatrix(*D);
139
140 // Reference element, and reference Gauss points
141 size_t numRefPoints, numGaussPoints;
142 std::vector<Point> refPoints, gaussPoints;
143 BuildReferencePoints(numRefPoints, refPoints, numGaussPoints, gaussPoints);
144
145 // Evaluate the B matrix for the reference element
146 size_t sDim = 8;
147 size_t bDim = 9;
148 std::vector<Memory2D> Bs(numGaussPoints);
149 std::vector<Memory2D> Ss(numGaussPoints);
150
151 for (size_t j = 0; j < numGaussPoints; j++) {
152 Memory2D& S = Ss[j];
153 S = Memory2D("S", sDim, nDim_);
154 EvalD(refPoints, gaussPoints[j], S);
155
156 Memory2D& B = Bs[j];
157 B = Memory2D("B", bDim, numDofPerElem);
158
159 for (size_t k = 0; k < numNodesPerElem; k++) {
160 B(0, numDofPerNode * k + 0) = S(k, 0);
161 B(1, numDofPerNode * k + 0) = S(k, 1);
162 B(2, numDofPerNode * k + 0) = S(k, 2);
163 B(3, numDofPerNode * k + 1) = S(k, 0);
164 B(4, numDofPerNode * k + 1) = S(k, 1);
165 B(5, numDofPerNode * k + 1) = S(k, 2);
166 B(6, numDofPerNode * k + 2) = S(k, 0);
167 B(7, numDofPerNode * k + 2) = S(k, 1);
168 B(8, numDofPerNode * k + 2) = S(k, 2);
169 }
170 }
171
172 // Construct reordering matrix (see 6.2-9 from Cook)
173 Memory2D R("R", D->extent(0), bDim);
174 R(0, 0) = R(1, 4) = R(2, 8) = R(3, 1) = R(3, 3) = R(4, 5) = R(4, 7) = R(5, 2) = R(5, 6) = 1;
175
176 this->A_ = MatrixTraits<Map, Matrix>::Build(this->Map_, numNodesPerElem * 8 * numDofPerElem);
177 this->A_->setObjectLabel(this->getObjectLabel());
178
179 SC one = Teuchos::ScalarTraits<SC>::one(), zero = Teuchos::ScalarTraits<SC>::zero();
180 Memory2D prevKE("prevKE", numDofPerElem, numDofPerElem), prevElementNodes("prevElementNodes", numNodesPerElem, nDim_); // cache
181 for (size_t i = 0; i < elements_.size(); i++) {
182 // Select nodes subvector
183 Memory2D elementNodes("elementNodes", numNodesPerElem, nDim_);
184 std::vector<LO>& elemNodes = elements_[i];
185 for (size_t j = 0; j < numNodesPerElem; j++) {
186 elementNodes(j, 0) = nodes_[elemNodes[j]].x;
187 elementNodes(j, 1) = nodes_[elemNodes[j]].y;
188 elementNodes(j, 2) = nodes_[elemNodes[j]].z;
189 }
190
191 // Check if element is a translation of the previous element
192 auto xMove = elementNodes(0, 0) - prevElementNodes(0, 0), yMove = elementNodes(0, 1) - prevElementNodes(0, 1), zMove = elementNodes(0, 2) - prevElementNodes(0, 2);
193 typename KAT::magnitudeType eps = 1e-15; // coordinate comparison criteria
194 bool recompute = false;
195 {
196 size_t j = 0;
197 for (j = 0; j < numNodesPerElem; j++)
198 if (KAT::magnitude(elementNodes(j, 0) - (prevElementNodes(j, 0) + xMove)) > eps ||
199 KAT::magnitude(elementNodes(j, 1) - (prevElementNodes(j, 1) + yMove)) > eps ||
200 KAT::magnitude(elementNodes(j, 2) - (prevElementNodes(j, 2) + zMove)) > eps)
201 break;
202 if (j != numNodesPerElem)
203 recompute = true;
204 }
205
206 Memory2D KE("KE", numDofPerElem, numDofPerElem);
207 Kokkos::View<SC**, Kokkos::HostSpace> KE2("KE2", KE.extent(0), KE.extent(1));
208 if (recompute == false) {
209 // If an element has the same form as previous element, reuse stiffness matrix
210 Kokkos::deep_copy(KE, prevKE);
211
212 } else {
213 // Evaluate new stiffness matrix for the element
214 Memory2D K0("K0", D->extent(0), numDofPerElem);
215 for (size_t j = 0; j < numGaussPoints; j++) {
216 Memory2D& B = Bs[j];
217 Memory2D& S = Ss[j];
218
219 Memory2D JAC("JAC", nDim_, nDim_);
220
221 for (size_t p = 0; p < nDim_; p++)
222 for (size_t q = 0; q < nDim_; q++) {
223 JAC(p, q) = zero;
224
225 for (size_t k = 0; k < numNodesPerElem; k++)
226 JAC(p, q) += S(k, p) * elementNodes(k, q);
227 }
228
229 auto detJ = JAC(0, 0) * JAC(1, 1) * JAC(2, 2) + JAC(2, 0) * JAC(0, 1) * JAC(1, 2) + JAC(0, 2) * JAC(2, 1) * JAC(1, 0) -
230 JAC(2, 0) * JAC(1, 1) * JAC(0, 2) - JAC(0, 0) * JAC(2, 1) * JAC(1, 2) - JAC(2, 2) * JAC(0, 1) * JAC(1, 0);
231
232 // J2 = inv([JAC zeros(3) zeros(3); zeros(3) JAC zeros(3); zeros(3) zeros(3) JAC])
233 Memory2D J2("J2", nDim_ * nDim_, nDim_ * nDim_);
234 J2(0, 0) = J2(3, 3) = J2(6, 6) = (JAC(2, 2) * JAC(1, 1) - JAC(2, 1) * JAC(1, 2)) / detJ;
235 J2(0, 1) = J2(3, 4) = J2(6, 7) = -(JAC(2, 2) * JAC(0, 1) - JAC(2, 1) * JAC(0, 2)) / detJ;
236 J2(0, 2) = J2(3, 5) = J2(6, 8) = (JAC(1, 2) * JAC(0, 1) - JAC(1, 1) * JAC(0, 2)) / detJ;
237 J2(1, 0) = J2(4, 3) = J2(7, 6) = -(JAC(2, 2) * JAC(1, 0) - JAC(2, 0) * JAC(1, 2)) / detJ;
238 J2(1, 1) = J2(4, 4) = J2(7, 7) = (JAC(2, 2) * JAC(0, 0) - JAC(2, 0) * JAC(0, 2)) / detJ;
239 J2(1, 2) = J2(4, 5) = J2(7, 8) = -(JAC(1, 2) * JAC(0, 0) - JAC(1, 0) * JAC(0, 2)) / detJ;
240 J2(2, 0) = J2(5, 3) = J2(8, 6) = (JAC(2, 1) * JAC(1, 0) - JAC(2, 0) * JAC(1, 1)) / detJ;
241 J2(2, 1) = J2(5, 4) = J2(8, 7) = -(JAC(2, 1) * JAC(0, 0) - JAC(2, 0) * JAC(0, 1)) / detJ;
242 J2(2, 2) = J2(5, 5) = J2(8, 8) = (JAC(1, 1) * JAC(0, 0) - JAC(1, 0) * JAC(0, 1)) / detJ;
243
244 Memory2D B2("B2", J2.extent(0), B.extent(1));
245 KokkosBlas::gemm("N", "N", one, J2, B, zero, B2);
246
247 // KE = KE + t * J2B' * D * J2B * detJ
248 Memory2D J2B("J2B", R.extent(0), B2.extent(1));
249 KokkosBlas::gemm("N", "N", one, R, B2, zero, J2B);
250 KokkosBlas::gemm("N", "N", one, *D, J2B, zero, K0);
251 KokkosBlas::gemm("T", "N", t * detJ, J2B, K0, one, KE);
252 }
253
254 // Cache the matrix and nodes
255 Kokkos::deep_copy(prevKE, KE);
256 Kokkos::deep_copy(prevElementNodes, elementNodes);
257 }
258
259 Teuchos::Array<GO> elemDofs(numDofPerElem);
260 for (size_t j = 0; j < numNodesPerElem; j++) { // FIXME: this may be inconsistent with the map
261 elemDofs[numDofPerNode * j + 0] = local2Global_[elemNodes[j]] * numDofPerNode;
262 elemDofs[numDofPerNode * j + 1] = elemDofs[numDofPerNode * j + 0] + 1;
263 elemDofs[numDofPerNode * j + 2] = elemDofs[numDofPerNode * j + 0] + 2;
264 }
265
266 // Deal with Dirichlet nodes
267 bool isDirichlet = false;
268 for (size_t j = 0; j < numNodesPerElem; j++)
269 if (dirichlet_[elemNodes[j]])
270 isDirichlet = true;
271
272 if (isDirichlet) {
273 bool keepBCs = this->list_.get("keepBCs", false);
274 if (keepBCs) {
275 // Simple case: keep Dirichlet DOF
276 // We rewrite rows and columns corresponding to Dirichlet DOF with zeros
277 // The diagonal elements corresponding to Dirichlet DOF are set to 1.
278 for (size_t j = 0; j < numNodesPerElem; j++)
279 if (dirichlet_[elemNodes[j]]) {
280 LO j0 = numDofPerNode * j + 0;
281 LO j1 = numDofPerNode * j + 1;
282 LO j2 = numDofPerNode * j + 2;
283
284 for (size_t k = 0; k < numDofPerElem; k++)
285 KE(j0, k) = KE(k, j0) = KE(j1, k) = KE(k, j1) = KE(j2, k) = KE(k, j2) = zero;
286 KE(j0, j0) = KE(j1, j1) = KE(j2, j2) = one;
287 }
288
289 } else {
290 // Complex case: get rid of Dirichlet DOF
291 // The case is complex because if we simply reduce the size of the matrix, it would become inconsistent
292 // with maps. So, instead, we modify values of the boundary cells as if we had an additional cell close
293 // to the boundary.
294 for (int j = 0; j < (int)numNodesPerElem; j++)
295 if (dirichlet_[elemNodes[j]]) {
296 LO j0 = numDofPerNode * j, j1 = j0 + 1, j2 = j0 + 2;
297
298 // NOTE: had to make j & k int instead of size_t so that I can use subtraction without overflowing
299 for (int k = 0; k < (int)numNodesPerElem; k++)
300 if ((j == k) || (std::abs(j - k) < 4 && ((j + k) & 0x1)) || (std::abs(j - k) == 4)) {
301 // Nodes j and k are connected by an edge, or j == k
302 LO k0 = numDofPerNode * k, k1 = k0 + 1, k2 = k0 + 2;
303 SC f = Teuchos::as<SC>(pow(2, Teuchos::as<int>(std::min(dirichlet_[elemNodes[j]], dirichlet_[elemNodes[k]]))));
304
305 KE(j0, k0) *= f;
306 KE(j0, k1) *= f;
307 KE(j0, k2) *= f;
308 KE(j1, k0) *= f;
309 KE(j1, k1) *= f;
310 KE(j1, k2) *= f;
311 KE(j2, k0) *= f;
312 KE(j2, k1) *= f;
313 KE(j2, k2) *= f;
314 }
315 }
316 }
317 }
318
319 if constexpr (!std::is_same_v<SC, impl_scalar_type>)
320 Kokkos::deep_copy(KE2, KE);
321
322 // Insert KE into the global matrix
323 // NOTE: KE is symmetric, therefore it does not matter that it is in the CSC format
324 for (size_t j = 0; j < numDofPerElem; j++)
325 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
326 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
327 if constexpr (std::is_same_v<SC, impl_scalar_type>) {
328 auto vals = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
329 this->A_->insertGlobalValues(elemDofs[j], inds, vals);
330 } else {
331 auto vals = Kokkos::Compat::getArrayView(Kokkos::subview(KE2, j, Kokkos::ALL()));
332 this->A_->insertGlobalValues(elemDofs[j], inds, vals);
333 }
334 }
335 }
336 this->A_->fillComplete();
337
338 return this->A_;
339}

341template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
342RCP<typename Problem<Map, Matrix, MultiVector>::RealValuedMultiVector>
343Elasticity3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildCoords() {
344 // FIXME: map here is an extended map, with multiple DOF per node
345 // as we cannot construct a single DOF map in Problem, we repeat the coords
346 this->Coords_ = MultiVectorTraits<Map, RealValuedMultiVector>::Build(this->Map_, nDim_);
347
348 typedef typename RealValuedMultiVector::scalar_type real_type;
349
350 Teuchos::ArrayRCP<real_type> x = this->Coords_->getDataNonConst(0);
351 Teuchos::ArrayRCP<real_type> y = this->Coords_->getDataNonConst(1);
352 Teuchos::ArrayRCP<real_type> z = this->Coords_->getDataNonConst(2);
353
354 Teuchos::ArrayView<const GO> GIDs = this->Map_->getLocalElementList();
355
356 // NOTE: coordinates vector local ordering is consistent with that of the
357 // matrix map, as it is constructed by going through GIDs and translating
358 // those.
359 const typename KAT::magnitudeType hx = KAT::magnitude(stretch[0]),
360 hy = KAT::magnitude(stretch[1]),
361 hz = KAT::magnitude(stretch[2]);
362 for (GO p = 0; p < GIDs.size(); p += 3) { // FIXME: we assume that DOF for the same node are label consequently
363 GlobalOrdinal ind = GIDs[p] / 3;
364 size_t i = ind % nx_, k = ind / (nx_ * ny_), j = (ind - k * nx_ * ny_) / nx_;
365
366 x[p] = x[p + 1] = x[p + 2] = (i + 1) * hx;
367 y[p] = y[p + 1] = y[p + 2] = (j + 1) * hy;
368 z[p] = z[p + 1] = z[p + 2] = (k + 1) * hz;
369 }
370
371 return this->Coords_;
372}

374template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
375RCP<MultiVector> Elasticity3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildNullspace() {
376 typedef typename RealValuedMultiVector::scalar_type real_type;
377
378 const int numVectors = 6;
379 this->Nullspace_ = MultiVectorTraits<Map, MultiVector>::Build(this->Map_, numVectors);
380
381 if (this->Coords_ == Teuchos::null)
382 BuildCoords();
383
384 // Teuchos::ArrayView<const GO> GIDs = this->Map_->getLocalElementList();
385
386 size_t numDofs = this->Map_->getLocalNumElements();
387 Teuchos::ArrayRCP<real_type> x = this->Coords_->getDataNonConst(0);
388 Teuchos::ArrayRCP<real_type> y = this->Coords_->getDataNonConst(1);
389 Teuchos::ArrayRCP<real_type> z = this->Coords_->getDataNonConst(2);
390
391 auto one = KAT::one();
392
393 // NOTE: nullspace local ordering is consistent with that of the matrix
394 // map, as it inherits ordering from coordinates, which is consistent.
395
396 {
397 // Translations
398 Teuchos::ArrayRCP<SC> T0 = this->Nullspace_->getDataNonConst(0), T1 = this->Nullspace_->getDataNonConst(1), T2 = this->Nullspace_->getDataNonConst(2);
399 for (size_t i = 0; i < numDofs; i += nDim_) {
400 T0[i] = one;
401 T1[i + 1] = one;
402 T2[i + 2] = one;
403 }
404
405 // Calculate center
406 real_type cx = this->Coords_->getVector(0)->meanValue();
407 real_type cy = this->Coords_->getVector(1)->meanValue();
408 real_type cz = this->Coords_->getVector(2)->meanValue();
409
410 // Rotations
411 Teuchos::ArrayRCP<SC> R0 = this->Nullspace_->getDataNonConst(3), R1 = this->Nullspace_->getDataNonConst(4), R2 = this->Nullspace_->getDataNonConst(5);
412 for (size_t i = 0; i < numDofs; i += nDim_) {
413 // Rotate in Y-Z Plane (around Z axis): [ -y; x]
414 R0[i + 0] = -(y[i] - cy);
415 R0[i + 1] = (x[i] - cx);
416
417 // Rotate in Y-Z Plane (around Z axis): [ -z; y]
418 R1[i + 1] = -(z[i] - cz);
419 R1[i + 2] = (y[i] - cy);
420
421 // Rotate in Y-Z Plane (around Z axis): [ z; -x]
422 R2[i + 0] = (z[i] - cz);
423 R2[i + 2] = -(x[i] - cx);
424 }
425 }
426
427 // Equalize norms of all vectors to that of the first one
428 // We do not normalize them as a vector of ones seems nice
429 Teuchos::Array<typename KAT::magnitudeType> norms2(numVectors);
430 this->Nullspace_->norm2(norms2);
431 Teuchos::Array<SC> norms2scalar(numVectors);
432 for (int i = 0; i < numVectors; i++)
433 norms2scalar[i] = norms2[0] / norms2[i];
434 this->Nullspace_->scale(norms2scalar);
435
436 return this->Nullspace_;
437}

439template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
440void Elasticity3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMesh() {
441 const typename KAT::magnitudeType hx = KAT::magnitude(stretch[0]),
442 hy = KAT::magnitude(stretch[1]),
443 hz = KAT::magnitude(stretch[2]);
444
445 GO myPID = this->Map_->getComm()->getRank();
446 GO const& negOne = -1;
447 GO mx = this->list_.get("mx", negOne), my = this->list_.get("my", negOne), mz = this->list_.get("mz", negOne);
448
449 const GO mxy = mx * my;
450
451 GO startx, starty, startz, endx, endy, endz;
452 Utils::getSubdomainData(dims[0], mx, (myPID % mxy) % mx, startx, endx);
453 Utils::getSubdomainData(dims[1], my, (myPID % mxy) / mx, starty, endy);
454 Utils::getSubdomainData(dims[2], mz, myPID / mxy, startz, endz);
455
456 LO nx = endx - startx, ny = endy - starty, nz = endz - startz;
457
458 // Expand subdomain to do overlap
459 if (startx > 0) {
460 nx++;
461 startx--;
462 }
463 if (starty > 0) {
464 ny++;
465 starty--;
466 }
467 if (startz > 0) {
468 nz++;
469 startz--;
470 }
471 if (startx + nx < dims[0]) {
472 nx++;
473 }
474 if (starty + ny < dims[1]) {
475 ny++;
476 }
477 if (startz + nz < dims[2]) {
478 nz++;
479 }
480
481 nodes_.resize((nx + 1) * (ny + 1) * (nz + 1));
482 dirichlet_.resize((nx + 1) * (ny + 1) * (nz + 1), 0);
483 local2Global_.resize((nx + 1) * (ny + 1) * (nz + 1));
484 elements_.resize(nx * ny * nz);
485
486#define NODE(i, j, k) ((k) * (ny + 1) * (nx + 1) + (j) * (nx + 1) + (i))
487#define CELL(i, j, k) ((k)*ny * nx + (j)*nx + (i))
488 // NOTE: the fact that local ordering here is not consistent with that of
489 // the matrix map does not matter. The two things that matter are:
490 // local2Global_ assigns to a correct GID, and nodes_ contain correct
491 // coordinates
492 for (int k = 0; k <= nz; k++)
493 for (int j = 0; j <= ny; j++)
494 for (int i = 0; i <= nx; i++) {
495 int ii = startx + i, jj = starty + j, kk = startz + k;
496 int nodeID = NODE(i, j, k);
497 nodes_[nodeID] = Point((ii + 1) * hx, (jj + 1) * hy, (kk + 1) * hz);
498 local2Global_[nodeID] = kk * nx_ * ny_ + jj * nx_ + ii;
499
500 if (ii == 0 && (this->DirichletBC_ & DIR_LEFT)) dirichlet_[nodeID]++;
501 if (ii == nx_ && (this->DirichletBC_ & DIR_RIGHT)) dirichlet_[nodeID]++;
502 if (jj == 0 && (this->DirichletBC_ & DIR_FRONT)) dirichlet_[nodeID]++;
503 if (jj == ny_ && (this->DirichletBC_ & DIR_BACK)) dirichlet_[nodeID]++;
504 if (kk == 0 && (this->DirichletBC_ & DIR_BOTTOM)) dirichlet_[nodeID]++;
505 if (kk == nz_ && (this->DirichletBC_ & DIR_TOP)) dirichlet_[nodeID]++;
506 }
507
508 for (int k = 0; k < nz; k++)
509 for (int j = 0; j < ny; j++)
510 for (int i = 0; i < nx; i++) {
511 std::vector<LO>& element = elements_[CELL(i, j, k)];
512 element.resize(8);
513 element[0] = NODE(i, j, k);
514 element[1] = NODE(i + 1, j, k);
515 element[2] = NODE(i + 1, j + 1, k);
516 element[3] = NODE(i, j + 1, k);
517 element[4] = NODE(i, j, k + 1);
518 element[5] = NODE(i + 1, j, k + 1);
519 element[6] = NODE(i + 1, j + 1, k + 1);
520 element[7] = NODE(i, j + 1, k + 1);
521 }
522#undef NODE
523#undef CELL
524}

528 D = Memory2D("Material", 6, 6);

570template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
571void Elasticity3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalD(const std::vector<Point>& refPoints, Point& gaussPoint, Memory2D S) {
572 const Scalar one = Teuchos::ScalarTraits<Scalar>::one();
573 const Scalar eight = one + one + one + one + one + one + one + one;
574 const Scalar eighth = one / eight;
575 for (size_t j = 0; j < refPoints.size(); j++) {
576 // dxi
577 S(j, 0) = refPoints[j].x * (one + refPoints[j].y * gaussPoint.y) * (one + refPoints[j].z * gaussPoint.z) * eighth;
578 // deta
579 S(j, 1) = (one + refPoints[j].x * gaussPoint.x) * refPoints[j].y * (one + refPoints[j].z * gaussPoint.z) * eighth;
580 // dzeta
581 S(j, 2) = (one + refPoints[j].x * gaussPoint.x) * (one + refPoints[j].y * gaussPoint.y) * refPoints[j].z * eighth;
582 }
583}
