197// **********************************************************************
198template<typename DataT,int Rank,typename Layout>
199void PHX::Field<DataT,Rank,Layout>::setFieldData(const std::any& a)
200{
201#if defined( PHX_DEBUG)
202 TEUCHOS_TEST_FOR_EXCEPTION(m_tag.is_null(), std::logic_error, m_field_tag_error_msg);
203 m_data_set = true;
204#endif
205
206 // any object is always the non-const data type. To correctly cast
207 // the any object to the Kokkos::View, need to pull the const off
208 // the scalar type if this Field has a const scalar type.
209 typedef Kokkos::View<typename array_type::non_const_data_type,Layout,PHX::Device> non_const_view;
210 try {
211 non_const_view tmp = std::any_cast<non_const_view>(a);
212 m_field_data = tmp;
213 }
214 catch (std::exception& ) {
215 std::cout << "\n\nError in compiletime PHX::Field::setFieldData() in std::any_cast. Tried to cast the field \""
216 << this->fieldTag().name() << "\" with the identifier \"" << this->fieldTag().identifier()
217 << "\" to a type of \"" << Teuchos::demangleName(typeid(non_const_view).name())
218 << "\" from a std::any object containing a type of \""
219 << Teuchos::demangleName(a.type().name()) << "\"." << std::endl;
220 throw;
221 }
222}

235template<typename DataT,int Rank,typename Layout>
236void PHX::Field<DataT,Rank,Layout>::print(std::ostream& os, bool printValues) const
237{
238 os << "Field<" << Rank << ">(";
239 for (int i=0; i < Rank; ++i) {
240 if (i > 0)
241 os << ",";
242 os << m_field_data.extent(i);
243 }
244 os << "): ";
245
246 if (nonnull(m_tag))
247 os << *m_tag;
248
249 if (printValues)
250 os << "Error - Field no longer supports the \"printValues\" member of the Field::print() method. Values may be on a device that does not support printing (e.g. GPU). Please disconstinue the use of this call!" << std::endl;
251}
