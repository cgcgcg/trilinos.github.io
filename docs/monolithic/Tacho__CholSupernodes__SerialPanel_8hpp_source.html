54 const ordinal_type sid) {
55 typedef SupernodeInfoType supernode_info_type;
56
57 typedef typename supernode_info_type::value_type value_type;
58 typedef typename supernode_info_type::value_type_matrix value_type_matrix;
59
60 // algorithm choice
61 using CholAlgoType = typename CholAlgorithm::type;
62 using TrsmAlgoType = typename TrsmAlgorithm::type;
63
64 // get current supernode
65 const auto &s = info.supernodes(sid);
66
67 // get panel pointer
68 value_type *ptr = s.buf;
69
70 // panel (s.m x s.n) is divided into ATL (m x m) and ATR (m x n)
71 const ordinal_type m = s.m, n = s.n - s.m;
72
73 // m is available, then factorize the supernode block
74 if (m > 0) {
75 UnmanagedViewType<value_type_matrix> ATL(ptr, m, m);
76 ptr += m * m;
77 Chol<Uplo::Upper, CholAlgoType>::invoke(member, ATL);
78

82 Trsm<Side::Left, Uplo::Upper, Trans::ConjTranspose, TrsmAlgoType>::invoke(member, Diag::NonUnit(), 1.0, ATL,
83 ATR);
84 }
85 }
86 return 0;

346 template <typename MemberType, typename SupernodeInfoType>
347 KOKKOS_INLINE_FUNCTION static int factorize_recursive_serial(MemberType &member, const SupernodeInfoType &info,
348 const ordinal_type sid, const bool final,
349 typename SupernodeInfoType::value_type *buf,
350 const size_type bufsize, const size_type np) {
351 typedef SupernodeInfoType supernode_info_type;
352
353 typedef typename supernode_info_type::value_type value_type;
354
355 static constexpr bool runOnHost = run_tacho_on_host_v<typename supernode_info_type::exec_space>;
356
357 const auto &s = info.supernodes(sid);
358
359 if (final) {
360 // serial recursion
361 for (ordinal_type i = 0; i < s.nchildren; ++i)
362 factorize_recursive_serial(member, info, s.children[i], final, buf, bufsize, np);
363 }
364
365 {
366 const ordinal_type n = s.n - s.m;
367
368 size_type bufsize_required;
369 if constexpr(runOnHost)
370 bufsize_required = n * (min(np, n) + 1) * sizeof(value_type);
371 else
372 bufsize_required = n * (min(np, n) + member.team_size()) * sizeof(value_type);
373
374 TACHO_TEST_FOR_ABORT(bufsize < bufsize_required, "bufsize is smaller than required");
375
376 CholSupernodes<Algo::Workflow::SerialPanel>::factorize(member, info, sid);
377
378 for (ordinal_type offn = 0; offn < n; offn += np) {
379 CholSupernodes<Algo::Workflow::SerialPanel>::update(member, info, offn, np, sid, bufsize, (void *)buf);
380 }
381 }
382 return 0;
383 }
