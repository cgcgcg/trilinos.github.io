51template <typename ordinal_type, typename value_type>
52ordinal_type
53Stokhos::BlockPreconditioner<ordinal_type, value_type>::
54ApplyInverse(const Teuchos::SerialDenseMatrix<ordinal_type, value_type>& Input,
55 Teuchos::SerialDenseMatrix<ordinal_type, value_type>& Result,
56 ordinal_type n) const
57{ //Solve M*Result=Input
58 ordinal_type c=siz(p,m);
59 ordinal_type s = siz(p-1,m);
60
61 //Split residual
62 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r1(Teuchos::Copy, Input, s, 1);
63 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r2(Teuchos::Copy, Input, c-s, 1, s, 0);
64
65 //Split Result
66 Teuchos::SerialDenseMatrix<ordinal_type, value_type> u1(Teuchos::Copy, Result, s, 1);
67 Teuchos::SerialDenseMatrix<ordinal_type, value_type> u2(Teuchos::Copy, Result, c-s, 1, s, 0);
68
69 Teuchos::SerialDenseMatrix<ordinal_type, value_type> B(Teuchos::View, K, s, c-s, 0, s);
70 Teuchos::SerialDenseMatrix<ordinal_type, value_type> D(Teuchos::View, K, c-s, c-s, s,s);
71
72 //rD=inv(D)r2
73
74 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Dr(c-s,1);
75
76 for (ordinal_type i=0; i<c-s; i++)
77 Dr(i,0)=r2(i,0)/D(i,i);
78
79 ordinal_type ret = r1.multiply(Teuchos::NO_TRANS,Teuchos::NO_TRANS, -1.0, B, Dr, 1.0);
80 TEUCHOS_ASSERT(ret == 0);
81
82 //Compute S=A-B*inv(D)*Bt
83 Teuchos::SerialDenseMatrix<ordinal_type, value_type> S(Teuchos::Copy, K, s, s);
84 //Compute B*inv(D)
85 Teuchos::SerialDenseMatrix<ordinal_type, value_type> BinvD(s,c-s);
86 for (ordinal_type i=0; i<c-s; i++) //col
87 for (ordinal_type j=0; j<s; j++) //row
88 BinvD(j,i)=B(j,i)/D(i,i);
89
90 S.multiply(Teuchos::NO_TRANS,Teuchos::TRANS, -1.0, BinvD, B, 1.0);
91
92 Teuchos::RCP< Teuchos::SerialDenseMatrix<ordinal_type, value_type> > SS, w, rr;
93 SS = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (S));
94 w = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (s,1));
95 rr = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (r1));
96
97
98 // Setup solver
99 Teuchos::SerialDenseSolver<ordinal_type, value_type> solver;
100 solver.setMatrix(SS);
101 solver.setVectors(w, rr);
102 //Solve S*w=r1
103 if (solver.shouldEquilibrate()) {
104 solver.factorWithEquilibration(true);
105 solver.equilibrateMatrix();
106 }
107 solver.solve();
108
109 for (ordinal_type i=0; i<s; i++)
110 Result(i,0)=(*w)(i,0);
111
112 ret = r2.multiply(Teuchos::TRANS,Teuchos::NO_TRANS, -1.0, B, *w, 1.0);
113 TEUCHOS_ASSERT(ret == 0);
114
115 for (ordinal_type i=s; i<c; i++)
116 Result(i,0)=r2(-s+i,0)/D(-s+i, -s+i);
117
118 return 0;
119}
