71 ComplexT ret ;
72 cr = (Teuchos::ScalarTraits<ComplexT>::real(c)) ;
73 ci = (Teuchos::ScalarTraits<ComplexT>::imag(c)) ;
74 if (MY_SCALAR_ABS (cr) >= MY_SCALAR_ABS (ci))
75 {
76 r = ci / cr ;
77 den = cr + r * ci ;
78 ret = std::complex<T>(1.0 / den, -r / den) ;
79 }
80 else
81 {
82 r = cr / ci ;
83 den = r * cr + ci ;
84 ret = std::complex<T>(r / den, -1.0 / den) ;
85 }
86 return ret;
87 }
