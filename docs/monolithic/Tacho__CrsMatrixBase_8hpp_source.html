256template <typename CrsMatrixType, typename OrdinalTypeArray>
257inline static void applyPermutationToCrsMatrix(/* */ CrsMatrixType &A, const CrsMatrixType &B,
258 const OrdinalTypeArray &p, const OrdinalTypeArray &ip) {
259 const ordinal_type m = A.NumRows(); //, n = A.NumCols();
260 typedef typename CrsMatrixType::exec_space exec_space;
261 typedef typename CrsMatrixType::exec_memory_space exec_memory_space;
262
264 auto ap = A.RowPtr();
265 auto aj = A.Cols();
266 auto ax = A.Values();
267
268 auto perm = Kokkos::create_mirror_view(exec_memory_space(), p);
269 Kokkos::deep_copy(perm, p);
270 auto peri = Kokkos::create_mirror_view(exec_memory_space(), ip);
271 Kokkos::deep_copy(peri, ip);
272 {
273 Kokkos::RangePolicy<exec_space, Kokkos::Schedule<Kokkos::Static>> policy(0, m + 1);
274 Kokkos::parallel_scan(
275 policy, KOKKOS_LAMBDA(const ordinal_type &i, size_type &update, const bool &final) {
276 if (final)
277 ap(i) = update;
278
279 if (i < m) {
280 const ordinal_type row = perm(i);
281 update += (B.RowPtrEnd(row) - B.RowPtrBegin(row));
282 }
283 });
284 Kokkos::fence();
285 }
286 {
287 typedef Kokkos::TeamPolicy<exec_space> team_policy_type;
288 team_policy_type policy(m, 1, 1); // Kokkos::AUTO()); ///, Kokkos::AUTO());
289 Kokkos::parallel_for(
290 policy, KOKKOS_LAMBDA(const typename team_policy_type::member_type &member) {
291 const ordinal_type i = member.league_rank(),
292 kbeg = ap(i), kend = ap(i + 1), nk = kend - kbeg,
293 row = perm(i),
294 colbeg = B.RowPtrBegin(row);
295 // colend = B.RowPtrEnd(row);
296
297 Kokkos::parallel_for(
298 Kokkos::TeamVectorRange(member, nk),
299 [&, kbeg, colbeg](const ordinal_type &k) {
300 const ordinal_type tk = kbeg + k, sk = colbeg + k;
301 aj(tk) = peri(B.Col(sk));
302 ax(tk) = B.Value(sk);
303 });
304 });
305 Kokkos::fence();
306 }
307}

310template <typename CrsMatrixType, typename OrdinalTypeArray>
311inline static void applyPermutationToCrsMatrixLower(/* */ CrsMatrixType &A, const CrsMatrixType &B,
312 const OrdinalTypeArray &p, const OrdinalTypeArray &ip) {
313 const ordinal_type m = A.NumRows(); //, n = A.NumCols();
314 typedef typename CrsMatrixType::exec_space exec_space;
315 typedef typename CrsMatrixType::exec_memory_space exec_memory_space;
316
318 auto ap = A.RowPtr();
319 auto aj = A.Cols();
320 auto ax = A.Values();
321
322 auto perm = Kokkos::create_mirror_view(exec_memory_space(), p);
323 Kokkos::deep_copy(perm, p);
324 auto peri = Kokkos::create_mirror_view(exec_memory_space(), ip);
325 Kokkos::deep_copy(peri, ip);
326
327 {
328 Kokkos::RangePolicy<exec_space, Kokkos::Schedule<Kokkos::Static>> policy(0, m + 1);
329 Kokkos::parallel_scan(
330 policy, KOKKOS_LAMBDA(const ordinal_type &i, size_type &update, const bool &final) {
331 if (final)
332 ap(i) = update;
333
334 if (i < m) {
335 ordinal_type count(0);
336 const ordinal_type row = perm(i);
337 for (ordinal_type k = B.RowPtrBegin(row), kend = B.RowPtrEnd(row); k < kend; ++k) {
338 const ordinal_type j = peri(B.Col(k));
339 count += (i >= j);
340 }
341 update += count;
342 }
343 });
344 Kokkos::fence();
345 }
346 {
347 Kokkos::RangePolicy<exec_space, Kokkos::Schedule<Kokkos::Static>> policy(0, m);
348 Kokkos::parallel_for(
349 policy, KOKKOS_LAMBDA(const ordinal_type &i) {
350 const ordinal_type kbeg = ap(i),
351 row = perm(i),
352 colbeg = B.RowPtrBegin(row), colend = B.RowPtrEnd(row), nk = colend - colbeg;
353
354 for (ordinal_type k = 0, t = 0; k < nk; ++k) {
355 const ordinal_type tk = kbeg + t, sk = colbeg + k;
356 const ordinal_type j = peri(B.Col(sk));
357 if (i >= j) {
358 aj(tk) = j;
359 ax(tk) = B.Value(sk);
360 ++t;
361 }
362 }
363 });
364 Kokkos::fence();
365 }
366
368 A.setNumNonZeros();

371template <typename ValueType, typename DeviceType>
372inline double computeRelativeResidual(const CrsMatrixBase<ValueType, DeviceType> &A,
373 const Kokkos::View<ValueType **, Kokkos::LayoutLeft, DeviceType> &x,
374 const Kokkos::View<ValueType **, Kokkos::LayoutLeft, DeviceType> &b,
375 const bool verbose = false) {
376 const bool test = (size_t(A.NumRows()) != size_t(A.NumCols()) || size_t(A.NumRows()) != size_t(b.extent(0)) ||
377 size_t(x.extent(0)) != size_t(b.extent(0)) || size_t(x.extent(1)) != size_t(b.extent(1)));
378 if (test)
379 throw std::logic_error("A,x and b dimensions are not compatible");
380
381 typedef ValueType value_type;
382 typedef typename UseThisDevice<Kokkos::DefaultHostExecutionSpace>::type host_device_type;
383 typedef typename host_device_type::memory_space host_memory_space;
384
385 CrsMatrixBase<value_type, host_device_type> h_A;
386 h_A.createMirror(A);
387 h_A.copy(A);
388
389 auto h_x = Kokkos::create_mirror_view(host_memory_space(), x);
390 Kokkos::deep_copy(h_x, x);
391 auto h_b = Kokkos::create_mirror_view(host_memory_space(), b);
392 Kokkos::deep_copy(h_b, b);
393
394 typedef ArithTraits<value_type> arith_traits;
395 const ordinal_type m = h_A.NumRows(), k = h_b.extent(1);
396 double diff = 0, norm = 0;
397 for (ordinal_type p = 0; p < k; ++p) {
398 double diff_p = 0, norm_p = 0;
399 for (ordinal_type i = 0; i < m; ++i) {
400 value_type s = 0;
401 const ordinal_type jbeg = h_A.RowPtrBegin(i), jend = h_A.RowPtrEnd(i);
402 for (ordinal_type j = jbeg; j < jend; ++j) {
403 const ordinal_type col = h_A.Col(j);
404 s += h_A.Value(j) * h_x(col, p);
405 }
406 norm_p += arith_traits::real(h_b(i, p) * arith_traits::conj(h_b(i, p)));
407 diff_p += arith_traits::real((h_b(i, p) - s) * arith_traits::conj(h_b(i, p) - s));
408
409 norm += arith_traits::real(h_b(i, p) * arith_traits::conj(h_b(i, p)));
410 diff += arith_traits::real((h_b(i, p) - s) * arith_traits::conj(h_b(i, p) - s));
411 }
412 if (verbose)
413 std::cout << " * Relative residual norm(" << p << ") = " << sqrt(diff_p) << " / " << sqrt(norm_p) << " = " << sqrt(diff_p/norm_p) << std::endl;
414 }
415 if (verbose)
416 std::cout << " Relative residual norm = " << sqrt(diff) << " / " << sqrt(norm) << " = " << sqrt(diff/norm) << std::endl;
417 return sqrt(diff / norm);
418}
