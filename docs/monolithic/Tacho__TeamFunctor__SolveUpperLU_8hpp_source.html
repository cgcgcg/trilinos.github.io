75 KOKKOS_INLINE_FUNCTION void solve_var0(MemberType &member, const supernode_type &s, value_type *bptr) const {
76 using TrsvAlgoType = typename TrsvAlgorithm_Team::type;
77 using GemvAlgoType = typename GemvAlgorithm_Team::type;
78
79 const value_type minus_one(-1), one(1);
80 {
81 const ordinal_type m = s.m, n = s.n, n_m = n - m;
82 if (m > 0) {
83 value_type *uptr = s.u_buf;
84 // solve
85 const UnmanagedViewType<value_type_matrix> ATL(uptr, m, m);
86 uptr += m * m;
87 const ordinal_type offm = s.row_begin;
88 const auto tT = Kokkos::subview(_t, range_type(offm, offm + m), Kokkos::ALL());
89
90 if (n_m > 0) {
91 // update
92 const UnmanagedViewType<value_type_matrix> ATR(uptr, m, n_m); // uptr += m*n;
93 const UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
94 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, minus_one, ATR, bB, one, tT);
95 member.team_barrier();
96 }
97 Trsv<Uplo::Upper, Trans::NoTranspose, TrsvAlgoType>::invoke(member, Diag::NonUnit(), ATL, tT);
98 }
99 }

202 KOKKOS_INLINE_FUNCTION void update_var2(MemberType &member, const supernode_type &s, value_type *bptr) const {
203 {
204 const ordinal_type m = s.m, n = s.n;
205 UnmanagedViewType<value_type_matrix> b(bptr, n, _nrhs);
206 if (n > 0) {
207 const ordinal_type offm = s.row_begin;
208 const ordinal_type goffset = s.gid_col_begin + s.m;
209 Kokkos::parallel_for(
210 Kokkos::TeamVectorRange(member, n),
211 [&, m, goffset,
212 offm](const ordinal_type &i) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
213 for (ordinal_type j = 0; j < _nrhs; ++j) {
214 if (i < m) {
215 b(i, j) = _t(offm + i, j);
216 } else {
217 const ordinal_type row = _gid_colidx(i - m + goffset);
218 b(i, j) = _t(row, j);
219 }
220 }
221 });
222 }

236 const ordinal_type p = _pbeg + member.league_rank();
237 const ordinal_type sid = _level_sids(p);
238 const ordinal_type mode = _compute_mode(sid);
239 if (p < _pend && mode == 1) {
240 using solve_tag_type = SolveTag<Var>;
241
242 const supernode_type &s = _supernodes(sid);
243 value_type *bptr = _buf.data() + _buf_ptr(member.league_rank());
244 if (solve_tag_type::variant == 0) {
245 solve_var0(member, s, bptr);
246 } else if (solve_tag_type::variant == 1) {
247 solve_var1(member, s, bptr);
248 } else if (solve_tag_type::variant == 2 || solve_tag_type::variant == 3) {
249 solve_var2(member, s, bptr);

252 Kokkos::printf("Error: TeamFunctorSolveUpperChol::SolveTag, computing mode is not determined\n");
