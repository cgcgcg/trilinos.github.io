157template <typename ordinal_type, typename value_type, typename node_type>
158void
159Stokhos::CGDivisionExpansionStrategy<ordinal_type,value_type,node_type>::
160divide(Stokhos::OrthogPolyApprox<ordinal_type, value_type, node_type>& c,
161 const value_type& alpha,
162 const Stokhos::OrthogPolyApprox<ordinal_type, value_type, node_type>& a,
163 const Stokhos::OrthogPolyApprox<ordinal_type, value_type, node_type>& b,
164 const value_type& beta)
165{
166#ifdef STOKHOS_TEUCHOS_TIME_MONITOR
167 TEUCHOS_FUNC_TIME_MONITOR("Stokhos::CGDivisionStrategy::divide()");
168#endif
169
170
171 ordinal_type sz = basis->size();
172 ordinal_type pa = a.size();
173 ordinal_type pb = b.size();
174
175 ordinal_type pc;
176 if (pb > 1)
177 pc = sz;
178 else
179 pc = pa;
180 if (c.size() != pc)
181 c.resize(pc);
182
183 const value_type* ca = a.coeff();
184 const value_type* cb = b.coeff();
185
186
187 value_type* cc = c.coeff();
188
189 if (pb > 1) {
190 // Compute A
191 A->putScalar(0.0);
192 typename Cijk_type::k_iterator k_begin = Cijk->k_begin();
193 typename Cijk_type::k_iterator k_end = Cijk->k_end();
194
195 if (pb < Cijk->num_k())
196 k_end = Cijk->find_k(pb);
197 value_type cijk;
198 ordinal_type i,j,k;
199 for (typename Cijk_type::k_iterator k_it=k_begin; k_it!=k_end; ++k_it) {
200 k = index(k_it);
201 for (typename Cijk_type::kj_iterator j_it = Cijk->j_begin(k_it);
202 j_it != Cijk->j_end(k_it); ++j_it) {
203 j = index(j_it);
204 for (typename Cijk_type::kji_iterator i_it = Cijk->i_begin(j_it);
205 i_it != Cijk->i_end(j_it); ++i_it) {
206 i = index(i_it);
207 cijk = value(i_it);
208 (*A)(i,j) += cijk*cb[k];
209 }
210 }
211 }
212
213 // Compute B
214 B->putScalar(0.0);
215 for (i=0; i<pa; i++)
216 (*B)(i,0) = ca[i]*basis->norm_squared(i);
217
218 Teuchos::SerialDenseMatrix<ordinal_type,value_type> D(sz, 1);
219 //Equilibrate the linear system
220 if (equil == 1){
221 //Create diag mtx of max row entries
222 for (i=0; i<sz; i++){
223 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r(Teuchos::View, *A, 1, sz, i, 0);
224 D(i,0)=sqrt(r.normOne());
225 }
226
227
228 //Compute inv(D)*A*inv(D)
229 for (i=0; i<sz; i++){
230 for (j=0; j<sz; j++){
231 (*A)(i,j)=(*A)(i,j)/(D(i,0)*D(j,0));
232 }
233 }
234
235 //Scale b by inv(D)
236 for (i=0; i<sz; i++){
237 (*B)(i,0)=(*B)(i,0)/D(i,0);
238 }
239
240 }
241
242 if (linear == 1){
243 //Compute M, the linear matrix to be used in the preconditioner
244
245 pb = basis->dimension()+1;
246
247 M->putScalar(0.0);
248 if (pb < Cijk->num_k())
249 k_end = Cijk->find_k(pb);
250 for (typename Cijk_type::k_iterator k_it=k_begin; k_it!=k_end; ++k_it) {
251 k = index(k_it);
252 for ( typename Cijk_type::kj_iterator j_it = Cijk->j_begin(k_it);
253 j_it != Cijk->j_end(k_it); ++j_it) {
254 j = index(j_it);
255 for ( typename Cijk_type::kji_iterator i_it = Cijk->i_begin(j_it);
256 i_it != Cijk->i_end(j_it); ++i_it) {
257 i = index(i_it);
258 cijk = value(i_it);
259 (*M)(i,j) += cijk*cb[k];
260 }
261 }
262 }
263
264 //Scale M
265 if (equil == 1){
266 //Compute inv(D)*M*inv(D)
267 for (i=0; i<sz; i++){
268 for (j=0; j<sz; j++){
269 (*M)(i,j)=(*M)(i,j)/(D(i,0)*D(j,0));
270 }
271 }
272 }
273 CG(*A,*X,*B, max_it, tol, prec_iter, basis->order(), basis->dimension(), PrecNum, *M, diag);
274 }
275
276 else{
277
278 CG(*A,*X,*B, max_it, tol, prec_iter, basis->order(), basis->dimension(), PrecNum, *A, diag);
279 }
280
281 if (equil == 1 ) {
282 //Rescale X
283 for (i=0; i<sz; i++){
284 (*X)(i,0)=(*X)(i,0)/D(i,0);
285 }
286 }
287
288 // Compute c
289 for (i=0; i<pc; i++)
290 cc[i] = alpha*(*X)(i,0) + beta*cc[i];
291 }
292 else {
293 for (ordinal_type i=0; i<pc; i++)
294 cc[i] = alpha*ca[i]/cb[0] + beta*cc[i];
295 }
296}

299template <typename ordinal_type, typename value_type, typename node_type>
300ordinal_type
301Stokhos::CGDivisionExpansionStrategy<ordinal_type,value_type,node_type>::
302CG(const Teuchos::SerialDenseMatrix<ordinal_type, value_type> & aA,
303 Teuchos::SerialDenseMatrix<ordinal_type,value_type> & aX,
304 const Teuchos::SerialDenseMatrix<ordinal_type,value_type> & aB,
305 ordinal_type max_iter,
306 value_type tolerance,
307 ordinal_type aprec_iter,
308 ordinal_type order ,
309 ordinal_type m,
310 ordinal_type aPrecNum,
311 const Teuchos::SerialDenseMatrix<ordinal_type, value_type> & aM,
312 ordinal_type adiag)
313
314{
315 ordinal_type n = aA.numRows();
316 ordinal_type k=0;
317 value_type resid;
318 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Ax(n,1);
319 Ax.multiply(Teuchos::NO_TRANS,Teuchos::NO_TRANS,1.0, aA, aX, 0.0);
320 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r(Teuchos::Copy,aB);
321 r-=Ax;
322 resid=r.normFrobenius();
323 Teuchos::SerialDenseMatrix<ordinal_type, value_type> p(r);
324 Teuchos::SerialDenseMatrix<ordinal_type, value_type> rho(1,1);
325 Teuchos::SerialDenseMatrix<ordinal_type, value_type> oldrho(1,1);
326 Teuchos::SerialDenseMatrix<ordinal_type, value_type> pAp(1,1);
327 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Ap(n,1);
328 value_type b;
329 value_type a;
330 while (resid > tolerance && k < max_iter){
331 Teuchos::SerialDenseMatrix<ordinal_type, value_type> z(r);
332 //Solve Mz=r
333 if (aPrecNum != 0){
334 if (aPrecNum == 1){
335 Stokhos::DiagPreconditioner<ordinal_type, value_type> precond(aM);
336 precond.ApplyInverse(r,z,aprec_iter);
337 }
338 else if (aPrecNum == 2){
339 Stokhos::JacobiPreconditioner<ordinal_type, value_type> precond(aM);
340 precond.ApplyInverse(r,z,2);
341 }
342 else if (aPrecNum == 3){
343 Stokhos::GSPreconditioner<ordinal_type, value_type> precond(aM,0);
344 precond.ApplyInverse(r,z,1);
345 }
346 else if (aPrecNum == 4){
347 Stokhos::SchurPreconditioner<ordinal_type, value_type> precond(aM, order, m, adiag);
348 precond.ApplyInverse(r,z,aprec_iter);
349 }
350 }
351 rho.multiply(Teuchos::TRANS,Teuchos::NO_TRANS,1.0, r, z, 0.0);
352
353
354 if (k==0){
355 p.assign(z);
356 rho.multiply(Teuchos::TRANS, Teuchos::NO_TRANS, 1.0, r, z, 0.0);
357 }
358 else {
359 b=rho(0,0)/oldrho(0,0);
360 p.scale(b);
361 p+=z;
362 }
363 Ap.multiply(Teuchos::NO_TRANS,Teuchos::NO_TRANS,1.0, aA, p, 0.0);
364 pAp.multiply(Teuchos::TRANS,Teuchos::NO_TRANS,1.0, p, Ap, 0.0);
365 a=rho(0,0)/pAp(0,0);
366 Teuchos::SerialDenseMatrix<ordinal_type, value_type> scalep(p);
367 scalep.scale(a);
368 aX+=scalep;
369 Ap.scale(a);
370 r-=Ap;
371 oldrho.assign(rho);
372 resid=r.normFrobenius();
373 k++;
374 }
375
376 //std::cout << "iteration count " << k << std::endl;
377 return 0;
378}
