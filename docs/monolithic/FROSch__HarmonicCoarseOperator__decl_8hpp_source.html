199 KOKKOS_INLINE_FUNCTION
200 void operator()(const ScaleTag &, const int j) const {
201 scale(j) = STS::zero();
202 for (UN i = 0; i < numRows; i++) {
203 scale(j) += localMVBasis(i,j)*localMVBasis(i,j);
204 }
205 scale(j) = RTS::one()/RTS::sqrt(STS::abs(scale(j)));
206 }
207
208 KOKKOS_INLINE_FUNCTION
209 void operator()(const CountNnzTag &, const int i) const {
210 LO rowID = indicesGammaDofsAll[i];
211 GO iGlobal = localRepeatedMap.getGlobalElement(rowID);
212 LO iLocal = localRowMap.getLocalElement(iGlobal);
213 if (iLocal!=-1) { // This should prevent duplicate entries on the interface
214 for (UN j=0; j<numCols; j++) {
215 SC valueTmp=localMVBasis(i,j);
216 if (fabs(valueTmp)>tresholdDropping) {
217 Rowptr(iLocal+1) ++;
218 }
219 }
220 }
221 }
222
223
224 KOKKOS_INLINE_FUNCTION
225 void operator()(const TotalNnzTag&, const size_t i, UN &lsum) const {
226 lsum += Rowptr[i];
227 }
228
229 KOKKOS_INLINE_FUNCTION
230 void operator()(const FillNzEntriesTag &, const int i) const {
231 LO rowID = indicesGammaDofsAll[i];
232 GO iGlobal = localRepeatedMap.getGlobalElement(rowID);
233 LO iLocal = localRowMap.getLocalElement(iGlobal);
234 if (iLocal!=-1) { // This should prevent duplicate entries on the interface
235 UN nnz_i = Rowptr(iLocal);
236 for (UN j=0; j<numCols; j++) {
237 SC valueTmp=localMVBasis(i,j);
238 if (fabs(valueTmp)>tresholdDropping) {
239 Indices(nnz_i) = j; //localBasisMap.getGlobalElement(j);
240 Values(nnz_i) = valueTmp*scale(j);
241 nnz_i ++;
242 }
243 }
244 }
245 }
