33 template <class Int, class Entry, class Exe_Space>
34 BASKER_INLINE
35 int Basker<Int,Entry,Exe_Space>::nfactor_domain_error
36 (
37 INT_1DARRAY threads_start
38 )
39 {
40 Int nthread_remalloc = 0;
41 for(Int ti = 0; ti < num_threads; ti++)
42 {
43 //Note: jdb we can make this into a switch
44 if(thread_array(ti).error_type == BASKER_ERROR_NOERROR)
45 {
46 threads_start(ti) = BASKER_MAX_IDX;
47 continue;
48 } else if(thread_array(ti).error_type == BASKER_ERROR_SINGULAR)
49 {
50 if(Options.verbose == BASKER_TRUE)
51 {
52 std::cout << "ERROR THREAD: " << ti
53 << " DOMBLK SINGULAR: blk=" << thread_array(ti).error_blk
54 << std::endl;
55 }
56 return BASKER_ERROR;
57 } else if(thread_array(ti).error_type == BASKER_ERROR_NOMALLOC)
58 {
59 if(Options.verbose == BASKER_TRUE)
60 {
61 std::cout << "ERROR THREAD: " << ti
62 << " DOMBLK NOMALLOC : blk=" << thread_array(ti).error_blk
63 << std::endl;
64 }
65 return BASKER_ERROR;
66 } else if(thread_array(ti).error_type == BASKER_ERROR_REMALLOC)
67 {
68 BASKER_ASSERT(thread_array(ti).error_blk >= 0, "nfactor_dom_error error_blk");
69 if(Options.verbose == BASKER_TRUE)
70 {
71 std::cout << " > THREAD: " << ti
72 << " DOMBLK MALLOC : blk=" << thread_array(ti).error_blk
73 << " subblk=" << thread_array(ti).error_subblk
74 << " newsize=" << thread_array(ti).error_info
75 << std::endl << std::flush;
76 }
77
78 //If on diagonal, want to compare L and U
79 Int resize_L = BASKER_MAX_IDX;
80 Int resize_U = BASKER_MAX_IDX;
81 if(thread_array(ti).error_subblk != BASKER_MAX_IDX)
82 {
83 BASKER_ASSERT(thread_array(ti).error_info > 0, "L) newsize not big enough");
84 resize_L = thread_array(ti).error_info;
85
86 //if L is already bigger and U,
87 //We will want re size U as, well
88 if(thread_array(ti).error_subblk == 0)
89 {
90 Int blkcol = thread_array(ti).error_blk;
91 Int blkUrow = LU_size(blkcol)-1;
92 if(LL(blkcol)(0).nnz >=
93 LU(blkcol)(blkUrow).nnz)
94 {
95 resize_U = thread_array(ti).error_info;
96 }
97 }//if - a domain
98 }
99 //We don't care about the other way since,
100 //L is already checked before U.
101 if(thread_array(ti).error_subblk == -1)
102 {
103 resize_U = thread_array(ti).error_info;
104 }
105
106 //Resize L, if resize_L != -1 (meaning realloc-L is requested)
107 if(resize_L != BASKER_MAX_IDX)
108 {
109 if(Options.verbose == BASKER_TRUE)
110 {
111 std::cout << " ++ resize L( tid = " << ti << " ): new size = " << resize_L << std::endl << std::flush;
112 }
113 BASKER_MATRIX &L =
114 LL(thread_array(ti).error_blk)(thread_array(ti).error_subblk);
115 REALLOC_INT_1DARRAY(L.row_idx,
116 L.nnz,
117 resize_L);
118 REALLOC_ENTRY_1DARRAY(L.val,
119 L.nnz,
120 resize_L);
121 L.clear_pend();
122 if(Options.incomplete == BASKER_TRUE)
123 {
124 REALLOC_INT_1DARRAY(L.inc_lvl,
125 L.nnz,
126 resize_L);
127 }
128 L.mnnz = resize_L;
129 L.nnz = resize_L;
130 }
131
132 //Resize U, if resize_U != -1 (meaning realloc-U is requested)
133 if(resize_U != BASKER_MAX_IDX)
134 {
135 if(Options.verbose == BASKER_TRUE)
136 {
137 std::cout << " ++ resize U( tid = " << ti << " ): new size = " << resize_U << std::endl << std::flush;
138 }
139 BASKER_MATRIX &U =
140 LU(thread_array(ti).error_blk)(0);
141 REALLOC_INT_1DARRAY(U.row_idx,
142 U.nnz,
143 resize_U);
144 REALLOC_ENTRY_1DARRAY(U.val,
145 U.nnz,
146 resize_U);
147 U.mnnz = resize_U;
148 U.nnz = resize_U;
149 //Still need to clear pend
150 BASKER_MATRIX &L =
151 LL(thread_array(ti).error_blk)(0);
152 L.clear_pend();
153 }
154
155 //clean up workspace
156 //if(LL(thread_array(ti).error_blk)(0).w_fill ==
157 // BASKER_TRUE)
158 {
159 //Clear workspace, whole column
160 for(Int sb = 0;
161 sb < LL_size(thread_array(ti).error_blk);
162 sb++)
163 {
164 BASKER_MATRIX &SL =
165 LL(thread_array(ti).error_blk)(sb);
166 for(Int i = 0; i < SL.iws_size*SL.iws_mult; ++i)
167 {
168 SL.iws(i) = (Int) 0;
169 }
170 for(Int i = 0; i < SL.ews_size*SL.ews_mult; ++i)
171 {
172 SL.ews(i) = (Entry) 0;
173 }
174 if(sb == 0)
175 {
176 //Clear perm
177 Int scol_top = btf_tabs[btf_top_tabs_offset]; // the first column index of A
178 for(Int i = scol_top+SL.srow; i < scol_top+SL.srow+SL.nrow; ++i)
179 {
180 //printf( " x gperm(%d) = %d\n",i,BASKER_MAX_IDX );
181 gperm(i) = BASKER_MAX_IDX;
182 }
183
184 //Clear incomplete ws
185 if(Options.incomplete == BASKER_TRUE)
186 {
187 for(Int i = SL.srow; i < SL.srow+SL.nrow; ++i)
188 {
189 INC_LVL_TEMP(i) = BASKER_MAX_IDX;
190 }
191 }
192 }
193 }//for - sb (subblks)
194 }//if ws is filled
195
196 threads_start(ti) = thread_array(ti).error_blk;
197
198
199 //Reset
200 thread_array(ti).error_type = BASKER_ERROR_NOERROR;
201 thread_array(ti).error_blk = BASKER_MAX_IDX;
202 thread_array(ti).error_info = BASKER_MAX_IDX;
203
204 nthread_remalloc++;
205 }//if REMALLOC
206 }//for all threads
207
208 if(nthread_remalloc == 0)
209 return BASKER_SUCCESS;
210 else
211 return nthread_remalloc;
212
213 }//end nfactor_domain_error

218 template <class Int, class Entry, class Exe_Space>
219 BASKER_INLINE
220 int Basker<Int, Entry, Exe_Space>::nfactor_sep_error
221 (
222 INT_1DARRAY thread_start
223 )
224 {
225 Int nthread_remalloc = 0;
226 for(Int ti = 0; ti < num_threads; ti++)
227 {
228 //Note: jdb we can make this into a switch
229 if(thread_array(ti).error_type == BASKER_ERROR_NOERROR)
230 {
231 thread_start(ti) = BASKER_MAX_IDX;
232 continue;
233 }
234 else if(thread_array(ti).error_type == BASKER_ERROR_SINGULAR)
235 {
236 if(Options.verbose == BASKER_TRUE)
237 {
238 std::cout << "ERROR THREAD: " << ti
239 << " SEPBLK SINGULAR: blk=" << thread_array(ti).error_blk
240 << std::endl;
241 }
242 return BASKER_ERROR;
243 } else if(thread_array(ti).error_type == BASKER_ERROR_NOMALLOC)
244 {
245 if(Options.verbose == BASKER_TRUE)
246 {
247 std::cout << "ERROR THREADS: " << ti
248 << " SEPBLK NOMALLOC: blk=" << thread_array(ti).error_blk
249 << std::endl;
250 }
251 return BASKER_ERROR;
252 }//end if NOMALLOC
253
254 //Find lvl in sep error happend
255 Int error_sep_lvl = BASKER_MAX_IDX;
256 for(Int l = 1; l < tree.nlvls+1; l++)
257 {
258 if(thread_array(ti).error_blk == S(l)(ti))
259 {
260 error_sep_lvl = l;
261 break;
262 }
263 }
264
265 if(thread_array(ti).error_type == BASKER_ERROR_REMALLOC)
266 {
267 BASKER_ASSERT(thread_array(ti).error_blk >= 0, "nfactor_SEP_error error_blk");
268 if(Options.verbose == BASKER_TRUE)
269 {
270 std::cout << " > THREADS: " << ti
271 << " SEPBLK MALLOC: blk=" << thread_array(ti).error_blk
272 << " subblk=" << thread_array(ti).error_subblk
273 << " newsize=" << thread_array(ti).error_info
274 << std::endl;
275 std::cout << " > SEPLVL: " << error_sep_lvl << std::endl;
276 }
277
278 //If on diagonal, want to compare L and U
279 Int resize_L = BASKER_MAX_IDX;
280 Int resize_U = BASKER_MAX_IDX;
281 if(thread_array(ti).error_subblk <= -1)
282 {
283 resize_L = thread_array(ti).error_info;
284 if(Options.verbose == BASKER_TRUE)
285 {
286 std::cout << " ++ L size: " << resize_L << std::endl;
287 }
288 }
289 //We don't care about the other way since,
290 //L is already checked before U.
291 if(thread_array(ti).error_subblk > -1)
292 {
293 resize_U = thread_array(ti).error_info;
294 if(Options.verbose == BASKER_TRUE)
295 {
296 std::cout << " ++ U size: " << resize_U << std::endl;
297 }
298 }
299
300 //Resize L, if resize_L != -1 (meaning realloc-L is requested)
301 if(resize_L != BASKER_MAX_IDX)
302 {
303 const Int tsb = (-1*thread_array(ti).error_subblk)-1;
304 BASKER_MATRIX &L =
305 LL(thread_array(ti).error_blk)(tsb);
306 REALLOC_INT_1DARRAY(L.row_idx,
307 L.nnz,
308 resize_L);
309 REALLOC_ENTRY_1DARRAY(L.val,
310 L.nnz,
311 resize_L);
312 L.mnnz = resize_L;
313 L.nnz = resize_L;
314 L.clear_pend();
315 }
316
317 //Resize U, if resize_U != -1 (meaning realloc-U is requested)
318 if(resize_U != BASKER_MAX_IDX)
319 {
320 const Int tsb = thread_array(ti).error_subblk;
321 BASKER_MATRIX &U =
322 LU(thread_array(ti).error_blk)(tsb);
323 REALLOC_INT_1DARRAY(U.row_idx,
324 U.nnz,
325 resize_U);
326 REALLOC_ENTRY_1DARRAY(U.val,
327 U.nnz,
328 resize_U);
329 if(Options.incomplete == BASKER_TRUE)
330 {
331 REALLOC_INT_1DARRAY(U.inc_lvl,
332 U.nnz,
333 resize_U);
334 }
335 U.mnnz = resize_U;
336 U.nnz = resize_U;
337 }
338
339 //clean up workspace
340 //No nice way to do this since multiple threads
341 //Though this could be done in parallel in the future
342 for(Int p = 0; p < num_threads; p++)
343 {
344 Int blk = S(0)(p);
345 //if(LL(blk)(0).w_fill == BASKER_TRUE)
346 {
347 //Clear workspace, whole column
348 for(Int sb = 0; sb < LL_size(blk); sb++)
349 {
350 BASKER_MATRIX &SL = LL(blk)(sb);
351 for(Int i = 0; i < SL.iws_size*SL.iws_mult; ++i)
352 {
353 SL.iws(i) = (Int) 0;
354 }
355 for(Int i = 0; i < SL.ews_size*SL.ews_mult; ++i)
356 {
357 SL.ews(i) = (Entry) 0;
358 }
359 }//for - sb (subblks)
360 }//if ws is filled
361 }//for-other all threads
362
363 //Clear perm
364 Int scol_top = btf_tabs[btf_top_tabs_offset]; // the first column index of A
365 for(Int p = 0; p < num_threads; p++)
366 {
367 Int blk = S(error_sep_lvl)(p);
368 //if(LL(blk)(0).w_fill == BASKER_TRUE)
369 {
370 BASKER_MATRIX &TM = LL(blk)(0);
371 //printf( " > p=%d: scol_top = %d, scol = %d, ncol = %d\n",p,scol_top,TM.scol,TM.ncol );
372 for(Int i = scol_top + TM.scol; i < scol_top + (TM.scol+TM.ncol); i++)
373 {
374 //printf( " > gperm(%d) = %d\n",i,BASKER_MAX_IDX );
375 gperm(i) = BASKER_MAX_IDX;
376 }
377 }//if ws is filled
378 }//for-other all threads
379
380
381 //Note, will have to clear the perm in all sep blk in that level
382 //Clear permuation
383 BASKER_MATRIX &SL =
384 LL(thread_array(ti).error_blk)(0);
385 //printf( " + scol_top = %d, srow = %d, nrowl = %d\n",scol_top,SL.srow,SL.nrow );
386 for(Int i = scol_top + SL.srow; i < scol_top + (SL.srow+SL.nrow); i++)
387 {
388 //printf( " + gperm(%d) = %d\n",i,BASKER_MAX_IDX );
389 gperm(i) = BASKER_MAX_IDX;
390 }//for--to clear perm
391
392 thread_start(ti) = thread_array(ti).error_blk;
393
394 //Reset
395 thread_array(ti).error_type = BASKER_ERROR_NOERROR;
396 thread_array(ti).error_blk = BASKER_MAX_IDX;
397 thread_array(ti).error_info = BASKER_MAX_IDX;
398
399 for(Int i = 0; i < num_threads; i++)
400 {
401 basker_barrier.ExitSet(i,BASKER_FALSE);
402 }
403
404 nthread_remalloc++;
405
406 }//if REMALLOC
407
408 //Reset Inc vector
409 if(Options.inc_lvl == BASKER_TRUE)
410 {
411 //for(Int i = 0; i < INC_LVL_TEMP.extent(0); i++) //NDE - warning: comparison s and us
412 for(Int i = 0; i < static_cast<Int>( INC_LVL_TEMP.extent(0) ); i++)
413 {
414 INC_LVL_TEMP(i) = BASKER_MAX_IDX;
415 }
416 }
417 }//for all threads
418
419 if(nthread_remalloc == 0)
420 {
421 return BASKER_SUCCESS;
422 }
423 else
424 {
425 return nthread_remalloc;
426 }
427
428 //Should never be here
429 BASKER_ASSERT(0==1, "nfactor_sep_error, should never");
430 return BASKER_SUCCESS;
431
432 }//end nfactor_sep_error

436 template <class Int, class Entry, class Exe_Space>
437 BASKER_INLINE
438 int Basker<Int,Entry,Exe_Space>::nfactor_diag_error
439 (
440 INT_1DARRAY thread_start_top,
441 INT_1DARRAY thread_start
442 )
443 {
444 Entry zero (0.0);
445 Int nthread_remalloc = 0;
446 Int btab = btf_tabs_offset;
447 for(Int ti = 0; ti < num_threads; ti++)
448 {
449 Int c = thread_array(ti).error_blk;
450 //Note: jdb we can make this into a switch
451 if(thread_array(ti).error_type == BASKER_ERROR_NOERROR)
452 {
453 if (c >= btab) {
454 thread_start(ti) = BASKER_MAX_IDX;
455 } else {
456 thread_start_top(ti) = BASKER_MAX_IDX;
457 }
458 continue;
459 }//end if NOERROR
460
461 if(thread_array(ti).error_type == BASKER_ERROR_SINGULAR)
462 {
463 if(Options.verbose == BASKER_TRUE)
464 {
465 std::cout << "ERROR_THREAD: " << ti
466 << " DIAGBLK SINGULAR " << c
467 << std::endl;
468 }
469 return BASKER_ERROR;
470 }//end if SINGULAR
471
472 if(thread_array(ti).error_type == BASKER_ERROR_NOMALLOC)
473 {
474 std::cout << "ERROR_THREADS: " << ti
475 << " DIAGBLK NOMALLOC blk=" << c
476 << std::endl;
477 return BASKER_ERROR;
478 }//end if NOMALLOC
479
480 if(thread_array(ti).error_type == BASKER_ERROR_REMALLOC)
481 {
482 Int liwork = thread_array(ti).iws_size*thread_array(ti).iws_mult;
483 Int lework = thread_array(ti).ews_size*thread_array(ti).ews_mult;
484 BASKER_ASSERT(c >= 0, "nfactor_diag_error error_blk");
485 if(Options.verbose == BASKER_TRUE)
486 {
487 std::cout << " > THREADS: " << ti
488 << " DIAGBLK MALLOC blk=" << c
489 << " newsize=" << thread_array(ti).error_info
490 << " for both L( " << c << " ) and U( " << c << " )"
491 << std::endl;
492
493 //Clean the workspace
494 std::cout << "clean workspace of size "
495 << liwork
496 << ", " << lework
497 << std::endl;
498 }
499
500 for(Int i = 0; i < liwork; i++)
501 {
502 thread_array(ti).iws(i) = (Int) 0;
503 }
504 for(Int i = 0; i < lework; i++)
505 {
506 thread_array(ti).ews(i) = zero;
507 }
508
509 //Resize L
510 BASKER_MATRIX &L = (c >= btab ? LBTF(c-btab) : L_D(c));
511 L.clear_pend();
512 REALLOC_INT_1DARRAY(L.row_idx,
513 L.nnz,
514 thread_array(ti).error_info);
515 REALLOC_ENTRY_1DARRAY(L.val,
516 L.nnz,
517 thread_array(ti).error_info);
518 L.mnnz = thread_array(ti).error_info;
519 L.nnz = thread_array(ti).error_info;
520 for(Int i = 0; i < L.ncol; i++)
521 {
522 L.col_ptr(i) = 0;
523 }
524 // srow for D & c are global..
525 for(Int i = L.srow; i < (L.srow+L.nrow); i++)
526 {
527 gperm(i) = BASKER_MAX_IDX;
528 }
529
530 //Resize U
531 BASKER_MATRIX &U = (c >= btab ? UBTF(c-btab) : U_D(c));
532 REALLOC_INT_1DARRAY(U.row_idx,
533 U.nnz,
534 thread_array(ti).error_info);
535 REALLOC_ENTRY_1DARRAY(U.val,
536 U.nnz,
537 thread_array(ti).error_info);
538 U.mnnz = thread_array(ti).error_info;
539 U.nnz = thread_array(ti).error_info;
540 for(Int i = 0; i < U.ncol; i++)
541 {
542 U.col_ptr(i) = 0;
543 }
544
545 // set the thread start (skip the diagonal blocks that have been factored without error)
546 if(Options.verbose == BASKER_TRUE)
547 {
548 std::cout << "Setting thread start(" << ti << ") to be "
549 << c
550 << std::endl;
551 }
552 if (c >= btab) {
553 thread_start(ti) = c;
554 } else {
555 thread_start_top(ti) = c;
556 }
557
558 //Reset
559 thread_array(ti).error_type = BASKER_ERROR_NOERROR;
560 thread_array(ti).error_blk = BASKER_MAX_IDX;
561 thread_array(ti).error_info = BASKER_MAX_IDX;
562
563 nthread_remalloc++;
564
565 }//if REMALLOC
566
567 }//for all threads
568
569 if(nthread_remalloc == 0)
570 {
571 return BASKER_SUCCESS;
572 }
573 else
574 {
575 return nthread_remalloc;
576 }
577
578 //Should never be here
579 BASKER_ASSERT(0==1, "nfactor_diag_error, should never");
580 return BASKER_SUCCESS;
581 }//end nfactor_diag_error
