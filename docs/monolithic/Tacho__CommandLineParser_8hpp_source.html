50 template <typename T> void set_option(const std::string opt, const std::string desc, T *val) {
51 const auto in = Option<T>(desc, *val);

54 void print_option(const std::string argv0) {
55 std::cout << "Usage: " << argv0 << " [options]\n";
56 std::cout << " options:\n";
57
58 const std::string prefix = "--";
59 for (auto it = _map.begin(); it != _map.end(); ++it) {
60 auto key = prefix + it->first;
61 auto val = it->second;
62 std::cout << std::left << " " << std::setw(30) << key << std::setw(10) << std::get<1>(val) << std::get<0>(val)
63 << "\n";
64 if (std::get<1>(val) == "bool") {
65 if (it->first != "help" && it->first != "echo-command-line")
66 std::cout << std::setw(42) << " "
67 << "(default: " << key << "=" << (std::get<2>(val) == "0" ? "false" : "true") << ")\n";
68 } else {
69 std::cout << std::setw(42) << " "
70 << "(default: " << key << "=" << std::get<2>(val) << ")\n";
71 }
72 }
73 std::cout << "Description:\n";
74 std::cout << " " << _desc << "\n\n";

76 bool parse(int argc, char **argv) {
77 bool help = false, echo = false;
78 this->set_option<bool>("help", "Print this help message", &help);
79 this->set_option<bool>("echo-command-line", "Echo the command-line but continue as normal", &echo);
80
81 // check help
82 for (int i = 1; i < argc; ++i) {
83 std::string s = argv[i];
84 if (s == "--help")
85 help = true;
86 if (s == "--echo-command-line")
87 echo = true;
88 }
89
90 if (help) {
91 print_option(argv[0]);
92 } else {
93 // parse
94 for (int i = 1; i < argc; ++i) {
95 std::string s = argv[i];
96 for (auto it = _map.begin(); it != _map.end(); ++it) {
97 const auto t = it->second;
98 // find option starting with --
99 if (s.find("--") != std::string::npos && s.length() != 2) {
100 std::string desc = std::get<0>(t);
101 std::string type = std::get<1>(t);
102
103 size_t pos = s.find("=");
104 if (pos != std::string::npos) {
105 // --opt=val
106 std::string key = s.substr(2, pos - 2);
107 if (key == it->first) {
108 std::string sval = s.substr(pos + 1, s.length());
109 void *tval = std::get<3>(t);
110 if (tval != NULL) {
111 if (type == "int") {
112 *((int *)tval) = atoi(sval.c_str());
113 } else if (type == "string") {
114 *((std::string *)tval) = sval;
115 } else if (type == "bool") {
116 *((bool *)tval) = (sval == "true");
117 } else {
118 std::cout << " int somethng wrong\n";
119 }
120 _map[it->first] = std::make_tuple(desc, type, sval, (void *)NULL);
121 }
122 }
123 } else {
124 // --opt (bool)
125 std::string key = s.substr(2, s.length());
126 if (key == it->first) {
127 void *tval = std::get<3>(t);
128 if (tval != NULL) {
129 if (type == "bool") {
130 *((bool *)tval) = true;
131 } else {
132 std::cout << " bool somethng wrong\n";
133 }
134 _map[it->first] = std::make_tuple(desc, type, "1", (void *)NULL);
135 }
136 }
137 }
138 }
139 }
140 }
141
142 // print out unused options
143 if (echo) {
144 // echo command
145 std::cout << "Echoing the command-line:\n\n";
146 for (int i = 0; i < argc; ++i)
147 std::cout << argv[i] << " ";
148 std::cout << "\n\n";
149
150 // used options
151 std::cout << "Used options:\n\n";
152 for (auto it = _map.begin(); it != _map.end(); ++it) {
153 std::string key = it->first;
154 std::string type = std::get<1>(it->second);
155 std::string val = std::get<2>(it->second);
156 bool used = std::get<3>(it->second) == NULL;
157 if (used) {
158 std::cout << " ";
159 if (type == "bool")
160 std::cout << "--" << key << "\n";
161 else
162 std::cout << "--" << key << "=" << val << "\n";
163 }
164 }
165 std::cout << "\n";
166
167 // not used options
168 std::cout << "Not used options:\n\n";
169 for (auto it = _map.begin(); it != _map.end(); ++it) {
170 std::string key = it->first;
171 std::string type = std::get<1>(it->second);
172 std::string val = std::get<2>(it->second);
173 bool used = std::get<3>(it->second) == NULL;
174 if (!used) {
175 std::cout << " ";
176 if (type == "bool")
177 std::cout << "--" << key << "\n";
178 else
179 std::cout << "--" << key << "=" << val << " (default)\n";
180 }
181 }
182 std::cout << "\n";
183 }
184 }
185 return help;
186 }
