60 public:
61 AdelusHandle( const int comm_id_,
62 MPI_Comm comm_,
63 const int matrix_size_,
64 const int num_procsr_,
65 const int num_rhs_,
66 const int blksz_ = 128 )
67 : comm_id(comm_id_),
68 comm(comm_),
69 nrows_matrix(matrix_size_),
70 ncols_matrix(matrix_size_),
71 nprocs_row(num_procsr_),
72 nrhs(num_rhs_),
73 blksz(blksz_) {
74 // Determine who I am (myrank) and the total number of processes (nprocs_cube)
75 MPI_Comm_size(comm, &nprocs_cube);
76 MPI_Comm_rank(comm, &myrank);
77 nprocs_col = nprocs_cube/nprocs_row;
78
79 // Set up communicators for rows and columns
80 mycol = myrank % nprocs_row;
81 myrow = myrank / nprocs_row;
82
83 MPI_Comm_split(comm, myrow, mycol, &row_comm);
84
85 MPI_Comm_split(comm, mycol, myrow, &col_comm);
86
87 // Distribution for the matrix on myrank
88 my_first_col = myrank % nprocs_row;
89 my_first_row = myrank / nprocs_row;
90
91 my_rows = nrows_matrix / nprocs_col;
92 if (my_first_row < nrows_matrix % nprocs_col) my_rows++;
93 my_cols = ncols_matrix / nprocs_row;
94 if (my_first_col < ncols_matrix % nprocs_row) my_cols++;
95
96 // Distribution for the rhs on myrank
97 my_rhs = nrhs / nprocs_row;
98 if (my_first_col < nrhs % nprocs_row) my_rhs++;
99 }
