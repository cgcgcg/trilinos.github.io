30/* Default traits */
31/* These traits work for the following couples of (Map,Matrix):
32 - Map = Tpetra::Map<...>, and Matrix = Tpetra::CrsMatrix<...>
33 - Map = Xpetra::TpetraMap<...> and Matrix = Xpetra::TpetraCrsMatrix<...>
34*/
35template <class Map, class Matrix>
36class MatrixTraits {
37 using scalar_type = typename Matrix::scalar_type;
38 using local_ordinal_type = typename Matrix::local_ordinal_type;
39 using global_ordinal_type = typename Matrix::global_ordinal_type;
40 using node_type = typename Matrix::node_type;
41 using local_matrix_type = typename ::Tpetra::CrsMatrix<scalar_type, local_ordinal_type, global_ordinal_type, node_type>::local_matrix_device_type;
42
43 public:
44 static Teuchos::RCP<Matrix> Build(const Teuchos::RCP<const Map> &rowMap, size_t maxNumEntriesPerRow) { return Teuchos::rcp(new Matrix(rowMap, maxNumEntriesPerRow)); };
45 static Teuchos::RCP<Matrix> Build(local_matrix_type &lclMatrix, const Teuchos::RCP<const Map> &rowMap, const Teuchos::RCP<const Map> &colMap, const Teuchos::RCP<const Map> &domainMap, const Teuchos::RCP<const Map> &rangeMap) { return Teuchos::rcp(new Matrix(lclMatrix, rowMap, colMap, domainMap, rangeMap)); };
46};
