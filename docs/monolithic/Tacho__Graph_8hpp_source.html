41 template <typename SizeTypeArray, typename OrdinalTypeArray>
42 inline void init(const ordinal_type m, const size_type nnz, const SizeTypeArray &ap, const OrdinalTypeArray &aj) {
43 _rptr = size_type_array(do_not_initialize_tag("Graph::rptr"), m + 1);
44 _cidx = ordinal_type_array(do_not_initialize_tag("Graph::cidx"), nnz);
45
46 _m = m;
47 _nnz = 0;
48 for (ordinal_type i = 0; i < _m; ++i) {
49 const size_type jbeg = ap(i), jend = ap(i + 1);
50 _rptr(i) = _nnz;
51 for (size_type j = jbeg; j < jend; ++j) {
52 // skip diagonal
53 const ordinal_type col = aj(j);
54 if (i != col)
55 _cidx(_nnz++) = col;
56 }
57 }
58 _rptr(_m) = _nnz;
59 }
