54 template<int OuterViewRank,typename InnerViewType,typename MemorySpace>
55 class ViewOfViews {
56
57 public:
58 // Layout of the outer view doesn't matter for performance so we
59 // use a raw Kokkos::View instead of PHX::View. The inner views are
60 // what is important for performance.
61 using OuterDataType = typename PHX::v_of_v_utils::add_pointer<InnerViewType,OuterViewRank>::type;
62 using OuterViewType = Kokkos::View<OuterDataType,MemorySpace>;
63
64 private:
65 typename OuterViewType::host_mirror_type view_host_;
66 OuterViewType view_device_;
67 bool device_view_is_synced_;
68
69 public:
70 template<typename... Extents>
71 ViewOfViews(const std::string name,
72 Extents... extents)
73 : view_host_(name,extents...),
74 view_device_(name,extents...),
75 device_view_is_synced_(false)
76 {
77 // Inner view must be unmanaged if the outerview is not using UVM!
78 static_assert(InnerViewType::memory_traits::is_unmanaged,
79 "ERROR: PHX::ViewOfViews - Inner view must be unmanaged!");
80 }
81
82 ~ViewOfViews()
83 {
84 // Make sure there is not another object pointing to device view
85 // since the host view will delete the inner views on exit.
86 if (view_device_.impl_track().use_count() != 1)
87 Kokkos::abort("\n ERROR - PHX::ViewOfViews - please free all instances of device ViewOfView \n before deleting the host ViewOfView!\n\n");
88 }
89
90 template<typename... Indices>
91 void addView(InnerViewType v,Indices... i)
92 {
93 view_host_(i...) = v;
94 device_view_is_synced_ = false;
95 }
96
97 void syncHostToDevice()
98 {
99 Kokkos::deep_copy(view_device_,view_host_);
100 device_view_is_synced_ = true;
101 }
102
103 auto getViewHost()
104 {
105 return view_host_;
106 }
107
108 auto getViewDevice()
109 {
110 TEUCHOS_ASSERT(device_view_is_synced_);
111 return view_device_;
112 }

291 public:
292 // Layout of the outer view doesn't matter for performance so we
293 // use a raw Kokkos::View instead of PHX::View. The inner views are
294 // what is important for performance.
295 using OuterDataType = typename PHX::v_of_v_utils::add_pointer<InnerViewType,OuterViewRank>::type;
296 using OuterViewType = Kokkos::View<OuterDataType,OuterViewProps...>;
297
298 private:
299 // Inner views are mananged - used to prevent early deletion
300 typename OuterViewType::host_mirror_type view_host_;
301 // Inner views are unmanaged by runtime construction with pointer
302 // (avoids template parameter). Used to correctly initialize outer
303 // device view on device.
304 typename OuterViewType::host_mirror_type view_host_unmanaged_;
305 // Device view
306 OuterViewType view_device_;
307 // True if the host view has not been synced to device
308 bool device_view_is_synced_;
309 // True if the outer view has been initialized
310 bool is_initialized_;
311 // Use count of device view after initialization. This changes based on whether the view_device_ is accessible to host space. If not accessible, the use_count_ is 1. If it is accessible, the value is 2 due to using create_mirror_view in initialization if view_host_unmanaged_.
312 int use_count_;
313 // A safety check. If true, this makes sure there are no external references to the device view of views.
314 bool check_use_count_;
315
316 public:
318 template<typename... Extents>
319 ViewOfViews3(const std::string name,Extents... extents)
320 : view_host_(name,extents...),
321 view_device_(name,extents...),
322 device_view_is_synced_(false),
323 is_initialized_(true),
324 use_count_(0),
325 check_use_count_(true)
326 {
327 view_host_unmanaged_ = Kokkos::create_mirror_view(view_device_);
328 use_count_ = view_device_.impl_track().use_count();
329 }

361 // in objects that contain ViewOfViews3 that are copied to device.

363 ~ViewOfViews3()
364 {
365 // Make sure there is not another object pointing to the device
366 // view if the host view is about to be deleted. The host view
367 // may delete the inner views if it is the last owner.

372 }
373 }
374 ))

418 template<typename... Indices>
419 void addView(InnerViewType v,Indices... i)
420 {
421 static_assert(sizeof...(Indices)==OuterViewRank,
422 "Error: PHX::ViewOfViews3::addView() - the number of indices must match the outer view rank!");
423
424 TEUCHOS_ASSERT(is_initialized_);
425
426 // Store the managed version so inner views don't get deleted.
427 view_host_(i...) = v;
428
429 // Store a runtime unmanaged view for deep_copy to
430 // device. Unmanaged is required to prevent double deletion on
431 // device. For FAD types, we need to adjust the layout to
432 // account for the derivative array. The layout() method reutrns
433 // the non-fad adjusted layout.
434 if (Sacado::IsADType<typename InnerViewType::value_type>::value) {
435 auto layout = v.layout();
436 layout.dimension[InnerViewType::rank] = Kokkos::dimension_scalar(v);
437 view_host_unmanaged_(i...) = InnerViewType(v.data(),layout);
438 }
439 else
440 view_host_unmanaged_(i...) = InnerViewType(v.data(),v.layout());
441
442 device_view_is_synced_ = false;
443 }

453 device_view_is_synced_ = true;

511 template<typename InnerViewDataType,typename... InnerProps,typename... OuterProps>
512 auto createHostHostViewOfViews(const Kokkos::View<Kokkos::View<InnerViewDataType,InnerProps...>*,OuterProps...>& v_of_v) {
513 // Host outer view pointing to device inner views
514 auto host_device = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),v_of_v);
515
516 // Host outer view point to host inner views
517 using HostInnerViewType = decltype(Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),host_device(0)));
518 using HostDeviceMirrorType = decltype(host_device);
519 Kokkos::View<HostInnerViewType *,
520 typename HostDeviceMirrorType::host_mirror_type::array_layout,
521 typename HostDeviceMirrorType::host_mirror_type::device_type> host_host(host_device.label(),
522 host_device.extent(0));
523
524 for (std::size_t i=0; i < host_device.extent(0); ++i) {
525 auto tmp = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),host_device(i));
526 // HostInnerViewType t = tmp;
527 host_host(i) = tmp;
528 }
529 return host_host;
530 }
