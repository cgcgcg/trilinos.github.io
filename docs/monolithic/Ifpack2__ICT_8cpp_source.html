110int Ifpack2_ICT::Compute() {
111 if (!IsInitialized())
112 IFPACK2_CHK_ERR(Initialize());
113
114 Time_.ResetStartTime();
115 IsComputed_ = false;
116
117 NumMyRows_ = A_.NumMyRows();
118 int Length = A_.MaxNumEntries();
119 vector<int> RowIndices(Length);
120 vector<double> RowValues(Length);
121
122 bool distributed = (Comm().NumProc() > 1) ? true : false;
123
124 if (distributed) {
125 SerialComm_ = Teuchos::rcp(new Tpetra_SerialComm);
126 SerialMap_ = Teuchos::rcp(new Tpetra_Map(NumMyRows_, 0, *SerialComm_));
127 assert(SerialComm_.get() != 0);
128 assert(SerialMap_.get() != 0);
129 } else
130 SerialMap_ = Teuchos::rcp(const_cast<Tpetra_Map*>(&A_.RowMatrixRowMap()), false);
131
132 int RowNnz;
133 double flops = 0.0;
134
135 H_ = Teuchos::rcp(new Tpetra_CrsMatrix(Copy, *SerialMap_, 0));
136 if (H_.get() == 0)
137 IFPACK2_CHK_ERR(-5); // memory allocation error
138
139 // get A(0,0) element and insert it (after sqrt)
140 IFPACK2_CHK_ERR(A_.ExtractMyRowCopy(0, Length, RowNnz,
141 &RowValues[0], &RowIndices[0]));
142
143 // skip off-processor elements
144 if (distributed) {
145 int count = 0;
146 for (int i = 0; i < RowNnz; ++i) {
147 if (RowIndices[i] < NumMyRows_) {
148 RowIndices[count] = RowIndices[i];
149 RowValues[count] = RowValues[i];
150 ++count;
151 } else
152 continue;
153 }
154 RowNnz = count;
155 }
156
157 // modify diagonal
158 double diag_val = 0.0;
159 for (int i = 0; i < RowNnz; ++i) {
160 if (RowIndices[i] == 0) {
161 double& v = RowValues[i];
162 diag_val = AbsoluteThreshold() * EPETRA_SGN(v) +
163 RelativeThreshold() * v;
164 break;
165 }
166 }
167
168 diag_val = sqrt(diag_val);
169 int diag_idx = 0;
170 EPETRA_CHK_ERR(H_->InsertGlobalValues(0, 1, &diag_val, &diag_idx));
171
172 int oldSize = RowNnz;
173
174 // this is a bit magic
175 int hash_size = 1024;
176 while (hash_size < (int)(A_.MaxNumEntries() * LevelOfFill()))
177 hash_size *= 2;
178 Ifpack2_HashTable Hash(hash_size - 1, 1);
179
180 // start factorization for line 1
181 for (int row_i = 1; row_i < NumMyRows_; ++row_i) {
182 // get row `row_i' of the matrix
183 IFPACK2_CHK_ERR(A_.ExtractMyRowCopy(row_i, Length, RowNnz,
184 &RowValues[0], &RowIndices[0]));
185
186 // skip off-processor elements
187 if (distributed) {
188 int count = 0;
189 for (int i = 0; i < RowNnz; ++i) {
190 if (RowIndices[i] < NumMyRows_) {
191 RowIndices[count] = RowIndices[i];
192 RowValues[count] = RowValues[i];
193 ++count;
194 } else
195 continue;
196 }
197 RowNnz = count;
198 }
199
200 // number of nonzeros in this row are defined as the nonzeros
201 // of the matrix, plus the level of fill
202 int LOF = (int)(LevelOfFill() * RowNnz);
203 if (LOF == 0) LOF = 1;
204
205 // convert line `row_i' into hash for fast access
206 Hash.reset();
207
208 double h_ii = 0.0;
209 for (int i = 0; i < RowNnz; ++i) {
210 if (RowIndices[i] == row_i) {
211 double& v = RowValues[i];
212 h_ii = AbsoluteThreshold() * EPETRA_SGN(v) + RelativeThreshold() * v;
213 } else if (RowIndices[i] < row_i) {
214 Hash.set(RowIndices[i], RowValues[i], true);
215 }
216 }
217
218 // form element (row_i, col_j)
219 // I start from the first row that has a nonzero column
220 // index in row_i.
221 for (int col_j = RowIndices[0]; col_j < row_i; ++col_j) {
222 double h_ij = 0.0, h_jj = 0.0;
223 // note: get() returns 0.0 if col_j is not found
224 h_ij = Hash.get(col_j);
225
226 // get pointers to row `col_j'
227 int* ColIndices;
228 double* ColValues;
229 int ColNnz;
230 H_->ExtractGlobalRowView(col_j, ColNnz, ColValues, ColIndices);
231
232 for (int k = 0; k < ColNnz; ++k) {
233 int col_k = ColIndices[k];
234
235 if (col_k == col_j)
236 h_jj = ColValues[k];
237 else {
238 double xxx = Hash.get(col_k);
239 if (xxx != 0.0) {
240 h_ij -= ColValues[k] * xxx;
241 flops += 2.0;
242 }
243 }
244 }
245
246 h_ij /= h_jj;
247
248 if (std::abs(h_ij) > DropTolerance_) {
249 Hash.set(col_j, h_ij);
250 }
251
252 // only approx
253 ComputeFlops_ += 2.0 * flops + 1.0;
254 }
255
256 int size = Hash.getNumEntries();
257
258 vector<double> AbsRow(size);
259 int count = 0;
260
261 // +1 because I use the extra position for diagonal in insert
262 vector<int> keys(size + 1);
263 vector<double> values(size + 1);
264
265 Hash.arrayify(&keys[0], &values[0]);
266
267 for (int i = 0; i < size; ++i) {
268 AbsRow[i] = std::abs(values[i]);
269 }
270 count = size;
271
272 double cutoff = 0.0;
273 if (count > LOF) {
274 nth_element(AbsRow.begin(), AbsRow.begin() + LOF, AbsRow.begin() + count,
275 greater<double>());
276 cutoff = AbsRow[LOF];
277 }
278
279 for (int i = 0; i < size; ++i) {
280 h_ii -= values[i] * values[i];
281 }
282
283 if (h_ii < 0.0) h_ii = 1e-12;
284 ;
285
286 h_ii = sqrt(h_ii);
287
288 // only approx, + 1 == sqrt
289 ComputeFlops_ += 2 * size + 1;
290
291 double DiscardedElements = 0.0;
292
293 count = 0;
294 for (int i = 0; i < size; ++i) {
295 if (std::abs(values[i]) > cutoff) {
296 values[count] = values[i];
297 keys[count] = keys[i];
298 ++count;
299 } else
300 DiscardedElements += values[i];
301 }
302
303 if (RelaxValue() != 0.0) {
304 DiscardedElements *= RelaxValue();
305 h_ii += DiscardedElements;
306 }
307
308 values[count] = h_ii;
309 keys[count] = row_i;
310 ++count;
311
312 H_->InsertGlobalValues(row_i, count, &(values[0]), (int*)&(keys[0]));
313
314 oldSize = size;
315 }
316
317 IFPACK2_CHK_ERR(H_->FillComplete());
318
319#if 0
320 // to check the complete factorization
321 Tpetra_Vector LHS(Matrix().RowMatrixRowMap());
322 Tpetra_Vector RHS1(Matrix().RowMatrixRowMap());
323 Tpetra_Vector RHS2(Matrix().RowMatrixRowMap());
324 Tpetra_Vector RHS3(Matrix().RowMatrixRowMap());
325 LHS.Random();
326
327 Matrix().Multiply(false,LHS,RHS1);
328 H_->Multiply(true,LHS,RHS2);
329 H_->Multiply(false,RHS2,RHS3);
330
331 RHS1.Update(-1.0, RHS3, 1.0);
332 cout << endl;
333 cout << RHS1;
334#endif
335 int MyNonzeros = H_->NumGlobalNonzeros();
336 Comm().SumAll(&MyNonzeros, &GlobalNonzeros_, 1);
337
338 IsComputed_ = true;
339 double TotalFlops; // sum across all the processors
340 A_.Comm().SumAll(&flops, &TotalFlops, 1);
341 ComputeFlops_ += TotalFlops;
342 ++NumCompute_;
343 ComputeTime_ += Time_.ElapsedTime();
344
345 return (0);
346}

349int Ifpack2_ICT::ApplyInverse(const Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& X,
350 Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& Y) const {
351 if (!IsComputed())
352 IFPACK2_CHK_ERR(-3); // compute preconditioner first
353
354 if (X.NumVectors() != Y.NumVectors())
355 IFPACK2_CHK_ERR(-2); // Return error: X and Y not the same size
356
357 Time_.ResetStartTime();
358
359 // AztecOO gives X and Y pointing to the same memory location,
360 // need to create an auxiliary vector, Xcopy
361 Teuchos::RCP<const Tpetra_MultiVector> Xcopy;
362 if (X.Pointers()[0] == Y.Pointers()[0])
363 Xcopy = Teuchos::rcp(new Tpetra_MultiVector(X));
364 else
365 Xcopy = Teuchos::rcp(&X, false);
366
367 // NOTE: H_ is based on SerialMap_, while Xcopy is based
368 // on A.Map()... which are in general different. However, Solve()
369 // does not seem to care... which is fine with me.
370 //
371 EPETRA_CHK_ERR(H_->Solve(false, false, false, *Xcopy, Y));
372 EPETRA_CHK_ERR(H_->Solve(false, true, false, Y, Y));
373
374 // these are global flop count
375 ApplyInverseFlops_ += 4.0 * GlobalNonzeros_;
376
377 ++NumApplyInverse_;
378 ApplyInverseTime_ += Time_.ElapsedTime();
379
380 return (0);
381}

390double Ifpack2_ICT::Condest(const Ifpack2_CondestType CT,
391 const int MaxIters, const double Tol,
392 Tpetra_RowMatrix* Matrix_in) {
393 if (!IsComputed()) // cannot compute right now
394 return (-1.0);
395
396 // NOTE: this is computing the *local* condest
397 if (Condest_ == -1.0)
398 Condest_ = Ifpack2_Condest(*this, CT, MaxIters, Tol, Matrix_in);
399
400 return (Condest_);
401}

435 os << "ApplyInverse() " << std::setw(5) << NumApplyInverse()
436 << " " << std::setw(15) << ApplyInverseTime()

440 else
441 os << " " << std::setw(15) << 0.0 << endl;
442 os << "================================================================================" << endl;
443 os << endl;
444 }
445
446 return (os);
447}
