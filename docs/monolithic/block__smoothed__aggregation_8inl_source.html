68template <typename MatrixType>
69struct Dinv_A : public cusp::linear_operator<typename MatrixType::value_type, typename MatrixType::memory_space>
70{
71 const MatrixType& A;
72 const cusp::precond::diagonal<typename MatrixType::value_type, typename MatrixType::memory_space> Dinv;
73
74 Dinv_A(const MatrixType& A)
75 : A(A), Dinv(A),
76 cusp::linear_operator<typename MatrixType::value_type, typename MatrixType::memory_space>(A.num_rows, A.num_cols, A.num_entries + A.num_rows)
77 {}
78
79 template <typename Array1, typename Array2>
80 void operator()(const Array1& x, Array2& y) const
81 {
82 cusp::multiply(A,x,y);
83 cusp::multiply(Dinv,y,y);
84 }
85};

90 typedef typename MatrixType::value_type ValueType;

93 Dinv_A<MatrixType> Dinv_A(A);

144template <typename IndexType, typename ValueType, typename MemorySpace, typename SmootherType, typename SolverType>
145template <typename MatrixType, typename ArrayType>
146void block_smoothed_aggregation<IndexType,ValueType,MemorySpace,SmootherType,SolverType>
147::init(const MatrixType& A, const ArrayType& B)
148{
149 CUSP_PROFILE_SCOPED();
150 Parent* ML = this;
151 ML->levels.reserve(20); // avoid reallocations which force matrix copies
152
153 sa_levels.push_back(sa_level<SetupMatrixType>());
154 ML->levels.push_back(typename Parent::level());
155
156 sa_levels.back().B = B;
157 sa_levels.back().A_ = A; // copy
158 while (sa_levels.back().A_.num_rows > 100){
159 extend_hierarchy();
160 }
161 ML->solver = SolverType(sa_levels.back().A_);
162
163 // Setup solve matrix for each level
164 for( size_t lvl = 0; lvl < sa_levels.size(); lvl++ )
165 detail::setup_level_matrix( ML->levels[lvl].A, sa_levels[lvl].A_ );
166}

168template <typename IndexType, typename ValueType, typename MemorySpace, typename SmootherType, typename SolverType>
169void block_smoothed_aggregation<IndexType,ValueType,MemorySpace,SmootherType,SolverType>
170::extend_hierarchy(void)
171{
172 CUSP_PROFILE_SCOPED();
173 const SetupMatrixType& A = sa_levels.back().A_;
174 Parent* ML = this;
175
176 cusp::array1d<IndexType,MemorySpace> aggregates;
177 {
178 // compute stength of connection matrix"
179 SetupMatrixType C;
180
181 symmetric_strength_of_connection(A, C, theta);
182 // compute aggregates
183 aggregates.resize(C.num_rows);
184 cusp::blas::fill(aggregates,IndexType(0));
185 cusp::coo_matrix<IndexType, ValueType, MemorySpace> CC;
186 cusp::convert(C, CC);
187 standard_aggregation(CC, aggregates);
188 cusp::convert(CC, C);
189 }
190
191 // compute spectral radius of diag(C)^-1 * C "
192 sa_levels.back().rho_DinvA = detail::estimate_rho_Dinv_A(A);
193 SetupMatrixType P;
194 cusp::array1d<ValueType,MemorySpace> B_coarse;
195 {
196 // compute tenative prolongator and coarse nullspace vector
197 SetupMatrixType T;
198 fit_candidates(aggregates, sa_levels.back().B, T, B_coarse);
199
200
201 // compute prolongation operator
202 cusp::coo_matrix<IndexType, ValueType, MemorySpace> AA;
203 cusp::coo_matrix<IndexType, ValueType, MemorySpace> TT;
204 cusp::coo_matrix<IndexType, ValueType, MemorySpace> PP;
205 cusp::convert(A, AA);
206 cusp::convert(T, TT);
207 cusp::convert(P, PP);
208 smooth_prolongator(AA, TT, PP, ValueType(4.0/3.0), sa_levels.back().rho_DinvA); // TODO if C != A then compute rho_Dinv_C
209 cusp::convert(PP, P);
210 }
211 // compute restriction operator (transpose of prolongator)
212 SetupMatrixType R;
213 cusp::transpose(P,R);
214
215 // construct Galerkin product R*A*P
216 SetupMatrixType RAP;
217 {
218 // TODO test speed of R * (A * P) vs. (R * A) * P
219 SetupMatrixType AP;
220 cusp::multiply(A, P, AP);
221 cusp::multiply(R, AP, RAP);
222 }
223 ML->levels.back().smoother = SmootherType(sa_levels.back());
224 sa_levels.back().aggregates.swap(aggregates);
225 detail::setup_level_matrix( ML->levels.back().R, R );
226 detail::setup_level_matrix( ML->levels.back().P, P );
227 ML->levels.back().residual.resize(sa_levels.back().A_.num_rows, numRHS);
228 ML->levels.push_back(typename Parent::level());
229 sa_levels.push_back(sa_level<SetupMatrixType>());
230
231 sa_levels.back().A_.swap(RAP);
232 sa_levels.back().B.swap(B_coarse);
233 ML->levels.back().x.resize(sa_levels.back().A_.num_rows, numRHS);
234 ML->levels.back().b.resize(sa_levels.back().A_.num_rows, numRHS);
235}
