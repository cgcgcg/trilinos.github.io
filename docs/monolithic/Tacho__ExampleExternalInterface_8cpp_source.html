16double solutionError(const int numRows, const double *rhs, const double *sol, const int *rowBegin, const int *columns,
17 const double *values) {
18 double normRhsSquared(0), normErrorSquared(0);
19 for (int i = 0; i < numRows; i++) {
20 double resid = rhs[i];
21 for (int j = rowBegin[i]; j < rowBegin[i + 1]; j++) {
22 const int col = columns[j];
23 resid -= values[j] * sol[col];
24 }
25 double absVal = std::abs(rhs[i]);
26 normRhsSquared += absVal * absVal;
27 absVal = std::abs(resid);
28 normErrorSquared += absVal * absVal;
29 }
30
31 double normError = std::sqrt(normErrorSquared);
32 double normRhs = std::sqrt(normRhsSquared);
33 return normError / normRhs;
34}
