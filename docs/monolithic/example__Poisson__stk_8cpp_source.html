306template<class crs_matrix_type, class multivector_type1, class multivector_type2, class solution_type>
307void Apply_Dirichlet_BCs(std::vector<int> &BCNodes, crs_matrix_type & A, multivector_type1 & x, multivector_type2 & b, solution_type & solution_values) {
308 using SC = typename multivector_type1::scalar_type;
309 using LO = typename multivector_type1::local_ordinal_type;
310 int N=(int)BCNodes.size();
311 Teuchos::ArrayRCP<SC> xdata = x.getDataNonConst(0);
312 Teuchos::ArrayRCP<SC> bdata = b.getDataNonConst(0);
313 if (b.getMap()->getComm()->getRank() == 0)
314 std::cout<<"Apply in Dirichlet BCs to "<<BCNodes.size() << " nodes"<<std::endl;
315
316 Tpetra::beginModify(A,b);
317
318 for(int i=0; i<N; i++) {
319 LO lrid = BCNodes[i];
320
321 xdata[lrid]=bdata[lrid] = solution_values[i];
322
323 size_t numEntriesInRow = A.getNumEntriesInLocalRow(lrid);
324 typename crs_matrix_type::nonconst_local_inds_host_view_type cols("cols", numEntriesInRow);
325 typename crs_matrix_type::nonconst_values_host_view_type vals("vals", numEntriesInRow);
326 A.getLocalRowCopy(lrid, cols, vals, numEntriesInRow);
327
328 for(int j=0; j<vals.extent_int(0); j++)
329 vals(j) = (cols(j) == lrid) ? 1.0 : 0.0;
330
331 A.replaceLocalValues(lrid, cols, vals);
332 }
333
334 Tpetra::endModify(A,b);
335
336}

1190template<typename Scalar>
1191const Scalar sourceTerm(Scalar& x, Scalar& y, Scalar& z){
1192
1193 Scalar u;
1194 Scalar grad_u[3];
1195 Scalar flux[3] = {0.0, 0.0, 0.0};
1196 Scalar material[3][3];
1197 Scalar f = 0.;
1198
1199 // Indicate the independent variables
1200 x.diff(0,3);
1201 y.diff(1,3);
1202 z.diff(2,3);
1203
1204 // Get exact solution and its gradient
1205 u = exactSolution(x, y, z);
1206 exactSolutionGrad(grad_u, x, y, z);
1207
1208 // Get material tensor
1209 materialTensor<Scalar>(material, x, y, z);
1210
1211 // Compute total flux = (A.grad u)
1212 for(int i = 0; i < 3; i++){
1213
1214 // Add diffusive flux
1215 for(int j = 0; j < 3; j++){
1216 flux[i] += material[i][j]*grad_u[j];
1217 }
1218 }
1219
1220 // Compute source term (right hand side): f = -div(A.grad u)
1221 f = -(flux[0].dx(0) + flux[1].dx(1) + flux[2].dx(2));
1222
1223 return f;
1224}
