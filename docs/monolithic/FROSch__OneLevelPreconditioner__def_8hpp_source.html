22 template <class SC,class LO,class GO,class NO>
23 OneLevelPreconditioner<SC,LO,GO,NO>::OneLevelPreconditioner(ConstXMatrixPtr k,
24 ParameterListPtr parameterList) :
25 SchwarzPreconditioner<SC,LO,GO,NO> (parameterList,k->getRangeMap()->getComm()),
26 K_ (k),
27 SumOperator_ (new SumOperator<SC,LO,GO,NO>(k->getRangeMap()->getComm())),
28 MultiplicativeOperator_ (new MultiplicativeOperator<SC,LO,GO,NO>(k,parameterList)),
29 OverlappingOperator_ ()
30 {
31 FROSCH_DETAILTIMER_START_LEVELID(oneLevelPreconditionerTime,"OneLevelPreconditioner::OneLevelPreconditioner");
32 if (!this->ParameterList_->get("OverlappingOperator Type","AlgebraicOverlappingOperator").compare("AlgebraicOverlappingOperator")) {
33 // Set the LevelID in the sublist
34 parameterList->sublist("AlgebraicOverlappingOperator").set("Level ID",this->LevelID_);
35 OverlappingOperator_ = AlgebraicOverlappingOperatorPtr(new AlgebraicOverlappingOperator<SC,LO,GO,NO>(k,sublist(parameterList,"AlgebraicOverlappingOperator")));
36 } else {
37 FROSCH_ASSERT(false,"OverlappingOperator Type unkown.");
38 }
39 if (!this->ParameterList_->get("Level Combination","Additive").compare("Multiplicative")) {
40 UseMultiplicative_ = true;
41 }
42 if (UseMultiplicative_) {
43 MultiplicativeOperator_->addOperator(OverlappingOperator_);
44 } else {
45 SumOperator_->addOperator(OverlappingOperator_);
46 }
47
48 }

71 template <class SC,class LO,class GO,class NO>
72 int OneLevelPreconditioner<SC,LO,GO,NO>::initialize(int overlap,
73 ConstXMapPtr repeatedMap)
74 {
75 FROSCH_TIMER_START_LEVELID(initializeTime,"OneLevelPreconditioner::initialize");
76 int ret = 0;
77 if (overlap<0) {
78 overlap = this->ParameterList_->get("Overlap",1);
79 }
80 if (!this->ParameterList_->get("OverlappingOperator Type","AlgebraicOverlappingOperator").compare("AlgebraicOverlappingOperator")) {
81 AlgebraicOverlappingOperatorPtr algebraicOverlappigOperator = rcp_static_cast<AlgebraicOverlappingOperator<SC,LO,GO,NO> >(OverlappingOperator_);
82 ret = algebraicOverlappigOperator->initialize(overlap,repeatedMap);
83 } else {
84 FROSCH_ASSERT(false,"OverlappingOperator Type unkown.");
85 }
86 return ret;
87 }
