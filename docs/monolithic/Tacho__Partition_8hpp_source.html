23KOKKOS_INLINE_FUNCTION void Part_2x2(const MatView A, MatView &ATL, MatView &ATR,
24 /**************/ MatView &ABL, MatView &ABR, const ordinal_type bm,
25 const ordinal_type bn, const int quadrant) {
26 ordinal_type bmm, bnn;
27
28 switch (quadrant) {
29 case Partition::TopLeft:
30 bmm = min(bm, A.extent(0));
31 bnn = min(bn, A.extent(1));
32
33 ATL.set_view(A, A.offset_0(), bmm, A.offset_1(), bnn);
34 break;
35 case Partition::TopRight:
36 case Partition::BottomLeft:
37 TACHO_TEST_FOR_ABORT(true, MSG_NOT_IMPLEMENTED);
38 break;
39 case Partition::BottomRight:
40 bmm = A.extent(0) - min(bm, A.extent(0));
41 bnn = A.extent(1) - min(bn, A.extent(1));
42
43 ATL.set_view(A, A.offset_0(), bmm, A.offset_1(), bnn);
44 break;
45 default:
46 TACHO_TEST_FOR_ABORT(true, MSG_INVALID_INPUT);
47 break;
48 }
49
50 ATR.set_view(A, A.offset_0(), ATL.extent(0), A.offset_1() + ATL.extent(1), A.extent(1) - ATL.extent(1));
51
52 ABL.set_view(A, A.offset_0() + ATL.extent(0), A.extent(0) - ATL.extent(0), A.offset_1(), ATL.extent(1));
53
54 ABR.set_view(A, A.offset_0() + ATL.extent(0), A.extent(0) - ATL.extent(0), A.offset_1() + ATL.extent(1),
55 A.extent(1) - ATL.extent(1));
