54double NOX::Thyra::WeightedMeritFunction::
55computef(const NOX::Abstract::Group& group) const
56{
57 using Teuchos::RCP;
58 using Teuchos::rcp;
59
60 TEUCHOS_ASSERT(group.isF());
61
62 RCP<const NOX::Thyra::Vector> f =
63 Teuchos::rcp_dynamic_cast<const NOX::Thyra::Vector>(group.getFPtr(),true);
64
65 ::Thyra::copy(f->getThyraVector(), outArg(tmp1_->getThyraVector()));
66 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp1_->getThyraVector()));
67
68 double tmp = tmp1_->norm();
69
70 double value = 0.5 * tmp * tmp;
71
72 return value;
73}

75double NOX::Thyra::WeightedMeritFunction::
76computeSlope(const NOX::Abstract::Vector& dir,
77 const NOX::Abstract::Group& group) const
78{
79 double value = 0.0;
80
81 //const NOX::Thyra::Vector& epDir = dynamic_cast<const NOX::Thyra::Vector&>(dir);
82
83 if (optimizeSlopeCalc_) { // use matrix-free to evaluate F^T Js
84
85 if ( is_null(tmpGrpPtr) ) {
86 tmpGrpPtr = Teuchos::rcp_dynamic_cast<NOX::Thyra::Group>
87 (group.clone(NOX::ShapeCopy),true);
88 }
89
90 // these norms are implicitly weighted
91 double numerator = group.getX().norm();
92 double denominator = dir.norm();
93
94 if (denominator < 1.0e-12) {
95 std::cout << "WARNING: NOX::Thyra::WeightedMeritFunction::computeSlope()\n"
96 << "denominator in perturbation < 1.0e-12! Setting to 1.0!" << std::endl;
97 denominator = 1.0;
98 }
99
100 double eta = 1.0e-6 * (1.0e-6 + numerator/denominator);
101
102 *tmp1_ = dir;
103
104 tmpGrpPtr->computeX(group, dir, eta);
105
106 tmpGrpPtr->computeF();
107
108 *tmp1_ = tmpGrpPtr->getF();
109
110 tmp1_->update(-1.0, group.getF(), 1.0);
111
112 tmp1_->scale(1.0/eta);
113
114// scalingPtr->applyLeftScaling(tmp1_->getEpetraVector(),
115// tmp1_->getEpetraVector());
116
117 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp1_->getThyraVector()));
118
119 *tmp2_ = group.getF();
120
121// scalingPtr->applyLeftScaling(tmp2_->getEpetraVector(),
122// tmp2_->getEpetraVector());
123
124 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp2_->getThyraVector()));
125
126 value = tmp2_->innerProduct(*tmp1_);
127
128 }
129
130 else { // explicitly compute F^T Js
131
132 group.applyJacobian(dir, *tmp1_);
133
134// scalingPtr->applyLeftScaling(tmp1_->getEpetraVector(),
135// tmp1_->getEpetraVector());
136
137 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp1_->getThyraVector()));
138
139 *tmp2_ = group.getF();
140
141// scalingPtr->applyLeftScaling(tmp2_->getEpetraVector(),
142// tmp2_->getEpetraVector());
143
144 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp2_->getThyraVector()));
145
146 value = tmp2_->innerProduct(*tmp1_);
147
148 }
149
150 return value;
151}

182double NOX::Thyra::WeightedMeritFunction::
183computeQuadraticModel(const NOX::Abstract::Vector& dir,
184 const NOX::Abstract::Group& group) const
185{
186 double f = this->computef(group);
187
188 this->computeGradient(group, *tmp2_);
189
190 double gradfTd = tmp2_->innerProduct(dir);
191
192 group.applyJacobian(dir, *tmp2_);
193
194// scalingPtr->applyLeftScaling(tmp2_->getEpetraVector(),
195// tmp2_->getEpetraVector());
196
197 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp2_->getThyraVector()));
198
199 double dTgrad2fd = 0.5 * tmp2_->innerProduct(*tmp2_);
200
201 double m = f + gradfTd + dTgrad2fd;
202
203 return m;
204}

214bool NOX::Thyra::WeightedMeritFunction::
215computeSteepestDescentDir(const NOX::Abstract::Group& group,
216 NOX::Abstract::Vector& result) const
217{
218 TEUCHOS_ASSERT(group.isF());
219 TEUCHOS_ASSERT(group.isJacobian());
220
221 NOX::Thyra::Vector& dir = dynamic_cast<NOX::Thyra::Vector&>(result);
222
223 NOX::Abstract::Group::ReturnType status;
224
225 // Compute Steepest Descent direction
226 this->computeGradient(group, dir);
227
228 // Scale by the quadratic minimization model
229 status = group.applyJacobian(dir, *tmp2_);
230 TEUCHOS_TEST_FOR_EXCEPTION(status != NOX::Abstract::Group::Ok, std::logic_error,
231 "applyJacobian() failed!");
232
233// scalingPtr->applyLeftScaling(tmp2_->getEpetraVector(),
234// tmp2_->getEpetraVector());
235
236 ::Thyra::ele_wise_scale(weights_->getThyraVector(), outArg(tmp2_->getThyraVector()));
237
238 dir.scale( -1.0 * dir.innerProduct(dir) / tmp2_->innerProduct(*tmp2_) );
239
240 return true;
