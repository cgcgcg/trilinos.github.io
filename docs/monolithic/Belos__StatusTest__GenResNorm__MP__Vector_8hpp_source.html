256 std::string resFormStr() const
257 {
258 std::ostringstream oss;
259 oss << "(";
260 oss << ((resnormtype_==OneNorm) ? "1-Norm" : (resnormtype_==TwoNorm) ? "2-Norm" : "Inf-Norm");
261 oss << ((restype_==Explicit) ? " Exp" : " Imp");
262 oss << " Res Vec) ";
263
264 // If there is no residual scaling, return current string.
265 if (scaletype_!=None)
266 {
267 // Insert division sign.
268 oss << "/ ";
269
270 // Determine output string for scaling, if there is any.
271 if (scaletype_==UserProvided)
272 oss << " (User Scale)";
273 else {
274 oss << "(";
275 oss << ((scalenormtype_==OneNorm) ? "1-Norm" : (resnormtype_==TwoNorm) ? "2-Norm" : "Inf-Norm");
276 if (scaletype_==NormOfInitRes || scaletype_==NormOfFullInitRes || scaletype_==NormOfFullScaledInitRes)
277 oss << " Res0";
278 else if (scaletype_==NormOfPrecInitRes || scaletype_==NormOfFullPrecInitRes || scaletype_==NormOfFullScaledPrecInitRes)
279 oss << " Prec Res0";
280 else
281 oss << " RHS ";
282 oss << ")";
283 }
284 }
285
286 return oss.str();

443template <class StorageType, class MV, class OP>
444StatusType StatusTestGenResNorm<Sacado::MP::Vector<StorageType>,MV,OP>::checkStatus( Iteration<ScalarType,MV,OP>* iSolver )
445{
446 MagnitudeType zero = Teuchos::ScalarTraits<MagnitudeType>::zero();
447 const LinearProblem<ScalarType,MV,OP>& lp = iSolver->getProblem();
448 // Compute scaling term (done once for each block that's being solved)
449 if (firstcallCheckStatus_) {
450 StatusType status = firstCallCheckStatusSetup(iSolver);
451 if(status==Failed) {
452 status_ = Failed;
453 return(status_);
454 }
455 }
456 //
457 // This section computes the norm of the residual std::vector
458 //
459 if ( curLSNum_ != lp.getLSNumber() ) {
460 //
461 // We have moved on to the next rhs block
462 //
463 curLSNum_ = lp.getLSNumber();
464 curLSIdx_ = lp.getLSIndex();
465 curBlksz_ = (int)curLSIdx_.size();
466 int validLS = 0;
467 for (int i=0; i<curBlksz_; ++i) {
468 if (curLSIdx_[i] > -1 && curLSIdx_[i] < numrhs_)
469 validLS++;
470 }
471 curNumRHS_ = validLS;
472 curSoln_ = Teuchos::null;
473 //
474 } else {
475 //
476 // We are in the same rhs block, return if we are converged
477 //
478 if (status_==Passed) { return status_; }
479 }
480 if (restype_==Implicit) {
481 //
482 // get the native residual norms from the solver for this block of right-hand sides.
483 // If the residual is returned in multivector form, use the resnormtype to compute the residual norms.
484 // Otherwise the native residual is assumed to be stored in the resvector_.
485 //
486 std::vector<MagnitudeType> tmp_resvector( curBlksz_ );
487 Teuchos::RCP<const MV> residMV = iSolver->getNativeResiduals( &tmp_resvector );
488 if ( residMV != Teuchos::null ) {
489 tmp_resvector.resize( MVT::GetNumberVecs( *residMV ) );
490 MVT::MvNorm( *residMV, tmp_resvector, resnormtype_ );
491 typename std::vector<int>::iterator p = curLSIdx_.begin();
492 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
493 // Check if this index is valid
494 if (*p != -1)
495 resvector_[*p] = tmp_resvector[i];
496 }
497 } else {
498 typename std::vector<int>::iterator p = curLSIdx_.begin();
499 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
500 // Check if this index is valid
501 if (*p != -1)
502 resvector_[*p] = tmp_resvector[i];
503 }
504 }
505 }
506 else if (restype_==Explicit) {
507 //
508 // Request the true residual for this block of right-hand sides.
509 //
510 Teuchos::RCP<MV> cur_update = iSolver->getCurrentUpdate();
511 curSoln_ = lp.updateSolution( cur_update );
512 Teuchos::RCP<MV> cur_res = MVT::Clone( *curSoln_, MVT::GetNumberVecs( *curSoln_ ) );
513 lp.computeCurrResVec( &*cur_res, &*curSoln_ );
514 std::vector<MagnitudeType> tmp_resvector( MVT::GetNumberVecs( *cur_res ) );
515 MVT::MvNorm( *cur_res, tmp_resvector, resnormtype_ );
516 typename std::vector<int>::iterator p = curLSIdx_.begin();
517 for (int i=0; p<curLSIdx_.end(); ++p, ++i) {
518 // Check if this index is valid
519 if (*p != -1)
520 resvector_[*p] = tmp_resvector[i];
521 }
522 }
523 //
524 // Compute the new linear system residuals for testing.
525 // (if any of them don't meet the tolerance or are NaN, then we exit with that status)
526 //
527 if ( scalevector_.size() > 0 ) {
528 typename std::vector<int>::iterator p = curLSIdx_.begin();
529 for (; p<curLSIdx_.end(); ++p) {
530 // Check if this index is valid
531 if (*p != -1) {
532 // Scale the std::vector accordingly
533 if ( scalevector_[ *p ] != zero ) {
534 // Don't intentionally divide by zero.
535 testvector_[ *p ] = resvector_[ *p ] / scalevector_[ *p ] / scalevalue_;
536 } else {
537 testvector_[ *p ] = resvector_[ *p ] / scalevalue_;
538 }
539 }
540 }
541 }
542 else {
543 typename std::vector<int>::iterator p = curLSIdx_.begin();
544 for (; p<curLSIdx_.end(); ++p) {
545 // Check if this index is valid
546 if (*p != -1)
547 testvector_[ *p ] = resvector_[ *p ] / scalevalue_;
548 }
549 }
550
551 // Check status of new linear system residuals and see if we have the quorum.
552 int have = 0;
553 ind_.resize( curLSIdx_.size() );
554 typename std::vector<int>::iterator p = curLSIdx_.begin();
555 for (; p<curLSIdx_.end(); ++p) {
556 // Check if this index is valid
557 if (*p != -1) {
558 // Check if any of the residuals are larger than the tolerance.
559 const int ensemble_size = StorageType::static_size;
560 bool all_converged = true;
561 for (int i=0; i<ensemble_size; ++i) {
562 if (!ensemble_converged[i]) {
563 if (testvector_[ *p ].coeff(i) > tolerance_) {
564 ++ensemble_iterations[i];
565 all_converged = false;
566 }
567 else if (testvector_[ *p ].coeff(i) <= tolerance_) {
568 ensemble_converged[i] = 1;
569 }
570 else {
571 // Throw an std::exception if a NaN is found.
572 status_ = Failed;
573 TEUCHOS_TEST_FOR_EXCEPTION(true,StatusTestError,"StatusTestGenResNorm::checkStatus(): NaN has been detected.");
574 }
575 }
576 }
577 if (all_converged) {
578 ind_[have] = *p;
579 have++;
580 }
581 }
582 }
583 ind_.resize(have);
584 int need = (quorum_ == -1) ? curNumRHS_: quorum_;
585 status_ = (have >= need) ? Passed : Failed;
586
587 // Return the current status
588 return status_;
589}

591template <class StorageType, class MV, class OP>
592void StatusTestGenResNorm<Sacado::MP::Vector<StorageType>,MV,OP>::print(std::ostream& os, int indent) const
593{
594 for (int j = 0; j < indent; j ++)
595 os << ' ';
596 printStatus(os, status_);
597 os << resFormStr();
598 if (status_==Undefined)
599 os << ", tol = " << tolerance_ << std::endl;
600 else {
601 os << std::endl;
602 if(showMaxResNormOnly_ && curBlksz_ > 1) {
603 const MagnitudeType maxRelRes = *std::max_element(
604 testvector_.begin()+curLSIdx_[0],testvector_.begin()+curLSIdx_[curBlksz_-1]
605 );
606 for (int j = 0; j < indent + 13; j ++)
607 os << ' ';
608 os << "max{residual["<<curLSIdx_[0]<<"..."<<curLSIdx_[curBlksz_-1]<<"]} = " << maxRelRes
609 << ( maxRelRes <= tolerance_ ? " <= " : " > " ) << tolerance_ << std::endl;
610 }
611 else {
612 for ( int i=0; i<numrhs_; i++ ) {
613 for (int j = 0; j < indent + 13; j ++)
614 os << ' ';
615 os << "residual [ " << i << " ] = " << testvector_[ i ];
616 os << ((testvector_[i]<tolerance_) ? " < " : (testvector_[i]==tolerance_) ? " == " : (testvector_[i]>tolerance_) ? " > " : " " ) << tolerance_ << std::endl;
617 }
618 }
619 }
620 os << std::endl;
621}

643template <class StorageType, class MV, class OP>
644StatusType StatusTestGenResNorm<Sacado::MP::Vector<StorageType>,MV,OP>::firstCallCheckStatusSetup( Iteration<ScalarType,MV,OP>* iSolver )
645{
646 int i;
647 MagnitudeType zero = Teuchos::ScalarTraits<MagnitudeType>::zero();
648 MagnitudeType one = Teuchos::ScalarTraits<MagnitudeType>::one();
649 const LinearProblem<ScalarType,MV,OP>& lp = iSolver->getProblem();
650 // Compute scaling term (done once for each block that's being solved)
651 if (firstcallCheckStatus_) {
652 //
653 // Get some current solver information.
654 //
655 firstcallCheckStatus_ = false;
656
657 if (scaletype_== NormOfRHS) {
658 Teuchos::RCP<const MV> rhs = lp.getRHS();
659 numrhs_ = MVT::GetNumberVecs( *rhs );
660 scalevector_.resize( numrhs_ );
661 MVT::MvNorm( *rhs, scalevector_, scalenormtype_ );
662 }
663 else if (scaletype_==NormOfInitRes || scaletype_==NormOfFullInitRes || scaletype_==NormOfFullScaledInitRes) {
664 Teuchos::RCP<const MV> init_res = lp.getInitResVec();
665 numrhs_ = MVT::GetNumberVecs( *init_res );
666 scalevector_.resize( numrhs_ );
667 MVT::MvNorm( *init_res, scalevector_, scalenormtype_ );
668 }
669 else if (scaletype_==NormOfPrecInitRes || scaletype_==NormOfFullPrecInitRes || scaletype_==NormOfFullScaledPrecInitRes) {
670 Teuchos::RCP<const MV> init_res = lp.getInitPrecResVec();
671 numrhs_ = MVT::GetNumberVecs( *init_res );
672 scalevector_.resize( numrhs_ );
673 MVT::MvNorm( *init_res, scalevector_, scalenormtype_ );
674 }
675 else {
676 numrhs_ = MVT::GetNumberVecs( *(lp.getRHS()) );
677 }
678
679 resvector_.resize( numrhs_ );
680 testvector_.resize( numrhs_ );
681
682 curLSNum_ = lp.getLSNumber();
683 curLSIdx_ = lp.getLSIndex();
684 curBlksz_ = (int)curLSIdx_.size();
685 int validLS = 0;
686 for (i=0; i<curBlksz_; ++i) {
687 if (curLSIdx_[i] > -1 && curLSIdx_[i] < numrhs_)
688 validLS++;
689 }
690 curNumRHS_ = validLS;
691 //
692 // Initialize the testvector.
693 for (i=0; i<numrhs_; i++) { testvector_[i] = one; }
694
695 // Return an error if the scaling is zero.
696 if (scalevalue_ == zero) {
697 return Failed;
698 }
699 }
700 return Undefined;
701}
