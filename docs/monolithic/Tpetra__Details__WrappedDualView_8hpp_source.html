160 WrappedDualView(const t_dev deviceView) {
161 TEUCHOS_TEST_FOR_EXCEPTION(
162 deviceView.data() != nullptr && deviceView.use_count() == 0,
163 std::invalid_argument,
164 "Tpetra::Details::WrappedDualView: cannot construct with a device view that\n"
165 "does not own its memory (i.e. constructed with a raw pointer and dimensions)\n"
166 "because the WrappedDualView needs to assume ownership of the memory.");
167 // If the provided view is default-constructed (null, 0 extent, 0 use count),
168 // leave the host mirror default-constructed as well in order to have a matching use count of 0.
169 t_host hostView;
170 if (deviceView.use_count() != 0) {
171 hostView = Kokkos::create_mirror_view(
172 Kokkos::WithoutInitializing,
173 typename t_host::memory_space(),
174 deviceView);
175 }
176 originalDualView = DualViewType(deviceView, hostView);
177 originalDualView.clear_sync_state();
178 originalDualView.modify_device();
179 dualView = originalDualView;
180 }

613 bool needsSyncPath() const {
614 if (!wdvTrackingEnabled)
615 return false;
616
617 // We check to see if the memory is not aliased *or* if it is a supported
618 // (heterogeneous memory) accelerator (for shared host/device memory).
619 if constexpr (Spaces::is_gpu_exec_space<typename DualViewType::execution_space>()) {
620 return true;
621 } else {
622 if constexpr (!deviceMemoryIsHostAccessible) {
623 return true;
624 } else {
625 return dualView.view_host().data() != dualView.view_device().data();
626 }
627 }

630 void throwIfViewsAreDifferentSizes() const {
631 // Here we check *size* (the product of extents) rather than each extent individually.
632 // This is mostly designed to catch people resizing one view, but not the other.
633 if (getRawDeviceView().size() != getRawHostView().size()) {
634 std::ostringstream msg;
635 msg << "Tpetra::Details::WrappedDualView (name = " << getRawDeviceView().label()
636 << "; host and device views are different sizes: "
637 << getRawHostView().size() << " vs " << getRawHostView().size();
638 throw std::runtime_error(msg.str());
639 }
640 }
641
642 void throwIfHostViewAlive() const {
643 throwIfViewsAreDifferentSizes();
644 if (getRawHostView().use_count() > getRawDeviceView().use_count()) {
645 std::ostringstream msg;
646 msg << "Tpetra::Details::WrappedDualView (name = " << getRawDeviceView().label()
647 << "; host use_count = " << getRawHostView().use_count()
648 << "; device use_count = " << getRawDeviceView().use_count() << "): "
649 << "Cannot access data on device while a host view is alive";
650 throw std::runtime_error(msg.str());
651 }
652 }
653
654 void throwIfDeviceViewAlive() const {
655 throwIfViewsAreDifferentSizes();
656 if (getRawDeviceView().use_count() > getRawHostView().use_count()) {
657 std::ostringstream msg;
658 msg << "Tpetra::Details::WrappedDualView (name = " << getRawDeviceView().label()
659 << "; host use_count = " << getRawHostView().use_count()
660 << "; device use_count = " << getRawDeviceView().use_count() << "): "
661 << "Cannot access data on host while a device view is alive";
662 throw std::runtime_error(msg.str());
663 }
664 }
