41 template<class SC,class LO,class GO,class NO>
42 void ThyraPreconditioner<SC,LO,GO,NO>::apply(const XMultiVector &x,
43 XMultiVector &y,
44 ETransp mode,
45 SC alpha,
46 SC beta) const
47 {
48 FROSCH_TIMER_START_SOLVER(applyTime,"ThyraPreconditioner::apply");
49 FROSCH_ASSERT(this->IsComputed_,"FROSch::ThyraPreconditioner: !this->IsComputed_.");
50
51 ConstMultiVectorBasePtr xThyra = ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(rcpFromRef(x));
52 MultiVectorBasePtr yThyra = rcp_const_cast<MultiVectorBase<SC> >(ThyraUtils<SC,LO,GO,NO>::toThyraMultiVector(rcpFromRef(y)));
53
54 EOpTransp tMode = NOTRANS;
55 switch (mode) {
56 case Teuchos::NO_TRANS: tMode = Thyra::NOTRANS; break;
57 case Teuchos::TRANS: tMode = Thyra::TRANS; break;
58 case Teuchos::CONJ_TRANS: tMode = Thyra::CONJTRANS; break;
59 default: FROSCH_ASSERT(false,"FROSch::ThyraPreconditioner: mode unknown.");
60 }
61
62 ThyraPreconditioner_->getUnspecifiedPrecOp()->apply(tMode,*xThyra,yThyra.ptr(),alpha,beta);
63
64 // It seems that we have to convert the Thyra vector back to Xpetra. Is there a cheaper/more elegant way?
65 // Same for ThyraSolver
66 XMultiVectorPtr yXpetra = ThyraUtils<SC,LO,GO,NO>::toXpetra(yThyra,y.getMap()->getComm());
67 y = *yXpetra;
68 }
