50template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal>
51Scalar VelocityModel<Scalar, LocalOrdinal, GlobalOrdinal>::getVelocity(double x, double y, double z) {
52 Scalar c = (Scalar)1.0;
53 if (dim_ == 2) {
54 if (modelType_ == 0) {
55 // constant wavespeed
56 c = (Scalar)1.0;
57 } else if (modelType_ == 1) {
58 // converging lens
59 c = (Scalar)(4.0 / 3.0) * (1.0 - 0.5 * exp(-32.0 * ((x - 0.5) * (x - 0.5) + (y - 0.5) * (y - 0.5))));
60 } else if (modelType_ == 2) {
61 // wedge
62 c = (Scalar)1.0;
63 if (x < 0.8 - 0.2 * y)
64 c = (Scalar)0.7;
65 if (x < 0.4 + 0.1 * y)
66 c = (Scalar)1.3;
67 }
68 } else { // if dim_==3
69 if (modelType_ == 0) {
70 // constant wavespeed
71 c = (Scalar)1.0;
72 } else if (modelType_ == 1) {
73 // converging lens
74 c = (Scalar)(4.0 / 3.0) * (1.0 - 0.5 * exp(-32.0 * ((x - 0.5) * (x - 0.5) + (y - 0.5) * (y - 0.5) + (z - 0.5) * (z - 0.5))));
75 } else if (modelType_ == 2) {
76 // wedge
77 c = (Scalar)1.0;
78 if (x < 0.8 - 0.2 * z)
79 c = (Scalar)0.7;
80 if (x < 0.4 + 0.1 * z)
81 c = (Scalar)1.3;
82 }
83 }
84 return c;
85}
