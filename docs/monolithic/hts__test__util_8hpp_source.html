154 // y = a*x.
155 static void mvp (const Data& a, const bool transp, const bool conj,
156 const Sclr* const x, Sclr* const y) {
157 const Size* const ir = &a.ir[0];
158 const Int* const jc = &a.jc[0];
159 const Sclr* const d = &a.v[0];
160 if (transp) {
161 for (Int i = 0; i < a.m; ++i)
162 y[i] = 0;
163 for (Int i = 0; i < a.m; ++i) {
164 const Sclr xi = x[i];
165 const Size iri = ir[i], irip1 = ir[i+1];
166 for (Size j = iri; j < irip1; ++j) {
167 Sclr dj = d[j];
168 if (conj) conjugate(dj);
169 y[jc[j]] += dj*xi;
170 }
171 }
172 } else {
173 for (Int i = 0; i < a.m; ++i) {
174 Sclr acc = 0;
175 const Size iri = ir[i], irip1 = ir[i+1];
176 for (Size j = iri; j < irip1; ++j) {
177 Sclr dj = d[j];
178 if (conj) conjugate(dj);
179 acc += dj*x[jc[j]];
180 }
181 y[i] = acc;
182 }
183 }

192 reldif (const Sclr* const a, const Sclr* const b, const Size n) {
193 double num = 0, den = 0;
194 for (Size i = 0; i < n; ++i) {
195 num += square(a[i] - b[i]);
196 den += square(a[i]);
197 }
198 return std::sqrt(num/den);

205 // Remove or add an entry to make the matrix non-triangular. This will test
206 // the shape-detection code.
207 static void
208 test_shape_checking (const TestOptions& to, Data& d, const int r) {
209 if (to.matrix_type == TestOptions::missing_some_diag && r == to.n/2) {
210 // Remove the previous diagonal entry.
211 --d.ir[r+1];
212 d.jc.pop_back();
213 d.v.pop_back();
214 } else if (is_not_tri(to.matrix_type) && r == to.n/2) {
215 // Add an entry on the wrong tri side.
216 d.jc.push_back(to.upper ? std::max(0, r-1) : std::min(to.n-1, r+1));
217 d.v.push_back(0.5);
218 ++d.ir[r+1];
219 }
220 }

222 // Make a random sparse matrix with probability of an off-diag entry set to
223 // p. If p is not specified or is negative, set it to min(1, 20/n).
224 static void gen_tri_sparse_matrix (const TestOptions& to, Data& d, double p) {
225 if (p < 0) p = std::min(0.4, 20.0/to.n);
226 std::vector<Sclr> row_val;
227 std::vector<Int> row_col;
228 for (int r = 0; r < to.n; ++r) {
229 Sclr diag;
230 {
231 // Generate the off-diag row and a diag. Scale the off-diag row to keep
232 // the condition number reasonable.
233 row_val.clear();
234 row_col.clear();
235 const int ntrial = to.upper ? to.n - r - 1 : r;
236 const int base = to.upper ? r + 1 : 0;
237 for (int i = 0; i < ntrial; ++i)
238 if (urand() < p) {
239 row_val.push_back(rand_scalar<Sclr>());
240 row_col.push_back(static_cast<Int>(base + i));
241 }
242 // If testing shape checking, make sure there's at least one entry on
243 // the correct side.
244 if (is_not_tri(to.matrix_type) && to.n > 2 && r == to.n/2 &&
245 row_val.empty()) {
246 row_val.push_back(rand_scalar<Sclr>());
247 row_col.push_back(static_cast<Int>(base + ntrial - 1));
248 }
249 double sum = 0;
250 for (std::size_t i = 0; i < row_val.size(); ++i)
251 sum += std::abs(row_val[i]);
252 diag = gen_diag();
253 const double scale = 0.1*std::abs(diag)/std::max(1.0, sum);
254 for (std::size_t i = 0; i < row_val.size(); ++i)
255 row_val[i] *= scale;
256 }
257 d.ir[r+1] = d.ir[r];
258 if (to.upper) {
259 if (is_not_tri(to.matrix_type))
260 test_shape_checking(to, d, r);
261 ++d.ir[r+1];
262 d.jc.push_back(r);
263 d.v.push_back(diag);
264 if (to.matrix_type == TestOptions::missing_some_diag)
265 test_shape_checking(to, d, r);
266 d.ir[r+1] += static_cast<Int>(row_val.size());
267 d.jc.insert(d.jc.end(), row_col.begin(), row_col.end());
268 d.v.insert(d.v.end(), row_val.begin(), row_val.end());
269 } else {
270 d.ir[r+1] += static_cast<Int>(row_val.size());
271 d.jc.insert(d.jc.end(), row_col.begin(), row_col.end());
272 d.v.insert(d.v.end(), row_val.begin(), row_val.end());
273 ++d.ir[r+1];
274 d.jc.push_back(r);
275 d.v.push_back(diag);
276 test_shape_checking(to, d, r);
277 }
278 }
279 }

286 static void gen_tri_block_matrix_from_tri_matrix (
287 const Data& src, const Int blksz, const bool upper, Data& dst)
288 {
289 assert(blksz > 1);
290 const Size nnz =
291 blksz*blksz*src.ir.back() - ((blksz*(blksz-1))/2)*src.m // full blocks
292 - src.ir.back(); // remove 1 from each block
293 dst.m = blksz*src.m;
294 dst.ir.resize(dst.m+1);
295 dst.jc.resize(nnz);
296 dst.v.resize(nnz);
297
298 dst.ir[0] = 0;
299 Size dk = 0;
300 for (Int sr = 0, dr = 0; sr < src.m; ++sr) {
301 for (Int br = 0; br < blksz; ++br, ++dr) {
302 dst.ir[dr+1] = dst.ir[dr];
303 const Real scale_odb = 1.0/(src.ir[sr+1] - src.ir[sr]), scale = 1.0/blksz;
304 for (Size j = src.ir[sr]; j < src.ir[sr+1]; ++j) {
305 const bool on_diag = (( upper && j == src.ir[sr]) ||
306 ( ! upper && j == src.ir[sr+1] - 1));
307 const Int sc = src.jc[j];
308 const Sclr& val = src.v[j];
309 const Int
310 bcs = on_diag && upper ? br : 0,
311 bce = on_diag ? (upper ? blksz : br+1) : blksz;
312 for (Int bc = bcs; bc < bce; ++bc) {
313 // Skip one entry in the block.
314 if (on_diag && upper) {
315 if (br == 0 && bc+1 == bce) continue;
316 } else if (br+1 == blksz && bc == bcs) continue;
317
318 ++dst.ir[dr+1];
319 dst.jc[dk] = blksz*sc + bc;
320 dst.v[dk] = val*2.0*(urand() - 0.5);
321 // Scale the off-diag elements of the on-diag block.
322 if (on_diag && bc != br)
323 dst.v[dk] *= scale_odb;
324 // Scale the off-diag elements of the tri.
325 if ( ! (on_diag && bc == br))
326 dst.v[dk] *= scale;
327 ++dk;
328 }
329 }
330 }
331 }
332 assert(dk == nnz);
333 }

335 static void gen_implicit_unit_diag_from_tri_matrix (
336 const bool upper, const Data& src, Data& dst)
337 {
338 dst.m = src.m;
339 dst.ir = src.ir;
340 dst.jc.resize(src.jc.size() - dst.m);
341 dst.v.resize(dst.jc.size());
342 dst.ir[0] = 0;
343 Size z = 0;
344 for (Int r = 0; r < dst.m; ++r) {
345 dst.ir[r+1] = src.ir[r+1] - (r + 1);
346 const Sclr diag = upper ? src.v[src.ir[r]] : src.v[src.ir[r+1] - 1];
347 Size
348 K_beg = src.ir[r] + (upper ? 1 : 0),
349 K_end = src.ir[r+1] - (upper ? 0 : 1);
350 for (Size K = K_beg; K < K_end; ++K) {
351 dst.jc[z] = src.jc[K];
352 dst.v[z] = src.v[K] / diag;
353 ++z;
354 }
355 }

358 // Make a random triangular matrix according to the TestOptions.
359 static void gen_tri_matrix (const TestOptions& to, Data& d) {
360 d.clear();
361 d.m = to.n;
362 d.ir.resize(d.m+1, 0);
363 switch (to.matrix_type) {
364 case TestOptions::diag: gen_diag_matrix(to, d); break;
365 case TestOptions::dense: gen_tri_dense_matrix(to, d); break;
366 case TestOptions::sparse:
367 case TestOptions::implicit_unit_diag:
368 case TestOptions::missing_some_diag:
369 case TestOptions::not_tri:
370 gen_tri_sparse_matrix(to, d, to.density); break;
371 case TestOptions::block_sparse:
372 case TestOptions::block_sparse_implicit_unit_diag: {
373 Data point_sparse(d);
374 gen_tri_sparse_matrix(to, point_sparse, to.density);
375 gen_tri_block_matrix_from_tri_matrix(point_sparse, to.block_size,
376 to.upper, d);
377 } break;
378 case TestOptions::not_tri_almost_diag:
379 gen_tri_sparse_matrix(to, d, 0); break;
380 }
381 if (to.has_unit_diag()) {
382 Data src(d);
383 d.clear();
384 gen_implicit_unit_diag_from_tri_matrix(to.upper, src, d);
385 }
386 assert((std::size_t) d.ir.back() == d.v.size());
387 assert((std::size_t) d.ir.back() == d.jc.size());
388 }

400 static void transpose (
401 const std::vector<Size>& ir, const std::vector<Int>& jc,
402 const std::vector<Sclr>& d, std::vector<Size>& irt,
403 std::vector<Int>& jct, std::vector<Sclr>& dt)
404 {
405 const Int m = static_cast<Int>(ir.size()) - 1;
406 // 1. Count the number of entries in each col.
407 for (Int i = 0; i <= m; ++i) irt[i] = 0;
408 const Size nnz = ir[m];
409 for (Size k = 0; k < nnz; ++k) {
410 // Store shifted up by 1. This is the trick that makes extra workspace
411 // unnecessary.
412 ++irt[jc[k]+1];
413 }
414 // 2. Cumsum to get the col pointers.
415 Size sum = 0;
416 for (Int i = 0; i < m; ++i) {
417 const Size incr = irt[i+1];
418 irt[i+1] = sum;
419 sum += incr;
420 }
421 assert(sum == nnz);
422 // At this point, At.ir[i+1] gives what is normally in At.ir[i].
423 // 3. Fill in jc and d.
424 for (Int r = 0; r < m; ++r)
425 for (Size j = ir[r], jlim = ir[r+1]; j < jlim; ++j) {
426 const Int c = jc[j];
427 // Points to next entry to be filled.
428 // 3a. Increment to next available entry.
429 const Size p = irt[c+1]++;
430 jct[p] = r;
431 dt[p] = d[j];
432 }
433 assert(irt[m] == ir[m]);
434 // 4. Step 3a made it such that At.ir[i] is now what we need except for
435 // i = 0. At.ir[0] was already set to 0 in step 1, so we're done.
436 }
