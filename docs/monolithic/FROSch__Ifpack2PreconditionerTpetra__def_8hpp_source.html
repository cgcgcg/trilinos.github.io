50 template<class SC,class LO,class GO,class NO>
51 void Ifpack2PreconditionerTpetra<SC,LO,GO,NO>::apply(const XMultiVector &x,
52 XMultiVector &y,
53 ETransp mode,
54 SC alpha,
55 SC beta) const
56 {
57 FROSCH_TIMER_START_SOLVER(applyTime,"Ifpack2PreconditionerTpetra::apply");
58 FROSCH_ASSERT(this->IsComputed_,"FROSch::Ifpack2PreconditionerTpetra: !this->IsComputed_.");
59
60 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorX = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(&x);
61 FROSCH_ASSERT(xTpetraMultiVectorX,"FROSch::Ifpack2PreconditionerTpetra: dynamic_cast failed.");
62 TMultiVectorPtr tpetraMultiVectorX = xTpetraMultiVectorX->getTpetra_MultiVector();
63
64 const TpetraMultiVector<SC,LO,GO,NO> * xTpetraMultiVectorY = dynamic_cast<const TpetraMultiVector<SC,LO,GO,NO> *>(&y);
65 FROSCH_ASSERT(xTpetraMultiVectorY,"FROSch::Ifpack2PreconditionerTpetra: dynamic_cast failed.");
66 TMultiVectorPtr tpetraMultiVectorY = xTpetraMultiVectorY->getTpetra_MultiVector();
67
68#ifdef HAVE_SHYLU_DDFROSCH_ZOLTAN2
69 if (this->useZoltan2 && this->useRILUK) {
70 auto A = Ifpack2Preconditioner_->getMatrix();
71 auto filteredA = rcp_dynamic_cast<const TRowMatrixFilterType> (A);
72
73 // make copies of X & Y
74 TMultiVector ReorderedX (*tpetraMultiVectorX, Teuchos::Copy);
75 TMultiVector ReorderedY (*tpetraMultiVectorY, Teuchos::Copy);
76
77 // permute X & Y
78 filteredA->permuteOriginalToReordered (*tpetraMultiVectorX, ReorderedX);
79 filteredA->permuteOriginalToReordered (*tpetraMultiVectorY, ReorderedY);
80
81 // solve
82 Ifpack2Preconditioner_->apply(ReorderedX,ReorderedY,mode,alpha,beta);
83
84 // permute X back
85 filteredA->permuteReorderedToOriginal (ReorderedY, *tpetraMultiVectorY);
86 } else
87#endif
88 {
89 Ifpack2Preconditioner_->apply(*tpetraMultiVectorX,*tpetraMultiVectorY,mode,alpha,beta);
90 }
91 }

93 template<class SC,class LO,class GO,class NO>
94 int Ifpack2PreconditionerTpetra<SC,LO,GO,NO>::updateMatrix(ConstXMatrixPtr k,
95 bool reuseInitialize)
96 {
97 if (this->useRILUK) {
98 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*this->K_);
99 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
100 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
101
102 auto RILUPreconditioner = rcp_dynamic_cast<Ifpack2::RILUK<TRowMatrix>>(Ifpack2Preconditioner_);
103#ifdef HAVE_SHYLU_DDFROSCH_ZOLTAN2
104 if (this->useZoltan2) {
105 // if K is replaced, then we need to re-wrap into matrix filter
106 auto rowMat = rcp_dynamic_cast<const TRowMatrix>(tpetraMat);
107 auto filteredMat = rcp(new TRowMatrixFilterType(rowMat, this->perm, this->revperm));
108 RILUPreconditioner->setMatrix(filteredMat);
109 } else
110#endif
111 {
112 RILUPreconditioner->setMatrix(tpetraMat);
113 }
114 return 0;
115 }
116 FROSCH_ASSERT(false,"FROSch::Ifpack2PreconditionerTpetra: updateMatrix() is not implemented for the Ifpack2PreconditionerTpetra yet.");
117 }

119 template<class SC,class LO,class GO,class NO>
120 Ifpack2PreconditionerTpetra<SC,LO,GO,NO>::Ifpack2PreconditionerTpetra(ConstXMatrixPtr k,
121 ParameterListPtr parameterList,
122 string description) :
123 Solver<SC,LO,GO,NO> (k,parameterList,description)
124 {
125 FROSCH_TIMER_START_SOLVER(Ifpack2PreconditionerTpetraTime,"Ifpack2PreconditionerTpetra::Ifpack2PreconditionerTpetra");
126 FROSCH_ASSERT(!this->K_.is_null(),"FROSch::Ifpack2PreconditionerTpetra: K_ is null.");
127
128 const CrsMatrixWrap<SC,LO,GO,NO>& crsOp = dynamic_cast<const CrsMatrixWrap<SC,LO,GO,NO>&>(*this->K_);
129 const TpetraCrsMatrix<SC,LO,GO,NO>& xTpetraMat = dynamic_cast<const TpetraCrsMatrix<SC,LO,GO,NO>&>(*crsOp.getCrsMatrix());
130 ConstTCrsMatrixPtr tpetraMat = xTpetraMat.getTpetra_CrsMatrix();
131 TEUCHOS_TEST_FOR_EXCEPT(tpetraMat.is_null());
132
133 auto solverName = this->ParameterList_->get("Solver","RILUK");
134 this->useRILUK = (solverName == "RILUK");
135 this->useZoltan2 = this->ParameterList_->get("RILUK: use reordering", false);
136
137 Ifpack2::Details::OneLevelFactory<TRowMatrix> ifpack2Factory;
138#ifdef HAVE_SHYLU_DDFROSCH_ZOLTAN2
139 if (this->useZoltan2 && this->useRILUK) {
140 // pre-ordering matrix, before constructing Ifpack2Preconditioner_ with the matrix
141 {
142 typedef Tpetra::RowGraph<LO, GO, NO> row_graph_type;
143 typedef Zoltan2::TpetraRowGraphAdapter<row_graph_type> z2_adapter_type;
144 typedef Zoltan2::OrderingProblem<z2_adapter_type> z2_ordering_problem_type;
145 typedef Zoltan2::LocalOrderingSolution<LO> z2_ordering_solution_type;
146
147 auto comm = tpetraMat->getRowMap()->getComm();
148 auto constActiveGraph = Teuchos::rcp_const_cast<const row_graph_type>(tpetraMat->getGraph());
149 z2_adapter_type Zoltan2Graph (constActiveGraph);
150
151 Teuchos::ParameterList zoltan2_params = this->ParameterList_->sublist ("RILUK: reordering list");
152 z2_ordering_problem_type MyOrderingProblem (&Zoltan2Graph, &zoltan2_params, comm);
153
154 MyOrderingProblem.solve ();
155 z2_ordering_solution_type sol (*MyOrderingProblem.getLocalOrderingSolution());
156 this->perm = sol.getPermutationRCPConst (true);
157 this->revperm = sol.getPermutationRCPConst ();
158 }
159
160 // wrap into matrix filter
161 auto rowMat = rcp_dynamic_cast<const TRowMatrix>(tpetraMat);
162 auto filteredMat = rcp(new TRowMatrixFilterType(rowMat, this->perm, this->revperm));
163 Ifpack2Preconditioner_ = ifpack2Factory.create(solverName,filteredMat);
164 } else
165#endif
166 {
167 Ifpack2Preconditioner_ = ifpack2Factory.create(solverName,tpetraMat);
168 }
169
170 // set ifpack2 parameters
171 ParameterListPtr ifpack2ParameterList = sublist(this->ParameterList_,"Ifpack2");
172 if (ifpack2ParameterList->isSublist(solverName)) {
173 ifpack2ParameterList = sublist(ifpack2ParameterList, solverName);
174 }
175 ifpack2ParameterList->setName("Ifpack2");
176 Ifpack2Preconditioner_->setParameters(*ifpack2ParameterList);
177 }
178
179}
