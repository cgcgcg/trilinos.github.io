62template <typename View1, typename View2, typename View3>
63void unfill_crs(View1& row_ptrs, View2& cols, View3& values)
64{
65 using Scalar = typename View3::non_const_value_type;
66 using Ordinal = typename View1::non_const_value_type;
67#if KOKKOS_VERSION >= 40799
68 using STS = KokkosKernels::ArithTraits<Scalar>;
69#else
70 using STS = Kokkos::ArithTraits<Scalar>;
71#endif
72 const Scalar zero = STS::zero();
73
74 const auto nrows = row_ptrs.size() - 1;
75 Ordinal real_nnzs = 0;
76 for (size_t row = 0; row < nrows; ++row) {
77 const Ordinal row_nnz_begin = row_ptrs[row];
78 row_ptrs[row] = real_nnzs;
79 for (Ordinal row_nnz = row_nnz_begin; row_nnz < row_ptrs[row+1]; ++row_nnz) {
80 const Ordinal col_idx = cols[row_nnz];
81 const Scalar value = values[row_nnz];
82 if (value != zero) {
83 cols[real_nnzs] = col_idx;
84 values[real_nnzs] = value;
85 ++real_nnzs;
86 }
87 }
88 }
89 row_ptrs[nrows] = real_nnzs;
90 Kokkos::resize(cols, real_nnzs);
91 Kokkos::resize(values, real_nnzs);
92}

94template <typename View1, typename View2, typename View3>
95void unblock(View1& row_map_host, View2& col_inds_host, View3& values_host, const int block_size)
96{
97 using tmap_type = Tpetra::Map<>;
98 using LO = tmap_type::local_ordinal_type;
99 using GO = tmap_type::global_ordinal_type;
100 using Node = Tpetra::KokkosCompat::KokkosSerialWrapperNode;
101 using Scalar = typename View3::non_const_value_type;
102 using BCrsMatrix = Tpetra::BlockCrsMatrix<Scalar, LO, GO, Node>;
103 using map_type = typename BCrsMatrix::map_type;
104 using graph_type = typename BCrsMatrix::crs_graph_type;
105 using row_vtype = typename graph_type::local_graph_device_type::row_map_type::non_const_type;
106
107 // Create new TCrsMatrix with the new filled data
108 row_vtype rv("temp", row_map_host.extent(0));
109 Kokkos::deep_copy(rv, row_map_host);
110 const auto nrows = row_map_host.size() - 1;
111
112 Teuchos::SerialComm<LO> SerialComm;
113 Teuchos::RCP<const Teuchos::Comm<LO> > comm_ptr = Teuchos::RCP(&SerialComm, false);
114 map_type proc_map(nrows, 0, comm_ptr, Tpetra::LocallyReplicated);
115 Teuchos::RCP<map_type> map_ptr = Teuchos::RCP(&proc_map, false);
116 graph_type graph(map_ptr, map_ptr, rv, col_inds_host);
117 graph.fillComplete();
118 BCrsMatrix bcrs_matrix(graph, values_host, block_size);
119
120 auto A = Tpetra::convertToCrsMatrix(bcrs_matrix);
121 auto localA = A->getLocalMatrixDevice();
122 Kokkos::resize(row_map_host, localA.graph.row_map.size());
123 Kokkos::deep_copy(row_map_host, localA.graph.row_map);
124 Kokkos::resize(col_inds_host, localA.graph.entries.size());
125 Kokkos::deep_copy(col_inds_host, localA.graph.entries);
126 Kokkos::resize(values_host, localA.values.size());
127 Kokkos::deep_copy(values_host, localA.values);
128 unfill_crs(row_map_host, col_inds_host, values_host);
129}

131template <typename View1, typename View2, typename View3, typename View4>
132void reblock(View1& row_map, View2& row_idx, View3& col_inds, View4& values, const int block_size)
133{
134 using LO = Tpetra::Map<>::local_ordinal_type;
135 using GO = Tpetra::Map<>::global_ordinal_type;
136 using Node = Tpetra::KokkosCompat::KokkosSerialWrapperNode;
137 using Scalar = typename View4::non_const_value_type;
138 using TCrsMatrix = Tpetra::CrsMatrix<Scalar, LO, GO, Node>;
139 using map_type = typename TCrsMatrix::map_type;
140 using graph_type = typename TCrsMatrix::crs_graph_type;
141 using row_vtype = typename graph_type::local_graph_device_type::row_map_type::non_const_type;
142
143 row_vtype rv("temp", row_map.extent(0));
144 Kokkos::deep_copy(rv, row_map);
145 const auto nrows = row_map.size() - 1;
146 Teuchos::SerialComm<LO> SerialComm;
147 Teuchos::RCP<const Teuchos::Comm<LO> > comm_ptr = Teuchos::RCP(&SerialComm, false);
148 map_type proc_map(nrows, 0, comm_ptr, Tpetra::LocallyReplicated);
149 Teuchos::RCP<map_type> map_ptr = Teuchos::RCP(&proc_map, false);
150 TCrsMatrix crs_matrix(map_ptr, map_ptr, rv, col_inds, values);
151 crs_matrix.fillComplete();
152 auto crs_matrix_block_filled = Tpetra::fillLogicalBlocks(crs_matrix, block_size);
153
154 auto bcrs_matrix = Tpetra::convertToBlockCrsMatrix(*crs_matrix_block_filled, block_size);
155 auto localA = bcrs_matrix->getLocalMatrixDevice();
156 auto rowptrs = localA.graph.row_map;
157 auto colinds = localA.graph.entries;
158 auto vals = localA.values;
159
160 Kokkos::resize(row_map, rowptrs.size());
161 Kokkos::deep_copy(row_map, rowptrs);
162 Kokkos::resize(col_inds, colinds.size());
163 Kokkos::deep_copy(col_inds, colinds);
164 Kokkos::resize(values, vals.size());
165 Kokkos::deep_copy(values, vals);
166
167 // Now do row idx
168 const auto nrows_blocked = row_map.size() -1;
169 Kokkos::resize(row_idx, colinds.size());
170 LO nnz = 0;
171 for (size_t row = 0; row < nrows_blocked; ++row) {
172 for (LO row_nnz = row_map(row); row_nnz < row_map(row+1); ++row_nnz) {
173 row_idx(nnz++) = row;
174 }
175 }
176 assert(nnz == row_map(nrows_blocked));
177}

721 void findFills(int levfill, OrdinalArrayMirror aRowMap, OrdinalArrayMirror aColIdx,
722 int& nzl, std::vector<int> &lRowMap, std::vector<int> &lColIdx, std::vector<int> &lLevel,
723 int& nzu, std::vector<int> &uRowMap, std::vector<int> &uColIdx, std::vector<int> &uLevel) {
724 using std::vector;
725 using std::sort;
726
727 const Ordinal n = lRowMap.size() - 1;
728
729 Ordinal row = 0, i = 0;
730 vector<int> lnklst(n);
731 vector<int> curlev(n);
732 vector<int> iwork(n);
733
734 int knzl = 0;
735 int knzu = 0;
736
737 lRowMap[0] = 0;
738 uRowMap[0] = 0;
739
740 for (i=0; i<n; i++)
741 {
742 int first, next, j;
743 row = (m_useMetis ? m_permMetisHost(i) : i);
744
745 /* copy column indices of row into workspace and sort them */
746 int len = aRowMap[row+1] - aRowMap[row];
747 next = 0;
748 for (j=aRowMap[row]; j<aRowMap[row+1]; j++) {
749 iwork[next++] = (m_useMetis ? m_ipermMetisHost(aColIdx[j]) : aColIdx[j]);
750 }
751 // sort column indices in non-descending (ascending) order
752 sort(iwork.begin(), iwork.begin() + len);
753
754 /* construct implied linked list for row */
755 first = iwork[0];
756 curlev[first] = 0;
757
758 for (j=0; j<=len-2; j++)
759 {
760 lnklst[iwork[j]] = iwork[j+1];
761 curlev[iwork[j]] = 0;
762 }
763
764 lnklst[iwork[len-1]] = n;
765 curlev[iwork[len-1]] = 0;
766
767 /* merge with rows in U */
768 next = first;
769 while (next < i)
770 {
771 int oldlst = next;
772 int nxtlst = lnklst[next];
773 int inner_row = next;
774 int ii;
775
776 /* scan row */
777 for (ii=uRowMap[inner_row]+1; ii<uRowMap[inner_row+1]; /*nop*/)
778 {
779 if (uColIdx[ii] < nxtlst)
780 {
781 /* new fill-in */
782 int newlev = curlev[inner_row] + uLevel[ii] + 1;
783 if (newlev <= levfill)
784 {
785 lnklst[oldlst] = uColIdx[ii];
786 lnklst[uColIdx[ii]] = nxtlst;
787 oldlst = uColIdx[ii];
788 curlev[uColIdx[ii]] = newlev;
789 }
790 ii++;
791 }
792 else if (uColIdx[ii] == nxtlst)
793 {
794 int newlev;
795 oldlst = nxtlst;
796 nxtlst = lnklst[oldlst];
797 newlev = curlev[inner_row] + uLevel[ii] + 1;
798 //curlev[uColIdx[ii]] = MIN(curlev[uColIdx[ii]], newlev);
799 if (curlev[uColIdx[ii]] > newlev)
800 {
801 curlev[uColIdx[ii]] = newlev;
802 }
803 ii++;
804 }
805 else /* (jau[ii] > nxtlst) */
806 {
807 oldlst = nxtlst;
808 nxtlst = lnklst[oldlst];
809 }
810 }
811 next = lnklst[next];
812 }
813
814 /* gather the pattern into L and U */
815 /* L (no diagonal) */
816 next = first;
817 while (next < i)
818 {
819 assert(knzl < nzl);
820 lLevel[knzl] = curlev[next];
821 lColIdx[knzl++] = next;
822 if (knzl >= nzl)
823 {
824 nzl = nzl + n;
825 lColIdx.resize(nzl);
826 lLevel.resize(nzl);
827 }
828 next = lnklst[next];
829 }
830 lRowMap[i+1] = knzl;
831 assert(next == i);
832 /* U (with diagonal) */
833 while (next < n)
834 {
835 assert(knzu < nzu);
836 uLevel[knzu] = curlev[next];
837 uColIdx[knzu++] = next;
838 if (knzu >= nzu)
839 {
840 nzu = nzu + n;
841 uColIdx.resize(nzu);
842 uLevel.resize(nzu);
843 }
844 next = lnklst[next];
845 }
846 uRowMap[i+1] = knzu;
847 }
848
849 nzl = knzl;
850 nzu = knzu;
851 }

1763 void compute()
1764 {
1765 Kokkos::Timer timer;
1766 FASTILU_CREATE_TIMER(Timer);
1767 if ((m_level > 0) && (m_guessFlag !=0))
1768 {
1769 m_initGuessPrec->compute();
1770 FASTILU_FENCE_REPORT_TIMER(Timer, ExecSpace(), " > initGuess");
1771 }
1772
1773 numericILU();
1774 FASTILU_FENCE_REPORT_TIMER(Timer, ExecSpace(), " > numericILU ");
1775
1776 FastILUFunctor<Ordinal, Scalar, ExecSpace> iluFunctor(m_aRowMapHost[m_nRows], m_blkSzILU,
1777 m_aRowMap, m_aRowIdx, m_aColIdx, m_aVal,
1778 m_lRowMap, m_lColIdx, m_lVal, m_uRowMap, m_uColIdx, m_uVal, m_diagElems, m_omega, m_blockCrsSize, m_level);
1779 Ordinal extent = m_aRowMapHost[m_nRows]/m_blkSzILU;
1780 if (m_aRowMapHost[m_nRows]%m_blkSzILU != 0)
1781 {
1782 extent++;
1783 }
1784
1785 for (int i = 0; i < m_nFact; i++)
1786 {
1787#ifdef FASTILU_ONE_TO_ONE_UNBLOCKED
1788 // Force serialization to avoid races but still perform operation on device
1789 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, 1), KOKKOS_LAMBDA(const int) {
1790 for (Ordinal j = 0; j < extent; ++j) {
1791 iluFunctor(j);
1792 }
1793 });
1794#else
1795 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, extent), iluFunctor);
1796#endif
1797 }
1798 FASTILU_FENCE_REPORT_TIMER(Timer, ExecSpace(), " > iluFunctor (" << m_nFact << ")");
1799
1800 // transpose u
1801 Kokkos::deep_copy(m_utRowMap, 0);
1802 if (m_blockCrsSize == 1) {
1803 KokkosSparse::Impl::transpose_matrix<OrdinalArray, OrdinalArray, ScalarArray, OrdinalArray, OrdinalArray, ScalarArray, OrdinalArray, ExecSpace>
1804 (m_nRows, m_nRows, m_uRowMap, m_uColIdx, m_uVal, m_utRowMap, m_utColIdx, m_utVal);
1805 }
1806 else {
1807 KokkosSparse::Impl::transpose_bsr_matrix<OrdinalArray, OrdinalArray, ScalarArray, OrdinalArray, OrdinalArray, ScalarArray, ExecSpace>
1808 (m_nRows, m_nRows, m_blockCrsSize, m_uRowMap, m_uColIdx, m_uVal, m_utRowMap, m_utColIdx, m_utVal);
1809 }
1810
1811 // sort, if the triangular solve algorithm requires a sorted matrix.
1812 bool sortRequired = m_sptrsv_algo != FastILU::SpTRSV::Fast && m_sptrsv_algo != FastILU::SpTRSV::StandardHost;
1813 if (sortRequired) {
1814 if (m_blockCrsSize == 1) {
1815 KokkosSparse::sort_crs_matrix<ExecSpace, OrdinalArray, OrdinalArray, ScalarArray>
1816 (m_utRowMap, m_utColIdx, m_utVal);
1817 }
1818 else {
1819 KokkosSparse::sort_bsr_matrix<ExecSpace, OrdinalArray, OrdinalArray, ScalarArray>(
1820 m_blockCrsSize, m_utRowMap, m_utColIdx, m_utVal);
1821 }
1822 }
1823
1824 if (m_sptrsv_algo == FastILU::SpTRSV::StandardHost) {
1825 // deep-copy to host
1826 Kokkos::deep_copy(m_lColIdxHost, m_lColIdx);
1827 Kokkos::deep_copy(m_lValHost, m_lVal);
1828
1829 Kokkos::deep_copy(m_utRowMapHost, m_utRowMap);
1830 Kokkos::deep_copy(m_utColIdxHost, m_utColIdx);
1831 Kokkos::deep_copy(m_utValHost, m_utVal);
1832 }
1833 FASTILU_FENCE_REPORT_TIMER(Timer, ExecSpace(), " > transposeU");
1834
1835 if (m_sptrsv_algo == FastILU::SpTRSV::Standard) {
1836 assert(m_blockCrsSize == 1); // Not yet supported for block crs
1837 #if defined(KOKKOSKERNELS_ENABLE_TPL_CUSPARSE)
1838 KokkosSparse::Experimental::SPTRSVAlgorithm algo = KokkosSparse::Experimental::SPTRSVAlgorithm::SPTRSV_CUSPARSE;
1839 #else
1840 KokkosSparse::Experimental::SPTRSVAlgorithm algo = KokkosSparse::Experimental::SPTRSVAlgorithm::SEQLVLSCHD_TP1;
1841 #endif
1842 // setup L solve
1843 khL.create_sptrsv_handle(algo, m_nRows, true);
1844 #if defined(KOKKOSKERNELS_ENABLE_TPL_CUSPARSE)
1845 KokkosSparse::sptrsv_symbolic(&khL, m_lRowMap, m_lColIdx, m_lVal);
1846 #else
1847 KokkosSparse::sptrsv_symbolic(&khL, m_lRowMap, m_lColIdx);
1848 #endif
1849 // setup U solve
1850 khU.create_sptrsv_handle(algo, m_nRows, false);
1851 #if defined(KOKKOSKERNELS_ENABLE_TPL_CUSPARSE)
1852 KokkosSparse::sptrsv_symbolic(&khU, m_utRowMap, m_utColIdx, m_utVal);
1853 #else
1854 KokkosSparse::sptrsv_symbolic(&khU, m_utRowMap, m_utColIdx);
1855 #endif
1856 FASTILU_FENCE_REPORT_TIMER(Timer, ExecSpace(),
1857 " > sptrsv_symbolic : nnz(L)=" << m_lColIdx.extent(0) << " nnz(U)=" << m_utColIdx.extent(0));
1858 } else if (m_sptrsv_algo == FastILU::SpTRSV::StandardHost && m_doUnitDiag_TRSV) {
1859 // Prepare L for TRSV by removing unit-diagonals
1860 assert(m_blockCrsSize == 1); // Not yet supported for block crs
1861 m_lVal_trsvHost = ScalarArrayHost ("lVal_trsv", m_lRowMapHost[m_nRows]-m_nRows);
1862 m_lColIdx_trsvHost = OrdinalArrayHost("lColIdx_trsv", m_lRowMapHost[m_nRows]-m_nRows);
1863 m_lRowMap_trsvHost = OrdinalArrayHost("lRowMap_trsv", m_nRows+1);
1864
1865 size_t nnzL = 0;
1866 m_lRowMap_trsvHost(0) = 0;
1867 for (Ordinal i = 0; i < m_nRows; i++) {
1868 for (Ordinal k = m_lRowMapHost(i); k < m_lRowMapHost[i+1]; k++) {
1869 if (m_lColIdxHost(k) != i) {
1870 m_lVal_trsvHost(nnzL) = m_lValHost(k);
1871 m_lColIdx_trsvHost(nnzL) = m_lColIdxHost(k);
1872 nnzL++;
1873 }
1874 }
1875 m_lRowMap_trsvHost(i+1)=nnzL;
1876 }
1877
1878 // Prepare U by extracting and scaling D
1879 m_dVal_trsvHost = ScalarArrayHost ("dVal_trsv", m_nRows);
1880 m_utVal_trsvHost = ScalarArrayHost ("utVal_trsv", m_utRowMapHost[m_nRows]-m_nRows);
1881 m_utColIdx_trsvHost = OrdinalArrayHost("utColIdx_trsv", m_utRowMapHost[m_nRows]-m_nRows);
1882 m_utRowMap_trsvHost = OrdinalArrayHost("utRowMap_trsv", m_nRows+1);
1883
1884 size_t nnzU = 0;
1885 m_utRowMap_trsvHost(0) = 0;
1886 for (Ordinal i = 0; i < m_nRows; i++) {
1887 for (Ordinal k = m_utRowMapHost(i); k < m_utRowMapHost[i+1]; k++) {
1888 if (m_utColIdxHost(k) == i) {
1889 m_dVal_trsvHost(i) = m_utValHost(k);
1890 } else {
1891 m_utVal_trsvHost(nnzU) = m_utValHost(k);
1892 m_utColIdx_trsvHost(nnzU) = m_utColIdxHost(k);
1893 nnzU++;
1894 }
1895 }
1896 m_utRowMap_trsvHost(i+1)=nnzU;
1897 }
1898 for (Ordinal i = 0; i < m_nRows; i++) {
1899 for (Ordinal k = m_utRowMap_trsvHost(i); k < m_utRowMap_trsvHost[i+1]; k++) {
1900 m_utVal_trsvHost(k) = m_utVal_trsvHost(k) / m_dVal_trsvHost(i);
1901 }
1902 m_dVal_trsvHost(i) = STS::one() / m_dVal_trsvHost(i);
1903 }
1904 } else if (m_sptrsv_KKSpMV) {
1905 FastILUPrec_Functor functor(SwapDiagTag(), m_lVal, m_lRowMap, m_lColIdx, m_utVal, m_utRowMap, m_utColIdx, m_diagElems, m_blockCrsSize);
1906 Kokkos::RangePolicy<SwapDiagTag, ExecSpace> swap_policy (0, m_nRows);
1907 Kokkos::parallel_for(
1908 "numericILU::swapDiag", swap_policy, functor);
1909 }
1910 ExecSpace().fence(); // Fence so computeTime is accurate
1911 m_computeTime = timer.seconds();
1912 FASTILU_REPORT_TIMER(timer, " >> compute done\n");
1913 }

1915 template <typename CRS>
1916 void sptrsv_impl(ScalarArray &x, ScalarArray &y, CRS& crsmatL, CRS& crsmatU)
1917 {
1918 const Scalar one(1.0);
1919 const Scalar minus_one(-1.0);
1920
1921 const auto nrows_unblocked = m_xOld.extent(0);
1922 Scalar2dArray x2d_old (const_cast<Scalar*>(m_xOld.data()), nrows_unblocked, 1);
1923
1924 Scalar2dArray x2d (const_cast<Scalar*>(m_xTemp.data()), nrows_unblocked, 1);
1925 Scalar2dArray y2d (const_cast<Scalar*>(y.data()), nrows_unblocked, 1);
1926
1927 // 1) approximately solve, y = L^{-1}*x
1928 // functor to copy RHS x into y (for even iteration)
1929 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_x2y(y, m_xTemp);
1930 // functor to copy RHS x into xold (for odd iteration)
1931 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_x2xold(m_xOld, m_xTemp);
1932
1933 // functor to copy x_old to y (final iteration)
1934 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_xold2y(y, m_xOld);
1935
1936 // xold = zeros
1937 ParInitZeroFunctor<Ordinal, Scalar, ExecSpace> initZeroX(m_xOld);
1938 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), initZeroX);
1939 //Kokkos::deep_copy(x2d_old, zero);
1940 for (Ordinal i = 0; i < m_nTrisol; i++)
1941 {
1942 if (i%2 == 0) {
1943 // y = y - L*x_old
1944 // > y = x
1945 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_x2y);
1946 // > y = y - L*x_old
1947 KokkosSparse::spmv("N", minus_one, crsmatL, x2d_old, one, y2d);
1948 } else {
1949 // x_old = x_old - L*y
1950 // > x_old = x
1951 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_x2xold);
1952 // > x_old = x_old - L*y
1953 KokkosSparse::spmv("N", minus_one, crsmatL, y2d, one, x2d_old);
1954
1955 if (i == m_nTrisol-1) {
1956 // y = x_old
1957 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_xold2y);
1958 }
1959 }
1960 }
1961
1962 // 2) approximately solve, x = U^{-1}*y
1963 // functor to copy y into x
1964 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_y2x(m_xTemp, y);
1965 // functor to copy y into xold
1966 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_y2xold(m_xOld, y);
1967
1968 // functor to scale x
1969 ParScalFunctor<Ordinal, Scalar, Scalar, ExecSpace> scal_x(m_xTemp, m_xTemp, m_diagElems);
1970 ParScalFunctor<Ordinal, Scalar, Scalar, ExecSpace> scal_xold(m_xOld, m_xOld, m_diagElems);
1971
1972 // functor to copy x_old to x (final iteration)
1973 ParCopyFunctor<Ordinal, Scalar, ExecSpace> copy_xold2x(m_xTemp, m_xOld);
1974
1975 // xold = zeros
1976 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), initZeroX);
1977 //Kokkos::deep_copy(x2d_old, zero);
1978 for (Ordinal i = 0; i < m_nTrisol; i++)
1979 {
1980 if (i%2 == 0) {
1981 // x = y - U*x_old
1982 // > x = y
1983 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_y2x);
1984 // > x = x - U*x_old
1985 KokkosSparse::spmv("N", minus_one, crsmatU, x2d_old, one, x2d);
1986 // > scale x = inv(diag(U))*x
1987 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), scal_x);
1988 } else {
1989 // xold = y - U*x
1990 // > xold = y
1991 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_y2xold);
1992 // > x = x - U*x_old
1993 KokkosSparse::spmv("N", minus_one, crsmatU, x2d, one, x2d_old);
1994 // > scale x = inv(diag(U))*x
1995 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), scal_xold);
1996
1997 if (i == m_nTrisol-1) {
1998 // x = x_old
1999 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, nrows_unblocked), copy_xold2x);
2000 }
2001 }
2002 }
2003 }

2005 //Preconditioner application. Note that this does
2006 //*not* support multiple right hand sizes.
2007 void apply(ScalarArray &x, ScalarArray &y)
2008 {
2009 Kokkos::Timer timer;
2010
2011 //required to prevent contamination of the input.
2012 ParCopyFunctor<Ordinal, Scalar, ExecSpace> parCopyFunctor(m_xTemp, x);
2013 Kokkos::parallel_for(Kokkos::RangePolicy<ExecSpace>(0, x.extent(0)), parCopyFunctor);
2014
2015 //apply D
2016 if (m_useMetis) {
2017 applyD_Perm(x, m_xTemp);
2018 } else {
2019 applyD(x, m_xTemp);
2020 }
2021
2022 if (m_sptrsv_algo == FastILU::SpTRSV::Standard) {
2023 assert(m_blockCrsSize == 1); // Not yet supported for block crs
2024 // solve with L
2025 KokkosSparse::sptrsv_solve(&khL, m_lRowMap, m_lColIdx, m_lVal, m_xTemp, y);
2026 // solve with U
2027 KokkosSparse::sptrsv_solve(&khU, m_utRowMap, m_utColIdx, m_utVal, y, m_xTemp);
2028 } else {
2029 // wrap x and y into 2D views
2030 Scalar2dArray x2d (const_cast<Scalar*>(m_xTemp.data()), m_nRows, 1);
2031 Scalar2dArray y2d (const_cast<Scalar*>(y.data()), m_nRows, 1);
2032
2033 if (m_sptrsv_algo == FastILU::SpTRSV::StandardHost) {
2034 assert(m_blockCrsSize == 1); // Not yet supported for block crs
2035
2036 // copy x to host

2038 // don't need to initialize y_ because KokkosSparse::trsv always overwrites the output vectors
2039 // (there is no alpha/beta, it's as if beta is always 0)
2040 auto y_ = Kokkos::create_mirror_view(Kokkos::WithoutInitializing, y2d);
2041 Kokkos::deep_copy(x_, x2d);
2042
2043 if (m_doUnitDiag_TRSV) {
2044 using crsmat_host_t = KokkosSparse::CrsMatrix<Scalar, Ordinal, HostSpace, void, Ordinal>;
2045 using graph_host_t = typename crsmat_host_t::StaticCrsGraphType;

2057 // solve with D
2058 for (Ordinal i = 0; i < m_nRows; i++) {
2059 y_(i, 0) = m_dVal_trsvHost(i) * y_(i, 0);
2060 }
2061 // solve with U, unit-diag
2062 KokkosSparse::trsv ("U", "N", "U", crsmatU, y_, x_);
2063 } else {
2064 using crsmat_mirror_t = KokkosSparse::CrsMatrix<Scalar, Ordinal, MirrorSpace, void, Ordinal>;
2065 using graph_mirror_t = typename crsmat_mirror_t::StaticCrsGraphType;
2066
2067 // wrap L into crsmat on host
2068 graph_mirror_t static_graphL(m_lColIdxHost, m_lRowMapHost);
2069 crsmat_mirror_t crsmatL("CrsMatrix", m_nRows, m_lValHost, static_graphL);

2073 crsmat_mirror_t crsmatU("CrsMatrix", m_nRows, m_utValHost, static_graphU);
2074
2075 // solve with L
2076 KokkosSparse::trsv ("L", "N", "N", crsmatL, x_, y_);

2078 KokkosSparse::trsv ("U", "N", "N", crsmatU, y_, x_);
2079 }
2080 // copy x to device
2081 Kokkos::deep_copy(x2d, x_);
2082 } else {
2083 if (m_sptrsv_KKSpMV) {
2084 if (m_blockCrsSize == 1) {
2085 using crsmat_t = KokkosSparse::CrsMatrix<Scalar, Ordinal, ExecSpace, void, Ordinal>;
2086 using graph_t = typename crsmat_t::StaticCrsGraphType;
2087
2088 graph_t static_graphL(m_lColIdx, m_lRowMap);
2089 crsmat_t crsmatL("CrsMatrix", m_nRows, m_lVal, static_graphL);
2090
2091 graph_t static_graphU(m_utColIdx, m_utRowMap);
2092 crsmat_t crsmatU("CrsMatrix", m_nRows, m_utVal, static_graphU);
2093
2094 sptrsv_impl(x, y, crsmatL, crsmatU);
2095 }
2096 else {
2097 using crsmat_t = KokkosSparse::Experimental::BsrMatrix<Scalar, Ordinal, ExecSpace, void, Ordinal>;
2098 using graph_t = typename crsmat_t::StaticCrsGraphType;
2099
2100 graph_t static_graphL(m_lColIdx, m_lRowMap);
2101 crsmat_t crsmatL("BsrMatrix", m_nRows, m_lVal, static_graphL, m_blockCrsSize);
2102

2107 }
2108 } else {

2110
2111 //apply L^{-1} to xTemp
2112 applyL(m_xTemp, y);
2113 //apply U^{-1} to y
2114 applyU(y, m_xTemp);
2115 }
2116 }
2117 }

2119 //symmetric for now).
2120 if (m_useMetis) {
2121 applyD_iPerm(m_xTemp, y);
2122 } else {
2123 applyD(m_xTemp, y);
2124 }

2126 ExecSpace().fence();
2127 m_applyTime = timer.seconds();

2172 //Compute the L2 norm of the nonlinear residual (A - LU) on sparsity pattern
2173 void checkILU() const
2174 {
2175 assert(m_blockCrsSize == 1); // Not yet supported for block crs
2176 Scalar sum = 0.0;
2177 Scalar sum_diag = 0.0;
2178 for (int i = 0 ; i < m_nRows; i++)
2179 {
2180 // Each row in A matrix (result)
2181 for (int k = m_aRowMap[i]; k < m_aRowMap[i+1]; k++)
