21bool supportsUnsymmetric(const std::string& prec_type) {
22 bool result = false;
23 if (prec_type == "RELAXATION" ||
24 prec_type == "CHEBYSHEV" ||
25 prec_type == "DIAGONAL" ||
26 prec_type == "MDF" ||
27 prec_type == "RILUK" ||
28 prec_type == "RBILUK" ||
29 prec_type == "ILUT" ||
30 prec_type == "SCHWARZ" ||
31 prec_type == "KRYLOV"
32#ifdef HAVE_IFPACK2_HYPRE

34#endif
35 ) {
36 result = true;
37 } else {
38 TEUCHOS_TEST_FOR_EXCEPTION(
39 true, std::invalid_argument,
40 "Ifpack2::supportsUnsymmetric: "
41 "Unrecognized preconditioner type prec_type = \""
42 << prec_type
43 << "\"");
44 }
45 return result;
46}
