36// Customized elimination on the rhs that I own
37template<class ZView, class RHSView, class DView>
38void elimination_rhs(int N, ZView& ptr2, RHSView& ptr3, DView& ptr4, int act_col) {
39#if defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP)
40 Kokkos::parallel_for(Kokkos::RangePolicy<typename ZView::device_type::execution_space>(0,N), KOKKOS_LAMBDA (const int i) {
41 ptr4(0,i) = ptr3(i)/ptr2(act_col);
42 ptr3(i) = ptr4(0,i);
43 });
44#else//OpenMP //Note: FOR loop is faster than parallel_for when execution_space is OpenMP
45 for(int i=0;i<N;i++){
46 ptr4(0,i) = ptr3(i)/ptr2(act_col);
47 ptr3(i) = ptr4(0,i);
48 }
49#endif
50}

52template<class HandleType, class ZViewType, class RHSViewType>

54void back_solve_rhs_pipelined_comm(HandleType& ahandle, ZViewType& Z, RHSViewType& RHS)
55{
56 using value_type = typename ZViewType::value_type;
57#ifdef PRINT_STATUS
58 using execution_space = typename ZViewType::device_type::execution_space;
59#endif
60 using memory_space = typename ZViewType::device_type::memory_space;
61 using View2DType = Kokkos::View<value_type**, Kokkos::LayoutLeft, memory_space>;
62
63#if defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)
64#if defined(KOKKOS_ENABLE_CUDA)
65 using View2DHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace>;//CudaHostPinnedSpace
66#elif defined(KOKKOS_ENABLE_HIP)
67 using View2DHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace>;//HIPHostPinnedSpace
68#endif
69#endif
70
71 MPI_Comm comm = ahandle.get_comm();
72 MPI_Comm col_comm = ahandle.get_col_comm();
73 int me = ahandle.get_myrank();
74 int nprocs_row = ahandle.get_nprocs_row();
75 int nprocs_col = ahandle.get_nprocs_col();
76 int ncols_matrix = ahandle.get_ncols_matrix();
77 int my_rows = ahandle.get_my_rows();
78 int my_cols = ahandle.get_my_cols();
79 int my_first_row = ahandle.get_my_first_row();
80 int my_first_col = ahandle.get_my_first_col();
81 int nrhs = ahandle.get_nrhs();
82 int my_rhs = ahandle.get_my_rhs();
83
84 int j; // loop counters
85 int end_row; // row num to end column operations
86 int bytes[16]; // number of bytes in messages
87 int root; // root processor for fanout
88 int type[16]; // mesage type for messages
89 int dest[16]; // dest for message sends
90
91 int one = 1;
92
93 value_type d_one = 1.0;
94 value_type d_min_one = -1.0;
95
96 int j2;
97
98 int n_rhs_this; // num rhs that I currently own
99 int col_offset; // which processor starts the pipeline
100 int my_pos; // my position in the new linup
101 int extra; // extra loop to realign data after pipeline
102 int act_col; // act this column (that I own)
103 int on_col; // on this collection of rhs's
104 int global_col; // global col number for act_col
105 int max_bytes; // max number of bytes of rhs I can receive
106
107 int my_col_id, my_row_id, id_temp;
108 int dest_right, dest_left;
109
110#ifdef GET_TIMING
111 double t1,t2;
112 double allocviewtime,eliminaterhstime,bcastrowtime,updrhstime,xchgrhstime;
113 double totalsolvetime;
114#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
115 double copyhostpinnedtime;
116#endif
117#endif
118
119 MPI_Request msgrequest;
120 MPI_Status msgstatus;
121
122#ifdef GET_TIMING
123 t2 = MPI_Wtime();
124#endif
125
126 // find left, right destination procs
127
128 my_col_id = mesh_col(me);
129 my_row_id = mesh_row(me);
130
131 id_temp = my_col_id + 1;
132 if (id_temp >= nprocs_row) id_temp = 0;
133 dest_right = proc_num(my_row_id,id_temp);
134
135 id_temp = my_col_id - 1;
136 if (id_temp < 0) id_temp = nprocs_row-1;
137 dest_left = proc_num(my_row_id,id_temp);
138
139 // set j2 to be first column in last group of columns
140 max_bytes = nrhs/nprocs_row;
141 if (nrhs%nprocs_row > 0) max_bytes++;
142 max_bytes = max_bytes*sizeof(ADELUS_DATA_TYPE)*my_rows;
143
144#ifdef GET_TIMING
145 allocviewtime=eliminaterhstime=bcastrowtime=updrhstime=xchgrhstime=0.0;
146#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
147 copyhostpinnedtime=0.0;
148#endif
149
150 t1 = MPI_Wtime();
151#endif
152
153 View2DType row1( "row1", one, nrhs ); // row1: diagonal row (temp variables)
154#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
155 View2DHostPinnType h_row2( "h_row2", my_rows, max_bytes/sizeof(ADELUS_DATA_TYPE)/my_rows );
156#else
157 View2DType row2( "row2", my_rows, max_bytes/sizeof(ADELUS_DATA_TYPE)/my_rows );
158#endif
159#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
160 View2DHostPinnType h_row1( "h_row1", one, nrhs );
161 View2DHostPinnType h_rhs ( "h_rhs", my_rows, nrhs );
162#endif
163
164 Kokkos::fence();
165
166#ifdef GET_TIMING
167 allocviewtime += (MPI_Wtime()-t1);
168#endif
169
170 n_rhs_this = my_rhs;// why "n_rhs_this" is changing after the first iteration (need to as Joe???)
171 j2 = ncols_matrix-1;
172 col_offset = (j2%nprocs_row);
173 my_pos = my_first_col - col_offset;
174 if (my_pos < 0) my_pos += nprocs_row;
175 extra = (nprocs_row - (col_offset-1))%nprocs_row;
176
177 act_col = my_cols-1;
178 if (my_pos != 0) act_col++;
179
180 on_col = my_pos;
181
182#ifdef PRINT_STATUS
183 printf("Rank %i -- back_solve6() Begin back solve, execution_space %s, memory_space %s\n",me, typeid(execution_space).name(), typeid(memory_space).name());
184#endif
185
186 for (j = j2; j >= 1-nprocs_row-extra; j--) {
187
188 if ((j+nprocs_row-1 >= 0) && (n_rhs_this > 0)) {
189
190 if ((act_col < my_cols) && (act_col >= 0)) {
191
192 global_col = act_col*nprocs_row + my_first_col;
193
194 end_row = global_col/nprocs_col;
195 if (my_first_row <= global_col%nprocs_col) ++end_row;
196
197 // do an elimination step on the rhs that I own
198
199 //auto ptr2_view = subview(Z, end_row-1, Kokkos::ALL());
200
201 root = row_owner(global_col);
202
203 if (me == root) {
204#ifdef GET_TIMING
205 t1 = MPI_Wtime();
206#endif
207 auto ptr2_view = subview(Z, end_row-1, Kokkos::ALL());
208 auto ptr3_view = subview(RHS, end_row-1, Kokkos::make_pair(0, n_rhs_this));
209 elimination_rhs(n_rhs_this, ptr2_view, ptr3_view, row1, act_col);//note: row1 = ptr4
210//Workaround for GPU-aware MPI issue on HIP: fence here to make sure "elimination_rhs" completes before MPI_Bcast
211#if defined(KOKKOS_ENABLE_HIP) && !defined(ADELUS_HOST_PINNED_MEM_MPI)
212 Kokkos::fence();
213#endif
214 end_row--;
215#ifdef GET_TIMING
216 eliminaterhstime += (MPI_Wtime()-t1);
217#endif
218 }
219
220#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
221#ifdef GET_TIMING
222 t1 = MPI_Wtime();
223#endif
224 Kokkos::deep_copy(h_row1,row1);
225#ifdef GET_TIMING
226 copyhostpinnedtime += (MPI_Wtime()-t1);
227#endif
228#endif
229
230#ifdef GET_TIMING
231 t1 = MPI_Wtime();
232#endif
233 bytes[0] = n_rhs_this*sizeof(ADELUS_DATA_TYPE);
234 type[0] = SOCOLTYPE+j;
235
236 //MPI_Bcast((char *) row1, bytes[0], MPI_CHAR, mesh_row(root), col_comm);
237#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
238 MPI_Bcast(reinterpret_cast<char *>(h_row1.data()), bytes[0], MPI_CHAR, mesh_row(root), col_comm);
239#else //GPU-aware MPI
240 MPI_Bcast(reinterpret_cast<char *>(row1.data()), bytes[0], MPI_CHAR, mesh_row(root), col_comm);
241#endif
242 // added this barrier for CPLANT operation
243
244 MPI_Barrier(col_comm);
245#ifdef GET_TIMING
246 bcastrowtime += (MPI_Wtime()-t1);
247#endif
248
249#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
250#ifdef GET_TIMING
251 t1 = MPI_Wtime();
252#endif
253 Kokkos::deep_copy(row1,h_row1);
254#ifdef GET_TIMING
255 copyhostpinnedtime += (MPI_Wtime()-t1);
256#endif
257#endif
258
259#ifdef GET_TIMING
260 t1 = MPI_Wtime();
261#endif
262
263 auto A_view = subview(Z, Kokkos::make_pair(0, end_row), Kokkos::make_pair(act_col, act_col+one));
264 auto C_view = subview(RHS, Kokkos::make_pair(0, end_row), Kokkos::make_pair(0, n_rhs_this));
265 auto B_view = subview(row1, Kokkos::ALL(), Kokkos::make_pair(0, n_rhs_this));
266
267 KokkosBlas::gemm("N","N",d_min_one,
268 A_view,
269 B_view,
270 d_one,
271 C_view);
272//Workaround for GPU-aware MPI issue on HIP: fence here to make sure "gemm" completes before MPI_Send
273#if defined(KOKKOS_ENABLE_HIP) && !defined(ADELUS_HOST_PINNED_MEM_MPI)
274 Kokkos::fence();
275#endif
276#ifdef GET_TIMING
277 updrhstime += (MPI_Wtime()-t1);
278#endif
279 }
280 }
281
282#ifdef GET_TIMING
283 t1 = MPI_Wtime();
284#endif
285 if (j != 1-nprocs_row-extra) {
286 dest[0] = dest_right;
287 if (me != dest[0]) {
288 bytes[0] = max_bytes;
289 type[0] = SOROWTYPE+j;
290
291#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
292 MPI_Irecv(reinterpret_cast<char *>(h_row2.data()), bytes[0], MPI_CHAR, MPI_ANY_SOURCE, type[0], comm, &msgrequest);
293#else
294 MPI_Irecv(reinterpret_cast<char *>( row2.data()), bytes[0], MPI_CHAR, MPI_ANY_SOURCE, type[0], comm, &msgrequest);
295#endif
296
297 n_rhs_this = bytes[0]/sizeof(ADELUS_DATA_TYPE)/my_rows;
298
299#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
300 Kokkos::deep_copy(subview(h_rhs, Kokkos::ALL(), Kokkos::make_pair(0, n_rhs_this)),
301 subview(RHS, Kokkos::ALL(), Kokkos::make_pair(0, n_rhs_this)));
302#endif
303
304 dest[1] = dest_left;
305 bytes[1] = n_rhs_this * sizeof(ADELUS_DATA_TYPE) * my_rows;
306 type[1] = SOROWTYPE+j;
307
308#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
309 MPI_Send(reinterpret_cast<char *>(h_rhs.data()), bytes[1], MPI_CHAR, dest[1], type[1], comm);
310#else //GPU-aware MPI
311 MPI_Send(reinterpret_cast<char *>(RHS.data()), bytes[1], MPI_CHAR, dest[1], type[1], comm);
312#endif
313
314 MPI_Wait(&msgrequest,&msgstatus);
315
316 // Copy row2 -> rhs
317 int blas_length = n_rhs_this*my_rows;
318#if (defined(ADELUS_HOST_PINNED_MEM_MPI) || defined(IBM_MPI_WRKAROUND2)) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP)) //Use memcpy for now, can use deep_copy in the future //deep_copy is slower than BLAS XCOPY
319#if defined(KOKKOS_ENABLE_CUDA)
320 //Kokkos::deep_copy(subview(RHS, Kokkos::ALL(), Kokkos::make_pair(0, n_rhs_this)), subview(h_row2, Kokkos::ALL(), Kokkos::make_pair(0, n_rhs_this)));
321 cudaMemcpy(reinterpret_cast<ADELUS_DATA_TYPE *>(RHS.data()), reinterpret_cast<ADELUS_DATA_TYPE *>(h_row2.data()), blas_length*sizeof(ADELUS_DATA_TYPE), cudaMemcpyHostToDevice);
322#elif defined(KOKKOS_ENABLE_HIP)
323 hipMemcpy(reinterpret_cast<ADELUS_DATA_TYPE *>(RHS.data()), reinterpret_cast<ADELUS_DATA_TYPE *>(h_row2.data()), blas_length*sizeof(ADELUS_DATA_TYPE), hipMemcpyHostToDevice);
324#endif
325#else
326#if defined(KOKKOS_ENABLE_CUDA)
327 cudaMemcpy(reinterpret_cast<ADELUS_DATA_TYPE *>(RHS.data()), reinterpret_cast<ADELUS_DATA_TYPE *>(row2.data()), blas_length*sizeof(ADELUS_DATA_TYPE), cudaMemcpyDeviceToDevice);
328#elif defined(KOKKOS_ENABLE_HIP)
329 hipMemcpy(reinterpret_cast<ADELUS_DATA_TYPE *>(RHS.data()), reinterpret_cast<ADELUS_DATA_TYPE *>(row2.data()), blas_length*sizeof(ADELUS_DATA_TYPE), hipMemcpyDeviceToDevice);
330#else
331 memcpy(reinterpret_cast<ADELUS_DATA_TYPE *>(RHS.data()), reinterpret_cast<ADELUS_DATA_TYPE *>(row2.data()), blas_length*sizeof(ADELUS_DATA_TYPE));
332#endif
333#endif
334 }
335 on_col++;
336 if (on_col >= nprocs_row) {
337 on_col = 0;
338 act_col--;
339 }
340 }
341#ifdef GET_TIMING
342 xchgrhstime += (MPI_Wtime()-t1);
343#endif
344
345 }
346
347#ifdef GET_TIMING
348 totalsolvetime = MPI_Wtime() - t2;
349#endif
350#ifdef GET_TIMING
351 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to alloc view", &allocviewtime);
352 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to eliminate rhs",&eliminaterhstime);
353 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to bcast temp row",&bcastrowtime);
354 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to update rhs",&updrhstime);
355#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
356 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to copy host pinned mem <--> dev mem",&copyhostpinnedtime);
357#endif
358 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Time to xchg rhs",&xchgrhstime);
359 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Total time in solve",&totalsolvetime);
360#endif
361}

363template<class HandleType, class ZViewType, class RHSViewType>
364inline
365void back_solve_currcol_bcast(HandleType& ahandle, ZViewType& Z, RHSViewType& RHS)
366{
367 using value_type = typename ZViewType::value_type;
368#ifdef PRINT_STATUS
369 using execution_space = typename ZViewType::device_type::execution_space;
370#endif
371 using memory_space = typename ZViewType::device_type::memory_space;
372 using View2DType = Kokkos::View<value_type**, Kokkos::LayoutLeft, memory_space>;
373
374#if defined(ADELUS_HOST_PINNED_MEM_MPI)
375#if defined(KOKKOS_ENABLE_CUDA)
376 using View2DHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace>;//CudaHostPinnedSpace
377#elif defined(KOKKOS_ENABLE_HIP)
378 using View2DHostPinnType = Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace>;//HIPHostPinnedSpace
379#endif
380#endif
381
382#if defined(GET_TIMING) || defined(PRINT_STATUS)
383 int me = ahandle.get_myrank();
384#ifdef GET_TIMING
385 MPI_Comm comm = ahandle.get_comm();
386#endif
387#endif
388 MPI_Comm col_comm = ahandle.get_col_comm();
389 MPI_Comm row_comm = ahandle.get_row_comm();
390 int myrow = ahandle.get_myrow();
391 int mycol = ahandle.get_mycol();
392 int nprocs_row = ahandle.get_nprocs_row();
393 int nprocs_col = ahandle.get_nprocs_col();
394 int ncols_matrix = ahandle.get_ncols_matrix();
395 int my_rows = ahandle.get_my_rows();
396 int my_rhs = ahandle.get_my_rhs();
397
398 value_type d_one = 1.0;
399 value_type d_min_one = -1.0;
400
401#ifdef GET_TIMING
402 double t1,t2;
403 double allocviewtime,eliminaterhstime,bcastrowtime,updrhstime,bcastcoltime,copycoltime;
404 double totalsolvetime;
405#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
406 double copyhostpinnedtime;
407#endif
408#endif
409
410#ifdef GET_TIMING
411 t2 = MPI_Wtime();
412#endif
413
414#ifdef GET_TIMING
415 allocviewtime=eliminaterhstime=bcastrowtime=updrhstime=bcastcoltime=copycoltime=0.0;
416#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
417 copyhostpinnedtime=0.0;
418#endif
419
420 t1 = MPI_Wtime();
421#endif
422
423 View2DType curr_col( "curr_col", my_rows, 1 ); //current column
424 View2DType rhs_row ( "rhs_row", 1, my_rhs ); //current row of RHS to hold the elimination results (i.e row of solution)
425#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
426 View2DHostPinnType h_curr_col( "h_curr_col", my_rows, 1 );
427 View2DHostPinnType h_rhs_row( "h_rhs_row", 1, my_rhs );
428#endif
429
430 //Kokkos::fence();//NOTE: Should we need this?
431
432#ifdef GET_TIMING
433 allocviewtime += (MPI_Wtime()-t1);
434#endif
435
436#ifdef PRINT_STATUS
437 printf("Rank %i -- back_solve6() Begin back solve, execution_space %s, memory_space %s\n",me, typeid(execution_space).name(), typeid(memory_space).name());
438#endif
439
440 for (int k = ncols_matrix-1; k >= 0; k--) {
441 int k_row = k%nprocs_col;//proc. id (in the col_comm) having global k
442 int k_col = k%nprocs_row;//proc. id (in the row_comm) having global k
443 int end_row = k/nprocs_col;
444 if (myrow <= k_row) end_row++;
445
446#ifdef GET_TIMING
447 t1 = MPI_Wtime();
448#endif
449 //Step 1: copy the current column of Z to a temporary view
450 if (mycol == k_col) { //only deep_copy if holding the current column
451 Kokkos::deep_copy( Kokkos::subview(curr_col, Kokkos::make_pair(0, end_row), 0),
452 Kokkos::subview(Z, Kokkos::make_pair(0, end_row), k/nprocs_row) );
453 }
454#ifdef GET_TIMING
455 copycoltime += (MPI_Wtime()-t1);
456#endif
457
458#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
459#ifdef GET_TIMING
460 t1 = MPI_Wtime();
461#endif
462 Kokkos::deep_copy(h_curr_col,curr_col);
463#ifdef GET_TIMING
464 copyhostpinnedtime += (MPI_Wtime()-t1);
465#endif
466#endif
467
468#ifdef GET_TIMING
469 t1 = MPI_Wtime();
470#endif
471 //Step 2: broadcast the current column to all ranks in the row_comm
472#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
473 MPI_Bcast(reinterpret_cast<char *>(h_curr_col.data()), end_row*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_col, row_comm);
474#else //GPU-aware MPI
475 MPI_Bcast(reinterpret_cast<char *>(curr_col.data()), end_row*sizeof(ADELUS_DATA_TYPE), MPI_CHAR, k_col, row_comm);
476#endif
477#ifdef GET_TIMING
478 bcastcoltime += (MPI_Wtime()-t1);
479#endif
480
481#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
482#ifdef GET_TIMING
483 t1 = MPI_Wtime();
484#endif
485 Kokkos::deep_copy(curr_col,h_curr_col);
486#ifdef GET_TIMING
487 copyhostpinnedtime += (MPI_Wtime()-t1);
488#endif
489#endif
490
491#ifdef GET_TIMING
492 t1 = MPI_Wtime();
493#endif
494 //Step 3: do rhs elimination to get solution x
495 if (myrow == k_row) {//only on ranks having row k
496 if (my_rhs > 0) { //only on ranks having some rhs
497 auto sub_curr_col = Kokkos::subview(curr_col, end_row-1, Kokkos::ALL());
498 auto sub_rhs = Kokkos::subview(RHS, end_row-1, Kokkos::make_pair(0, my_rhs));
