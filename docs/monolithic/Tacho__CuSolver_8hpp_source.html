98 template <typename arg_size_type_array, typename arg_ordinal_type_array>
99 int analyze(const ordinal_type m, const arg_size_type_array &ap, const arg_ordinal_type_array &aj) {
100 if (_verbose) {
101 printf("cuSolver: Analyze\n");
102 printf("=================\n");
103 }
104 Kokkos::Timer timer;
105
106 _m = m;
107
108 timer.reset();
109
110 _ap = Kokkos::create_mirror_view(exec_memory_space(), ap);
111 Kokkos::deep_copy(_ap, ap);
112 _aj = Kokkos::create_mirror_view(exec_memory_space(), aj);
113 Kokkos::deep_copy(_aj, aj);
114 Kokkos::fence();
115#if defined(TACHO_USE_INT_INT)
116 _ap_ordinal = ap;
117#else
119 {
120 ordinal_type_array l_ap_ordinal(do_not_initialize_tag("CuSolver::ap_ordinal"), _ap.extent(0));
121 auto l_ap = _ap;
122 Kokkos::RangePolicy<exec_space, Kokkos::Schedule<Kokkos::Static>> policy(0, l_ap.extent(0));
123 Kokkos::parallel_for(
124 policy, KOKKOS_LAMBDA(const int i) { l_ap_ordinal(i) = static_cast<int>(l_ap(i)); });
125 _ap_ordinal = l_ap_ordinal;
126 }
127#endif
128 auto last = Kokkos::subview(_ap, _m);
129 auto h_last = Kokkos::create_mirror_view(host_memory_space(), last);
130 Kokkos::deep_copy(h_last, last);
131 _nnz = h_last();
132 const double t_copy = timer.seconds();
133
134 timer.reset();
135#if defined(TACHO_HAVE_CUSPARSE)
136 _status = cusolverSpXcsrcholAnalysis(_handle, _m, _nnz, _desc, _ap_ordinal.data(), _aj.data(), _chol_info);
137#endif
138 checkStatus("cusolverSpXcsrcholAnalysis");
139 Kokkos::fence();
140 const double t_analyze = timer.seconds();
141
142 if (_verbose) {
143 printf(" Linear system A\n");
144 printf(" number of equations: %10d\n", _m);
145 printf(" number of nonzeros: %10ld\n",(long unsigned int) _nnz);
146 printf("\n");
147 printf(" Time\n");
148 printf(" time for copying A into U: %10.6f s\n", t_copy);
149 printf(" time for analysis: %10.6f s\n", t_analyze);
150 printf(" total time spent: %10.6f s\n", (t_copy + t_analyze));
151 printf("\n");
152 }
153
154 return 0;
155 }

157 int factorize(const value_type_array &ax) {
158 if (_verbose) {
159 printf("cuSolver: Factorize\n");
160 printf("===================\n");
161 }
162 Kokkos::Timer timer;
163
164 timer.reset();
165 size_t internalDataInBytes = 0, workspaceInBytes = 0;
166#if defined(TACHO_HAVE_CUSPARSE)
167 _status = cusolverSpDcsrcholBufferInfo(_handle, _m, _nnz, _desc, ax.data(), _ap_ordinal.data(), _aj.data(),
168 _chol_info, &internalDataInBytes, &workspaceInBytes);
169#endif
170 checkStatus("cusolverSpDcsrcholBufferInfo");
171
172 const size_t bufsize = workspaceInBytes / sizeof(value_type);
173 if (bufsize > _buf.extent(0))
174 _buf = value_type_array(do_not_initialize_tag("cusolver buf"), bufsize);
175 Kokkos::fence();
176 const double t_alloc = timer.seconds();
177
178 timer.reset();
179#if defined(TACHO_HAVE_CUSPARSE)
180 _status = cusolverSpDcsrcholFactor(_handle, _m, _nnz, _desc, ax.data(), _ap_ordinal.data(), _aj.data(), _chol_info,
181 _buf.data());
182#endif
183 checkStatus("cusolverSpDcsrcholFactor");
184 Kokkos::fence();
185 const double t_factor = timer.seconds();
186 if (_verbose) {
187 printf(" Time\n");
188 printf(" time for workspace allocation: %10.6f s\n", t_alloc);
189 printf(" time for numeric factorization: %10.6f s\n", t_factor);
190 printf(" total time spent: %10.6f s\n", (t_alloc + t_factor));
191 printf("\n");
192 printf(" Workspace\n");
193 printf(" internal data in MB: %10.3f MB\n",
194 double(internalDataInBytes) / 1.e6);
195 printf(" workspace in MB: %10.3f MB\n", double(workspaceInBytes) / 1.e6);
196 printf("\n");
197 }
198
199 return 0;
200 }

203 if (_verbose) {
204 printf("cuSolver: Solve\n");
205 printf("===============\n");
206 }
207 Kokkos::Timer timer;
208
209 timer.reset();
211 const ordinal_type len = x.extent(0), nrhs = x.extent(1);
212 for (ordinal_type i = 0; i < nrhs; ++i) {
213 _status = cusolverSpDcsrcholSolve(_handle, _m, b.data() + i * len, x.data() + i * len, _chol_info, _buf.data());
214 checkStatus("cusolverSpDcsrcholSolve");
215 }
216 Kokkos::fence();
217 const double t_solve = timer.seconds();
218 if (_verbose) {
219 printf(" Time\n");
220 printf(" time for solve: %10.6f s\n", t_solve);
221 printf("\n");
222 }
223
224 return 0;
225 }
