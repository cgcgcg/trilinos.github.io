26 // Computes C = this * B
27 Matrix3 operator*(const Matrix3& B) const {
28 Matrix3 C;
29 for(int i=0; i<3; i++)
30 for(int j=0; j<3; j++)
31 for(int k=0; k<3; k++)
32 C(i,j) += (*this)(i,k)* B(k,j);
33 return C;
34 }

58Matrix3 DiagonalMatrix(const std::vector<double> &d) {
59 Matrix3 rv;
60 for(int i=0; i<3; i++)
61 rv(i,i) = d[i];
62 return rv;
63}

65Matrix3 X_Rotation(double theta) {
66 std::vector<double> v {1.0, 0.0, 0.0, 0.0, cos(theta), -sin(theta), 0.0, sin(theta), cos(theta)};
67 return Matrix3(v);
68}
69
70Matrix3 Y_Rotation(double theta) {
71 std::vector<double> v {cos(theta), 0.0, sin(theta), 0.0, 1.0, 0.0, -sin(theta), 0.0, cos(theta)};
72 return Matrix3(v);
73}
74
75Matrix3 Z_Rotation(double theta) {
76 std::vector<double> v {cos(theta), -sin(theta), 0.0, sin(theta), cos(theta), 0.0, 0.0, 0.0, 1.0};
77 return Matrix3(v);
78}

80Matrix3 Transpose(const Matrix3 & A) {
81 Matrix3 B;
82 for(int i=0; i<3; i++)
83 for(int j=0; j<3; j++)
84 B(j,i) = A(i,j);
85 return B;

134void setDiffusionRotationAndStrength(const std::vector<double>& theta, const std::vector<double>& diagonal) {
135
136 rotation_ = Z_Rotation(theta[2]*M_PI/180.0) * Y_Rotation(theta[1]*M_PI/180.0) * X_Rotation(theta[0]*M_PI/180.0);
137 strength_ = DiagonalMatrix(diagonal);
138 diff_total_ = rotation_ * strength_ * Transpose(rotation_);
139 use_diffusion_ = true;
140}
