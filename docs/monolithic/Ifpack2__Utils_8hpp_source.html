105class Ifpack2_Element {
106 public:
107 Ifpack2_Element(){};
108
109 Ifpack2_Element(const Ifpack2_Element& rhs) {
110 i_ = rhs.Index();
111 val_ = rhs.Value();
112 aval_ = rhs.AbsValue();
113 }
114
115 inline int Index() const {
116 return (i_);
117 }
118
119 inline double Value() const {
120 return (val_);
121 }
122
123 inline double AbsValue() const {
124 return (aval_);
125 }
126
127 inline void SetIndex(const int i) {
128 i_ = i;
129 }
130
131 inline void SetValue(const double val) {
132 val_ = val;
133 aval_ = std::abs(val_);
134 }
135
136 inline bool operator<(const Ifpack2_Element& rhs) const {
137 if (rhs.AbsValue() > AbsValue())
138 return (false);
139 else if (rhs.AbsValue() < AbsValue())
140 return (true);
141 else if (rhs.Index() < Index())
142 return (true);
143 return (false);
144 }
145
146 private:
147 int i_;
148 double val_;
149 double aval_;
150};
