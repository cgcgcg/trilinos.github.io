402 template <typename arg_size_type_array, typename arg_ordinal_type_array>
403 int analyze(const ordinal_type m, const ordinal_type blk_size,
404 const arg_size_type_array &ap, const arg_ordinal_type_array &aj,
405 const bool duplicate = false) {
406
407 if (blk_size > 1) {
408 //condense graph before calling analyze
409 const size_type nnz = ap(m);
410 ordinal_type m_graph = m / blk_size;
411 size_type nnz_graph = nnz / (blk_size*blk_size);
412 TACHO_TEST_FOR_EXCEPTION((m != blk_size * m_graph || nnz != size_type(blk_size*blk_size) * nnz_graph),
413 std::logic_error, "Failed to initialize the condensed graph");
414
415 size_type_array_host ap_graph
416 (Kokkos::ViewAllocateWithoutInitializing("ap_graph"), 1+m_graph);
417 ordinal_type_array_host aj_graph
418 (Kokkos::ViewAllocateWithoutInitializing("aj_graph"), nnz_graph);
419 ordinal_type_array_host aw_graph
420 (Kokkos::ViewAllocateWithoutInitializing("wgs"), m_graph);
421 // condense the graph
422 nnz_graph = 0;
423 ap_graph(0) = 0;
424 for (ordinal_type i = 0; i < m; i += blk_size) {
425 for (size_type k = ap(i); k < ap(i+1); k++) {
426 if (aj(k)%blk_size == 0) {
427 aj_graph(nnz_graph) = aj(k)/blk_size;
428 nnz_graph++;
429 }
430 aw_graph(i/blk_size) = blk_size;
431 ap_graph((i/blk_size)+1) = nnz_graph;
432 }
433 }
434 TACHO_TEST_FOR_EXCEPTION((nnz != size_type(blk_size*blk_size) * nnz_graph),
435 std::logic_error, "Failed to condense graph");
436 return analyze(m, ap, aj, m_graph, ap_graph, aj_graph, aw_graph, duplicate);
437 } else {
438 return analyze(m, ap, aj, duplicate);
439 }
440 }
