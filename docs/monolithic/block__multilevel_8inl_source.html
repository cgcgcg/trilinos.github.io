71template <typename MatrixType, typename SmootherType, typename SolverType>
72template <typename Array1, typename Array2, typename Monitor>
73void block_multilevel<MatrixType,SmootherType,SolverType>
74::solve(const Array1& b, Array2& x, Monitor& monitor)
75{
76 CUSP_PROFILE_SCOPED();
77 const MatrixType& A = levels[0].A;
78 const size_t n = A.num_rows;
79 const size_t numRHS = b.num_cols;
80
81 // use simple iteration
82 cusp::array2d<ValueType,MemorySpace,Orientation> update(n, numRHS);
83 cusp::array2d<ValueType,MemorySpace,Orientation> residual(n, numRHS);
84
85 // compute initial residual
86 cusp::MVmultiply(A, x, residual);
87 cusp::subtract(b, residual, residual);
88 while(!monitor.finished(residual))
89 {
90 _solve(residual, update, 0);
91
92 // x += M * r
93 cusp::add(update, x, x);
94
95 // update residual
96 cusp::MVmultiply(A, x, residual);
97 cusp::axpby_array(ValueType(1.0), b, ValueType(-1.0), residual, residual);
98 ++monitor;
99 }
100}

102template <typename MatrixType, typename SmootherType, typename SolverType>
103template <typename Array1, typename Array2>
104void block_multilevel<MatrixType,SmootherType,SolverType>
105::_solve(const Array1& b, Array2& x, const size_t i)
106{
107 TEUCHOS_FUNC_TIME_MONITOR("CUSP Block Multilevel Solve");
108
109 CUSP_PROFILE_SCOPED();
110 const size_t numRHS = b.num_cols;
111
112 if (i + 1 == levels.size())
113 {
114 TEUCHOS_FUNC_TIME_MONITOR_DIFF("CUSP Coarse-grid Solve", coarse_grid);
115
116 // coarse grid solve
117 // copy to host first
118 cusp::array2d<ValueType,cusp::host_memory,Orientation> temp_b(b);
119 cusp::array2d<ValueType,cusp::host_memory,Orientation> temp_x(x.num_rows, numRHS);
120 solver(temp_b, temp_x);
121 cusp::copy(temp_x, x);
122 }
123 else
124 {
125 const MatrixType& A = levels[i].A;
126 // presmooth
127 levels[i].smoother.presmooth(A, b, x);
128
129 // compute residual <- b - A*x
130 cusp::MVmultiply(A, x, levels[i].residual);
131 cusp::axpby_array(ValueType(1.0), b, ValueType(-1.0), levels[i].residual, levels[i].residual);
132
133 // restrict to coarse grid
134 cusp::MVmultiply(levels[i].R, levels[i].residual, levels[i + 1].b);
135
136 // compute coarse grid solution
137 _solve(levels[i + 1].b, levels[i + 1].x, i + 1);
138
139 // apply coarse grid correction
140 cusp::MVmultiply(levels[i].P, levels[i + 1].x, levels[i].residual);
141 cusp::axpby_array(ValueType(1.0), levels[i].residual, ValueType(1.0), x, x);
142
143 // postsmooth
144 levels[i].smoother.postsmooth(A, b, x);
145 }
146}

148template <typename MatrixType, typename SmootherType, typename SolverType>
149void block_multilevel<MatrixType,SmootherType,SolverType>
150::print( void )
151{
152 size_t num_levels = levels.size();
153
154 std::cout << "\tNumber of Levels:\t" << num_levels << std::endl;
155 std::cout << "\tOperator Complexity:\t" << operator_complexity() << std::endl;
156 std::cout << "\tGrid Complexity:\t" << grid_complexity() << std::endl;
157 std::cout << "\tlevel\tunknowns\tnonzeros:\t" << std::endl;
158
159 double nnz = 0;
160
161 for(size_t index = 0; index < num_levels; index++)
162 nnz += levels[index].A.num_entries;
163
164 for(size_t index = 0; index < num_levels; index++)
165 {
166 double percent = levels[index].A.num_entries / nnz;
167 std::cout << "\t" << index << "\t" << levels[index].A.num_cols << "\t\t" \
168 << levels[index].A.num_entries << " \t[" << 100*percent << "%]" \
169 << std::endl;
170 }
171}

173template <typename MatrixType, typename SmootherType, typename SolverType>
174double block_multilevel<MatrixType,SmootherType,SolverType>
175::operator_complexity( void )
176{
177 size_t nnz = 0;
178
179 for(size_t index = 0; index < levels.size(); index++)
180 nnz += levels[index].A.num_entries;
181
182 return (double) nnz / (double) levels[0].A.num_entries;
183}

185template <typename MatrixType, typename SmootherType, typename SolverType>
186double block_multilevel<MatrixType,SmootherType,SolverType>
187::grid_complexity( void )
188{
189 size_t unknowns = 0;
190 for(size_t index = 0; index < levels.size(); index++)
191 unknowns += levels[index].A.num_rows;
192
193 return (double) unknowns / (double) levels[0].A.num_rows;
194}
