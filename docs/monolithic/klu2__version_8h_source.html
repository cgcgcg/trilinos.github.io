92/*
93SCALAR_IS_NAN(x):
94 True if x is NaN. False otherwise. The commonly-existing isnan(x)
95 function could be used, but it's not in Kernighan & Ritchie 2nd edition
96 (ANSI C). It may appear in <math.h>, but I'm not certain about
97 portability. The expression x != x is true if and only if x is NaN,
98 according to the IEEE 754 floating-point standard.
99
100SCALAR_IS_ZERO(x):
101 True if x is zero. False if x is nonzero, NaN, or +/- Inf.
102 This is (x == 0) if the compiler is IEEE 754 compliant.
103
104SCALAR_IS_NONZERO(x):
105 True if x is nonzero, NaN, or +/- Inf. False if x zero.
106 This is (x != 0) if the compiler is IEEE 754 compliant.
107
108SCALAR_IS_LTZERO(x):
109 True if x is < zero or -Inf. False if x is >= 0, NaN, or +Inf.
110 This is (x < 0) if the compiler is IEEE 754 compliant.
111*/
112
113/* These all work properly, according to the IEEE 754 standard ... except on */
114/* a PC with windows. Works fine in Linux on the same PC... */
115#define SCALAR_IS_NAN(x) ((x) != (x))
116#define SCALAR_IS_ZERO(x) ((x) == 0.)
117#define SCALAR_IS_NONZERO(x) ((x) != 0.)
118#define SCALAR_IS_LTZERO(x) ((x) < 0.)
119
120
121/* scalar absolute value macro. If x is NaN, the result is NaN: */
122#define SCALAR_ABS(x) ((SCALAR_IS_LTZERO (x)) ? -(x) : (x))
