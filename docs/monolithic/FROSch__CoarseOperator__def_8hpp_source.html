38 template <class SC,class LO,class GO,class NO>
39 int CoarseOperator<SC,LO,GO,NO>::compute()
40 {
41 FROSCH_TIMER_START_LEVELID(computeTime,"CoarseOperator::compute");
42 FROSCH_ASSERT(this->IsInitialized_,"FROSch::CoarseOperator: CoarseOperator has to be initialized before calling compute()");
43 // This is not optimal yet... Some work could be moved to Initialize
44 //if (this->Verbose_) cout << "FROSch::CoarseOperator : WARNING: Some of the operations could probably be moved from initialize() to Compute().\n";
45
46 bool reuseCoarseBasis = this->ParameterList_->get("Reuse: Coarse Basis",true);
47 bool reuseCoarseMatrix = this->ParameterList_->get("Reuse: Coarse Matrix",false);
48 if (!this->IsComputed_) {
49 reuseCoarseBasis = false;
50 reuseCoarseMatrix = false;
51 }
52
53 if (!reuseCoarseBasis) {
54 if (this->IsComputed_ && this->Verbose_) cout << "FROSch::CoarseOperator : Recomputing the Coarse Basis" << endl;
55 clearCoarseSpace(); // AH 12/11/2018: If we do not clear the coarse space, we will always append just append the coarse space
56 ConstXMapPtr subdomainMap = this->computeCoarseSpace(CoarseSpace_); // AH 12/11/2018: This map could be overlapping, repeated, or unique. This depends on the specific coarse operator
57 if (CoarseSpace_->hasUnassembledMaps()) { // If there is no unassembled basis, the current Phi_ should already be correct
58 CoarseSpace_->assembleCoarseSpace();
59 FROSCH_ASSERT(CoarseSpace_->hasAssembledBasis(),"FROSch::CoarseOperator : !CoarseSpace_->hasAssembledBasis()");
60 CoarseSpace_->buildGlobalBasisMatrix(this->K_->getRowMap(),this->K_->getRangeMap(),subdomainMap,this->ParameterList_->get("Phi: Dropping Threshold",1.e-8));
61 FROSCH_ASSERT(CoarseSpace_->hasGlobalBasisMatrix(),"FROSch::CoarseOperator : !CoarseSpace_->hasGlobalBasisMatrix()");
62 Phi_ = CoarseSpace_->getGlobalBasisMatrix();
63 }
64 }
65 if (!reuseCoarseMatrix) {
66 if (this->IsComputed_ && this->Verbose_) cout << "FROSch::CoarseOperator : Recomputing the Coarse Matrix" << endl;
67 this->setUpCoarseOperator();
68 }
69 this->IsComputed_ = true;
70
71 // Store current Phi in ParameterList_
72 if ( this->ParameterList_->get("Store Phi",false) ) {
73 FROSCH_NOTIFICATION("FROSch::CoarseOperator",this->Verbose_,"Storing current Phi in Parameterlist.");
74 this->ParameterList_->set("RCP(Phi)", Phi_);
75 }
76
77 // Store current Coarse Matrix in ParameterList_
78 if ( this->ParameterList_->get("Store Coarse Matrix",false) ) {
79 FROSCH_NOTIFICATION("FROSch::CoarseOperator",this->Verbose_,"Storing current Coarse Matrix in Parameterlist.");
80 this->ParameterList_->set("RCP(Coarse Matrix)", CoarseMatrix_);
81 this->ParameterList_->set("bool(CoarseSolveComm)", OnCoarseSolveComm_);
82 }
83
84 return 0;
85 }

93 template<class SC,class LO,class GO,class NO>
94 void CoarseOperator<SC,LO,GO,NO>::apply(const XMultiVector &x,
95 XMultiVector &y,
96 bool usePreconditionerOnly,
97 ETransp mode,
98 SC alpha,
99 SC beta) const
100 {
101 FROSCH_TIMER_START_LEVELID(applyTime,"CoarseOperator::apply");
102 static int i = 0;
103 if (!Phi_.is_null() && this->IsComputed_) {
104 if (XTmp_.is_null() || XTmp_->getNumVectors() != x.getNumVectors()) {
105 XTmp_ = MultiVectorFactory<SC,LO,GO,NO>::Build(x.getMap(),x.getNumVectors());
106 }
107 *XTmp_ = x;
108 if (!usePreconditionerOnly && mode == NO_TRANS) {
109 this->K_->apply(x,*XTmp_,mode,ScalarTraits<SC>::one(),ScalarTraits<SC>::zero());
110 }
111 if (XCoarseSolve_.is_null() || XCoarseSolve_->getNumVectors() != x.getNumVectors()) {
112 XCoarseSolve_ = MultiVectorFactory<SC,LO,GO,NO>::Build(GatheringMaps_[GatheringMaps_.size()-1],x.getNumVectors());
113 } else {
114 XCoarseSolve_->replaceMap(GatheringMaps_[GatheringMaps_.size()-1]); // The map is replaced in applyCoarseSolve(). If we do not build it from scratch, we should at least replace the map here. This may be important since the maps live on different communicators.
115 }
116 if (YCoarseSolve_.is_null() || YCoarseSolve_->getNumVectors() != y.getNumVectors()) {
117 YCoarseSolve_ = MultiVectorFactory<SC,LO,GO,NO>::Build(GatheringMaps_[GatheringMaps_.size()-1],y.getNumVectors());
118 }
119 applyPhiT(*XTmp_,*XCoarseSolve_);
120 applyCoarseSolve(*XCoarseSolve_,*YCoarseSolve_,mode);
121 applyPhi(*YCoarseSolve_,*XTmp_);
122 if (!usePreconditionerOnly && mode != NO_TRANS) {
123 this->K_->apply(*XTmp_,*XTmp_,mode,ScalarTraits<SC>::one(),ScalarTraits<SC>::zero());
124 }
125 y.update(alpha,*XTmp_,beta);
126 } else {
127 if (i==0) {
128 FROSCH_WARNING("FROSch::CoarseOperator",(this->Verbose_ && Phi_.is_null()),"Coarse Basis is empty => The CoarseOperator will just act as the identity...");
129 FROSCH_WARNING("FROSch::CoarseOperator",(this->Verbose_ && !this->IsComputed_),"CoarseOperator has not been computed yet => The CoarseOperator will just act as the identity...");
130 i++;
131 }
132 y.update(ScalarTraits<SC>::one(),x,ScalarTraits<SC>::zero());
133 }

235 int CoarseOperator<SC,LO,GO,NO>::setUpCoarseOperator()
236 {
237 FROSCH_DETAILTIMER_START_LEVELID(setUpCoarseOperatorTime,"CoarseOperator::setUpCoarseOperator");
238 if (!Phi_.is_null()) {
239 // Build CoarseMatrix_
240 XMatrixPtr k0 = buildCoarseMatrix();
241
242 //------------------------------------------------------------------------------------------------------------------------
243 // Communicate coarse matrix
244 FROSCH_DETAILTIMER_START_LEVELID(communicateCoarseMatrixTime,"communicate coarse matrix");
245 if (!DistributionList_->get("Type","linear").compare("linear")) {
246 XMatrixPtr tmpCoarseMatrix = MatrixFactory<SC,LO,GO,NO>::Build(GatheringMaps_[0]);
247 {
248#ifdef FROSCH_COARSEOPERATOR_DETAIL_TIMERS
249 FROSCH_DETAILTIMER_START_LEVELID(coarseMatrixExportTime,"Export Coarse Matrix");
250#endif
251 tmpCoarseMatrix->doExport(*k0,*CoarseSolveExporters_[0],INSERT);
252 }
253
254 for (UN j=1; j<GatheringMaps_.size(); j++) {
255 tmpCoarseMatrix->fillComplete();
256 k0 = tmpCoarseMatrix;
257 tmpCoarseMatrix = MatrixFactory<SC,LO,GO,NO>::Build(GatheringMaps_[j]);
258 {
259#ifdef FROSCH_COARSEOPERATOR_DETAIL_TIMERS
260 FROSCH_DETAILTIMER_START_LEVELID(coarseMatrixExportTime,"Export Coarse Matrix");
261#endif
262 tmpCoarseMatrix->doExport(*k0,*CoarseSolveExporters_[j],INSERT);
263 }
264 }
265 k0 = tmpCoarseMatrix;
266
267 } else if (!DistributionList_->get("Type","linear").compare("ZoltanDual")) {
268 //ZoltanDual
269 CoarseSolveExporters_[0] = Xpetra::ExportFactory<LO,GO,NO>::Build(k0->getMap(),GatheringMaps_[0]);
270 XMatrixPtr tmpCoarseMatrix = Xpetra::MatrixFactory<SC,LO,GO,NO>::Build(GatheringMaps_[0],k0->getGlobalMaxNumRowEntries());
271 tmpCoarseMatrix->doExport(*k0,*CoarseSolveExporters_[0],Xpetra::INSERT);
272
273 for (UN j=1; j<GatheringMaps_.size(); j++) {
274 tmpCoarseMatrix->fillComplete();
275 k0 = tmpCoarseMatrix;
276 CoarseSolveExporters_[j] = Xpetra::ExportFactory<LO,GO,NO>::Build(GatheringMaps_[j-1],GatheringMaps_[j]);
277 tmpCoarseMatrix = Xpetra::MatrixFactory<SC,LO,GO,NO>::Build(GatheringMaps_[j],k0->getGlobalMaxNumRowEntries());
278 tmpCoarseMatrix->doExport(*k0,*CoarseSolveExporters_[j],Xpetra::INSERT);
279 }
280
281 tmpCoarseMatrix->fillComplete();
282 k0 = tmpCoarseMatrix;
283
284 } else if (!DistributionList_->get("Type","linear").compare("Zoltan2")) {
285#ifdef HAVE_SHYLU_DDFROSCH_ZOLTAN2
286 GatheringMaps_[0] = rcp_const_cast<XMap> (BuildUniqueMap(k0->getRowMap()));
287 CoarseSolveExporters_[0] = ExportFactory<LO,GO,NO>::Build(CoarseSpace_->getBasisMapUnique(),GatheringMaps_[0]);
288
289 if (NumProcsCoarseSolve_ < this->MpiComm_->getSize()) {
290 XMatrixPtr k0Unique = MatrixFactory<SC,LO,GO,NO>::Build(GatheringMaps_[0]);
291 k0Unique->doExport(*k0,*CoarseSolveExporters_[0],INSERT);
292 k0Unique->fillComplete(GatheringMaps_[0],GatheringMaps_[0]);
293
294 if (NumProcsCoarseSolve_<this->MpiComm_->getSize()) {
295 ParameterListPtr tmpList = sublist(DistributionList_,"Zoltan2 Parameter");
296 tmpList->set("num_global_parts",NumProcsCoarseSolve_);
297 FROSch::RepartionMatrixZoltan2(k0Unique,tmpList);
298 }
299
300 k0 = k0Unique;
301 GatheringMaps_[0] = k0->getRowMap();
302 CoarseSolveExporters_[0] = ExportFactory<LO,GO,NO>::Build(CoarseSpace_->getBasisMapUnique(),GatheringMaps_[0]);
303
304 if (GatheringMaps_[0]->getLocalNumElements()>0) {
305 OnCoarseSolveComm_=true;
306 }
307 const GO INVALID = Teuchos::OrdinalTraits<GO>::invalid();
308 CoarseSolveComm_ = this->MpiComm_->split(!OnCoarseSolveComm_,this->MpiComm_->getRank());
309 CoarseSolveMap_ = MapFactory<LO,GO,NO>::Build(CoarseSpace_->getBasisMapUnique()->lib(),INVALID,GatheringMaps_[0]->getLocalElementList(),0,CoarseSolveComm_);
310 }
311#else
312 ThrowErrorMissingPackage("FROSch::CoarseOperator","Zoltan2");
313#endif
314 //------------------------------------------------------------------------------------------------------------------------
315 } else {
316 FROSCH_ASSERT(false,"Distribution Type unknown!");
317 }
318 FROSCH_DETAILTIMER_STOP(communicateCoarseMatrixTime);
319
320 //------------------------------------------------------------------------------------------------------------------------
321 // Matrix to the new communicator
322 if (OnCoarseSolveComm_) {
323 FROSCH_DETAILTIMER_START_LEVELID(replicateCoarseMatrixOnCoarseCommTime,"replicate coarse matrix on coarse comm");
324 LO numRows = k0->getLocalNumRows();
325 ArrayRCP<size_t> elemsPerRow(numRows);
326 LO numDiagonalsAdded = 0;
327 if (k0->isFillComplete()) {
328 ConstLOVecView indices;
329 ConstSCVecView values;
330 for (LO i = 0; i < numRows; i++) {
331 size_t numEntries;
332 numEntries = k0->getNumEntriesInLocalRow(i);
333 if (numEntries == 0) {
334 //Always add the diagonal for empty rows
335 numEntries = 1;
336 }
337 elemsPerRow[i] = numEntries;
338 }
339 CoarseMatrix_ = MatrixFactory<SC,LO,GO,NO>::Build(CoarseSolveMap_,elemsPerRow);
340 for (LO i = 0; i < numRows; i++) {
341 GO globalRow = CoarseSolveMap_->getGlobalElement(i);
342 k0->getLocalRowView(i,indices,values);
343 if (indices.size()>0) {
344 GOVec indicesGlob(indices.size());
345 for (UN j=0; j<indices.size(); j++) {
346 indicesGlob[j] = k0->getColMap()->getGlobalElement(indices[j]);
347 }
348 CoarseMatrix_->insertGlobalValues(globalRow,indicesGlob(),values);
349 } else { // Add diagonal unit for zero rows // Todo: Do you we need to sort the coarse matrix "NodeWise"?
350 GOVec indicesGlob(1,CoarseSolveMap_->getGlobalElement(i));
351 SCVec values2(1,ScalarTraits<SC>::one());
352 CoarseMatrix_->insertGlobalValues(globalRow,indicesGlob(),values2());
353 numDiagonalsAdded++;
354 }
355 }
356 CoarseMatrix_->fillComplete(CoarseSolveMap_,CoarseSolveMap_); //RCP<FancyOStream> fancy = fancyOStream(rcpFromRef(cout)); CoarseMatrix_->describe(*fancy,VERB_EXTREME);
357 } else {
358 ConstGOVecView indices;
359 ConstSCVecView values;
360 for (LO i = 0; i < numRows; i++) {
361 GO globalRow = CoarseSolveMap_->getGlobalElement(i);
362 size_t numEntries;
363 numEntries = k0->getNumEntriesInGlobalRow(globalRow);
364 if (numEntries == 0) {
365 //Always add the diagonal for empty rows
366 numEntries = 1;
367 }
368 elemsPerRow[i] = numEntries;
369 }
370 CoarseMatrix_ = MatrixFactory<SC,LO,GO,NO>::Build(CoarseSolveMap_,elemsPerRow);
371 for (LO i = 0; i < numRows; i++) {
372 GO globalRow = CoarseSolveMap_->getGlobalElement(i);
373 k0->getGlobalRowView(globalRow,indices,values);
374 if (indices.size()>0) {
375 CoarseMatrix_->insertGlobalValues(globalRow,indices,values);
376 } else { // Add diagonal unit for zero rows // Todo: Do you we need to sort the coarse matrix "NodeWise"?
377 GOVec indices2(1,globalRow);
378 SCVec values2(1,ScalarTraits<SC>::one());

380 numDiagonalsAdded++;
381 }
382 }
383 CoarseMatrix_->fillComplete(CoarseSolveMap_,CoarseSolveMap_); //RCP<FancyOStream> fancy = fancyOStream(rcpFromRef(cout)); CoarseMatrix_->describe(*fancy,VERB_EXTREME);
384 }
385 FROSCH_DETAILTIMER_STOP(replicateCoarseMatrixOnCoarseCommTime);
