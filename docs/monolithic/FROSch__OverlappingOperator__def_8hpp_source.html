44 template <class SC,class LO,class GO,class NO>
45 void OverlappingOperator<SC,LO,GO,NO>::apply(const XMultiVector &x,
46 XMultiVector &y,
47 bool usePreconditionerOnly,
48 ETransp mode,
49 SC alpha,
50 SC beta) const
51 {
52 FROSCH_TIMER_START_LEVELID(applyTime,"OverlappingOperator::apply");
53 FROSCH_ASSERT(this->IsComputed_,"FROSch::OverlappingOperator: OverlappingOperator has to be computed before calling apply()");
54 if (XTmp_.is_null() || XTmp_->getNumVectors() != x.getNumVectors()) {
55 XTmp_ = MultiVectorFactory<SC,LO,GO,NO>::Build(x.getMap(),x.getNumVectors());
56 }
57 *XTmp_ = x;
58 if (!usePreconditionerOnly && mode == NO_TRANS) {
59 this->K_->apply(x,*XTmp_,mode,ScalarTraits<SC>::one(),ScalarTraits<SC>::zero());
60 }
61 // AH 11/28/2018: For Epetra, XOverlap_ will only have a view to the values of XOverlapTmp_. Therefore, xOverlapTmp should not be deleted before XOverlap_ is used.
62 if (YOverlap_.is_null() || YOverlap_->getNumVectors() != x.getNumVectors()) {
63 YOverlap_ = MultiVectorFactory<SC,LO,GO,NO>::Build(OverlappingMatrix_->getDomainMap(),x.getNumVectors());
64 } else {
65 YOverlap_->replaceMap(OverlappingMatrix_->getDomainMap());
66 }
67 // AH 11/28/2018: replaceMap does not update the GlobalNumRows. Therefore, we have to create a new MultiVector on the serial Communicator. In Epetra, we can prevent to copy the MultiVector.
68 {
69 if (XOverlap_.is_null()) {
70 XOverlap_ = MultiVectorFactory<SC,LO,GO,NO>::Build(OverlappingMap_,x.getNumVectors());
71 } else {
72 XOverlap_->replaceMap(OverlappingMap_);
73 }
74 XOverlap_->doImport(*XTmp_,*Scatter_,INSERT);
75 XOverlap_->replaceMap(OverlappingMatrix_->getRangeMap());
76 }
77 SubdomainSolver_->apply(*XOverlap_,*YOverlap_,mode,ScalarTraits<SC>::one(),ScalarTraits<SC>::zero());
78 YOverlap_->replaceMap(OverlappingMap_);
79
80 XTmp_->putScalar(ScalarTraits<SC>::zero());
81 ConstXMapPtr yMap = y.getMap();
82 ConstXMapPtr yOverlapMap = YOverlap_->getMap();
83 if (Combine_ == Restricted) {
84 if (XTmp_->getMap()->lib() == UseTpetra) {
85 auto yLocalMap = yMap->getLocalMap();
86 auto yLocalOverlapMap = yOverlapMap->getLocalMap();
87 // run local restriction on execution space defined by local-map
88 using XMap2 = typename SchwarzOperator<SC,LO,GO,NO>::XMap;
89 using execution_space = typename XMap2::local_map_type::execution_space;
90 Kokkos::RangePolicy<execution_space> policy (0, yMap->getLocalNumElements());
91
92 using xTMVector = Xpetra::TpetraMultiVector<SC,LO,GO,NO>;
93 // Xpetra wrapper for Tpetra MV
94 auto yXTpetraMVector = rcp_dynamic_cast<const xTMVector>(YOverlap_, true);
95 auto xXTpetraMVector = rcp_dynamic_cast< xTMVector>(XTmp_, true);
96 // Tpetra MV
97 auto yTpetraMVector = yXTpetraMVector->getTpetra_MultiVector();
98 auto xTpetraMVector = xXTpetraMVector->getTpetra_MultiVector();
99 // View
100 auto yView = yTpetraMVector->getLocalViewDevice(Tpetra::Access::ReadOnly);
101 auto xView = xTpetraMVector->getLocalViewDevice(Tpetra::Access::ReadWrite);
102 for (UN j=0; j<y.getNumVectors(); j++) {
103 Kokkos::parallel_for(
104 "FROSch_OverlappingOperator::applyLocalRestriction", policy,
105 KOKKOS_LAMBDA(const int i) {
106 GO gID = yLocalMap.getGlobalElement(i);
107 LO lID = yLocalOverlapMap.getLocalElement(gID);
108 xView(i, j) = yView(lID, j);
109 });
110 }
111 Kokkos::fence();
112 } else
113 {
114 GO globID = 0;
115 LO localID = 0;
116 for (UN i=0; i<y.getNumVectors(); i++) {
117 ConstSCVecPtr yOverlapData_i = YOverlap_->getData(i);
118 for (UN j=0; j<yMap->getLocalNumElements(); j++) {
119 globID = yMap->getGlobalElement(j);
120 localID = yOverlapMap->getLocalElement(globID);
121 XTmp_->getDataNonConst(i)[j] = yOverlapData_i[localID];
122 }
123 }
124 }
125 } else {
126 XTmp_->doExport(*YOverlap_,*Scatter_,ADD);
127 }
128 if (Combine_ == Averaging) {
129 ConstSCVecPtr scaling = Multiplicity_->getData(0);
130 for (UN j=0; j<XTmp_->getNumVectors(); j++) {
131 SCVecPtr values = XTmp_->getDataNonConst(j);
132 for (UN i=0; i<values.size(); i++) {
133 values[i] = values[i] / scaling[i];
134 }
135 }
136 }
137
138 if (!usePreconditionerOnly && mode != NO_TRANS) {
139 this->K_->apply(*XTmp_,*XTmp_,mode,ScalarTraits<SC>::one(),ScalarTraits<SC>::zero());
140 }
141 y.update(alpha,*XTmp_,beta);
142 }

144 template <class SC,class LO,class GO,class NO>
145 int OverlappingOperator<SC,LO,GO,NO>::initializeOverlappingOperator()
146 {
147 FROSCH_DETAILTIMER_START_LEVELID(initializeOverlappingOperatorTime,"OverlappingOperator::initializeOverlappingOperator");
148 Scatter_ = ImportFactory<LO,GO,NO>::Build(this->getDomainMap(),OverlappingMap_);
149 if (Combine_ == Averaging) {
150 Multiplicity_ = MultiVectorFactory<SC,LO,GO,NO>::Build(this->getRangeMap(),1);
151 XMultiVectorPtr multiplicityRepeated;
152 multiplicityRepeated = MultiVectorFactory<SC,LO,GO,NO>::Build(OverlappingMap_,1);
153 multiplicityRepeated->putScalar(ScalarTraits<SC>::one());
154 XExportPtr multiplicityExporter = ExportFactory<LO,GO,NO>::Build(multiplicityRepeated->getMap(),this->getRangeMap());
155 Multiplicity_->doExport(*multiplicityRepeated,*multiplicityExporter,ADD);
156 }
157 return 0; // RETURN VALUE
158 }

171 template <class SC,class LO,class GO,class NO>
172 int OverlappingOperator<SC,LO,GO,NO>::computeOverlappingOperator()
173 {
174 FROSCH_DETAILTIMER_START_LEVELID(computeOverlappingOperatorTime,"OverlappingOperator::computeOverlappingOperator");
175
176 updateLocalOverlappingMatrices();
177 bool reuseSymbolicFactorization = this->ParameterList_->get("Reuse: Symbolic Factorization",true);
178 if (!reuseSymbolicFactorization || SubdomainSolver_.is_null()) {
179 // initializeSubdomainSolver is called during symbolic only if reuseSymbolicFactorization=true
180 // so if reuseSymbolicFactorization=false, we always call initializeSubdomainSolver
181 if (this->IsComputed_ && this->Verbose_) cout << "FROSch::OverlappingOperator : Recomputing the Symbolic Factorization" << endl;
182 initializeSubdomainSolver(this->OverlappingMatrix_);
183 } else if (this->IsComputed_) {
184 // if !IsComputed, then this is the first timing calling "compute" after initializeSubdomainSolver is called in symbolic phase
185 // so no need to do anything
186 SubdomainSolver_->updateMatrix(this->OverlappingMatrix_,true);
187 }
188 this->IsComputed_ = true;
189 return SubdomainSolver_->compute();
190 }
