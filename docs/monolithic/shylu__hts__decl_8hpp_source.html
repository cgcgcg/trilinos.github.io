269 struct Options {
270 // - Obvious user parameters.
271 // Tile rows and cols into nxn blocks. For scalar problems, n=1 (default) is
272 // sensible. For PDEs with multiple equations per node (e.g. 3 in 3D
273 // elasticity), n = ndof/node is useful.
274 Int levelset_block_size;
275 // Minimum level set size. Any smaller is considered a bottleneck. Set to 0
276 // to make hts solve the whole problem using level scheduling.
277 Int min_lset_size;
278
279 // - Nonobvious performance tuning parameters.
280 // Recurse until blocks are this size.
281 Int min_block_size;
282 // Guide to how to break up blocks for parallel execution.
283 Int min_parallel_rows;
284 // In the symbolic analysis phase, serial block sizes need to be larger
285 // because memory access patterns are terrible.
286 Int pp_min_block_size;
287 // Where sparse and dense representations are decided dynamically, choose
288 // dense if nonzero density is >= min_dense_density.
289 Real min_dense_density;
290 // Max fraction of rows in the LS block that are in too-small level sets.
291 Real lset_max_bad_fraction;
292 // Scale min_lset_size with nthreads. If true, effective minimum size is
293 // nthreads*lset_min_size
294 // rather than simply min_lset_size.
295 bool lset_min_size_scale_with_nthreads;
296
297 // - Debug/output parameters.
298 // Turn on profiling. Increases computation time.
299 bool profile;
300 // 0 for no printing.
301 int print_level;
302
303 // Sets default values.
304 Options();
305 };
