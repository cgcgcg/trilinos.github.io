29class InnerProductSpaceTraits< Sacado::MP::Vector<S> > {
30public:

37
38#if KOKKOS_VERSION >= 40799
39 typedef typename KokkosKernels::ArithTraits<val_type>::mag_type mag_type;
40#else
41 typedef typename Kokkos::ArithTraits<val_type>::mag_type mag_type;
42#endif
43 typedef base_dot_type dot_type;
44
45 static KOKKOS_FORCEINLINE_FUNCTION
46 mag_type norm (const val_type& x) {
47 //return ArithTraits<val_type>::abs (x);
48 const ordinal_type sz = x.size();
49 mag_type nrm = mag_type(0);
50 for (ordinal_type i=0; i<sz; ++i) {
51 const mag_type n = BIT::norm( x.fastAccessCoeff(i) );
52 nrm += n*n;
53 }
54 return std::sqrt(nrm);
55 }
56
57 static KOKKOS_FORCEINLINE_FUNCTION
58 dot_type dot (const val_type& x, const val_type& y) {
59 const ordinal_type xsz = x.size();
60 const ordinal_type ysz = y.size();
61 const ordinal_type sz = xsz > ysz ? xsz : ysz;
62
63 dot_type r = dot_type(0);
64 if (x.hasFastAccess(sz) && y.hasFastAccess(sz))
65 for (ordinal_type i=0; i<sz; ++i)
66 r += BIT::dot( x.fastAccessCoeff(i), y.fastAccessCoeff(i) );
67 else
68 for (ordinal_type i=0; i<sz; ++i)
69 r += BIT::dot( x.coeff(i), y.coeff(i) );
70
71 return r;
72 }

74};
