21 ConstraintModelEvaluator::ConstraintModelEvaluator(const Teuchos::RCP<::Thyra::ModelEvaluator<double>>& model,
22 const LOCA::ParameterVector& pVec,
23 const std::vector<std::string>& constraintResponseNames,
24 const NOX::Abstract::Vector& cloneVec) :
25 model_(model),
26 pVec_(pVec),
27 gNames_(constraintResponseNames),
28 constraints_(constraintResponseNames.size(),1),
29 isValidConstraints_(false),
30 isValidDx_(false),
31 printDebug_(false)
32 {
33 x_ = cloneVec.clone(NOX::ShapeCopy);
34 dgdx_ = cloneVec.createMultiVector(constraintResponseNames.size(),NOX::ShapeCopy);
35
36 TEUCHOS_ASSERT(constraintResponseNames.size() <= size_t(pVec_.length()));
37
38 // Get the parameter indices
39 meParameterIndices_.clear();
40 const auto pNames = pVec_.getNamesVector();
41 for (size_t i=0; i < pNames.size(); ++i) {
42 const auto& p_name = pNames[i];
43 bool found = false;
44 for (int j=0; j < model_->Np(); ++j) {
45 const auto& names_vec = model_->get_p_names(j);
46 const auto search = std::find(names_vec->begin(),names_vec->end(),p_name);
47 if (search != names_vec->end()) {
48 meParameterIndices_.push_back(j);
49 found = true;
50 break;
51 }
52 }
53 TEUCHOS_TEST_FOR_EXCEPTION(!found,std::runtime_error,"ERROR LOCA::TpetraConstraintModelEvaluator::CTOR - could not find parameter named \"" << p_name << "\" in the model evaluator!");
54 }
55 TEUCHOS_ASSERT(pNames.size() == meParameterIndices_.size());
56
57 // Get the response indices
58 meResponseIndices_.clear();
59 for (size_t i=0; i < gNames_.size(); ++i) {
60 const auto& g_name = gNames_[i];
61 bool found = false;
62 for (int j=0; j < model_->Ng(); ++j) {
63 const auto& names_vec = model_->get_g_names(j);
64 const auto search = std::find(names_vec.begin(),names_vec.end(),g_name);
65 if (search != names_vec.end()) {
66 meResponseIndices_.push_back(j);
67 found = true;
68 break;
69 }
70 }
71 TEUCHOS_TEST_FOR_EXCEPTION(!found,std::runtime_error,"ERROR LOCA::TpetraConstraintModelEvaluator::CTOR - could not find response named \"" << g_name << "\" in the model evaluator!");
72 }
73 TEUCHOS_ASSERT(gNames_.size() == meResponseIndices_.size());
74
75 // Allocate all objects to pass into model evaluator. The
76 // parameters and responses are treated as separate objects.
77 me_p_.resize(pVec_.length());
78 for (size_t l=0; l < me_p_.size(); ++l)
79 me_p_[l] = ::Thyra::createMember(*model->get_p_space(meParameterIndices_[l]),"p_l");
80
81 const auto numResponses = gNames_.size();
82 me_g_.resize(numResponses);
83 me_dgdx_.resize(numResponses);
84 me_dgdp_.resize(numResponses);
85 for (auto& i : me_dgdp_)
86 i.resize(me_p_.size());
87 for (size_t j=0; j < numResponses; ++j) {
88 me_g_[j] = ::Thyra::createMember(*model->get_g_space(meResponseIndices_[j]),"g_j");
89 me_dgdx_[j] = Teuchos::rcp_dynamic_cast<::Thyra::MultiVectorBase<double>>(model->create_DgDx_op(meResponseIndices_[j]),false);
90 if (me_dgdx_[j].is_null()) {
91 // It might be wrapped in an Adjoint linear op for the transpose.
92 auto ptr = Teuchos::rcp_dynamic_cast<::Thyra::DefaultScaledAdjointLinearOp<double>>(model->create_DgDx_op(meResponseIndices_[j]),true);
93 me_dgdx_[j] = Teuchos::rcp_dynamic_cast<::Thyra::MultiVectorBase<double>>(ptr->getNonconstOp(),true);
94 }
95 for (size_t l=0; l < me_p_.size(); ++l)
96 me_dgdp_[j][l] = Teuchos::rcp_dynamic_cast<::Thyra::MultiVectorBase<double>>(model->create_DgDp_op(meResponseIndices_[j],
97 meParameterIndices_[l]),
98 true);
99 }
100
101 }

177 NOX::Abstract::Group::ReturnType ConstraintModelEvaluator::computeConstraints()
178 {
179 auto inArgs = model_->createInArgs();
180 auto x_thyra = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(x_)->getThyraRCPVector();
181 inArgs.set_x(x_thyra);
182 for (int i=0; i < pVec_.length(); ++i)
183 inArgs.set_p(meParameterIndices_[i],me_p_[i]);
184
185 auto outArgs = model_->createOutArgs();
186 for (size_t i=0; i < me_g_.size(); ++i)
187 outArgs.set_g(meResponseIndices_[i],::Thyra::ModelEvaluatorBase::Evaluation<::Thyra::VectorBase<double>>(me_g_[i]));
188
189 model_->evalModel(inArgs,outArgs);
190
191 // Now copy responses into the aggregated constraint object for
192 // loca.
193 using extractor = ::Thyra::TpetraOperatorVectorExtraction<NOX::Scalar,NOX::LocalOrdinal,NOX::GlobalOrdinal,NOX::NodeType>;
194 for (size_t i=0; i < me_g_.size(); ++i) {
195 auto tmp = extractor::getTpetraMultiVector(me_g_[i]);
196 auto val = tmp->getLocalViewHost(Tpetra::Access::ReadOnly);
197 constraints_(i,0) = val(0,0);
198 if (printDebug_)
199 std::cout << "LOCA::ConstraintME: constraints_(" << i << ")=" << val(0,0) << std::endl;
200 }
201
202 isValidConstraints_ = true;
203 return NOX::Abstract::Group::Ok;
204 }

206 NOX::Abstract::Group::ReturnType ConstraintModelEvaluator::computeDX()
207 {
208 auto inArgs = model_->createInArgs();
209 auto x_thyra = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(x_)->getThyraRCPVector();
210 inArgs.set_x(x_thyra);
211 for (int i=0; i < pVec_.length(); ++i)
212 inArgs.set_p(meParameterIndices_[i],me_p_[i]);
213
214 auto outArgs = model_->createOutArgs();
215 for (size_t i=0; i < me_dgdx_.size(); ++i)
216 outArgs.set_DgDx(meResponseIndices_[i],::Thyra::ModelEvaluatorBase::Derivative<NOX::Scalar>(me_dgdx_[i],::Thyra::ModelEvaluatorBase::DERIV_MV_GRADIENT_FORM));
217
218 model_->evalModel(inArgs,outArgs);
219
220 // Aggregate derivative columns into a single
221 // multivector. Copies entries out of temporary me_dgdx_ into
222 // the aggregated object loca expects.
223 for (size_t i=0; i < me_dgdx_.size(); ++i) {
224 auto tmp = NOX::Thyra::MultiVector(me_dgdx_[i]);
225 (*dgdx_)[i] = tmp[0];
226
227 if (printDebug_) {
228 std::cout << "LOCA::ConstraintME: computeDX:" << std::endl;
229 me_dgdx_[i]->describe(std::cout,Teuchos::VERB_EXTREME);
230 }
231 }
232
233 isValidDx_ = true;
234 return NOX::Abstract::Group::Ok;

237 NOX::Abstract::Group::ReturnType
238 ConstraintModelEvaluator::computeDP(const std::vector<int>& paramIDs,
239 NOX::Abstract::MultiVector::DenseMatrix& dgdp,
240 bool isValidG)
241 {
242 if (!isValidG) {
243 if (!this->isConstraints())
244 this->computeConstraints();
245
246 for (size_t i=0; i < me_g_.size(); ++i)
247 dgdp(i,0) = constraints_(i,0);
248
249 if (printDebug_) {
250 for (size_t i=0; i < me_g_.size(); ++i)
251 std::cout << "LOCA::ConstraintME: computeDP: dgdp_(" << i << ",0)=" << dgdp(i,0) << std::endl;
252 }
253 }
254
255 auto inArgs = model_->createInArgs();
256 auto x_thyra = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(x_)->getThyraRCPVector();
257 inArgs.set_x(x_thyra);
258
259 // Set all parameters from the parameter vector
260 for (int i=0; i < pVec_.length(); ++i)
261 inArgs.set_p(meParameterIndices_[i],me_p_[i]);
262
263 // Only request derivatives for the incoming paramIDs. NOTE:
264 // this could be, and most likely is, a subset of the total
265 // number of parameters in the pVec_.
266 TEUCHOS_ASSERT(size_t(dgdp.numRows()) == me_dgdp_.size());
267 TEUCHOS_ASSERT(size_t(dgdp.numCols()) == (1+paramIDs.size()));
268 auto outArgs = model_->createOutArgs();
269 for (size_t j=0; j < me_dgdp_.size(); ++j) {
270 for (auto& l : paramIDs) {
271 outArgs.set_DgDp(meResponseIndices_[j],meParameterIndices_[l],me_dgdp_[j][l]);
272
273 if (printDebug_) {
274 std::cout << "LOCA::ConstraintME: computeDP: dgdp_(" << j << "," << l << "):" << std::endl;
275 me_dgdp_[j][l]->describe(std::cout,Teuchos::VERB_EXTREME);
276 }
277 }
278 }
279
280 model_->evalModel(inArgs,outArgs);
281
282 // Extract values of individual elements and copy into
283 // aggregated object for loca.
284 using extractor = ::Thyra::TpetraOperatorVectorExtraction<NOX::Scalar,NOX::LocalOrdinal,NOX::GlobalOrdinal,NOX::NodeType>;
285 for (size_t j=0; j < me_dgdp_.size(); ++j) {
286 for (size_t l=0; l < paramIDs.size(); ++l) {
287 auto tmp = extractor::getTpetraMultiVector(me_dgdp_[j][paramIDs[l]]);
288 auto val = tmp->getLocalViewHost(Tpetra::Access::ReadOnly);
289 // first col contains g, so we shift the columns by one
290 dgdp(j,l+1) = val(0,0);
291 }
292 }
293
294 return NOX::Abstract::Group::Ok;
295 }
