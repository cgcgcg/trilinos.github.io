105 GO GetNumGlobalElements() const {
106 const Teuchos::ParameterList& pL = GetParameterList();
107
108 const std::string& matrixType = pL.get<std::string>("matrixType");
109 GO nx, ny, nz;
110 if (pL.isType<int>("nx"))
111 nx = Teuchos::as<GO>(pL.get<int>("nx"));
112 else
113 nx = pL.get<GO>("nx");
114 if (pL.isType<int>("ny"))
115 ny = Teuchos::as<GO>(pL.get<int>("ny"));
116 else
117 ny = pL.get<GO>("ny");
118 if (pL.isType<int>("nz"))
119 nz = Teuchos::as<GO>(pL.get<int>("nz"));
120 else
121 nz = pL.get<GO>("nz");
122
123 GO numGlobalElements = -1;
124 if (matrixType == "Laplace1D" || matrixType == "Helmholtz1D" || matrixType == "Identity")
125 numGlobalElements = nx;
126
127 else if (matrixType == "Laplace2D" ||
128 matrixType == "Star2D" ||
129 matrixType == "BigStar2D" ||
130 matrixType == "AnisotropicDiffusion" ||
131 matrixType == "Elasticity2D" ||
132 matrixType == "Helmholtz2D" ||
133 matrixType == "Recirc2D")
134 numGlobalElements = nx * ny;
135
136 else if (matrixType == "Laplace3D" ||
137 matrixType == "Brick3D" ||
138 matrixType == "BigStar2D" ||
139 matrixType == "Elasticity3D" ||
140 matrixType == "Helmholtz3D")
141 numGlobalElements = nx * ny * nz;
142
143 TEUCHOS_TEST_FOR_EXCEPTION(numGlobalElements < 0, std::runtime_error,
144 "Gallery: numGlobalElements < 0 (did you forget --ny (or --nz) for 2D (3D) problems?)");
145
146 return numGlobalElements;
147 }

212 void describe(Teuchos::FancyOStream& out, const Teuchos::EVerbosityLevel verbLevel = verbLevel_default) const {
213 int vl = (verbLevel == Teuchos::VERB_DEFAULT) ? Teuchos::VERB_LOW : verbLevel;
214 if (vl == Teuchos::VERB_NONE)
215 return;
216
217 if (vl == Teuchos::VERB_LOW)
218 out << description() << std::endl;
219 else
220 out << Teuchos::Describable::description() << std::endl;
221
222 if (vl == Teuchos::VERB_MEDIUM || vl == Teuchos::VERB_HIGH || vl == Teuchos::VERB_EXTREME) {
223 Teuchos::OSTab tab1(out);
224
225 const Teuchos::ParameterList& paramList = GetParameterList();
226 std::string matrixType = paramList.get<std::string>("matrixType");
227 GO nx, ny, nz;
228 if (paramList.isType<int>("nx"))
229 nx = Teuchos::as<GO>(paramList.get<int>("nx"));
230 else
231 nx = paramList.get<GO>("nx");
232 if (paramList.isType<int>("ny"))
233 ny = Teuchos::as<GO>(paramList.get<int>("ny"));
234 else
235 ny = paramList.get<GO>("ny");
236 if (paramList.isType<int>("nz"))
237 nz = Teuchos::as<GO>(paramList.get<int>("nz"));
238 else
239 nz = paramList.get<GO>("nz");
240
241 out << "Matrix type: " << matrixType << std::endl
242 << "Problem size: " << GetNumGlobalElements();
243
244 if (matrixType == "Laplace2D" || matrixType == "AnisotropicDiffusion" || matrixType == "Elasticity2D" || matrixType == "Helmholtz2D")
245 out << " (" << nx << "x" << ny << ")";
246 else if (matrixType == "Laplace3D" || matrixType == "Elasticity3D" || matrixType == "Helmholtz3D")
247 out << " (" << nx << "x" << ny << "x" << nz << ")";
248
249 out << std::endl;
250
251 if (matrixType == "AnisotropicDiffusion") {
252 double Kxx = paramList.get<double>("Kxx");
253 double Kxy = paramList.get<double>("Kxy");
254 double Kyy = paramList.get<double>("Kyy");
255 double dt = paramList.get<double>("dt");
256 out << "K = [[ " << Kxx << ", " << Kxy << " ], [ " << Kxy << ", " << Kyy << " ]]" << std::endl;
257 out << "dt = " << dt << std::endl;
258 }
259 }
