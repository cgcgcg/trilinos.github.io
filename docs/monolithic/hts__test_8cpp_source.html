38 static int are_same (const Data& A, const Data& B) {
39#define same(expr, ret) if ( ! (expr)) return ret;
40 same(A.m == B.m, 1);
41 same(A.ir[A.m] == B.ir[B.m], 3);
42 const Size nnz = A.ir[A.m];
43 for (Int k = 0; k <= A.m; ++k)
44 same(B.ir[k] == A.ir[k], 4);
45 for (Size k = 0; k < nnz; ++k) {
46 same(B.jc[k] == A.jc[k], 5);
47 same(B.v[k] == A.v[k], 6);
48 }
49 return 0;
50#undef same
51 }

70 // Run one valid triangle. Solve for x in
71 // P' T Q' x = R \ b,
72 // where P = diag(p) and similarly for Q and R, and T is a triangle.
73 static int test (const TestOptions& to, const bool print_options=false,
74 const bool exception_expected=false) {
75 int nerr = 0;
76
77 const Int max_nrhs = 3;
78 const Real tol = std::numeric_limits<Real>::epsilon()*1e6;
79
80 // Generate matrix data.
81 Data d;
82 Size nnz;
83 {
84 ut::gen_tri_matrix(to, d); // triangle
85 nnz = d.ir.back();
86 ut::gen_rand_perm(d.m, d.p); // row permutation vector
87 ut::gen_rand_perm(d.m, d.q); // col permutation vector
88 ut::gen_rand_vector(d.m, d.r); // row scaling
89 }
90
91 const Int ldb = d.m + 3, ldx = d.m + 4;
92 std::vector<Sclr> b(ldb*max_nrhs), xt(d.m*max_nrhs), x(ldx*max_nrhs);
93 {
94 // True x.
95 ut::gen_rand_vector(xt.size(), xt);
96 // Generate the rhs b.
97 std::vector<Sclr> y(d.m);
98 for (Int irhs = 0; irhs < max_nrhs; ++irhs) {
99 const Sclr* const xtp = xt.data() + irhs*d.m;
100 Sclr* const bp = b.data() + irhs*ldb;
101 for (Int i = 0; i < d.m; ++i)
102 x[i] = xtp[d.q[i]];
103 ut::mvp(d, to.transpose, to.conjugate, x.data(), y.data());
104 if (to.has_unit_diag())
105 for (Int i = 0; i < d.m; ++i)
106 y[i] += x[i];
107 for (Int i = 0; i < d.m; ++i)
108 bp[d.p[i]] = y[i];
109 for (Int i = 0; i < d.m; ++i)
110 bp[i] *= d.r[i];
111 }
112 }
113 std::vector<Sclr> bo(b);
114
115 typename ihts::CrsMatrix* T;
116 typename ihts::Options opts;
117 {
118 T = ihts::make_CrsMatrix(d.m, d.ir.data(), d.jc.data(), d.v.data(),
119 to.transpose, to.conjugate);
120 if ( ! to.reprocess && to.nthreads > 1)
121 ihts::register_Deallocator(T, &d);
122 if (to.solve_type == TestOptions::ls_only)
123 ihts::set_level_schedule_only(opts);
124 else if (to.solve_type == TestOptions::rb_only)
125 opts.min_lset_size = d.m + 1;
126 // To really test things well, choose very small block sizes. (This is bad
127 // for performance.) These parameters are not meant to be set by the user
128 // ordinarily, but they are exposed in case an expert is tuning performance
129 // or, as here, for testing.
130 opts.min_block_size = 6;
131 opts.min_parallel_rows = 2;
132 opts.pp_min_block_size = 12;
133 if (to.matrix_type == TestOptions::block_sparse)
134 opts.levelset_block_size = to.block_size;
135 }
136
137 {
138 typename ihts::Impl* impl;
139 try {
140 std::vector<Sclr> Td;
141 if (to.reprocess) {
142 // Save the true values.
143 Td = d.v;
144 // Zero the matrix values to simulate reprocessing.
145 d.v.assign(d.v.size(), 1);
146 }
147 impl = ihts::preprocess(T, max_nrhs - 1 /* For testing; see below. */,
148 to.nthreads, to.reprocess, d.p.data(), d.q.data(),
149 d.r.data(), &opts);
150 if (to.reprocess) {
151 // Restore the values.
152 d.v = Td;
153 }
154 } catch (...) {
155 if ( ! exception_expected) {
156 std::cerr << "Unexpected exception on ";
157 to.print(std::cerr);
158 std::cerr << "\n";
159 ut::write_matrixmarket(d, "unexpected_exception.mm");
160 }
161 ihts::delete_CrsMatrix(T);
162 throw;
163 }
164 if (print_options)
165 ihts::print_options(impl, std::cout);
166 if (to.reprocess) {
167 // Run 2 times to test idempotency.
168 for (int rep = 0; rep < 2; ++rep)
169 ihts::reprocess_numeric(impl, T, d.r.data());
170 }
171 // Exercise reset_max_nrhs.
172 ihts::reset_max_nrhs(impl, max_nrhs);
173 if (ihts::is_lower_tri(impl) &&
174 ((to.upper && ! to.transpose) || ( ! to.upper && to.transpose)) &&
175 d.m > 1 && nnz > static_cast<Size>(d.m) /* not diag */)
176 ++nerr;
177 for (int slv = 0; slv <= 2; ++slv) {
178 // Check each solve interface.
179 switch (slv) {
180 case 0:
181 ihts::solve_omp(impl, b.data(), to.nrhs, x.data(), ldb, ldx);
182 break;
183 case 1:
184 ihts::solve_omp(impl, b.data(), to.nrhs, x.data(), 0.0, 1.0,
185 ldb, ldx);
186 ihts::solve_omp(impl, b.data(), to.nrhs, x.data(), 2.0, -1.0,
187 ldb, ldx);
188 break;
189 case 2:
190 ihts::solve_omp(impl, b.data(), to.nrhs, ldb);
191 break;
192 }
193 double rd = 0;
194 for (Int i = 0; i < to.nrhs; ++i)
195 rd = std::max(
196 rd, ut::reldif(xt.data() + i*d.m,
197 (slv == 2 ? b.data() + i*ldb : x.data() + i*ldx),
198 d.m));
199 if (slv == 2) b = bo;
200 if (rd >= tol) {
201 ++nerr;
202 if (to.verbose) std::cout << "rd " << slv << ": " << rd << "\n";
203 }
204 }
205 ihts::delete_Impl(impl);
206 }
207
208 if (to.nthreads == 1) {
209 std::vector<Sclr> xb(d.m*to.nrhs), w(d.m);
210 for (Int irhs = 0; irhs < to.nrhs; ++irhs) {
211 const Sclr* const bp = b.data() + irhs*ldb;
212 Sclr* const xbp = xb.data() + irhs*d.m;
213 for (Int i = 0; i < d.m; ++i)
214 xbp[i] = bp[i];
215 }
216 ihts::solve_serial(T, ! to.upper, to.has_unit_diag(), xb.data(), to.nrhs,
217 d.p.data(), d.q.data(), d.r.data(), w.data());
218 const double rd = ut::reldif(xt.data(), xb.data(), d.m*to.nrhs);
219 if (rd >= tol) {
220 ++nerr;
221 if (to.verbose) std::cout << "serial rd: " << rd << "\n";
222 }
223 }
224
225 ihts::delete_CrsMatrix(T);
226
227 if (to.verbose) {
228 const bool print = to.verbose == 2 || (to.verbose == 1 && nerr);
229 if (print) {
230 std::cout << (nerr ? "fail" : "pass") << "ed: ";
231 to.print(std::cout);
232 std::cout << "\n";
233 }
234 }
235
236 return nerr;
237 }

239 // Run one invalid triangle and make sure an exception is thrown.
240 static int test_for_exception (const TestOptions& to) {
241 bool correct_exception_thrown = false;
242 try {
243 test(to, false, true);
244 } catch (const hts::NotFullDiagonalException&) {
245 correct_exception_thrown = to.matrix_type == TestOptions::missing_some_diag;
246 } catch (const hts::NotTriangularException&) {
247 correct_exception_thrown = ut::is_not_tri(to.matrix_type);
248 }
249 const int nerr = correct_exception_thrown ? 0 : 1;
250 if (nerr && to.verbose) {
251 std::cout << "test_for_exception failed for ";
252 to.print(std::cout);
253 std::cout << "\n";
254 }
255 return nerr;
256 }

259 // Run all the tests.
260 static int test (const int verbose) {
261 int nerr = 0;
262
263 // Test that we throw on an unsigned Int.
264 {
265 bool caught = false;
266 try {
267 HTS<unsigned int, int, double>::make_CrsMatrix(1, 0, 0, 0);
268 } catch (const hts::Exception& e) {
269 caught = true;
270 }
271 if ( ! caught) ++nerr;
272 }
273
274 // Test our own transpose to make sure it's OK for subsequent use.
275 nerr += test_transpose(verbose, 277);
276
277 const int ns[] = {1, 2, 3, 21, 300};
278 const int max_nthreads = omp_get_max_threads();
279 const int nthreads_step = max_nthreads > 40 ? 11 : 3;
280
281 TestOptions to;
282 to.block_size = 3; // only for block_sparse
283 to.verbose = verbose;
284 bool print_options = to.verbose;
285 for (int nthreads = 1; ; ) {
286 to.nthreads = nthreads;
287 if (to.verbose >= 1) {
288 std::cout << " " << nthreads;
289 std::cout.flush();
290 }
291 for (int ti = 0; ti < 4; ++ti) {
292 to.transpose = ti % 2;
293 if ( ! is_complex<Sclr>() && ti >= 2) break;
294 if (ti >= 2) to.conjugate = true;
295 for (size_t ni = 0; ni < sizeof(ns)/sizeof(*ns); ++ni) {
296 to.n = ns[ni];
297 for (size_t si = 0; si < TestOptions::n_solve_types; ++si) {
298 to.solve_type = static_cast<typename ut::TestOptions::SolveType>(si);
299 for (int ui = 0; ui < 2; ++ui) {
300 to.upper = ui;
301 for (int ri = 0; ri < 2; ++ri) {
302 for (int nrhs = 1; nrhs <= 3; nrhs += 2) {
303 to.nrhs = nrhs;
304 to.reprocess = ri;
305 to.matrix_type = TestOptions::diag;
306 nerr += test(to, print_options);
307 print_options = false;
308 to.matrix_type = TestOptions::dense; nerr += test(to);
309 to.matrix_type = TestOptions::sparse; nerr += test(to);
310 to.matrix_type = TestOptions::block_sparse; nerr += test(to);
311 to.matrix_type = TestOptions::implicit_unit_diag; nerr += test(to);
312 to.matrix_type = TestOptions::block_sparse_implicit_unit_diag; nerr += test(to);
313 if (to.n > 2) {
314 to.matrix_type = TestOptions::not_tri;
315 nerr += test_for_exception(to);
316 to.matrix_type = TestOptions::missing_some_diag;
317 nerr += test_for_exception(to);
318 to.matrix_type = TestOptions::not_tri_almost_diag;
319 nerr += test_for_exception(to);
320 }
321 }
322 }
323 }
324 }
325 }
326 }
327 if (nthreads == max_nthreads) break;
328 nthreads = std::min(max_nthreads, nthreads + nthreads_step);
329 }
330
331 return nerr;
332 }
