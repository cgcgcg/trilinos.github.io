191template<class Container>
192double distance(Container &nodeCoord, int i1, int i2) {
193 double dist = 0.0;
194 for(int j=0; j<3; j++)
195 dist+= SQR( nodeCoord(i1,j) - nodeCoord(i2,j) );
196 return sqrt(dist);
197}

2418int Multiply_Ones(const Epetra_CrsMatrix &A,const Epetra_Vector &x,Epetra_Vector &y){
2419 if(!A.Filled())
2420 EPETRA_CHK_ERR(-1); // Matrix must be filled.
2421
2422 double* xp = (double*) x.Values();
2423 double* yp = (double*) y.Values();
2424 const Epetra_Import* Importer_=A.Importer();
2425 const Epetra_Export* Exporter_=A.Exporter();
2426 Epetra_Vector *xcopy=0, *ImportVector_=0, *ExportVector_=0;
2427
2428 if (&x==&y && Importer_==0 && Exporter_==0) {
2429 xcopy = new Epetra_Vector(x);
2430 xp = (double *) xcopy->Values();
2431 }
2432 else if (Importer_)
2433 ImportVector_ = new Epetra_Vector(Importer_->TargetMap());
2434 else if (Exporter_)
2435 ExportVector_ = new Epetra_Vector(Exporter_->SourceMap());
2436
2437
2438 // If we have a non-trivial importer, we must import elements that are permuted or are on other processors
2439 if(Importer_ != 0) {
2440 EPETRA_CHK_ERR(ImportVector_->Import(x, *Importer_, Insert));
2441 xp = (double*) ImportVector_->Values();
2442 }
2443
2444 // If we have a non-trivial exporter, we must export elements that are permuted or belong to other processors
2445 if(Exporter_ != 0) yp = (double*) ExportVector_->Values();
2446
2447 // Do actual computation
2448 for(int i = 0; i < A.NumMyRows(); i++) {
2449 int NumEntries,*RowIndices;
2450 A.Graph().ExtractMyRowView(i,NumEntries,RowIndices);
2451 double sum = 0.0;
2452 for(int j = 0; j < NumEntries; j++)
2453 sum += xp[*RowIndices++];
2454 yp[i] = sum;
2455 }
2456
2457 if(Exporter_ != 0) {
2458 y.PutScalar(0.0); // Make sure target is zero
2459 EPETRA_CHK_ERR(y.Export(*ExportVector_, *Exporter_, Add)); // Fill y with Values from export vector
2460 }
2461 // Handle case of rangemap being a local replicated map
2462 if (!A.Graph().RangeMap().DistributedGlobal() && A.Comm().NumProc()>1) EPETRA_CHK_ERR(y.Reduce());

2464 delete xcopy;
2465 delete ImportVector_;
2466 delete ExportVector_;
