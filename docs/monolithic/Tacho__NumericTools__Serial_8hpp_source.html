79 inline void print_stat_factor() override {
80 base_type::print_stat_factor();
81 double flop = 0;
82 auto h_supernodes = Kokkos::create_mirror_view_and_copy(host_memory_space(), _supernodes);
83 switch (this->getSolutionMethod()) {
84 case 1: {
85 for (ordinal_type sid = 0; sid < _nsupernodes; ++sid) {
86 auto &s = h_supernodes(sid);
87 const ordinal_type m = s.m, n = s.n - s.m;
88 flop += DenseFlopCount<value_type>::Chol(m);
89 flop += DenseFlopCount<value_type>::Trsm(true, m, n);
90 flop += DenseFlopCount<value_type>::Syrk(m, n);
91 }
92 break;
93 }
94 case 2: {
95 for (ordinal_type sid = 0; sid < _nsupernodes; ++sid) {
96 auto &s = h_supernodes(sid);
97 const ordinal_type m = s.m, n = s.n - s.m;
98 flop += DenseFlopCount<value_type>::LDL(m);
99 flop += DenseFlopCount<value_type>::Trsm(true, m, n);
100 flop += DenseFlopCount<value_type>::Syrk(m, n);
101 }
102 break;
103 }
104 case 3: {
105 for (ordinal_type sid = 0; sid < _nsupernodes; ++sid) {
106 auto &s = h_supernodes(sid);
107 const ordinal_type m = s.m, n = s.n - s.m;
108 flop += DenseFlopCount<value_type>::LU(m, m);
109 flop += 2 * DenseFlopCount<value_type>::Trsm(true, m, n);
110 flop += DenseFlopCount<value_type>::Gemm(n, n, m);
111 }
112 break;
113 }
114 default: {
115 TACHO_TEST_FOR_EXCEPTION(true, std::logic_error, "The solution method is not supported");
116 }
117 }
118 const double kilo(1024);
119 printf(" FLOPs\n");
120 printf(" gflop for numeric factorization: %10.3f GFLOP\n", flop / kilo / kilo / kilo);
121 printf(" gflop/s for numeric factorization: %10.3f GFLOP/s\n",
122 flop / stat.t_factor / kilo / kilo / kilo);
123 printf("\n");
124 }

238 buf.data(), bufsize);
