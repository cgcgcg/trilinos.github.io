93 SACADO_INLINE_FUNCTION
94 VectorDynamicStorage& operator=(const VectorDynamicStorage& x) {
95 if (this != &x) {
96 *val_ = *x.val_;
97 if (sz_ != x.sz_) {
98 sz_ = x.sz_;
99 if (x.sz_ > len_) {
100#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ )
101 if (!owns_mem)
102 throw "Can\'t resize beyond original size when memory isn't owned!";
103#endif
104 if (len_ != 0)
105 ds_array<U>::destroy_and_release(dx_, len_);
106 len_ = x.sz_;
107 dx_ = ds_array<U>::strided_get_and_fill(x.dx_, x.stride_, sz_);
108 }
109 else
110 ds_array<U>::strided_copy(x.dx_, x.stride_, dx_, stride_, sz_);
111 }
112 else
113 ds_array<U>::strided_copy(x.dx_, x.stride_, dx_, stride_, sz_);
114 }
115 return *this;
116 }

150 SACADO_INLINE_FUNCTION
151 void resizeAndZero(int sz) {
152 if (sz > len_) {
153#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ )
154 if (!owns_mem)
155 throw "Can\'t resize beyond original size when memory isn't owned!";

172 SACADO_INLINE_FUNCTION
173 void expand(int sz) {
174 if (sz > len_) {
175#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ )
176 if (!owns_mem)
177 throw "Can\'t resize beyond original size when memory isn't owned!";
178#endif
179 U* dx_new = ds_array<U>::get_and_fill(sz);
180 ds_array<U>::copy(dx_, dx_new, sz_);
181 if (len_ > 0)
182 ds_array<U>::destroy_and_release(dx_, len_);
183 dx_ = dx_new;
184 len_ = sz;
185 }
186 else if (sz > sz_)
187 ds_array<U>::strided_zero(dx_+stride_*sz_, stride_, sz-sz_);
188 sz_ = sz;
189 }

198 SACADO_INLINE_FUNCTION
199 void setMemory(int sz, T* x, U* dx_p, int stride) {
200
201 // Destroy old memory
202 if (owns_mem) {
203 if (len_ != 0)
204 ds_array<U>::destroy_and_release(dx_, len_);
205 }
206
207 // Set new values
208 owns_mem = false;
209 sz_ = sz;
210 len_ = sz;
211 stride_ = stride;
212 val_ = x;
213 dx_ = dx_p;
214 }
