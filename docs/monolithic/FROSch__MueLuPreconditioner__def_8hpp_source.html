41 template<class SC,class LO,class GO,class NO>
42 void MueLuPreconditioner<SC,LO,GO,NO>::apply(const XMultiVector &x,
43 XMultiVector &y,
44 ETransp mode,
45 SC alpha,
46 SC beta) const
47 {
48 FROSCH_TIMER_START_SOLVER(applyTime,"MueLuPreconditioner::apply");
49 FROSCH_ASSERT(this->IsComputed_,"FROSch::MueLuPreconditioner: !this->IsComputed_.");
50
51 if (Y_.is_null()) Y_ = XMultiVectorFactory::Build(y.getMap(),y.getNumVectors());
52
53 int mgridSweeps = this->ParameterList_->sublist("MueLu").get("mgridSweeps",-1);
54 if (mgridSweeps>0) {
55 MueLuHierarchy_->Iterate(x,*Y_,mgridSweeps);
56 }
57 else{
58 typename ScalarTraits<SC>::magnitudeType tol = this->ParameterList_->sublist("MueLu").get("tol",1.e-6);
59 MueLuHierarchy_->Iterate(x,*Y_,tol);
60 }
61 y.update(alpha,*Y_,beta);
62 }

89 }
90 MueLuHierarchy_ = MueLuFactory_->CreateHierarchy(); // Das vor den if block
91 MueLuHierarchy_->GetLevel(0)->Set("A", Teuchos::rcp_const_cast<XMatrix>(this->K_)); // Das in den if block
92 MueLuHierarchy_->GetLevel(0)->Set("Nullspace", nullspace);
