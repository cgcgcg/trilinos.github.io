234 void transposeL()
235 {
236 auto ltRowMap_ = Kokkos::create_mirror_view(ltRowMap);
237 auto ltColIdx_ = Kokkos::create_mirror_view(ltColIdx);
238 auto ltVal_ = Kokkos::create_mirror_view(ltVal);
239
240 //Count the elements in each row of Lt
241 auto temp = OrdinalArrayHost("temp", nRows + 1);
242 auto rowPtrs = OrdinalArrayHost("rowPtrs", nRows);
243 for (Ordinal i = 0; i <= nRows; i++)
244 {
245 temp[i] = 0;
246 }
247 for (Ordinal i = 0; i < nRows; i++)
248 {
249 for (Ordinal k = lRowMap_[i]; k < lRowMap_[i+1]; k++)
250 {
251 temp[lColIdx_[k]+1]++;
252
253 }
254 }
255 //Perform an add scan to get the row map for
256 //the transpose
257 for (Ordinal i = 0; i <= nRows; i++)
258 {
259 ltRowMap_[i] = temp[i];
260 }
261 for (Ordinal i = 1; i <= nRows; i++)
262 {
263 ltRowMap_[i] += ltRowMap_[i-1];
264 }
265 //Set the row pointers to their initial places;
266 for (Ordinal i = 0; i < nRows; i++)
267 {
268 rowPtrs[i] = ltRowMap_[i];
269 }
270 //Copy the data
271 Kokkos::deep_copy(lVal_, lVal);
272 for (Ordinal i = 0; i < nRows; i++)
273 {
274 for (Ordinal k = lRowMap_[i]; k < lRowMap_[i+1]; k++)
275 {
276 Ordinal row = lColIdx_[k];
277 Scalar value = lVal_[k];
278 ltVal_[rowPtrs[row]] = value;
279 ltColIdx_[rowPtrs[row]] = i;
280 rowPtrs[row]++;
281 assert(rowPtrs[row] <= ltRowMap_[row + 1]);
282 }
283 }
284 Kokkos::deep_copy(ltRowMap, ltRowMap_);
285 Kokkos::deep_copy(ltColIdx, ltColIdx_);
286 Kokkos::deep_copy(ltVal,ltVal_);
287 }

289 void symbolicILU()
290 {
291 using std::vector;
292 using std::cout;
293 using std::stable_sort;
294 using std::sort;
295 OrdinalArrayMirror ia = aRowMapHost;
296 OrdinalArrayMirror ja = aColIdxHost;
297 int n = nRows;
298 int nzl = aRowMapHost[nRows] * (level + 2);
299 int nzu = aRowMapHost[nRows] * (level + 2);
300 Ordinal i;
301 int levfill = level;
302 vector<int> lnklst(n);
303 vector<int> curlev(n);
304 vector<int> levels(nzu);
305 vector<int> iwork(n);
306 vector<int> ial(nzl);
307 vector<int> jal(nzl);
308 vector<int> iau(nzu);
309 vector<int> jau(nzu);
310
311 int knzl = 0;
312 int knzu = 0;
313
314 ial[0] = 0;
315 iau[0] = 0;
316
317 for (i=0; i<n; i++)
318 {
319 int first, next, j;
320
321 /* copy column indices of row into workspace and sort them */
322
323 int len = ia[i+1] - ia[i];
324 next = 0;
325 for (j=ia[i]; j<ia[i+1]; j++)
326 iwork[next++] = ja[j];
327 // shell_sort(len, iwork);
328 //stable_sort(iwork.begin(), iwork.begin() + len);
329 sort(iwork.begin(), iwork.begin() + len);
330
331 /* construct implied linked list for row */
332
333 first = iwork[0];
334 curlev[first] = 0;
335
336 for (j=0; j<=len-2; j++)
337 {
338 lnklst[iwork[j]] = iwork[j+1];
339 curlev[iwork[j]] = 0;
340 }
341
342 lnklst[iwork[len-1]] = n;
343 curlev[iwork[len-1]] = 0;
344
345 /* merge with rows in U */
346
347 next = first;
348 while (next < i)
349 {
350 int oldlst = next;
351 int nxtlst = lnklst[next];
352 int row = next;
353 int ii;
354
355 /* scan row */
356
357 for (ii=iau[row]+1; ii<iau[row+1]; /*nop*/)
358 {
359 if (jau[ii] < nxtlst)
360 {
361 /* new fill-in */
362 int newlev = curlev[row] + levels[ii] + 1;
363 if (newlev <= levfill)
364 {
365 lnklst[oldlst] = jau[ii];
366 lnklst[jau[ii]] = nxtlst;
367 oldlst = jau[ii];
368 curlev[jau[ii]] = newlev;
369 }
370 ii++;
371 }
372 else if (jau[ii] == nxtlst)
373 {
374 int newlev;
375 oldlst = nxtlst;
376 nxtlst = lnklst[oldlst];
377 newlev = curlev[row] + levels[ii] + 1;
378 //curlev[jau[ii]] = MIN(curlev[jau[ii]], newlev);
379 if (curlev[jau[ii]] > newlev)
380 {
381 curlev[jau[ii]] = newlev;
382 }
383 ii++;
384 }
385 else /* (jau[ii] > nxtlst) */
386 {
387 oldlst = nxtlst;
388 nxtlst = lnklst[oldlst];
389 }
390 }
391 next = lnklst[next];
392 }
393
394 /* gather the pattern into L and U */
395
396 next = first;
397 while (next < i)
398 {
399 assert(knzl < nzl);
400 jal[knzl++] = next;
401 if (knzl >= nzl)
402 {
403 //(*nzl)++;
404 nzl += nRows;
405 jal.resize(nzl);
406 }
407 //jal.push_back(next);
408 // knzl++;
409 next = lnklst[next];
410 }
411 ial[i+1] = knzl;
412 assert(next == i);
413
414 while (next < n)
415 {
416 assert(knzu < nzu);
417 levels[knzu] = curlev[next];
418 jau[knzu++] = next;
419 if (knzu >= nzu)
420 {
421 //(*nzu)++;
422 nzu += nRows;
423 jau.resize(nzu);
424 levels.resize(nzu);
425 }
426 //jau.push_back(next);
427 //knzu++;
428 next = lnklst[next];
429 }
430 iau[i+1] = knzu;
431 }
432
433 nzl = knzl;
434 nzu = knzu;
435
436 #ifdef FASTIC_DEBUG_OUTPUT
437 std::cout << "knzl =" << knzl;
438 std::cout << "knzu =" << knzu;
439 std::cout << "ILU: nnz = "<< knzl + knzu << std::endl;
440 std::cout << "Actual nnz for ILU: " << nzl + nzu << std::endl;
441 #endif
442 //Initialize the A matrix that is to be used in the computation
443 aRowMap = OrdinalArray("aRowMap", nRows + 1);
444 aColIdx = OrdinalArray("aColIdx", knzl + knzu);
445 aRowIdx = OrdinalArray("aRowIds", knzl + knzu);
446 aRowMap_ = Kokkos::create_mirror_view(aRowMap);
447 aColIdx_ = Kokkos::create_mirror_view(aColIdx);
448 aRowIdx_ = Kokkos::create_mirror_view(aRowIdx);
449
450 Ordinal aRowPtr = 0;
451 aRowMap_[0] = 0;
452 for (i = 0; i < nRows; i++)
453 {
454 #ifdef FASTIC_DEBUG_OUTPUT
455 std::cout << "***row:" << i << std::endl;
456 #endif
457 for(Ordinal k = ial[i]; k < ial[i+1]; k++)
458 {
459 #ifdef FASTIC_DEBUG_OUTPUT
460 std::cout << "jal[k]=" << jal[k] << std::endl;
461 #endif
462 aColIdx_[aRowPtr] = jal[k];
463 aRowIdx_[aRowPtr] = i;
464 aRowPtr++;
465 }
466 for(Ordinal k = iau[i]; k < iau[i+1]; k++)
467 {
468 aColIdx_[aRowPtr] = jau[k];
469 aRowIdx_[aRowPtr] = i;
470 aRowPtr++;
471 }
472 aRowMap_[i+1] = aRowPtr;
473 }
474 Kokkos::deep_copy(aRowMap, aRowMap_);
475 Kokkos::deep_copy(aColIdx, aColIdx_);
476 Kokkos::deep_copy(aRowIdx, aRowIdx_);
477
478 //Now allocate memory for L and U.
479 //
480 lRowMap = OrdinalArray("lRowMap", nRows + 1);
481 ltRowMap = OrdinalArray("ltRowMap", nRows + 1);
482 countL();
483 //Allocate memory and initialize pattern for L, U (transpose).
484 lColIdx = OrdinalArray("lColIdx", lRowMap_[nRows]);
485 ltColIdx = OrdinalArray("ltColIdx", lRowMap_[nRows]);
486 #ifdef FASTIC_DEBUG_OUTPUT
487 std::cout << "nnz L = " << lRowMap[nRows] << std::endl;
488 #endif
489 }

547 void fillL()
548 {
549 lVal_ = Kokkos::create_mirror_view(lVal);
550 lColIdx_ = Kokkos::create_mirror_view(lColIdx);
551 diagElems_ = Kokkos::create_mirror_view(diagElems);
552
553 Ordinal lPtr = 0;
554 for (Ordinal i = 0; i < nRows; i++)
555 {
556 for (Ordinal k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
557 {
558 Ordinal row = i;
559 Ordinal col = aColIdx_[k];
560
561 if (row >= col)
562 {
563 if (row == col)
564 {
565 diagElems_[row] = aVal_[k];
566 }
567 lVal_[lPtr] = aVal_[k];
568 lColIdx_[lPtr] = aColIdx_[k];
569 lPtr++;
570 }
571 }
572 }
573 Kokkos::deep_copy(diagElems, diagElems_);
574 assert(lPtr == lRowMap_[nRows]);

576 if ((level > 0) && (guessFlag !=0))
577 {
578 OrdinalArray lGRowMap;
579 OrdinalArray lGColIdx;
580 ScalarArray lGVal;
581 ScalarArray gD;

583
584 initGuessPrec->getL(lGRowMap, lGColIdx, lGVal);
585 initGuessPrec->getD(gD);
586

602

612 #endif
613 }
614 }
615 Kokkos::deep_copy(lColIdx, lColIdx_);
616 Kokkos::deep_copy(lVal, lVal_);
617 }

619 void applyDiagonalScaling()
620 {
621 const Real one = RTS::one();
622 int anext = 0;
623 //First fill Aj and extract the diagonal scaling factors
624 //Use diag array to store scaling factors since
625 //it gets set to the correct value by findFactorPattern anyway.
626 auto diagFact_ = Kokkos::create_mirror_view(diagFact);
627 for (int i = 0; i < nRows; i++)
628 {
629 for(int k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
630 {
631 aRowIdx_[anext++] = i;
632 if (aColIdx_[k] == i)
633 {
634 diagFact_[i] = one/RTS::sqrt(STS::abs(aVal_[k]));
635 #ifdef FASTIC_DEBUG_OUTPUT
636 std::cout << "diagFact["<<i<<"]="<<aVal[k]<<std::endl;
637 #endif
638 }
639 }
640 }
641
642 //Now go through each element of A and apply the scaling
643 int row;
644 int col;
645 Real sc1, sc2;
646 for (int i = 0; i < nRows; i++)
647 {
648 for (int k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
649 {
650 row = aRowIdx_[k];
651 col = aColIdx_[k];
652
653 sc1 = diagFact_[row];
654 sc2 = diagFact_[col];
655 aVal_[k] = aVal_[k]*sc1*sc2;
656 }
657 }
658 Kokkos::deep_copy(diagFact, diagFact_);
659 }

661 void applyManteuffelShift()
662 {
663 const Scalar one = STS::one();
664 //Scalar shift = 0.05;
665 for (Ordinal i = 0; i < nRows; i++)
666 {
667 for (Ordinal k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
668 {
669 Ordinal row = i;
670 Ordinal col = aColIdx_[k];
671 if (row != col)
672 {
673 aVal_[k] = (one/(one + shift))*aVal_[k];
674 }
675 }
676 }
677 }

696 {
697 ParInitZeroFunctor<Ordinal, Scalar, ExecSpace> parInitZero(xOld);
698 Kokkos::parallel_for(RangePolicy(0, nRows), parInitZero);
699 ExecSpace().fence();
700#if 0
701 JacobiIterFunctor<Ordinal, Scalar, ExecSpace> jacIter(nRows, lRowMap, lColIdx, lVal,
702 x, y, xOld, diagElems);
703#endif
704 BlockJacobiIterFunctorL<Ordinal, Scalar, ExecSpace> jacIter(nRows, blkSz, lRowMap,
705 lColIdx, lVal,
706 x, y, xOld, diagElems);
707 ParCopyFunctor<Ordinal, Scalar, ExecSpace> parCopy(xOld, y);
708 Ordinal extent = nRows/blkSz;
709 if(nRows%blkSz != 0)
710 {
711 extent++;
712 }
713 ExecSpace().fence();
714 for (Ordinal i = 0; i < nTrisol; i++)
715 {
716 Kokkos::parallel_for(RangePolicy(0, extent), jacIter);
717 ExecSpace().fence();
718 Kokkos::parallel_for(RangePolicy(0, nRows), parCopy);
719 ExecSpace().fence();
720 }
721 return;
722 }

914 void checkIC() const
915 {
916 //Compute the L2 norm of the nonlinear residual (A - LLt) on sparsity pattern
917 //
918 Scalar sum = STS::zero();
919 for (int i = 0 ; i < nRows; i++)
920 {
921 int row = i;
922 // Each row in A matrix (result)
923 for (int k = aRowMap[i]; k < aRowMap[i+1]; k++)
924 {
925
926 int col = aColIdx[k];
927
928 if (row >= col) {
929 Scalar acc_val = aVal[k];
930 Ordinal lptr, uptr;
931 for ( lptr = lRowMap[i], uptr = lRowMap[aColIdx[k]] ;
932 lptr < lRowMap[i+1] && uptr < lRowMap[aColIdx[k]+1] ; )
933 {
934 if (lColIdx[lptr] == lColIdx[uptr])
935 {
936 acc_val -= lVal[lptr] * lVal[uptr];
937 lptr++;
938 uptr++;
939 }
940 else if (lColIdx[lptr] < lColIdx[uptr])
941 lptr++;
942 else
943 uptr++;
944 }
945
946 sum += acc_val * acc_val;
947 }
948 }
949 }
950 #ifdef FASTIC_DEBUG_OUTPUT
951 std::cout << "l2 norm of nonlinear residual = " << std::sqrt(sum) << std::endl;
952 #endif
