77 static inline ComplexT reciprocal (ComplexT c)
78 {
79 T r, den, cr, ci ;
80 ComplexT ret ;
81 cr = (Teuchos::ScalarTraits<ComplexT>::real(c)) ;
82 ci = (Teuchos::ScalarTraits<ComplexT>::imag(c)) ;
83 if (SCALAR_ABS (cr) >= SCALAR_ABS (ci))
84 {
85 r = ci / cr ;
86 den = cr + r * ci ;
87 ret = ::std::complex<T>(1.0 / den, -r / den) ;
88 }
89 else
90 {
91 r = cr / ci ;
92 den = r * cr + ci ;
93 ret = ::std::complex<T>(r / den, -1.0 / den) ;
94 }
95 return ret;
96 }
97
98 static inline ComplexT divide (ComplexT a, ComplexT b)
99 {
100 T r, den, ar, ai, br, bi ;
101 ComplexT ret;
102
103 br = (Teuchos::ScalarTraits<ComplexT>::real(b)) ;
104 bi = (Teuchos::ScalarTraits<ComplexT>::imag(b)) ;
105 ar = (Teuchos::ScalarTraits<ComplexT>::real(a)) ;
106 ai = (Teuchos::ScalarTraits<ComplexT>::imag(a)) ;
107 if (SCALAR_ABS (br) >= SCALAR_ABS (bi))
108 {
109 r = bi / br ;
110 den = br + r * bi ;
111 ret = ::std::complex<T>((ar + ai * r) / den, (ai - ar * r) / den) ;
112 }
113 else
114 {
115 r = br / bi ;
116 den = r * br + bi ;
117 ret = ::std::complex<T>((ar * r + ai) / den, (ai * r - ar) / den) ;
118 }
119 return ret;
120 }
121
122 static inline ComplexT divideConjugate (ComplexT a, ComplexT b)
123 {
124 T r, den, ar, ai, br, bi ;
125 ComplexT ret;
126
127 br = (Teuchos::ScalarTraits<ComplexT>::real(b)) ;
128 bi = (Teuchos::ScalarTraits<ComplexT>::imag(b)) ;
129 ar = (Teuchos::ScalarTraits<ComplexT>::real(a)) ;
130 ai = (Teuchos::ScalarTraits<ComplexT>::imag(a)) ;
131 if (SCALAR_ABS (br) >= SCALAR_ABS (bi))
132 {
133 r = (-bi) / br ;
134 den = br - r * bi ;
135 ret = ::std::complex<T>((ar + ai * r) / den, (ai - ar * r) / den) ;
136 }
137 else
138 {
139 r = br / (-bi) ;
140 den = r * br - bi;
141 ret = ::std::complex<T>((ar * r + ai) / den, (ai * r - ar) / den) ;
142 }
143 return ret;
144 }
145
146 static inline magnitudeType approxABS (ComplexT a)
147 {
148 return ( SCALAR_ABS (Teuchos::ScalarTraits<ComplexT>::real(a)) +
149 SCALAR_ABS (Teuchos::ScalarTraits<ComplexT>::imag(a)) ) ;
150 }
151
152 static inline magnitudeType abs (ComplexT a)
153 {
154 T r, ar, ai ;
155 magnitudeType s;
156
157 ar = SCALAR_ABS (Teuchos::ScalarTraits<ComplexT>::real(a)) ;
158 ai = SCALAR_ABS (Teuchos::ScalarTraits<ComplexT>::imag(a)) ;
159 if (ar >= ai)
160 {
161 if (ar + ai == ar)
162 {
163 (s) = ar ;
164 }
165 else
166 {
167 r = ai / ar ;
168 (s) = ar * sqrt (1.0 + r*r) ;
169 }
170 }
171 else
172 {
173 if (ai + ar == ai)
174 {
175 (s) = ai ;
176 }
177 else
178 {
179 r = ar / ai ;
180 (s) = ai * sqrt (1.0 + r*r) ;
181 }
182 }
183 return s;
184 }
