48template <class Scalar,
49 class LocalOrdinal,

51 class Node>
52typename Rebalance_LinearProblem<Scalar, LocalOrdinal, GlobalOrdinal, Node>::NewType
53Rebalance_LinearProblem<Scalar, LocalOrdinal, GlobalOrdinal, Node>::operator()( OriginalType const & origProblem )
54{
55 using cg_t = Tpetra::CrsGraph < LocalOrdinal, GlobalOrdinal, Node>;
56 using mv_t = Tpetra::MultiVector <Scalar, LocalOrdinal, GlobalOrdinal, Node>;
57 using cm_t = Tpetra::CrsMatrix <Scalar, LocalOrdinal, GlobalOrdinal, Node>;
58 using lp_t = Tpetra::LinearProblem<Scalar, LocalOrdinal, GlobalOrdinal, Node>;
59
60 // Save original object
61 this->origObj_ = origProblem;
62
63 Teuchos::RCP<cm_t> origMatrix = Teuchos::rcp<cm_t>( dynamic_cast<cm_t *>(origProblem->getMatrix().get()), false );
64 Teuchos::RCP<cg_t const> origGraph = Teuchos::rcp<cg_t const>( dynamic_cast<cg_t const *>(origMatrix->getGraph().get()), false );
65 Teuchos::RCP<mv_t> origLHS = origProblem->getLHS();
66 Teuchos::RCP<mv_t> origRHS = origProblem->getRHS();
67
68 // ****************************************************************
69 // Instantiate a partitioning problem and solve it.
70 // ****************************************************************
71 using GraphAdapter_t = Zoltan2::XpetraCrsGraphAdapter<cg_t>;
72 GraphAdapter_t graphAdapter(origGraph);
73 Zoltan2::PartitioningProblem<GraphAdapter_t> partitioningProblem(&graphAdapter, paramListForZoltan2PartitioningProblem_.get());
74 partitioningProblem.solve();
75
76 // ****************************************************************
77 // Rebalance the graph
78 // ****************************************************************
79 Teuchos::RCP<cg_t> newGraph( Teuchos::null );
80 graphAdapter.applyPartitioningSolution(*origGraph, newGraph, partitioningProblem.getSolution());
81
82 // ****************************************************************
83 // Rebalance the matrix
84 // ****************************************************************
85 using MatrixAdapter_t = Zoltan2::XpetraCrsMatrixAdapter<cm_t>;
86 MatrixAdapter_t matrixAdapter(origMatrix);
87
88 Teuchos::RCP<cm_t> newMatrix( Teuchos::null );
89 matrixAdapter.applyPartitioningSolution(*origMatrix, newMatrix, partitioningProblem.getSolution());
90
91 // ****************************************************************
92 // Rebalance the lhs vector
93 // ****************************************************************
94 using MultiVectorAdapter_t = Zoltan2::XpetraMultiVectorAdapter<mv_t>;
95 MultiVectorAdapter_t lhsAdapter(origLHS);
96
97 Teuchos::RCP<mv_t> newLHS( Teuchos::null );
98 lhsAdapter.applyPartitioningSolution(*origLHS, newLHS, partitioningProblem.getSolution());
99
100 // ****************************************************************
101 // Rebalance the rhs vector
102 // ****************************************************************
103 MultiVectorAdapter_t rhsAdapter(origRHS);
104
105 Teuchos::RCP<mv_t> newRHS( Teuchos::null );
106 rhsAdapter.applyPartitioningSolution(*origRHS, newRHS, partitioningProblem.getSolution());
107
108 this->newObj_ = Teuchos::rcp<lp_t>( new lp_t(newMatrix, newLHS, newRHS) );
109
110 return this->newObj_;
111}
