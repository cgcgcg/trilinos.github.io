28 template <typename TpetraLocalOrdinal, typename TpetraGlobalOrdinal, typename TpetraNode,
29 typename StokhosOrdinal, typename StokhosValue>
30 Teuchos::RCP< Tpetra::CrsGraph<TpetraLocalOrdinal,TpetraGlobalOrdinal,TpetraNode > >
31 sparse3Tensor2TpetraCrsGraph(
32 const Stokhos::OrthogPolyBasis<StokhosOrdinal,StokhosValue>& basis,
33 const Stokhos::Sparse3Tensor<StokhosOrdinal,StokhosValue>& Cijk,
34 const Teuchos::RCP<const Teuchos::Comm<int> >& comm)
35 {
36 typedef Stokhos::Sparse3Tensor<StokhosOrdinal,StokhosValue> Cijk_type;
37 typedef Tpetra::Map<TpetraLocalOrdinal,TpetraGlobalOrdinal,TpetraNode> Map;
38 typedef Tpetra::CrsGraph<TpetraLocalOrdinal,TpetraGlobalOrdinal,TpetraNode> Graph;
39 using Teuchos::RCP;
40 using Teuchos::arrayView;
41
42 // Number of stochastic rows
43 StokhosOrdinal num_rows = basis.size();
44
45 // Compute max num entries -- note this is a substantial overestimate in most cases because it doesn't
46 // take into account overlap in j entries for each k. Also, while Cijk is symmetric with respect to i and
47 // j, it is not necessarily symmetric with respect to k. So this is the best we can do without adding
48 // and i -> j -> k traversal to the Cijk tensor.
49 size_t max_num_entry = 0;
50 for (typename Cijk_type::i_iterator i_it=Cijk.i_begin(); i_it!=Cijk.i_end(); ++i_it) {
51 size_t num_entry = 0;
52 for (typename Cijk_type::ik_iterator k_it = Cijk.k_begin(i_it); k_it != Cijk.k_end(i_it); ++k_it) {
53 for (typename Cijk_type::ikj_iterator j_it = Cijk.j_begin(k_it); j_it != Cijk.j_end(k_it); ++j_it) {
54 ++num_entry;
55 }
56 max_num_entry = (num_entry > max_num_entry) ? num_entry : max_num_entry;
57 }
58 }
59
60 // Replicated local map
61 RCP<const Map> map =
62 Tpetra::createLocalMapWithNode<TpetraLocalOrdinal,TpetraGlobalOrdinal,TpetraNode>(num_rows, comm);
63
64 // Graph to be created
65 RCP<Graph> graph = Tpetra::createCrsGraph(map, max_num_entry);
66
67 // Loop over Cijk entries including a non-zero in the graph at
68 // indices (i,j) if there is any k for which Cijk is non-zero
69 for (typename Cijk_type::i_iterator i_it=Cijk.i_begin(); i_it!=Cijk.i_end(); ++i_it) {
70 TpetraGlobalOrdinal i = index(i_it);
71 for (typename Cijk_type::ik_iterator k_it = Cijk.k_begin(i_it); k_it != Cijk.k_end(i_it); ++k_it) {
72 for (typename Cijk_type::ikj_iterator j_it = Cijk.j_begin(k_it); j_it != Cijk.j_end(k_it); ++j_it) {
73 TpetraGlobalOrdinal j = index(j_it);
74 graph->insertGlobalIndices(i, arrayView(&j, 1));
75 }
76 }
77 }
78
79 // Sort, remove redundencies, transform to local, ...
80 graph->fillComplete();
81
82 return graph;
83 }
