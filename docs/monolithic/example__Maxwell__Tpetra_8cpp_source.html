225template<class Container>
226double distance(Container &nodeCoord, int i1, int i2) {
227 double dist = 0.0;
228 for(int j=0; j<3; j++)
229 dist+= SQR( nodeCoord(i1,j) - nodeCoord(i2,j) );
230 return sqrt(dist);
231}
