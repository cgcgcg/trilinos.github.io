108 typedef typename LocalScalarType<
109 Fad::Exp::GeneralFad<Fad::Exp::DynamicStorage<T, U>>, Stride>::type

113 const int size = (x.size() + blockDim.x - threadIdx.x - 1) / blockDim.x;
114 const int offset = threadIdx.x;

116

120 // regardless
121 const T *dx = x.dx();

169struct ThreadLocalScalarType<

171 typename std::enable_if<is_view_fad_contiguous<ViewType>::value>::type> {
172 typedef typename ViewType::traits TraitsType;
173 // typedef Impl::ViewMapping<TraitsType, typename TraitsType::specialize>
174 // MappingType; typedef typename MappingType::thread_local_scalar_type type;
175
176 using fad_type = typename ViewType::value_type;
177 enum { FadStaticDimension = Sacado::StaticSize<fad_type>::value };
178 enum { PartitionedFadStride = TraitsType::array_layout::scalar_stride };
179
180 // The partitioned static size -- this will be 0 if ParitionedFadStride
181 // does not evenly divide FadStaticDimension
182 enum {
183 PartitionedFadStaticDimension =
184 Impl::computeFadPartitionSize(FadStaticDimension, PartitionedFadStride)
185 };
186#ifdef KOKKOS_ENABLE_CUDA
187 typedef typename Sacado::LocalScalarType<
188 fad_type, unsigned(PartitionedFadStride)>::type strided_scalar_type;
189 typedef typename std::conditional<
190 std::is_same<typename TraitsType::execution_space, Kokkos::Cuda>::value,
191 strided_scalar_type, fad_type>::type thread_local_scalar_type;
192#elif defined(KOKKOS_ENABLE_HIP)
193 typedef typename Sacado::LocalScalarType<
194 fad_type, unsigned(PartitionedFadStride)>::type strided_scalar_type;
195 typedef typename std::conditional<
196 std::is_same<typename TraitsType::execution_space, Kokkos::HIP>::value,
197 strided_scalar_type, fad_type>::type thread_local_scalar_type;
198#else
199 typedef fad_type thread_local_scalar_type;
200#endif
201 typedef thread_local_scalar_type type;
202};
