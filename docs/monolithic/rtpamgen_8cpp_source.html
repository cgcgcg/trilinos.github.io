138/*****************************************************************************/
139int main(int argc, char** argv)
140/*****************************************************************************/
141{
142 int issz;
143 int start_rank = 0;
144 int end_rank = 0;
145 char * out_file_name = NULL;
146 FILE * infile = NULL;
147 long size;
148 char *file_char_array = NULL;
149
150 // Teuchos-based optins
151 Teuchos::CommandLineProcessor clp(false);
152
153 int rank =0;
154 clp.setOption("rank", &rank, "rank of processor");
155 int num_procs=1;
156 clp.setOption("num_procs", &num_procs, "rank of processor");
157 int all = 0;
158 clp.setOption("all", &all, "generate all meshes for number of processors");
159 int dimension = 3;
160 clp.setOption("dimension", &dimension, "dimension");
161 std::string inFileName;
162 clp.setOption("file", &inFileName, "mesh description file");
163
164 switch (clp.parse(argc, argv)) {
165 case Teuchos::CommandLineProcessor::PARSE_HELP_PRINTED: return EXIT_SUCCESS;
166 case Teuchos::CommandLineProcessor::PARSE_ERROR:
167 case Teuchos::CommandLineProcessor::PARSE_UNRECOGNIZED_OPTION: return EXIT_FAILURE;
168 case Teuchos::CommandLineProcessor::PARSE_SUCCESSFUL: break;
169 }
170
171 const char * file_name = inFileName.c_str();
172
173 if(rank < 0 || rank >= num_procs){
174 printf("rtpamgen: rank must be positive and between 0 and the number of processors %i\n",rank);
175 exit(1);
176 }
177
178 if(num_procs < 0){
179 printf("rtpamgen: number of processors must be positive %i\n", num_procs);
180 exit(1);
181 }
182
183
184 /*deal with all switch*/
185 end_rank = num_procs;
186 if(!all){
187 start_rank = rank;
188 end_rank = start_rank+1;
189 }
190
191 infile = fopen(file_name,"r");
192 if(!infile){
193 printf("unable to open input file ");
194 return 1;
195 }
196 fseek(infile, 0, SEEK_END);
197 size = ftell(infile);
198 fseek(infile, 0, SEEK_SET);
199 file_char_array = (char *)malloc(size + 1);
200 file_char_array[size] = '\0';
201 fread(file_char_array, sizeof(char), size, infile);
202 fclose(infile);
203
204 /*create the out_file_name*/
205 out_file_name = (char*)malloc(MAX_STR_LENGTH+1);
206
207 for( rank = start_rank; rank != end_rank; rank ++){
208 int cr_result;
209 sprintf(out_file_name,"%s.exo.%i.%i",file_name,num_procs,rank);
210
211 cr_result = Create_Pamgen_Mesh(file_char_array,
212 dimension,
213 rank,
214 num_procs,
215 INT_MAX);
216
217
218 if (cr_result == ERROR_PARSING_DEFINITION){
219 int essz = getPamgenEchoStreamSize();
220 char * echo_char_array = (char *)malloc(essz+1);
221 printf("PARSE ERROR\n");
222 echo_char_array[essz] = '\0';
223 echo_char_array = getPamgenEchoStream(echo_char_array);
224 if(echo_char_array)printf(echo_char_array);
225 if(cr_result == ERROR_CREATING_IMD)printf("ERROR Failure to create Inline_Mesh_Desc creation\n");
226 if(echo_char_array)free(echo_char_array);
227 return 1;
228 }
229
230 if(cr_result == ERROR_CREATING_MS){
231 int essz = getPamgenErrorStreamSize();
232 char * error_char_array = (char *)malloc(essz+1);
233 error_char_array[essz] = '\0';
234 error_char_array = getPamgenErrorStream(error_char_array);
235 if(error_char_array)printf(error_char_array);
236 printf("\nERROR Failure to create Mesh_Specification\n");
237 if(error_char_array)free(error_char_array);
238 return 1;
239 }
240
241
242 {
243 int wssz = getPamgenWarningStreamSize();
244 if(wssz){
245 char * warning_char_array = (char *)malloc(wssz+1);
246 warning_char_array[wssz] = '\0';
247 warning_char_array = getPamgenWarningStream(warning_char_array);
248 printf("WARNING Records\n");
249 printf(warning_char_array);
250 free(warning_char_array);
251 }
252 }
253
254
255 {
256 issz = getPamgenInfoStreamSize();
257 if(issz){
258 char * info_char_array = (char *)malloc(issz+1);
259 info_char_array[issz] = '\0';
260 info_char_array = getPamgenInfoStream(info_char_array);
261 printf("INFO Records\n");
262 printf(info_char_array);
263 free(info_char_array);
264 }
265 }
266
267 read_mesh_to_memory();
268
269 write_mesh_to_stdout();
270
271 write_to_exodus(rank,num_procs,out_file_name);
272
273 Delete_Pamgen_Mesh();
274 free_memory();
275 }/* end loop over all output ranks*/
276
277 if(file_char_array)free(file_char_array);
278 if(out_file_name)free(out_file_name);
279}

424/*****************************************************************************/
425void read_mesh_to_memory()
426/*****************************************************************************/
427{
428 int idum = 0;
429 float fdum;
430 int i;
431 int j;
432 int b;
433 char * cdum = NULL;
434 int error = 0;
435 int id = 0;
436
437 mss.bptr[0] = mss.buffer[0];
438 mss.bptr[1] = mss.buffer[1];
439 mss.bptr[2] = mss.buffer[2];
440
441 for(i = 0; i < 100; i++)
442 for(j=0; j<4; j++) mss.qaRecord[i][j] = (char*)malloc(MAX_STR_LENGTH+1) ;
443
444
445 error += im_ex_get_init ( id,
446 mss.title,
447 &mss.num_dim,
448 &(mss.num_nodes),
449 &mss.num_elem,
450 &mss.num_elem_blk,
451 &mss.num_node_sets,
452 &mss.num_side_sets);
453
454
455 error += im_ex_inquire(id, IM_EX_INQ_NS_NODE_LEN, (int*)&mss.num_node_set_nodes,
456 &fdum, cdum);
457 error += im_ex_inquire(id, IM_EX_INQ_NS_DF_LEN, (int*)&mss.num_node_set_dfs,
458 &fdum, cdum);
459 error += im_ex_inquire(id, IM_EX_INQ_SS_ELEM_LEN, (int*)&mss.num_side_set_elements,
460 &fdum, cdum);
461 error += im_ex_inquire(id, IM_EX_INQ_SS_NODE_LEN, (int*)&mss.num_side_set_nodes,
462 &fdum, cdum);
463 error += im_ex_inquire(id, IM_EX_INQ_SS_DF_LEN, (int*)&mss.num_side_set_dfs,
464 &fdum, cdum);
465
466 /* get version number*/
467
468 error += im_ex_inquire(id, IM_EX_INQ_API_VERS, &idum, &fdum, cdum);
469
470 mss.version_number = (double) fdum;
471
472 mss.version = (int) mss.version_number;
473
474 /* get genesis-II parameters*/
475
476 error += im_ex_inquire(id, IM_EX_INQ_EB_PROP, (int*)&mss.num_block_properties, &fdum, cdum);
477
478 error += im_ex_inquire(id, IM_EX_INQ_NS_PROP, (int*)&mss.num_node_set_properties,
479 &fdum, cdum);
480
481 error += im_ex_inquire(id, IM_EX_INQ_SS_PROP, (int*)&mss.num_side_set_properties,
482 &fdum, cdum);
483
484 mss.coord = (double *)malloc(mss.num_nodes*mss.num_dim*sizeof(double));
485
486 error += im_ex_get_coord(id,mss.coord,mss.coord+mss.num_nodes,mss.coord+2*mss.num_nodes);
487
488
489 error += im_ex_get_coord_names (id, mss.bptr);
490
491 if (mss.num_elem){
492 mss.element_order_map = (int *)malloc(mss.num_elem * sizeof(int));
493 error += im_ex_get_map(id, mss.element_order_map);
494
495 if (mss.num_elem){
496 mss.global_element_numbers = (int *)malloc(mss.num_elem*sizeof(int));
497 error += im_ex_get_elem_num_map(id, mss.global_element_numbers);
498 }
499
500 if (mss.num_nodes){
501 mss.global_node_numbers = (int *)malloc(mss.num_nodes * sizeof(int));
502 error += im_ex_get_node_num_map(id, mss.global_node_numbers);
503 }
504
505
506 /*block info*/
507
508 mss.block_id = (int *)malloc(mss.num_elem_blk*sizeof(int));
509 mss.nodes_per_element = (int *)malloc(mss.num_elem_blk*sizeof(int));
510 mss.element_attributes = (int *)malloc(mss.num_elem_blk*sizeof(int));
511 mss.elements = (int *)malloc(mss.num_elem_blk*sizeof(int));
512 mss.element_types = (char **)malloc(mss.num_elem_blk*sizeof(char *));
513 mss.elmt_node_linkage = (int **)malloc(mss.num_elem_blk*sizeof(int*));
514
515 error += im_ex_get_elem_blk_ids(id, mss.block_id);
516
517 for(i = 0; i < mss.num_elem_blk; i ++){
518 mss.element_types[i] = (char *)malloc((MAX_STR_LENGTH + 1)*sizeof(char));
519 error += im_ex_get_elem_block(id,
520 mss.block_id[i],
521 mss.element_types[i],
522 (int*)&(mss.elements[i]),
523 (int*)&(mss.nodes_per_element[i]),
524 (int*)&(mss.element_attributes[i]));
525 }
526
527 /*connectivity*/
528 for(b = 0; b < mss.num_elem_blk; b++){
529 mss.elmt_node_linkage[b] = (int*)malloc(mss.nodes_per_element[b]*mss.elements[b]*sizeof(int));
530 error += im_ex_get_elem_conn(id,mss.block_id[b],mss.elmt_node_linkage[b]);
531 }
532
533
534 if(mss.num_node_sets){
535 mss.node_set_id = (int *) malloc(mss.num_node_sets*sizeof(int));
536 mss.num_nodes_in_node_set = (int *) malloc(mss.num_node_sets*sizeof(int));
537 mss.node_set_nodes = (int **)malloc(mss.num_node_sets*sizeof(int*));
538 mss.num_df_in_node_set = (int *) malloc(mss.num_node_sets*sizeof(int*));
539
540 error += im_ex_get_node_set_ids(id, mss.node_set_id);
541
542
543 for(i = 0; i < mss.num_node_sets; i ++){
544 error += im_ex_get_node_set_param(id, mss.node_set_id[i],
545 (int*)&mss.num_nodes_in_node_set[i],
546 (int*)&mss.num_df_in_node_set[i]);
547
548 mss.node_set_nodes[i] = NULL;
549
550 if(mss.num_nodes_in_node_set[i]) {
551 mss.node_set_nodes[i] = (int *)malloc(mss.num_nodes_in_node_set[i]*sizeof(int));
552 error += im_ex_get_node_set(id, mss.node_set_id[i], mss.node_set_nodes[i]);
553 }
554 }
555 }
556
557 /*side sets*/
558 if(mss.num_side_sets){
559 mss.side_set_id = (int*)malloc(mss.num_side_sets*sizeof(int));
560 mss.num_elements_in_side_set = (int*)malloc(mss.num_side_sets*sizeof(int));
561 mss.num_df_in_side_set = (int*)malloc(mss.num_side_sets*sizeof(int));
562 mss.side_set_elements = (int**)malloc(mss.num_side_sets*sizeof(int *));
563 mss.side_set_faces = (int **)malloc(mss.num_side_sets*sizeof(int*));
564
565 error += im_ex_get_side_set_ids(id, mss.side_set_id);
566 for(i = 0; i < mss.num_side_sets; i ++){
567 int ne = 0;
568 error += im_ex_get_side_set_param(id, mss.side_set_id[i],
569 (int*)&mss.num_elements_in_side_set[i],
570 (int*)&mss.num_df_in_side_set[i]);
571
572 ne = mss.num_elements_in_side_set[i];
573 mss.side_set_elements[i] = (int*)malloc(ne*sizeof(int));
574 mss.side_set_faces[i] = (int*)malloc(ne*sizeof(int));
575 if(ne){
576 error += im_ex_get_side_set(id, mss.side_set_id[i],
577 mss.side_set_elements[i],
578 mss.side_set_faces[i]);
579
580 }
581 }
582 }
583
584 error += im_ex_inquire(id, IM_EX_INQ_QA, (int*)&mss.num_qa_records, &fdum, cdum);
585
586 if(mss.num_qa_records)error += im_ex_get_qa(id,mss.qaRecord);
587
588
589 error += im_ex_inquire(id, IM_EX_INQ_INFO, (int*)&mss.num_info_records, &fdum, cdum);
590 if(mss.num_info_records) {
591 mss.info_records = (char **)malloc(mss.num_info_records*sizeof(char *));/*new std::string[num_info_records];*/
592 for(i = 0; i < mss.num_info_records; i ++){
593 mss.info_records[i] = (char *)malloc(MAX_STR_LENGTH+1);
594 }
595 error += im_ex_get_info(id, mss.info_records);
596 }
597
598
599 /*nemesis data*/
600 /* global info*/
601 if ( im_ne_get_init_global(id, &mss.num_nodes_global, &mss.num_elems_global,
602 &mss.num_elm_blks_global, &mss.num_node_sets_global,
603 &mss.num_side_sets_global) < 0 )
604 ++error;
605
606
607
608 if ( im_ne_get_init_info(id, &mss.num_total_proc, &mss.num_proc_in_file, mss.type) < 0 )
609 ++error;
610
611 mss.elem_blk_ids_global = (int*)malloc(mss.num_elm_blks_global*sizeof(int));
612 mss.elem_blk_cnts_global = (int*)malloc(mss.num_elm_blks_global*sizeof(int));
613
614 if ( im_ne_get_eb_info_global(id,mss.elem_blk_ids_global,mss.elem_blk_cnts_global) < 0 )
615 ++error;
616
617 mss.ns_ids_global = (int *)malloc(mss.num_node_sets_global*sizeof(int));
618 mss.ns_cnts_global = (int *)malloc(mss.num_node_sets_global*sizeof(int));
619 mss.ns_df_cnts_global = (int *)malloc(mss.num_node_sets_global*sizeof(int));
620 mss.ss_ids_global = (int *)malloc(mss.num_side_sets_global*sizeof(int));
621 mss.ss_cnts_global = (int *)malloc(mss.num_side_sets_global*sizeof(int));
622 mss.ss_df_cnts_global = (int *)malloc(mss.num_side_sets_global*sizeof(int));
623
624
625 if ( mss.num_node_sets_global > 0 ) {
626 if ( im_ne_get_ns_param_global(id,mss.ns_ids_global,mss.ns_cnts_global,
627 mss.ns_df_cnts_global) < 0 )++error;
628 }
629
630 if ( mss.num_side_sets_global > 0 ) {
631 if ( im_ne_get_ss_param_global(id,mss.ss_ids_global,mss.ss_cnts_global,
632 mss.ss_df_cnts_global) < 0 ) ++error;
633 }
634
635 /*parallel info*/
636 if ( im_ne_get_loadbal_param( id,
637 &mss.num_internal_nodes,
638 &mss.num_border_nodes,
639 &mss.num_external_nodes,
640 &mss.num_internal_elems,
641 &mss.num_border_elems,
642 &mss.num_node_comm_maps,
643 &mss.num_elem_comm_maps,
644 0/*unused*/ ) < 0 )++error;
645
646 mss.internal_elements = (int *)malloc(mss.num_internal_elems*sizeof(int));
647 mss.border_elements = (int *)malloc(mss.num_border_elems*sizeof(int));
648 mss.internal_nodes = (int *)malloc(mss.num_internal_nodes*sizeof(int));
649 mss.border_nodes = (int *)malloc(mss.num_border_nodes*sizeof(int));
650 mss.external_nodes = (int *)malloc(mss.num_external_nodes*sizeof(int));
