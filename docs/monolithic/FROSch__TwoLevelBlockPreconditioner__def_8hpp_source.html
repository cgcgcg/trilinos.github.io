21 template <class SC,class LO,class GO,class NO>

23 ParameterListPtr parameterList) :

26 FROSCH_DETAILTIMER_START_LEVELID(twoLevelBlockPreconditionerTime,"TwoLevelBlockPreconditioner::TwoLevelBlockPreconditioner");
27 if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("IPOUHarmonicCoarseOperator")) {
28 // Set the LevelID in the sublist
29 parameterList->sublist("IPOUHarmonicCoarseOperator").set("Level ID",this->LevelID_);
30 // FROSCH_ASSERT(false,"not implemented for block.");
31 // this->ParameterList_->sublist("IPOUHarmonicCoarseOperator").sublist("InterfacePartitionOfUnity").set("Test Unconnected Interface",true);
32 CoarseOperator_ = IPOUHarmonicCoarseOperatorPtr(new IPOUHarmonicCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"IPOUHarmonicCoarseOperator")));
33 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("GDSWCoarseOperator")) {
34 // Set the LevelID in the sublist
35 parameterList->sublist("GDSWCoarseOperator").set("Level ID",this->LevelID_);
36 // this->ParameterList_->sublist("GDSWCoarseOperator").set("Test Unconnected Interface",true);
37 CoarseOperator_ = GDSWCoarseOperatorPtr(new GDSWCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"GDSWCoarseOperator")));
38 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("RGDSWCoarseOperator")) {
39 // Set the LevelID in the sublist
40 parameterList->sublist("RGDSWCoarseOperator").set("Level ID",this->LevelID_);
41 // this->ParameterList_->sublist("RGDSWCoarseOperator").set("Test Unconnected Interface",true);
42 CoarseOperator_ = RGDSWCoarseOperatorPtr(new RGDSWCoarseOperator<SC,LO,GO,NO>(k,sublist(parameterList,"RGDSWCoarseOperator")));
43 } else {
44 FROSCH_ASSERT(false,"CoarseOperator Type unkown.");
45 } // TODO: Add ability to disable individual levels
46 if (this->UseMultiplicative_) {
47 this->MultiplicativeOperator_->addOperator(CoarseOperator_);
48 }
49 else{
50 this->SumOperator_->addOperator(CoarseOperator_);
51 }

55 template <class SC,class LO,class GO,class NO>
56 int TwoLevelBlockPreconditioner<SC,LO,GO,NO>::initialize(UN dimension,
57 UNVecPtr dofsPerNodeVec,
58 DofOrderingVecPtr dofOrderingVec,
59 int overlap,
60 ConstXMapPtrVecPtr repeatedMapVec,
61 ConstXMultiVectorPtrVecPtr nullSpaceBasisVec,
62 ConstXMultiVectorPtrVecPtr nodeListVec,
63 ConstXMapPtrVecPtr2D dofsMapsVec,
64 GOVecPtr2D dirichletBoundaryDofsVec)
65 {
66 FROSCH_TIMER_START_LEVELID(initializeTime,"TwoLevelBlockPreconditioner::initialize");
67
69 // Checks //
71 UN nmbBlocks = dofsPerNodeVec.size();
72 for (UN i = 0; i < dofOrderingVec.size(); i++ ) {
73 DofOrdering dofOrdering = dofOrderingVec[i];
74 FROSCH_ASSERT(dofOrdering == NodeWise || dofOrdering == DimensionWise || dofOrdering == Custom,"ERROR: Specify a valid DofOrdering.");
75 }
76 int ret = 0;
77
79 // Maps //
81 FROSCH_ASSERT(!repeatedMapVec.is_null(),"repeatedMapVec.is_null() = true. Please provide the repeated maps vector. The maps itself can be null and will be constructed.");
82 for (UN i = 0; i < repeatedMapVec.size(); i++) {
83 if (repeatedMapVec[i].is_null()) {
84 FROSCH_ASSERT( i==0, "We can only construct a repeated map for a non block system");
85 repeatedMapVec[i] = BuildRepeatedMap( this->K_ );
86 // repeatedMapVec[i] = BuildRepeatedMap(this->K_, this->ParameterList_->get("Reduce approx repeated map",true) ); // Todo: Achtung, die UniqueMap könnte unsinnig verteilt sein. Falls es eine repeatedMap gibt, sollte dann die uniqueMap neu gebaut werden können. In diesem Fall, sollte man das aber basierend auf der repeatedNodesMap tun
87 }
88 }
89
90 // Build dofsMaps and repeatedNodesMap
91 ConstXMapPtrVecPtr repeatedNodesMapVec;
92 if (dofsMapsVec.is_null()) {
93 FROSCH_DETAILTIMER_START_LEVELID(buildDofMapsTime,"BuildDofMaps");
94 if (0>BuildDofMapsVec(repeatedMapVec,dofsPerNodeVec,dofOrderingVec,repeatedNodesMapVec,dofsMapsVec)) ret -= 100; // Todo: Rückgabewerte
95 } else {
96 FROSCH_ASSERT(dofsMapsVec.size()==dofsPerNodeVec.size(),"dofsMapsVec.size()!=dofsPerNodeVec.size()");
97 for (UN j=0; j<dofsMapsVec.size(); j++) {
98 FROSCH_ASSERT(dofsMapsVec[j].size()==dofsPerNodeVec[j],"dofsMapsVec[block].size()!=dofsPerNodeVec[block]");
99 for (UN i=0; i<dofsMapsVec.size(); i++) {
100 FROSCH_ASSERT(!dofsMapsVec[j][i].is_null(),"dofsMapsVec[block][i].is_null()");
101 }
102 }
103 }
104
106 // Communicate nodeList //
108 if (!nodeListVec.is_null()) {
109 if (this->MpiComm_->getRank() == 0) std::cout<<"Node List is NOT null\n";
110 FROSCH_DETAILTIMER_START_LEVELID(communicateNodeListTime,"Communicate Node List");
111 for (UN i=0; i<nodeListVec.size(); i++) {
112 if (!nodeListVec[i]->getMap()->isSameAs(*repeatedNodesMapVec[i])) {
113 RCP<MultiVector<SC,LO,GO,NO> > tmpNodeList = MultiVectorFactory<SC,LO,GO,NO>::Build(repeatedNodesMapVec[i],nodeListVec[i]->getNumVectors());
114 RCP<Import<LO,GO,NO> > scatter = ImportFactory<LO,GO,NO>::Build(nodeListVec[i]->getMap(),repeatedNodesMapVec[i]);
115 tmpNodeList->doImport(*nodeListVec[i],*scatter,INSERT);
116 nodeListVec[i] = tmpNodeList.getConst();
117 }
118 }
119 } else {
120 if (this->MpiComm_->getRank() == 0) std::cout<<"Node List is null\n";
121 nodeListVec.resize(nmbBlocks);
122 }
123
125 // Determine dirichletBoundaryDofs //
127 ConstXMapPtr repeatedMap = MergeMaps(repeatedMapVec);
128 if (dirichletBoundaryDofsVec.is_null()) {
129 FROSCH_DETAILTIMER_START_LEVELID(determineDirichletRowsTime,"Determine Dirichlet Rows");
130 dirichletBoundaryDofsVec.resize(repeatedMapVec.size());
131 LOVecPtr counterSub(repeatedMapVec.size(),0);
132 for (UN j=0; j<dirichletBoundaryDofsVec.size(); j++) {
133 dirichletBoundaryDofsVec[j] = GOVecPtr(repeatedMapVec[j]->getLocalNumElements());
134 }
135#ifdef FindOneEntryOnlyRowsGlobal_Matrix
136 GOVecPtr dirichletBoundaryDofs = FindOneEntryOnlyRowsGlobal(this->K_.getConst(),repeatedMap);
137#else
138 GOVecPtr dirichletBoundaryDofs = FindOneEntryOnlyRowsGlobal(this->K_->getCrsGraph(),repeatedMap);
139#endif
140 for (UN i=0; i<dirichletBoundaryDofs.size(); i++) {
141 LO subNumber = -1;
142 for (UN j = dofsMapsVec.size(); j > 0 ; j--) {
143 for (UN k=0; k<dofsMapsVec[j-1].size(); k++) {
144 if ( dirichletBoundaryDofs[i] <= dofsMapsVec[j-1][k]->getMaxAllGlobalIndex() ) {
145 subNumber = j-1;
146 }
147 }
148 }
149 dirichletBoundaryDofsVec[subNumber][counterSub[subNumber]] = dirichletBoundaryDofs[i];
150 counterSub[subNumber]++;
151 }
152
153 //dirichletBoundaryDofsVec = GOVecPtr2D(repeatedMapVec.size());
154 for (UN i=0; i<dirichletBoundaryDofsVec.size(); i++) {
155 dirichletBoundaryDofsVec[i].resize(counterSub[i]);
156 }
157
158 }
159
161 // Initialize OverlappingOperator //
163 if (!this->ParameterList_->get("OverlappingOperator Type","AlgebraicOverlappingOperator").compare("AlgebraicOverlappingOperator")) {
164 AlgebraicOverlappingOperatorPtr algebraicOverlappigOperator = rcp_static_cast<AlgebraicOverlappingOperator<SC,LO,GO,NO> >(this->OverlappingOperator_);
165 if (0>algebraicOverlappigOperator->initialize(overlap,repeatedMap)) ret -= 1;
166 } else {
167 FROSCH_ASSERT(false,"OverlappingOperator Type unkown.");
168 }
169
171 // Initialize CoarseOperator //
173 if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("IPOUHarmonicCoarseOperator")) {
174 // Build Null Space
175 if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Laplace")) {
176 nullSpaceBasisVec.resize(nmbBlocks);
177 for (UN i = 0; i < nmbBlocks; i++) {
178 nullSpaceBasisVec[i] = BuildNullSpace<SC,LO,GO,NO>(dimension,NullSpaceType::Laplace,repeatedMap,dofsPerNodeVec[i],dofsMapsVec[i]);
179 }
180 } else if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Linear Elasticity")) {
181 nullSpaceBasisVec.resize(repeatedMapVec.size());
182 for (int i = 0; i<repeatedMapVec.size(); i++) {
183 nullSpaceBasisVec[i] = BuildNullSpace(dimension,NullSpaceType::Elasticity,repeatedMap,dofsPerNodeVec[i],dofsMapsVec[i],nodeListVec[i]);
184 }
185 } else if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Stokes")) {
186 nullSpaceBasisVec.resize(2);
187 nullSpaceBasisVec[0] = BuildNullSpace<SC,LO,GO,NO>(dimension,NullSpaceType::Laplace,repeatedMap,dofsPerNodeVec[0],dofsMapsVec[0]);
188 nullSpaceBasisVec[1] = BuildNullSpace<SC,LO,GO,NO>(dimension,NullSpaceType::Laplace,repeatedMap,dofsPerNodeVec[1],dofsMapsVec[1]);
189 } else if (!this->ParameterList_->get("Null Space Type","Laplace").compare("Input")) {
190 FROSCH_ASSERT(!nullSpaceBasisVec.is_null(),"Null Space Type is 'Input', but nullSpaceBasis.is_null().");
191 } else {
192 FROSCH_ASSERT(false,"Null Space Type unknown.");
193 }
194
195 IPOUHarmonicCoarseOperatorPtr iPOUHarmonicCoarseOperator = rcp_static_cast<IPOUHarmonicCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
196 if (0>iPOUHarmonicCoarseOperator->initialize(dimension,dofsPerNodeVec,repeatedNodesMapVec,dofsMapsVec,nullSpaceBasisVec,nodeListVec,dirichletBoundaryDofsVec)) ret -=10;
197 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("GDSWCoarseOperator")) {
198 this->ParameterList_->sublist("GDSWCoarseOperator").sublist("CoarseSolver").sublist("MueLu").set("Dimension",(int)dimension);
199 GDSWCoarseOperatorPtr gDSWCoarseOperator = rcp_static_cast<GDSWCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
200 if (0>gDSWCoarseOperator->initialize(dimension,dofsPerNodeVec,repeatedNodesMapVec,dofsMapsVec,dirichletBoundaryDofsVec,nodeListVec)) ret -=10;
201 } else if (!this->ParameterList_->get("CoarseOperator Type","IPOUHarmonicCoarseOperator").compare("RGDSWCoarseOperator")) {
202 this->ParameterList_->sublist("RGDSWCoarseOperator").sublist("CoarseSolver").sublist("MueLu").set("Dimension",(int)dimension);
203 RGDSWCoarseOperatorPtr rGDSWCoarseOperator = rcp_static_cast<RGDSWCoarseOperator<SC,LO,GO,NO> >(CoarseOperator_);
204 if (0>rGDSWCoarseOperator->initialize(dimension,dofsPerNodeVec,repeatedNodesMapVec,dofsMapsVec,dirichletBoundaryDofsVec,nodeListVec)) ret -=10;
205 } else {
206 FROSCH_ASSERT(false,"CoarseOperator Type unkown.");
207 }
208 return ret;
209 }
