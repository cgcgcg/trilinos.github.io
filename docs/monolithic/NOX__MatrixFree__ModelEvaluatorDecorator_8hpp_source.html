50 {

53 // the underlying models.
54 ::Thyra::ModelEvaluatorBase::InArgs<ScalarT> modelInArgs = this->getUnderlyingModel()->createInArgs();
55 modelInArgs.setArgs(inArgs,false);
56
57 // model may not support W_op (hence JFNK)
58 ::Thyra::ModelEvaluatorBase::OutArgs<ScalarT> modelOutArgs = this->getUnderlyingModel()->createOutArgs();
59 modelOutArgs.setArgs(outArgs, true);
60
61 // In some cases is may still support W_op, be sure to disable for JFNK
62 if (modelOutArgs.supports( ::Thyra::ModelEvaluatorBase::OUT_ARG_W_op))
63 modelOutArgs.set_W_op(Teuchos::null);
