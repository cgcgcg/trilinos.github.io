82 int Initialize(int numRows,
84 int *rowBegin, int *columns, SX *values, int numGraphRows = 0, int *graphRowBegin = nullptr,
85 int *graphColumns = nullptr, int *graphWeights = nullptr) {
86 m_numRows = numRows;
87 if (m_numRows == 0)
88 return 0;
89
90 const int numTerms = rowBegin[numRows];
91 size_type_array_host ap_host((size_type *)rowBegin, numRows + 1);
92 ordinal_type_array_host aj_host((ordinal_type *)columns, numTerms);
93
94 size_type_array_host graph_ap_host;
95 ordinal_type_array_host graph_aj_host;
96 ordinal_type_array_host graph_aw_host;
97
98 if (numGraphRows > 0) {
99 if (graphRowBegin == nullptr || graphColumns == nullptr || graphWeights == nullptr) {
100 std::cout << "ExternalInterface::Error, with non-zero numGraphRows, graph pointers should not be nullptr\n";
101 std::logic_error("Error: one of graph pointers is nullptr");
102 } else {
103 const size_type nnz_graph = graph_ap_host(numGraphRows);
104 graph_ap_host = size_type_array_host((size_type *)graphRowBegin, numGraphRows + 1);
105 graph_aj_host = ordinal_type_array_host((ordinal_type *)graphColumns, nnz_graph);
106 graph_aw_host = ordinal_type_array_host((ordinal_type *)graphWeights, nnz_graph);
107 }
108 }
109
110#if defined(KOKKOS_ENABLE_CUDA)
112 value_type_array ax(Kokkos::ViewAllocateWithoutInitializing("ax"), numTerms);
113 value_type_array_host ax_host(values, numTerms);
114 Kokkos::deep_copy(ax, ax_host);
115#else
117 value_type_array ax(values, numTerms);
118#endif
119
120 Kokkos::Timer timer;
121 {
122 timer.reset();
123 if (numGraphRows > 0) {
124 m_Solver.analyze(numRows, ap_host, aj_host, numGraphRows, graph_ap_host, graph_aj_host, graph_aw_host);
125 } else {
126 m_Solver.analyze(numRows, ap_host, aj_host);
127 }
128 const double t = timer.seconds();
129 std::cout << "ExternalInterface:: analyze time " << t << std::endl;
130 }
131
132 {
133 timer.reset();
134 m_Solver.initialize();
135 const double t = timer.seconds();
136 std::cout << "ExternalInterface:: initialize time " << t << std::endl;
137 }
138
140 {
141 timer.reset();
142 m_Solver.factorize(ax);
143 const double t = timer.seconds();
144 std::cout << "ExternalInterface:: factorize time " << t << std::endl;
145 }
146
147 return 0;
148 }

248 void exportUpperTriangularFactorsToCrsMatrix(const int iSolver, std::vector<int> &rowBeginU,
249 std::vector<int> &columnsU, std::vector<SX> &valuesU,
250 std::vector<int> &perm) {
251 solver_type *solver = nullptr;
252 if (iSolver < 0) {
253 solver = &m_Solver;
254 } else {
255 const int solver_array_size(m_SolverArray.size());
256 if (iSolver < solver_array_size) {
257 solver = &m_SolverArray[iSolver];
258 } else {
259 std::cout << "ExternalInterface::Error, non-zero iSolver (" << iSolver
260 << ") is selected where m_SolverArray is sized by (" << solver_array_size << ")\n";
261 std::logic_error("Error: iSolver is out of range in m_SolverArray");
262 }
263 }
264
265 {
266 const auto perm_device = solver->getPermutationVector();
267 auto perm_host = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(), perm_device);
268 {
269 const int n = perm_host.extent(0);
270 perm.resize(n);
271 std::copy(perm_host.data(), perm_host.data() + n, perm.data());
272 }
273 }
274
275 {
276 typename solver_type::crs_matrix_type A;
277 solver->exportFactorsToCrsMatrix(A);
278
279 typename solver_type::crs_matrix_type_host A_host;
280 A_host.createMirror(A);
281 A_host.copy(A);

283 const auto ap = A_host.RowPtr();

308 void MySolve(int NRHS, value_type_matrix &b, value_type_matrix &x) {
309 if (m_numRows == 0)
310 return;
311
312 const int m = m_numRows;
313 if (static_cast<int>(m_TempRhs.extent(0)) < m || static_cast<int>(m_TempRhs.extent(1)) < NRHS)
314 m_TempRhs = value_type_matrix("temp rhs", m, NRHS);
315
316 if (m_SolverArray.empty()) {
318 m_Solver.solve(x, b, m_TempRhs);
319 } else {
321 using range_type = Kokkos::pair<int, int>;
322 const auto range_b = range_type(0, NRHS);
323 for (int i = 0, iend = m_SolverArray.size(); i < iend; ++i) {
324 const auto range_a = range_type(m * i, m * i + m);
325 m_SolverArray[i].solve(Kokkos::subview(x, range_a, range_b), Kokkos::subview(b, range_a, range_b), m_TempRhs);
326 }
327 }
328 }
