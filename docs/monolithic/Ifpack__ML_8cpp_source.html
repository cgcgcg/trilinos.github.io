53//==============================================================================
54int Ifpack_ML::Initialize()
55{
56
57 IsInitialized_ = false;
58 IsComputed_ = false;
59
60 if (Matrix_ == Teuchos::null)
61 IFPACK_CHK_ERR(-1);
62
63 // only square matrices
64 if (Matrix_->NumGlobalRows() != Matrix_->NumGlobalCols())
65 IFPACK_CHK_ERR(-1);
66
67 // at least one nonzero
68 if (Matrix_->NumMyNonzeros() == 0)
69 IFPACK_CHK_ERR(-1);
70
71 if (Time_ == Teuchos::null)
72 Time_ = Teuchos::rcp( new Epetra_Time(Comm()) );
73
74 MLPrec_ = Teuchos::rcp( new ML_Epetra::MultiLevelPreconditioner(Matrix(),
75 List_, false) );
76
77 // if empty, give up.
78 if (MLPrec_ == Teuchos::null)
79 IFPACK_CHK_ERR(-1);
80
81 IsInitialized_ = true;
82 ++NumInitialize_;
83 InitializeTime_ += Time_->ElapsedTime();
84 return(0);
85}

87//==============================================================================
88int Ifpack_ML::Compute()
89{
90
91 if (!IsInitialized())
92 IFPACK_CHK_ERR(Initialize());
93
94 IsComputed_ = false;
95 Time_->ResetStartTime();
96
97 if (Matrix_ == Teuchos::null)
98 IFPACK_CHK_ERR(-1);
99
100 IFPACK_CHK_ERR(MLPrec_->ComputePreconditioner());
101
102 IsComputed_ = true;
103 ++NumCompute_;
104 ComputeTime_ += Time_->ElapsedTime();
105 return(0);
106}

124//==============================================================================
125int Ifpack_ML::
126ApplyInverse(const Epetra_MultiVector& X, Epetra_MultiVector& Y) const
127{
128
129 if (IsComputed() == false)
130 IFPACK_CHK_ERR(-1);
131
132 int numVectors = X.NumVectors();
133 if (numVectors != Y.NumVectors())
134 IFPACK_CHK_ERR(-1); // wrong input
135
136
137 Time_->ResetStartTime();
138
139 // AztecOO gives X and Y pointing to the same memory location,
140 // need to create an auxiliary vector, Xcopy
141 Teuchos::RefCountPtr<const Epetra_MultiVector> Xcopy;
142 if (X.Pointers()[0] == Y.Pointers()[0])
143 Xcopy = Teuchos::rcp( new Epetra_MultiVector(X) );
144 else
145 Xcopy = Teuchos::rcp( &X, false );
146
147 for (int i=0; i<numVectors; i++) {
148
149 IFPACK_CHK_ERR(MLPrec_->ApplyInverse(*Xcopy,Y));
150 }
151
152 ++NumApplyInverse_;
153 ApplyInverseTime_ += Time_->ElapsedTime();
154
155 return(0);
156}

201double Ifpack_ML::Condest(const Ifpack_CondestType CT,
202 const int MaxIters, const double Tol,
203 Epetra_RowMatrix* Matrix)
204{
205
206 if (!IsComputed()) // cannot compute right now
207 return(-1.0);
208
209 if (Condest_ == -1.0)
210 Condest_ = Ifpack_Condest(*this, CT, MaxIters, Tol, Matrix);
211
212 return(Condest_);
213}
