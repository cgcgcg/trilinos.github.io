35 struct ViewOfViewsDeleter {
36 bool do_safety_check_ = false;
37
38 template <class D, class... P>
39 std::enable_if_t<
40 Kokkos::is_view_v<typename Kokkos::View<D, P...>::value_type>>
41 operator()(Kokkos::View<D, P...>* vov) const {
42 Kokkos::fence("PHX:ViewOfViewsDeleter: fence before host cleanup of View of Views");
43 if (do_safety_check_) {
44 if (vov->use_count() > 1) {
45 Kokkos::abort("\n\n********\n ERROR - PHX::ViewOfViews - please free all instances of device Kokkos::View<View<...>> \n before deleting the host ViewOfView!\n********\n\n");
46 }
47 }
48
49 // destroy inner views on host, outside of a parallel region
50 constexpr size_t rank = Kokkos::View<D, P...>::rank();
51 static constexpr bool device_view_is_accessible_from_host = Kokkos::SpaceAccessibility<Kokkos::HostSpace, typename Kokkos::View<D, P...>::memory_space>::accessible;
52 if (device_view_is_accessible_from_host) {
53 if constexpr (rank == 0) {
54 (*vov)() = {};
55 } else if constexpr (rank == 1) {
56 for (size_t i = 0; i < vov->extent(0); ++i) {
57 (*vov)(i) = {};
58 }
59 } else if constexpr (rank == 2) {
60 for (size_t i = 0; i < vov->extent(0); ++i) {
61 for (size_t j = 0; j < vov->extent(1); ++j) {
62 (*vov)(i, j) = {};
63 }
64 }
65 } else if constexpr (rank == 3) {
66 for (size_t i = 0; i < vov->extent(0); ++i) {
67 for (size_t j = 0; j < vov->extent(1); ++j) {
68 for (size_t k = 0; k < vov->extent(2); ++k) {
69 (*vov)(i, j, k) = {};
70 }
71 }
72 }
73 } else {
74 static_assert(std::is_void_v<decltype(vov)>, "\n\n********\n Error - PHX::ViewOfViews - ViewOfView with outer view greater than rank 3 is not supported!\n********\n\n");
75 }
76 }
77 // dispose of the outer view
78 delete vov;
79 }
80 };

126 template<int OuterViewRank,typename InnerViewType,typename... OuterViewProps>
127 class ViewOfViews {
128
129 public:
130 // Layout of the outer view doesn't matter for performance so we
131 // use a raw Kokkos::View instead of PHX::View. The inner views are
132 // what is important for performance.
133 using OuterDataType = typename PHX::v_of_v_utils::add_pointer<InnerViewType,OuterViewRank>::type;
134 using OuterViewType = Kokkos::View<OuterDataType,OuterViewProps...>;
135
136 private:
137 // Inner views are managed - used to prevent early deletion
138 std::shared_ptr<typename OuterViewType::host_mirror_type> view_host_;
139 // Device view
140 std::shared_ptr<OuterViewType> view_device_;
141 // Inner views are unmanaged by runtime construction with pointer
142 // (avoids template parameter). Used to correctly initialize outer
143 // device view on device.
144 static constexpr bool device_view_is_accessible_from_host = Kokkos::SpaceAccessibility<Kokkos::HostSpace, typename OuterViewType::memory_space>::accessible;
145 std::conditional_t<device_view_is_accessible_from_host, std::shared_ptr<OuterViewType>, std::shared_ptr<typename OuterViewType::host_mirror_type>> view_host_unmanaged_;
146 // True if the host view has not been synced to device
147 bool device_view_is_synced_;
148 // True if the outer view has been initialized
149 bool is_initialized_;
150 // A safety check. If true, this makes sure there are no external references to the device view of views.
151 bool check_use_count_;
152
153 public:
155 template<typename... Extents>
156 ViewOfViews(const std::string name,Extents... extents)
157 : view_host_(details::ViewOfViewsMaker<typename OuterViewType::host_mirror_type>::make_shared(name,extents...)),
158 view_device_(details::ViewOfViewsMaker<OuterViewType>::make_shared(name,extents...)),
159 device_view_is_synced_(false),
160 is_initialized_(true),
161 check_use_count_(true)
162 {
163 if constexpr (device_view_is_accessible_from_host) {
164 view_host_unmanaged_ = view_device_;
165 } else {
166 view_host_unmanaged_ = details::ViewOfViewsMaker<typename OuterViewType::host_mirror_type>::make_shared(Kokkos::create_mirror_view(*view_device_));
167 }
168
169 std::get_deleter<details::ViewOfViewsDeleter>(view_device_)->do_safety_check_ = check_use_count_;
170 }

262 template<typename... Indices>
263 void addView(InnerViewType v,Indices... i)
264 {
265 static_assert(sizeof...(Indices)==OuterViewRank,
266 "Error: PHX::ViewOfViews::addView() - the number of indices must match the outer view rank!");
267
268 TEUCHOS_ASSERT(is_initialized_);
269
270 // Store the managed version so inner views don't get deleted.
271 (*view_host_)(i...) = v;
272
273 // Store a runtime unmanaged view for deep_copy to
274 // device. Unmanaged is required to prevent double deletion on
275 // device. For FAD types, we need to adjust the layout to
276 // account for the derivative array. The layout() method reutrns
277 // the non-fad adjusted layout.
278 if (Sacado::IsADType<typename InnerViewType::value_type>::value) {
279 #ifdef KOKKOS_ENABLE_IMPL_VIEW_LEGACY
280 auto layout = v.layout();
281 layout.dimension[InnerViewType::rank] = Kokkos::dimension_scalar(v);
282 (*view_host_unmanaged_)(i...) = InnerViewType(v.data(),layout);
283 #else
284 (*view_host_unmanaged_)(i...) = InnerViewType(v.data(),v.mapping(),v.accessor());
285 #endif
286 }
287 else
288 (*view_host_unmanaged_)(i...) = InnerViewType(v.data(),v.layout());
289
290 device_view_is_synced_ = false;
291 }

301 device_view_is_synced_ = true;
302 }

335 template<typename InnerViewType,typename... OuterViewProps>
336 auto createHostHostViewOfViews(const PHX::ViewOfViews<1,InnerViewType,OuterViewProps...>& vov)
337 {
338 auto outer_host_inner_device_vov = vov.getViewHost();
339 using HostInnerViewType = decltype(Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),outer_host_inner_device_vov(0)));
340 PHX::ViewOfViews<1,HostInnerViewType,OuterViewProps...> host_host_vov("OuterHostInnerHost VOV: "+vov.getViewHost().label(),
341 vov.getViewHost().extent(0));
342 for (size_t i=0; i < outer_host_inner_device_vov.extent(0); ++i) {
343 host_host_vov.addView(Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),outer_host_inner_device_vov(i)),i);
344 }
345 return host_host_vov;
346 }
