244 supernodes_range_policy,
245 KOKKOS_LAMBDA(const ordinal_type &sid, SuperNodeInfoInitReducer::value_type &update) {
246 auto &s = supernodes_(sid);
247
248 s.row_begin = snodes_(sid);
249 s.m = snodes_(sid + 1) - snodes_(sid);

251
252 s.gid_col_begin = gid_ptr_(sid);
253 s.gid_col_end = gid_ptr_(sid + 1);
254 s.sid_col_begin = sid_ptr_(sid);
255 s.sid_col_end = sid_ptr_(sid + 1);
256
257 for (ordinal_type i = s.sid_col_begin; i < s.sid_col_end; ++i) {
258 sid_block_colidx_(i).first = sid_colidx_(i);
259 sid_block_colidx_(i).second = blk_colidx_(i);
260 }
261
262 s.nchildren = stree_ptr_(sid + 1) - stree_ptr_(sid);
263 if (s.nchildren > 0) {
264 s.children = &stree_children_(stree_ptr_(sid));
265 }
266 // const ordinal_type offset = stree_ptr_(sid);
267 // for (ordinal_type i=0;i<s.nchildren;++i)
268 // s.children[i] = stree_children_(offset + i);
269
270 update.max_nchildren = max(update.max_nchildren, s.nchildren);
271 update.max_supernode_size = max(update.max_supernode_size, s.m);
272 update.max_num_cols = max(update.max_num_cols, s.n);
273 update.max_schur_size = max(update.max_schur_size, s.n - s.m);
274
275 update.nnz += (s.m * s.n);
276 update.nnz += (allocate_l_buf_ ? (s.m * s.n) : 0);
277
278 s.max_decendant_supernode_size = s.m;
279 s.max_decendant_schur_size = s.n - s.m;
280 },
281 SuperNodeInfoInitReducer(init_reduce_val));

283 {
284 // finding max decendant supernode information is sequential
285 auto h_supernodes = Kokkos::create_mirror_view(host_memory_space(), supernodes_);
286 auto h_stree_parent = Kokkos::create_mirror_view(host_memory_space(), stree_parent_);
287 Kokkos::deep_copy(h_supernodes, supernodes_);
288 Kokkos::deep_copy(h_stree_parent, stree_parent_);
289 for (ordinal_type sid = 0; sid < nsupernodes; ++sid) {
290 auto &s = h_supernodes(sid);
291 const ordinal_type sidpar = h_stree_parent(sid);
292 if (sidpar != -1) {
293 auto &spar = h_supernodes(sidpar);
294 spar.max_decendant_supernode_size = max(s.max_decendant_supernode_size, spar.max_decendant_supernode_size);
295 spar.max_decendant_schur_size = max(s.max_decendant_schur_size, spar.max_decendant_schur_size);
296 }
297 }
298 Kokkos::deep_copy(supernodes_, h_supernodes);
299 }
300
301 self.max_nchildren = init_reduce_val.max_nchildren;
302 self.max_supernode_size = init_reduce_val.max_supernode_size;
303 self.max_num_cols = init_reduce_val.max_num_cols;
304 self.max_schur_size = init_reduce_val.max_schur_size;

306 // supernodal factor array; data is held outside with a managed view
307 // supernode does not include this view
308 // for the case that the same data structure is reused, the buffer will
309 // be zero'ed for each numeric factorization.
310 superpanel_buf_ = value_type_array(do_not_initialize_tag("superpanel_buf"), init_reduce_val.nnz);
311 Kokkos::parallel_scan(
312 supernodes_range_policy, KOKKOS_LAMBDA(const ordinal_type &sid, size_type &update, const bool &final) {
313 auto &s = supernodes_(sid);
314 const ordinal_type u_buf_size = s.m * s.n;
315 const ordinal_type l_buf_size = allocate_l_buf_ ? (s.m * s.n) : 0;
316 if (final) {
317 s.u_buf = &superpanel_buf_(update);
318 s.l_buf = (allocate_l_buf_ ? s.u_buf + u_buf_size : NULL);
319 }
320 update += (u_buf_size + l_buf_size);
321 });
322
323 self.supernodes = supernodes_; // unmanaged view, data is held outside
324 self.gid_colidx = gid_colidx_;
325 self.sid_block_colidx = sid_block_colidx_;

350 static inline void copySparseToSuperpanels(SupernodeInfo &self,
352 const bool copy_to_l_buf,
354 const size_type_array &ap, const ordinal_type_array &aj,
355 const value_type_array &ax, const ordinal_type_array &perm,
356 const ordinal_type_array &peri) {
357 const ordinal_type nsupernodes = self.supernodes.extent(0);
358 using policy_type = Kokkos::TeamPolicy<exec_space, Kokkos::Schedule<Kokkos::Static>>;
359
360 value_type_array axt;
361 if (copy_to_l_buf) {
362 axt = value_type_array("axt", ax.extent(0));
363 policy_type policy(ap.extent(0) - 1, Kokkos::AUTO());
364 Kokkos::parallel_for(
365 policy, KOKKOS_LAMBDA(const typename policy_type::member_type &member) {
366 const ordinal_type row = member.league_rank();
367 const ordinal_type kbeg = ap(row), kend = ap(row + 1);
368
369 Kokkos::parallel_for(Kokkos::TeamVectorRange(member, kbeg, kend), [&](const ordinal_type &k) {
370 const ordinal_type i = aj(k), j = row;
371 {
372 const ordinal_type lbeg = ap(i), lend = ap(i + 1);
373 ordinal_type *first = aj.data() + lbeg;
374 ordinal_type *last = aj.data() + lend;
375 ordinal_type *loc =
376 lower_bound(first, last, j, [](ordinal_type left, ordinal_type right) { return left < right; });
377 TACHO_TEST_FOR_ABORT(*loc != j, "transpose fail");
378 axt(lbeg + loc - first) = ax(k);
379 }
380 });
381 });
382 }
383
384 {
385 policy_type policy(nsupernodes, Kokkos::AUTO()); // team and vector sizes are AUTO selected.
386 Kokkos::parallel_for(
387 policy, KOKKOS_LAMBDA(const typename policy_type::member_type &member) {
388 const ordinal_type sid = member.league_rank();
389 const auto s = self.supernodes(sid);
391 {
392 UnmanagedViewType<value_type_matrix> tgt_u(s.u_buf, s.m, s.n);
393 UnmanagedViewType<value_type_matrix> tgt_lp(s.l_buf, s.n, s.m);
394 const auto tgt_l = Kokkos::subview(tgt_lp, range_type(s.m, s.n), Kokkos::ALL());
395
396 // row major access to sparse src
397 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, 0, s.m), [&](const ordinal_type &i) {
398 const ordinal_type ii = i + s.row_begin, // row in U
399 row = perm(ii), kbeg = ap(row), kend = ap(row + 1); // row in A
400
401 const ordinal_type jjbeg = (copy_to_l_buf ? s.row_begin : ii);
402 Kokkos::parallel_for(Kokkos::ThreadVectorRange(member, kbeg, kend),
403 [&, jjbeg](const ordinal_type &k) { // Value capture is a workaround for cuda +
404 // gcc-7.2 compiler bug w/c++14
405 const ordinal_type jj = peri(aj(k) /* col in A */); // col in U
406 if (jjbeg <= jj) {
407 ordinal_type *first = self.gid_colidx.data() + s.gid_col_begin;
408 ordinal_type *last = self.gid_colidx.data() + s.gid_col_end;
409 ordinal_type *loc =
410 lower_bound(first, last, jj, [](ordinal_type left, ordinal_type right) {
411 return left < right;
412 });
413 TACHO_TEST_FOR_ABORT(*loc != jj, "copy is wrong");
414 const ordinal_type j = loc - first;
415 tgt_u(i, j) = ax(k);
416 if (j >= s.m && copy_to_l_buf) {
417 tgt_l(j - s.m, i) = axt(k);
418 }
419 }
420 });
421 });
422 }
423 });
424 }
