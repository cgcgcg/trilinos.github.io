199 void LowRankUpdateRowMatrix::apply(const NOX::TMultiVector &X,
200 NOX::TMultiVector &Y,
201 Teuchos::ETransp mode,
202 NOX::Scalar alpha,
203 NOX::Scalar beta) const
204 {
205 TEUCHOS_ASSERT(alpha == Teuchos::ScalarTraits<NOX::Scalar>::one());
206 TEUCHOS_ASSERT(beta == Teuchos::ScalarTraits<NOX::Scalar>::zero());
207
208 // Number of input vectors
209 auto k = X.getNumVectors();
210
211 // Compute J*Input or J^T*input
212 J_rowMatrix->apply(X, Y);
213
214 // Create temporary matrix to store V^T*input or U^T*input
215 if (tmpMat.is_null() || tmpMat->getNumVectors() != k)
216 tmpMat = Teuchos::rcp(new NOX::TMultiVector(local_map, k, false));
217
218 // if (!useTranspose) {
219 if (mode == Teuchos::NO_TRANS) {
220
221 // Compute V^T*Input
222 // tmpMat->Multiply('T', 'N', 1.0, *V, Input, 0.0);
223 tmpMat->multiply(Teuchos::TRANS, Teuchos::NO_TRANS, 1.0, *nonconst_V, X, 0.0);
224
225 // Compute J*Input + U*(V^T*input)
226 // Result.Multiply('N', 'N', 1.0, *U, *tmpMat, 1.0);
227 Y.multiply(Teuchos::NO_TRANS, Teuchos::NO_TRANS, 1.0, *nonconst_U, *tmpMat, 1.0);
228
229 }
230 else {
231
232 // Compute U^T*Input
233 // tmpMat->Multiply('T', 'N', 1.0, *U, Input, 0.0);
234 tmpMat->multiply(Teuchos::TRANS, Teuchos::NO_TRANS, 1.0, *nonconst_U, X, 0.0);
235
236 // Compute J^T*Input + V*(U^T*input)
237 Y.multiply(Teuchos::NO_TRANS, Teuchos::NO_TRANS, 1.0, *nonconst_V, *tmpMat, 1.0);
238
239 }
240 }

242 template <typename ViewType>
243 KOKKOS_INLINE_FUNCTION
244 NOX::Scalar LowRankUpdateRowMatrix::computeUV(int u_row_lid, int v_row_lid) const
245 {
246 NOX::Scalar val = 0.0;
247 auto U_HostView=nonconst_U->getLocalViewHost(::Tpetra::Access::ReadOnly);
248 auto V_HostView=nonconst_V->getLocalViewHost(::Tpetra::Access::ReadOnly);
249
250 // val = sum_{k=0}^m U(i,k)*V(j,k)
251 for (int k=0; k<m; ++k)
252 val += U_HostView(u_row_lid,k) * V_HostView(v_row_lid,k);
253
254 return val;
255 }
