18 template <typename SerializerT,typename InnerViewType,typename... OuterProps>
19 void serialize(SerializerT& s, PHX::ViewOfViews3<2,InnerViewType,OuterProps...>& vov) {
20 std::string label = "";
21 bool is_initialized = false;
22 bool is_synced = false;
23 bool safety_check = true;
24 size_t extent_0 = 0;
25 size_t extent_1 = 0;
26
27 if (s.isSizing() || s.isPacking() || s.isFootprinting()) {
28 auto host_view = vov.getViewHost();
29 label = std::string(host_view.label());
30 is_initialized = vov.isInitialized();
31 is_synced = vov.deviceViewIsSynced();
32 safety_check = vov.safetyCheck();
33 extent_0 = host_view.extent(0);
34 extent_1 = host_view.extent(1);
35 }
36
37 s | label
38 | is_initialized
39 | is_synced
40 | safety_check
41 | extent_0
42 | extent_1;
43
44 if (!safety_check)
45 vov.disableSafetyCheck();
46
47 // If not initialized, then the inner views don't exist and we can
48 // completely ignore inner view data.
49 if (is_initialized) {
50
51 if (s.isSizing() || s.isPacking() || s.isFootprinting()) {
52 auto host_view = vov.getViewHost();
53 for (size_t i=0; i < extent_0; ++i) {
54 for (size_t j=0; j < extent_1; ++j) {
55 InnerViewType tmp = host_view(i,j);
56 TEUCHOS_ASSERT(tmp.extent(0)==10);
57 s | tmp;
58 }
59 }
60 }
61 else if (s.isUnpacking()) {
62 vov.initialize(label,extent_0,extent_1);
63 InnerViewType tmp;
64 for (size_t i=0; i < extent_0; ++i) {
65 for (size_t j=0; j < extent_1; ++j) {
66 s | tmp;
67 vov.addView(tmp,i,j);
68 }
69 }
70 if (is_synced)
71 vov.syncHostToDevice();
72 }
73 }
74
75 }
76}

78#endif
