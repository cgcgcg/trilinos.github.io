119 SACADO_INLINE_FUNCTION
120 VectorDynamicStorage& operator=(const VectorDynamicStorage& x) {
121 if (this != &x) {
122 *val_ = *x.val_;
123 if (sz_ != x.sz_) {
124 sz_ = x.sz_;
125 if (x.sz_ > len_) {
126#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__)
127 if (!owns_mem)
128 throw "Can\'t resize beyond original size when memory isn't owned!";
129#endif
130 if (len_ != 0)
131 ds_array<U>::destroy_and_release(dx_, len_);
132 len_ = x.sz_;
133 dx_ = ds_array<U>::strided_get_and_fill(x.dx_, x.stride_, sz_);
134 }
135 else
136 ds_array<U>::strided_copy(x.dx_, x.stride_, dx_, stride_, sz_);
137 }
138 else
139 ds_array<U>::strided_copy(x.dx_, x.stride_, dx_, stride_, sz_);
140 }
141 return *this;
142 }

180 SACADO_INLINE_FUNCTION
181 void resizeAndZero(int sz) {
182 if (sz > len_) {
183#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__)
184 if (!owns_mem)
185 throw "Can\'t resize beyond original size when memory isn't owned!";

189 len_ = sz;
190 dx_ = ds_array<U>::get_and_fill(len_);

202 SACADO_INLINE_FUNCTION
203 void expand(int sz) {
204 if (sz > len_) {
205#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__)
206 if (!owns_mem)
207 throw "Can\'t resize beyond original size when memory isn't owned!";
208#endif
209 U* dx_new = ds_array<U>::get_and_fill(sz);
210 ds_array<U>::copy(dx_, dx_new, sz_);
211 if (len_ > 0)
212 ds_array<U>::destroy_and_release(dx_, len_);
213 dx_ = dx_new;
214 len_ = sz;
215 }
216 else if (sz > sz_)
217 ds_array<U>::strided_zero(dx_+stride_*sz_, stride_, sz-sz_);
218 sz_ = sz;
219 }
