26 template<class HandleType, class PViewType>
27 inline
28 void exchange_pivots(HandleType& ahandle, PViewType& lpiv_view, PViewType& permute) {
29
30 using pival_type = typename PViewType::value_type;
31
32 MPI_Comm comm = ahandle.get_comm();
33 MPI_Comm row_comm = ahandle.get_row_comm();
34 int me = ahandle.get_myrank();
35 int myrow = ahandle.get_myrow();
36 int mycol = ahandle.get_mycol();
37 int nprocs_row = ahandle.get_nprocs_row();
38 int nprocs_col = ahandle.get_nprocs_col();
39 int nrows_matrix = ahandle.get_nrows_matrix();
40 int my_rows = ahandle.get_my_rows();
41
42 MPI_Status msgstatus;
43 int rank_row,k_row,pivot_col;
44
45 // First gather the permutation vector to processor 0 in row_comm
46 if (myrow == 0 || mycol == 0) {
47 for (int k=0;k<=nrows_matrix-1;k++) {
48 pivot_col = k%nprocs_row;
49 k_row = k%nprocs_col;
50 rank_row = k_row*nprocs_row;
51 if (rank_row == pivot_col) {//on the same rank
52 if (me == rank_row) {//I am the right process to do
53 int j=k/nprocs_row;
54 int i=k/nprocs_col;
55 permute(i) = lpiv_view(j);
56 }
57 }
58 else {//on different ranks
59 if (me == pivot_col) {
60 int j=k/nprocs_row;
61 MPI_Send(reinterpret_cast<char*>(lpiv_view.data()+j),sizeof(pival_type),MPI_CHAR,rank_row,0,comm);
62 }
63 if (me == rank_row) {
64 int i=k/nprocs_col;
65 MPI_Recv(reinterpret_cast<char*>(permute.data()+i),sizeof(pival_type),MPI_CHAR,pivot_col,0,comm,&msgstatus);
66 }
67 }
68 }
69 }
70 MPI_Barrier(comm);
71 // Broadcast to the rest of the processors in row_comm
72 MPI_Bcast(reinterpret_cast<char*>(permute.data()),my_rows*sizeof(pival_type),MPI_CHAR,0,row_comm);
73
74 }// End of function exchange_pivots

76 template<class HandleType, class ZViewType, class PViewType>
77 inline
78 void permute_mat(HandleType& ahandle, ZViewType& Z, PViewType& lpiv_view, PViewType& permute) {
79 using value_type = typename ZViewType::value_type;
80#ifndef ADELUS_PERM_MAT_FORWARD_COPY_TO_HOST
81 using execution_space = typename ZViewType::device_type::execution_space ;
82 using memory_space = typename ZViewType::device_type::memory_space ;
83 using ViewVectorType = Kokkos::View<value_type*, Kokkos::LayoutLeft, memory_space>;
84#ifdef ADELUS_HOST_PINNED_MEM_MPI
85 #if defined(KOKKOS_ENABLE_CUDA)
86 using ViewVectorHostPinnType = Kokkos::View<value_type*, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace>;//CudaHostPinnedSpace
87 #elif defined(KOKKOS_ENABLE_HIP)
88 using ViewVectorHostPinnType = Kokkos::View<value_type*, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace>;//HIPHostPinnedSpace
89 #endif
90#endif
91 MPI_Comm col_comm = ahandle.get_col_comm();
92 int myrow = ahandle.get_myrow();
93 int mycol = ahandle.get_mycol();
94 int nprocs_row = ahandle.get_nprocs_row();
95 int nprocs_col = ahandle.get_nprocs_col();
96 int nrows_matrix = ahandle.get_nrows_matrix();
97
98#ifdef PRINT_STATUS
99 printf("Rank %i -- permute_mat() Begin permute mat with myrow %d, mycol %d, nprocs_row %d, nprocs_col %d, nrows_matrix %d, ncols_matrix %d, my_rows %d, my_cols %d, my_rhs %d, nrhs %d, value_type %s, execution_space %s, memory_space %s\n", ahandle.get_myrank(), myrow, mycol, nprocs_row, nprocs_col, nrows_matrix, ahandle.get_ncols_matrix(), ahandle.get_my_rows(), ahandle.get_my_cols(), ahandle.get_my_rhs(), ahandle.get_nrhs(), typeid(value_type).name(), typeid(execution_space).name(), typeid(memory_space).name());
100#endif
101#endif
102
103 MPI_Status msgstatus;
104
105 int pivot_row, k_row;
106#ifdef ADELUS_PERM_MAT_FORWARD_COPY_TO_HOST
107 value_type tmpr, tmps;
108#else
109 ViewVectorType tmpr( "tmpr", Z.extent(1) );
110 ViewVectorType tmps( "tmps", Z.extent(1) );
111#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
112 ViewVectorHostPinnType h_tmpr( "h_tmpr", Z.extent(1) );
113 ViewVectorHostPinnType h_tmps( "h_tmps", Z.extent(1) );
114#endif
115#endif
116
117#ifdef GET_TIMING
118 double exchpivtime,permutemattime,t1;
119
120 t1 = MPI_Wtime();
121#endif
122
123 exchange_pivots(ahandle, lpiv_view, permute);
124
125#ifdef GET_TIMING
126 exchpivtime = MPI_Wtime()-t1;
127
128 t1 = MPI_Wtime();
129#endif
130
131#ifdef ADELUS_PERM_MAT_FORWARD_COPY_TO_HOST
132 for (int j=0;j<=ahandle.get_my_cols()-1;j++) {
133 int J=j*nprocs_row+mycol; // global column index
134 for (int k=J+1;k<=nrows_matrix-1;k++) {
135 k_row=k%nprocs_col;
136 if (myrow==k_row)
137 pivot_row = static_cast<int>(permute(k/nprocs_col));
138 MPI_Bcast(&pivot_row,1,MPI_INT,k_row,col_comm);
139 if (k != pivot_row) {
140 if (myrow == k_row) {
141 tmps = Z(k/nprocs_col, J/nprocs_row);
142 MPI_Send((char *)(&tmps),sizeof(value_type),MPI_CHAR,pivot_row%nprocs_col,2,col_comm);
143 }
144 if (myrow == pivot_row%nprocs_col) {
145 tmps = Z(pivot_row/nprocs_col, J/nprocs_row);
146 MPI_Send((char *)(&tmps),sizeof(value_type),MPI_CHAR,k_row,3,col_comm);
147 }
148 if (myrow == k_row) {
149 MPI_Recv((char *)(&tmpr),sizeof(value_type),MPI_CHAR,pivot_row%nprocs_col,3,col_comm,&msgstatus);
150 Z(k/nprocs_col, J/nprocs_row) = tmpr;
151 }
152 if (myrow == pivot_row%nprocs_col) {
153 MPI_Recv((char *)(&tmpr),sizeof(value_type),MPI_CHAR,k_row,2,col_comm,&msgstatus);
154 Z(pivot_row/nprocs_col, J/nprocs_row) = tmpr;
155 }
156 }// End of if (k != pivot_row)
157 }// End of for (k=J+1;k<=nrows_matrix-1;k++)
158 }// End of for (j=0;j<=my_cols-1;j++)
159#else
160 for (int k = 1 + mycol; k <= nrows_matrix - 1; k++) {
161 int max_gcol_k=k-1; // max. global column index in the k row
162 int max_lcol_k=0; // max. local column index in the k row
163 k_row=k%nprocs_col; // mesh row id (in the MPI process mesh) of the process that holds k
164
165 if (myrow==k_row) pivot_row = static_cast<int>(permute(k/nprocs_col));
166 MPI_Bcast(&pivot_row,1,MPI_INT,k_row,col_comm);
167
168 int max_gcol_pivot=pivot_row-1; // max. global column index in the pivot row
169 int max_lcol_pivot=0; // max. local column index in the pivot row
170 int pivot_row_pid = pivot_row%nprocs_col;// mesh row id (in the MPI process mesh) of the process that holds pivot_row
171
172 //Find max. local column index in the k row that covers the lower triangular part
173 if ( mycol <= max_gcol_k%nprocs_row)
174 max_lcol_k = max_gcol_k/nprocs_row;
175 else
176 max_lcol_k = max_gcol_k/nprocs_row - 1;//one element less
177
178 //Find max. local column index in the pivot row that covers the lower triangular part
179 if ( mycol <= max_gcol_pivot%nprocs_row)
180 max_lcol_pivot = max_gcol_pivot/nprocs_row;
181 else
182 max_lcol_pivot = max_gcol_pivot/nprocs_row - 1;//one element less
183
184 //Find the number of columns needs to be exchanged
185 int min_len = std::min(max_lcol_k,max_lcol_pivot) + 1;
186
187 if (k != pivot_row) {//k row is differrent from pivot_row, i.e. needs permutation
188 if (k_row == pivot_row_pid) {//pivot row is in the same rank
189 if (myrow == k_row) {//I am the right process to do permutation
190 int curr_lrid = k/nprocs_col;
191 int piv_lrid = pivot_row/nprocs_col;
192 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,min_len), KOKKOS_LAMBDA (const int i) {
193 value_type tmp = Z(curr_lrid,i);
194 Z(curr_lrid,i) = Z(piv_lrid,i);
195 Z(piv_lrid,i) = tmp;
196 });
197 Kokkos::fence();
198 }
199 }
200 else {//k row and pivot row are in different processes (rank)
201 if (myrow == k_row) {//I am holding k row
202 int curr_lrid = k/nprocs_col;
203 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,min_len), KOKKOS_LAMBDA (const int i) {
204 tmps(i) = Z(curr_lrid,i);
205 });
206
207#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
208 Kokkos::deep_copy(h_tmps,tmps);
209 MPI_Send(reinterpret_cast<char *>(h_tmps.data()),min_len*sizeof(value_type),MPI_CHAR,pivot_row_pid,2,col_comm);
210 MPI_Recv(reinterpret_cast<char *>(h_tmpr.data()),min_len*sizeof(value_type),MPI_CHAR,pivot_row_pid,3,col_comm,&msgstatus);
211 Kokkos::deep_copy(tmpr,h_tmpr);
212#else //GPU-aware MPI
213 Kokkos::fence();
214
215 MPI_Send(reinterpret_cast<char *>(tmps.data()),min_len*sizeof(value_type),MPI_CHAR,pivot_row_pid,2,col_comm);
216 MPI_Recv(reinterpret_cast<char *>(tmpr.data()),min_len*sizeof(value_type),MPI_CHAR,pivot_row_pid,3,col_comm,&msgstatus);
217#endif
218
219 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,min_len), KOKKOS_LAMBDA (const int i) {
220 Z(curr_lrid,i) = tmpr(i);
221 });
222 Kokkos::fence();
223 }
224 if (myrow == pivot_row_pid) {//I am holding the pivot row
225 int piv_lrid = pivot_row/nprocs_col;
226 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,min_len), KOKKOS_LAMBDA (const int i) {
227 tmps(i) = Z(piv_lrid,i);
228 });
229
230#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
231 Kokkos::deep_copy(h_tmps,tmps);
232 MPI_Recv(reinterpret_cast<char *>(h_tmpr.data()),min_len*sizeof(value_type),MPI_CHAR,k_row,2,col_comm,&msgstatus);
233 MPI_Send(reinterpret_cast<char *>(h_tmps.data()),min_len*sizeof(value_type),MPI_CHAR,k_row,3,col_comm);
234 Kokkos::deep_copy(tmpr,h_tmpr);
235#else // GPU-aware MPI
236 Kokkos::fence();
237
238 MPI_Recv(reinterpret_cast<char *>(tmpr.data()),min_len*sizeof(value_type),MPI_CHAR,k_row,2,col_comm,&msgstatus);
239 MPI_Send(reinterpret_cast<char *>(tmps.data()),min_len*sizeof(value_type),MPI_CHAR,k_row,3,col_comm);
240#endif
241
242 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,min_len), KOKKOS_LAMBDA (const int i) {
243 Z(piv_lrid,i) = tmpr(i);
244 });
245 Kokkos::fence();
246 }
247 }//End of k row and pivot row are in different processes (rank)
248 }// End of if (k != pivot_row)
249 }// End of for (int k=1+mycol;k<=nrows_matrix-1;k++) {
250#endif
251
252#ifdef GET_TIMING
253 permutemattime = MPI_Wtime()-t1;
254
255 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
256 "Time to exchange pivot information", &exchpivtime);
257 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
258 "Time to permute matrix", &permutemattime);
259#endif
260 }// End of function permute_mat
