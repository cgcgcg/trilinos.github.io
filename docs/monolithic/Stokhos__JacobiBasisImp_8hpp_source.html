47template <typename ordinal_type, typename value_type>
48bool
49Stokhos::JacobiBasis<ordinal_type, value_type>::
50computeRecurrenceCoefficients(ordinal_type n,
51 Teuchos::Array<value_type>& aalpha,
52 Teuchos::Array<value_type>& abeta,
53 Teuchos::Array<value_type>& adelta,
54 Teuchos::Array<value_type>& agamma) const
55{
56 value_type a = alphaIndex_;
57 value_type b = betaIndex_;
58
59 if (a==0.0 && b==0.0)
60 {
61 aalpha[0] = 0.0;
62 abeta[0] = 1.0;
63 adelta[0] = 1.0;
64 agamma[0] = 1.0;
65 }
66 else
67 {
68 aalpha[0] = getB(0)/getA(0);
69 abeta[0] = 1.0;
70 adelta[0] = getC(0)/getA(0);
71 agamma[0] = 1.0;
72 }
73 for (ordinal_type i=1; i<n; i++)
74 {
75 aalpha[i] = getB(i)/getA(i);
76 abeta[i] = getD(i)/getA(i);
77 adelta[i] = getC(i)/getA(i);
78 agamma[i] = 1.0;
79 }
80
81 return false;
82}

85template <typename ordinal_type, typename value_type>
86value_type Stokhos::JacobiBasis<ordinal_type, value_type>::getA(int n) const
87{
88 return 2*(n+1)*(n+alphaIndex_+betaIndex_+1)*(2*n+alphaIndex_+betaIndex_);
89}

91template <typename ordinal_type, typename value_type>
92value_type Stokhos::JacobiBasis<ordinal_type, value_type>::getB(int n) const
93{
94 value_type a = alphaIndex_;
95 value_type b = betaIndex_;
96 return -(2*n+a+b+1)*(a*a-b*b);
97}

99template <typename ordinal_type, typename value_type>
100value_type Stokhos::JacobiBasis<ordinal_type, value_type>::getC(int n) const
101{
102 value_type a = alphaIndex_;
103 value_type b = betaIndex_;

105}

107template <typename ordinal_type, typename value_type>
108value_type Stokhos::JacobiBasis<ordinal_type, value_type>::getD(int n) const
109{
110 value_type a = alphaIndex_;
111 value_type b = betaIndex_;
112 return 2*(n+a)*(n+b)*(2*n + a + b + 2);
