42 template <typename MemberType, typename SupernodeInfoType>
43 KOKKOS_INLINE_FUNCTION static int
44 factorize(MemberType &member, const SupernodeInfoType &info, const typename SupernodeInfoType::ordinal_type_array &P,
45 const typename SupernodeInfoType::value_type_matrix &D,
46 const typename SupernodeInfoType::value_type_array &W,
47 const typename SupernodeInfoType::value_type_matrix &ABR, const ordinal_type sid) {
48 using supernode_info_type = SupernodeInfoType;
49 using value_type = typename supernode_info_type::value_type;
50 using value_type_matrix = typename supernode_info_type::value_type_matrix;
51 using ordinal_type_array = typename supernode_info_type::ordinal_type_array;
52
53 // algorithm choice
54 using LDL_AlgoType = typename LDL_Algorithm::type;
55 using TrsmAlgoType = typename TrsmAlgorithm::type;
56 using GemmAlgoType = typename GemmAlgorithm::type;
57
58 // get current supernode
59 const auto &s = info.supernodes(sid);
60
61 // get panel pointer
62 value_type *ptr = s.u_buf;
63
64 // panel (s.m x s.n) is divided into ATL (m x m) and ATR (m x n)
65 const ordinal_type m = s.m, n = s.n - s.m;
66
67 // m and n are available, then factorize the supernode block
68 if (m > 0) {
70 UnmanagedViewType<value_type_matrix> ATL(ptr, m, m);
71 ptr += m * m;
72
73 Symmetrize<Uplo::Upper, Algo::Internal>::invoke(member, ATL);
74
75 LDL<Uplo::Lower, LDL_AlgoType>::invoke(member, ATL, P, W);
76 LDL<Uplo::Lower, LDL_AlgoType>::modify(member, ATL, P, D);
77
78 if (n > 0) {
79 const value_type one(1), zero(0);
80 UnmanagedViewType<value_type_matrix> ATR(ptr, m, n);
81 ptr += m * n;
82 UnmanagedViewType<value_type_matrix> STR(W.data(), m, n);
83
84 auto fpiv = ordinal_type_array(P.data() + m, m);
85 ApplyPivots<PivotMode::Flame, Side::Left, Direct::Forward, Algo::Internal>
86 ::invoke(member, fpiv, ATR);
87 Trsm<Side::Left, Uplo::Lower, Trans::NoTranspose, TrsmAlgoType>::invoke(member, Diag::Unit(), one, ATL, ATR);
88 Copy<Algo::Internal>::invoke(member, STR, ATR);
89 Scale2x2_BlockInverseDiagonals<Side::Left, Algo::Internal>
90 ::invoke(member, P, D, ATR);
91
92 TACHO_TEST_FOR_ABORT(static_cast<ordinal_type>(ABR.extent(0)) != n ||
93 static_cast<ordinal_type>(ABR.extent(1)) != n,
94 "ABR dimension does not match to supernodes");
95 GemmTriangular<Trans::Transpose, Trans::NoTranspose, Uplo::Upper, GemmAlgoType>::invoke(member, -one, ATR, STR,
96 zero, ABR);
97 }
98 }
99 return 0;
100 }

102 template <typename MemberType, typename SupernodeInfoType>
103 KOKKOS_INLINE_FUNCTION static int solve_lower(MemberType &member, const SupernodeInfoType &info,
104 const typename SupernodeInfoType::ordinal_type_array &P,
105 const typename SupernodeInfoType::value_type_matrix &xB,
106 const ordinal_type sid) {
107 using supernode_info_type = SupernodeInfoType;
108 using value_type = typename supernode_info_type::value_type;
109 using value_type_matrix = typename supernode_info_type::value_type_matrix;
110 using ordinal_type_array = typename supernode_info_type::ordinal_type_array;
111
112 using range_type = Kokkos::pair<ordinal_type, ordinal_type>;
113
114 const auto &s = info.supernodes(sid);
115
116 using TrsvAlgoType = typename TrsvAlgorithm::type;
117 using GemvAlgoType = typename GemvAlgorithm::type;
118
119 // get panel pointer
120 value_type *ptr = s.u_buf;
121
122 // panel is divided into diagonal and interface block
123 const ordinal_type m = s.m, n = s.n - s.m; //, nrhs = info.x.extent(1);
124
125 // m and n are available, then factorize the supernode block
126 if (m > 0) {
127 const value_type one(1), zero(0);
128 const ordinal_type offm = s.row_begin;
129 UnmanagedViewType<value_type_matrix> AL(ptr, m, m);
130 ptr += m * m;
131 const auto xT = Kokkos::subview(info.x, range_type(offm, offm + m), Kokkos::ALL());
132 const auto fpiv = ordinal_type_array(P.data() + m, m);
133
134 ApplyPivots<PivotMode::Flame, Side::Left, Direct::Forward, Algo::Internal>
135 ::invoke(member, fpiv, xT);
136
137 Trsv<Uplo::Lower, Trans::NoTranspose, TrsvAlgoType>::invoke(member, Diag::Unit(), AL, xT);
138
139 if (n > 0) {
140 UnmanagedViewType<value_type_matrix> AR(ptr, m, n); // ptr += m*n;
141 Gemv<Trans::Transpose, GemvAlgoType>::invoke(member, -one, AR, xT, zero, xB);
142 }
143 }
144 return 0;
145 }

197 template <typename MemberType, typename SupernodeInfoType>
198 KOKKOS_INLINE_FUNCTION static int
199 factorize_recursive_serial(MemberType &member, const SupernodeInfoType &info, const ordinal_type sid,
200 const bool final, typename SupernodeInfoType::ordinal_type_array::pointer_type piv,
201 typename SupernodeInfoType::value_type_array::pointer_type diag,
202 typename SupernodeInfoType::value_type_array::pointer_type buf, const size_type bufsize) {
203 using supernode_info_type = SupernodeInfoType;
204 using value_type = typename supernode_info_type::value_type;
205 using value_type_array = typename supernode_info_type::value_type_array;
206 using value_type_matrix = typename supernode_info_type::value_type_matrix;
207 using ordinal_type_array = typename supernode_info_type::ordinal_type_array;
208
209 const auto &s = info.supernodes(sid);
210 if (final) {
211 // serial recursion
212 for (ordinal_type i = 0; i < s.nchildren; ++i)
213 factorize_recursive_serial(member, info, s.children[i], final, piv, diag, buf, bufsize);
214 }
215
216 {
217 const ordinal_type m = s.m;
218 const ordinal_type rbeg = s.row_begin;
219 UnmanagedViewType<ordinal_type_array> ipiv(piv + rbeg * 4, 4 * m);
220 UnmanagedViewType<value_type_matrix> dblk(diag + rbeg * 2, m, 2);
221
222 const ordinal_type n = s.n - s.m;
223
224 const ordinal_type mm = m < 32 ? m : 32;
225 const ordinal_type mn = mm > n ? mm : n;
226
227 const size_type bufsize_required = (n * n + m * mn) * sizeof(value_type);
228 TACHO_TEST_FOR_ABORT(bufsize < bufsize_required, "bufsize is smaller than required");
229 value_type *bufptr = buf;
230 UnmanagedViewType<value_type_matrix> ABR(bufptr, n, n);
231 bufptr += ABR.span();
232 UnmanagedViewType<value_type_array> w(bufptr, m * mn);
233 bufptr += w.span();
234
235 LDL_Supernodes<Algo::Workflow::Serial>::factorize(member, info, ipiv, dblk, w, ABR, sid);
236
238 CholSupernodes<Algo::Workflow::Serial>::update(member, info, ABR, sid, bufsize - ABR.span() * sizeof(value_type),
239 (void *)(w.data()));
240 }
241 return 0;
242 }
