41 template <class SC,class LO,class GO,class NO>
42 int RGDSWPreconditioner<SC,LO,GO,NO>::initialize(ConstXMapPtr repeatedMap,
43 bool useDefaultParameters)
44 {
45 if (useDefaultParameters) {
46 return initialize(3,1,repeatedMap);
47 } else {
48 DofOrdering dofOrdering = NodeWise;
49 if (!this->ParameterList_->get("DofOrdering","NodeWise").compare("NodeWise")) {
50 dofOrdering = NodeWise;
51 } else if (!this->ParameterList_->get("DofOrdering","NodeWise").compare("DimensionWise")) {
52 dofOrdering = DimensionWise;
53 } else if (!this->ParameterList_->get("DofOrdering","NodeWise").compare("Custom")) {
54 dofOrdering = Custom;
55 } else {
56 FROSCH_ASSERT(false,"ERROR: Specify a valid DofOrdering.");
57 }
58
59 return initialize(this->ParameterList_->get("Dimension",1),this->ParameterList_->get("DofsPerNode",1),dofOrdering,this->ParameterList_->get("Overlap",1),repeatedMap);
60 }
61 }
