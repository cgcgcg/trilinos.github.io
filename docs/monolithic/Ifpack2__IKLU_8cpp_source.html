104int Ifpack2_IKLU::Initialize() {
105 // delete previously allocated factorization
106 Destroy();
107
108 Time_.ResetStartTime();
109
110 if (A_.Comm().NumProc() != 1) {
111 cout << " There are too many processors !!! " << endl;
112 cerr << "Ifpack2_IKLU can be used with Comm().NumProc() == 1" << endl;
113 cerr << "only. This class is a subdomain solver for Ifpack2_AdditiveSchwarz," << endl;
114 cerr << "and it is currently not meant to be used otherwise." << endl;
115 exit(EXIT_FAILURE);
116 }
117
118 // check dimensions of input matrix only in serial
119 if (Comm().NumProc() == 1 && Matrix().NumMyRows() != Matrix().NumMyCols())
120 IFPACK2_CHK_ERR(-2);
121
122 NumMyRows_ = Matrix().NumMyRows();
123 NumMyNonzeros_ = Matrix().NumMyNonzeros();
124
125 int RowNnz, Length = Matrix().MaxNumEntries();
126 vector<int> RowIndices(Length);
127 vector<double> RowValues(Length);
128
129 // cout << "Processor " << Comm().MyPID() << " owns " << NumMyRows_ << " rows and has " << NumMyNonzeros_ << " nonzeros " << endl;
130 // get general symbolic structure of the matrix
131 csrA_ = csr_spalloc(NumMyRows_, NumMyRows_, NumMyNonzeros_, 1, 0);
132
133 // copy the symbolic structure into csrA_
134 int count = 0;
135 csrA_->p[0] = 0;
136 for (int i = 0; i < NumMyRows_; ++i) {
137 IFPACK2_CHK_ERR(A_.ExtractMyRowCopy(i, Length, RowNnz,
138 &RowValues[0], &RowIndices[0]));
139 for (int j = 0; j < RowNnz; ++j) {
140 csrA_->j[count++] = RowIndices[j];
141 // cout << "Row = " << i << ", Column = " << RowIndices[j] << ", Value = " << RowValues[j] << endl;
142 }
143 csrA_->p[i + 1] = csrA_->p[i] + RowNnz;
144 }
145
146 // Perform symbolic analysis on the current matrix structure
147 int order = 1;
148 cssS_ = csr_sqr(order, csrA_);
149 for (int i = 0; i < NumMyRows_; ++i) {
150 cout << "AMD Perm (from inside KLU) [" << i << "] = " << cssS_->q[i] << endl;
151 }
152
153 // nothing else to do here
154 IsInitialized_ = true;
155 ++NumInitialize_;
156 InitializeTime_ += Time_.ElapsedTime();
157
158 return (0);
159}

171int Ifpack2_IKLU::Compute() {
172 if (!IsInitialized())
173 IFPACK2_CHK_ERR(Initialize());
174
175 Time_.ResetStartTime();
176 IsComputed_ = false;
177
178 NumMyRows_ = A_.NumMyRows();
179 int Length = A_.MaxNumEntries();
180
181 bool distributed = (Comm().NumProc() > 1) ? true : false;
182 if (distributed) {
183 SerialComm_ = rcp(new Tpetra_SerialComm);
184 SerialMap_ = rcp(new Tpetra_Map(NumMyRows_, 0, *SerialComm_));
185 assert(SerialComm_.get() != 0);
186 assert(SerialMap_.get() != 0);
187 } else
188 SerialMap_ = rcp(const_cast<Tpetra_Map*>(&A_.RowMatrixRowMap()), false);
189
190 int RowNnz;
191 vector<int> RowIndices(Length);
192 vector<double> RowValues(Length);
193
194 // copy the values from A_ into csrA_
195 int count = 0;
196 for (int i = 0; i < NumMyRows_; ++i) {
197 IFPACK2_CHK_ERR(A_.ExtractMyRowCopy(i, Length, RowNnz,
198 &RowValues[0], &RowIndices[0]));
199 // make sure each row has the same number of nonzeros
200 if (RowNnz != (csrA_->p[i + 1] - csrA_->p[i])) {
201 cout << "The number of nonzeros for this row does not math the expected number of nonzeros!!!" << endl;
202 }
203 for (int j = 0; j < RowNnz; ++j) {
204 csrA_->x[count++] = RowValues[j];
205 // cout << "Row = " << i << ", Column = " << RowIndices[j] << ", Value = " << RowValues[j] << endl;
206 }
207 }
208
209 // compute the lu factors
210 double tol = 0.1;
211 csrnN_ = csr_lu(&*csrA_, &*cssS_, tol);
212
213 // Create L and U as a view of the information stored in csrnN_->L and csrnN_->U
214 csr* L_tmp = csrnN_->L;
215 csr* U_tmp = csrnN_->U;
216 vector<int> numEntriesL(NumMyRows_), numEntriesU(NumMyRows_);
217 for (int i = 0; i < NumMyRows_; ++i) {
218 numEntriesL[i] = (L_tmp->p[i + 1] - L_tmp->p[i]);
219 numEntriesU[i] = (U_tmp->p[i + 1] - U_tmp->p[i]);
220 }
221 L_ = rcp(new Tpetra_CrsMatrix(View, *SerialMap_, &numEntriesL[0]));
222 U_ = rcp(new Tpetra_CrsMatrix(View, *SerialMap_, &numEntriesU[0]));
223
224 // Insert the values into L and U
225 for (int i = 0; i < NumMyRows_; ++i) {
226 L_->InsertGlobalValues(i, numEntriesL[i], &(L_tmp->x[L_tmp->p[i]]), &(L_tmp->j[L_tmp->p[i]]));
227 U_->InsertGlobalValues(i, numEntriesU[i], &(U_tmp->x[U_tmp->p[i]]), &(U_tmp->j[U_tmp->p[i]]));
228 }
229
230 IFPACK2_CHK_ERR(L_->FillComplete());
231 IFPACK2_CHK_ERR(U_->FillComplete());
232
233 int MyNonzeros = L_->NumGlobalNonzeros() + U_->NumGlobalNonzeros();
234 Comm().SumAll(&MyNonzeros, &GlobalNonzeros_, 1);
235
236 IsComputed_ = true;
237
238 ++NumCompute_;
239 ComputeTime_ += Time_.ElapsedTime();
240
241 return (0);
242}

245int Ifpack2_IKLU::ApplyInverse(const Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& X,
246 Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& Y) const {
247 if (!IsComputed())
248 IFPACK2_CHK_ERR(-2); // compute preconditioner first
249
250 if (X.NumVectors() != Y.NumVectors())
251 IFPACK2_CHK_ERR(-3); // Return error: X and Y not the same size
252
253 Time_.ResetStartTime();
254
255 // NOTE: L_ and U_ are based on SerialMap_, while Xcopy is based
256 // on A.Map()... which are in general different. However, Solve()
257 // does not seem to care... which is fine with me.
258 //
259 // AztecOO gives X and Y pointing to the same memory location,
260 // need to create an auxiliary vector, Xcopy and apply permutation.
261 vector<int> invq(NumMyRows_);
262
263 for (int i = 0; i < NumMyRows_; ++i) {
264 csrnN_->perm[csrnN_->pinv[i]] = i;
265 invq[cssS_->q[i]] = i;
266 }
267
268 Teuchos::RCP<Tpetra_MultiVector> Xcopy = Teuchos::rcp(new Tpetra_MultiVector(X.Map(), X.NumVectors()), false);
269 Teuchos::RCP<Tpetra_MultiVector> Ytemp = Teuchos::rcp(new Tpetra_MultiVector(Y.Map(), Y.NumVectors()));
270
271 for (int i = 0; i < NumMyRows_; ++i) {
272 for (int j = 0; j < X.NumVectors(); ++j) {
273 Xcopy->ReplaceMyValue(invq[i], j, (*X(j))[i]);
274 }
275 }
276
277 if (!UseTranspose_) {
278 // solves LU Y = X
279 IFPACK2_CHK_ERR(L_->Solve(false, false, false, *Xcopy, *Ytemp));
280 IFPACK2_CHK_ERR(U_->Solve(true, false, false, *Ytemp, *Ytemp));
281 } else {
282 // solves U(trans) L(trans) Y = X
283 IFPACK2_CHK_ERR(U_->Solve(true, true, false, *Xcopy, *Ytemp));
284 IFPACK2_CHK_ERR(L_->Solve(false, true, false, *Ytemp, *Ytemp));
285 }
286
287 // Reverse the permutation.
288 for (int i = 0; i < NumMyRows_; ++i) {
289 for (int j = 0; j < Y.NumVectors(); ++j) {
290 Y.ReplaceMyValue(csrnN_->perm[i], j, (*(*Ytemp)(j))[i]);
291 }
292 }
293
294 ++NumApplyInverse_;
295 ApplyInverseFlops_ += X.NumVectors() * 2 * GlobalNonzeros_;
296 ApplyInverseTime_ += Time_.ElapsedTime();
297
298 return (0);
299}

309 const int MaxIters, const double Tol,
