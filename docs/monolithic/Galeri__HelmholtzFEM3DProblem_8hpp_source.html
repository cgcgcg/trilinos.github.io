26template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
27class HelmholtzFEM3DProblem : public Problem_Helmholtz<Map, Matrix, MultiVector> {
28 public:
29 HelmholtzFEM3DProblem(Teuchos::ParameterList& list, const Teuchos::RCP<const Map>& map)
30 : Problem_Helmholtz<Map, Matrix, MultiVector>(list, map) {
31 // parameters
32 hx_ = list.get("stretchx", 1.0);
33 hy_ = list.get("stretchy", 1.0);
34 hz_ = list.get("stretchz", 1.0);
35 px_ = list.get("px", 1);
36 py_ = list.get("py", 1);
37 pz_ = list.get("pz", 1);
38 if (list.isType<int>("nx"))
39 nx_ = Teuchos::as<GO>(list.get("nx", -1));
40 else
41 nx_ = list.get<GO>("nx", -1);
42 if (list.isType<int>("ny"))
43 ny_ = Teuchos::as<GO>(list.get("ny", -1));
44 else
45 ny_ = list.get<GO>("ny", -1);
46 if (list.isType<int>("nz"))
47 nz_ = Teuchos::as<GO>(list.get("nz", -1));
48 else
49 nz_ = list.get<GO>("nz", -1);
50 if (list.isType<int>("mx"))
51 mx_ = Teuchos::as<GO>(list.get("mx", -1));
52 else
53 mx_ = list.get<GO>("mx", -1);
54 if (list.isType<int>("my"))
55 my_ = Teuchos::as<GO>(list.get("my", -1));
56 else
57 my_ = list.get<GO>("my", -1);
58 if (list.isType<int>("mz"))
59 mz_ = Teuchos::as<GO>(list.get("mz", -1));
60 else
61 mz_ = list.get<GO>("mz", -1);
62 omega_ = list.get("omega", 2.0 * M_PI);
63 shift_ = list.get("shift", 0.0);
64 delta_ = list.get("delta", 2.0);
65 model_ = list.get("model", 0);
66 PMLx_left = list.get("PMLx_left", 0);
67 PMLx_right = list.get("PMLx_right", 0);
68 PMLy_left = list.get("PMLy_left", 0);
69 PMLy_right = list.get("PMLy_right", 0);
70 PMLz_left = list.get("PMLz_left", 0);
71 PMLz_right = list.get("PMLz_right", 0);
72
73 // calculate info
74 Dx_ = hx_ * nx_;
75 Dy_ = hy_ * ny_;
76 Dz_ = hz_ * nz_;
77 LBx_ = PMLx_left * hx_;
78 RBx_ = Dx_ - PMLx_right * hx_;
79 LBy_ = PMLy_left * hy_;
80 RBy_ = Dy_ - PMLy_right * hy_;
81 LBz_ = PMLz_left * hz_;
82 RBz_ = Dz_ - PMLz_right * hz_;
83 PMLwidthx_ = std::max(LBx_, Dx_ - RBx_);
84 if (PMLwidthx_ == 0) {
85 PMLwidthx_ = 1.0;
86 }
87 PMLwidthy_ = std::max(LBy_, Dy_ - RBy_);
88 if (PMLwidthy_ == 0) {
89 PMLwidthy_ = 1.0;
90 }
91 PMLwidthz_ = std::max(LBz_, Dz_ - RBz_);
92 if (PMLwidthz_ == 0) {
93 PMLwidthz_ = 1.0;
94 }
95 nDim = 3;
96
97 // velocity model
98 velocitymodel_.setDim(nDim);
99 velocitymodel_.setModel(model_);
100
101 // currently AMG doesn't suport higher orders. set to linears
102 px_ = 1;
103 py_ = 1;
104 pz_ = 1;
105
106 // NOTE: galeri counts points, not elements
107 dims.push_back(nx_ * px_ - 1);
108 dims.push_back(ny_ * py_ - 1);
109 dims.push_back(nz_ * pz_ - 1);
110 TEUCHOS_TEST_FOR_EXCEPTION(nx_ <= 0 || ny_ <= 0 || nz_ <= 0, std::logic_error, "nx, ny, and nz must be positive");
111 }

113 Teuchos::RCP<Matrix> BuildMatrix();
114 std::pair<Teuchos::RCP<Matrix>, Teuchos::RCP<Matrix> > BuildMatrices();

116 private:
117 typedef Scalar SC;
118 typedef LocalOrdinal LO;
119 typedef GlobalOrdinal GO;
120
121 struct Point {
122 double x, y, z;
123 Point() { z = 0.0; }
124 Point(double x_, double y_, double z_ = 0.0)
125 : x(x_)
126 , y(y_)
127 , z(z_) {}
128 };
129
130 // General mesh parameters
131 GlobalOrdinal nx_, ny_, nz_;
132 int mx_, my_, mz_;
133 size_t nDim, px_, py_, pz_;
134 std::vector<GO> dims;
135 std::vector<Point> nodes;
136 std::vector<std::vector<LO> > elements;
137 std::vector<GO> local2Global_;
138 VelocityModel<SC, LO, GO> velocitymodel_;
139
140 // Helmholtz/PML parameters
141 double hx_, hy_, hz_, shift_, delta_, omega_;
142 int PMLx_left, PMLx_right;
143 int PMLy_left, PMLy_right;
144 int PMLz_left, PMLz_right;
145 double Dx_, Dy_, Dz_;
146 int model_;
147 double LBx_, RBx_, LBy_, RBy_, LBz_, RBz_;
148 double PMLwidthx_, PMLwidthy_, PMLwidthz_;
149
150 using Memory2D = Kokkos::View<SC**, Kokkos::HostSpace>;
151
152 void BuildMesh();
153 void BuildPoints(std::vector<Point>& quadPoints, std::vector<double>& quadWeights);
154 void EvalBasis(Point& quadPoint, std::vector<double>& vals, std::vector<double>& dxs, std::vector<double>& dys, std::vector<double>& dzs);
155 void EvalStretch(double& shiftx, double& shifty, double& shiftz, const std::vector<Point>& quadPoints, std::vector<Scalar>& sx, std::vector<Scalar>& sy, std::vector<Scalar>& sz, std::vector<Scalar>& cs);
156};

158template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
159Teuchos::RCP<Matrix> HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
160 BuildMesh();
161
162 const size_t numDofPerNode = 1;
163 const size_t numNodesPerElem = (px_ + 1) * (py_ + 1) * (pz_ + 1);
164 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
165
166 TEUCHOS_TEST_FOR_EXCEPTION(elements[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
167
168 // Compute quadrature points/rules
169 std::vector<Point> quadPoints;
170 std::vector<double> quadWeights;
171 BuildPoints(quadPoints, quadWeights);
172
173 // Compute basis function values and derivatives at each quadrature point
174 std::vector<std::vector<double> > vals, dxs, dys, dzs;
175 vals.resize(quadPoints.size());
176 dxs.resize(quadPoints.size());
177 dys.resize(quadPoints.size());
178 dzs.resize(quadPoints.size());
179 for (unsigned int i = 0; i < quadPoints.size(); i++) {
180 EvalBasis(quadPoints[i], vals[i], dxs[i], dys[i], dzs[i]);
181 }
182
183 this->A_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * (2 * pz_ + 1) * numDofPerNode);
184 SC cpxshift(1.0, shift_);
185
186 // iterate over elements
187 for (size_t i = 0; i < elements.size(); i++) {
188 Memory2D KE("KE", numDofPerElem, numDofPerElem);
189
190 // element domain is [shiftx,shiftx+hx] x [shifty,shifty+hy] x [shiftz,shiftz+hz]
191 std::vector<LO>& elemNodes = elements[i];
192 double shiftx = nodes[elemNodes[0]].x;
193 double shifty = nodes[elemNodes[0]].y;
194 double shiftz = nodes[elemNodes[0]].z;
195
196 // evaluate PML values at quadrature points for this element
197 std::vector<SC> stretchx, stretchy, stretchz, cs;
198 EvalStretch(shiftx, shifty, shiftz, quadPoints, stretchx, stretchy, stretchz, cs);
199
200 // Evaluate the stiffness matrix for the element
201 for (size_t j = 0; j < quadPoints.size(); j++) {
202 Scalar sx = stretchx[j];
203 Scalar sy = stretchy[j];
204 Scalar sz = stretchz[j];
205 Scalar cc = cs[j] * cs[j];
206 double qdwt = quadWeights[j];
207 std::vector<double> curvals = vals[j];
208 std::vector<double> curdxs = dxs[j];
209 std::vector<double> curdys = dys[j];
210 std::vector<double> curdzs = dzs[j];
211 Scalar mass = qdwt * cpxshift * omega_ * omega_ * sx * sy * sz / cc;
212 Scalar pml1 = qdwt * sy * sz / sx;
213 Scalar pml2 = qdwt * sx * sz / sy;
214 Scalar pml3 = qdwt * sx * sy / sz;
215 for (unsigned int m = 0; m < numDofPerElem; m++) {
216 for (unsigned int n = 0; n < numDofPerElem; n++) {
217 KE(m, n) += pml1 * curdxs[m] * curdxs[n] + pml2 * curdys[m] * curdys[n] + pml3 * curdzs[m] * curdzs[n] - mass * curvals[m] * curvals[n];
218 }
219 }
220 }
221
222 Teuchos::Array<GO> elemDofs(numDofPerElem);
223 for (size_t j = 0; j < numDofPerElem; j++) {
224 elemDofs[j] = local2Global_[elemNodes[j]];
225 }
226
227 // Insert KE into the global matrix
228 for (size_t j = 0; j < numDofPerElem; j++)
229 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
230 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
231 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
232 this->A_->insertGlobalValues(elemDofs[j], inds, values);
233 }
234 }
235
236 this->A_->fillComplete();
237 return this->A_;
238}

240template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
241std::pair<Teuchos::RCP<Matrix>, Teuchos::RCP<Matrix> > HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrices() {
242 BuildMesh();
243
244 const size_t numDofPerNode = 1;
245 const size_t numNodesPerElem = (px_ + 1) * (py_ + 1) * (pz_ + 1);
246 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
247
248 TEUCHOS_TEST_FOR_EXCEPTION(elements[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
249
250 // Compute quadrature points/rules
251 std::vector<Point> quadPoints;
252 std::vector<double> quadWeights;
253 BuildPoints(quadPoints, quadWeights);
254
255 // Compute basis function values and derivatives at each quadrature point
256 std::vector<std::vector<double> > vals, dxs, dys, dzs;
257 vals.resize(quadPoints.size());
258 dxs.resize(quadPoints.size());
259 dys.resize(quadPoints.size());
260 dzs.resize(quadPoints.size());
261 for (unsigned int i = 0; i < quadPoints.size(); i++) {
262 EvalBasis(quadPoints[i], vals[i], dxs[i], dys[i], dzs[i]);
263 }
264
265 this->K_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * (2 * pz_ + 1) * numDofPerNode);
266 this->M_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * (2 * pz_ + 1) * numDofPerNode);
267
268 // iterate over elements
269 for (size_t i = 0; i < elements.size(); i++) {
270 Memory2D KE("KE", numDofPerElem, numDofPerElem);
271 Memory2D ME("ME", numDofPerElem, numDofPerElem);
272
273 // element domain is [shiftx,shiftx+hx] x [shifty,shifty+hy] x [shiftz,shiftz+hz]
274 std::vector<LO>& elemNodes = elements[i];
275 double shiftx = nodes[elemNodes[0]].x;
276 double shifty = nodes[elemNodes[0]].y;
277 double shiftz = nodes[elemNodes[0]].z;
278
279 // evaluate PML values at quadrature points for this element
280 std::vector<SC> stretchx, stretchy, stretchz, cs;
281 EvalStretch(shiftx, shifty, shiftz, quadPoints, stretchx, stretchy, stretchz, cs);
282
283 // Evaluate the stiffness matrix for the element
284 for (size_t j = 0; j < quadPoints.size(); j++) {
285 Scalar sx = stretchx[j];
286 Scalar sy = stretchy[j];
287 Scalar sz = stretchz[j];
288 Scalar cc = cs[j] * cs[j];
289 double qdwt = quadWeights[j];
290 std::vector<double> curvals = vals[j];
291 std::vector<double> curdxs = dxs[j];
292 std::vector<double> curdys = dys[j];
293 std::vector<double> curdzs = dzs[j];
294 Scalar mass = qdwt * sx * sy * sz / cc;
295 Scalar pml1 = qdwt * sy * sz / sx;
296 Scalar pml2 = qdwt * sx * sz / sy;
297 Scalar pml3 = qdwt * sx * sy / sz;
298 for (unsigned int m = 0; m < numDofPerElem; m++) {
299 for (unsigned int n = 0; n < numDofPerElem; n++) {
300 KE(m, n) += pml1 * curdxs[m] * curdxs[n] + pml2 * curdys[m] * curdys[n] + pml3 * curdzs[m] * curdzs[n];
301 ME(m, n) += mass * curvals[m] * curvals[n];
302 }
303 }
304 }
305
306 Teuchos::Array<GO> elemDofs(numDofPerElem);
307 for (size_t j = 0; j < numDofPerElem; j++) {
308 elemDofs[j] = local2Global_[elemNodes[j]];
309 }
310
311 // Insert KE and ME into the global matrices
312 for (size_t j = 0; j < numDofPerElem; j++) {
313 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
314 {
315 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
316 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
317 this->K_->insertGlobalValues(elemDofs[j], inds, values);
318 }
319 {
320 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
321 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(ME, j, Kokkos::ALL()));
322 this->M_->insertGlobalValues(elemDofs[j], inds, values);
323 }
324 }
325 }
326 }
327
328 this->K_->fillComplete();
329 this->M_->fillComplete();
330 std::pair<Teuchos::RCP<Matrix>, Teuchos::RCP<Matrix> > system;
331 system = std::make_pair(this->K_, this->M_);
332 return system;
333}

335template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
336void HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMesh() {
337 GO myPID = this->Map_->getComm()->getRank();
338 GO mySize = this->Map_->getComm()->getSize();
339 GO mx = mx_, my = my_, mz = mz_;
340 if (mx * my * mz != mySize) {
341 const int cubeRoot = std::max(1, (int)std::floor(pow((double)mySize, 1. / 3.)));
342 mz = cubeRoot;
343 while (mySize % mz != 0)
344 ++mz;
345 const int xyCommSize = mySize / mz;
346 const int squareRoot = std::max(1, (int)std::floor(sqrt((double)xyCommSize)));
347 my = squareRoot;
348 while (xyCommSize % my != 0)
349 ++my;
350 mx = xyCommSize / my;
351 }
352
353 GO shiftx, shifty, shiftz, endx, endy, endz;
354 Utils::getSubdomainData(dims[0], mx, myPID % mx, shiftx, endx);
355 Utils::getSubdomainData(dims[1], my, ((myPID - (mx * my) * (myPID / (mx * my))) / mx), shifty, endy);
356 Utils::getSubdomainData(dims[2], mz, myPID / (mx * my), shiftz, endz);
357
358 GO nx = endx - shiftx, ny = endy - shifty, nz = endz - shiftz;
359
360 // Expand subdomain to do overlap
361 if (shiftx > 0) {
362 nx++;
363 shiftx--;
364 }
365 if (shifty > 0) {
366 ny++;
367 shifty--;
368 }
369 if (shiftz > 0) {
370 nz++;
371 shiftz--;
372 }
373 if (shiftx + nx < dims[0]) {
374 nx++;
375 }
376 if (shifty + ny < dims[1]) {
377 ny++;
378 }
379 if (shiftz + nz < dims[2]) {
380 nz++;
381 }
382
383 nodes.resize((nx + 1) * (ny + 1) * (nz + 1));
384 local2Global_.resize((nx + 1) * (ny + 1) * (nz + 1));
385 elements.resize(nx * ny * nz);
386
387#define NODE(i, j, k) ((k) * (ny + 1) * (nx + 1) + (j) * (nx + 1) + (i))
388#define CELL(i, j, k) ((k)*ny * nx + (j)*nx + (i))
389 for (int k = 0; k <= nz; k++)
390 for (int j = 0; j <= ny; j++)
391 for (int i = 0; i <= nx; i++) {
392 int ii = shiftx + i, jj = shifty + j, kk = shiftz + k;
393 int nodeID = NODE(i, j, k);
394 nodes[nodeID] = Point((ii + 1) * hx_, (jj + 1) * hy_, (kk + 1) * hz_);
395 local2Global_[nodeID] = kk * nx_ * ny_ + jj * nx_ + ii;
396 }
397
398 for (int k = 0; k < nz; k++)
399 for (int j = 0; j < ny; j++)
400 for (int i = 0; i < nx; i++) {
401 std::vector<int>& element = elements[CELL(i, j, k)];
402 element.resize(8);
403 element[0] = NODE(i, j, k);
404 element[1] = NODE(i + 1, j, k);
405 element[2] = NODE(i, j + 1, k);
406 element[3] = NODE(i + 1, j + 1, k);
407 element[4] = NODE(i, j, k + 1);
408 element[5] = NODE(i + 1, j, k + 1);
409 element[6] = NODE(i, j + 1, k + 1);
410 element[7] = NODE(i + 1, j + 1, k + 1);
411 }
412#undef NODE
413#undef CELL
414}

416template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
417void HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildPoints(std::vector<Point>& quadPoints, std::vector<double>& quadWeights) {
418 // Use nonstandard quadrature for Helmholtz from the following paper:
419 // M. Ainsworth, H.A. Wajid. Optimally blended spectral-finite element scheme for wave propagation
420 // and nonstandard reduced integration. SIAM J. Numer. Analysis, 48(1): 346-371, 2010.
421
422 // Make reference domain [0,hx_] x [0,hy_] x [0,hz_].
423 double qdpt = 0.816496580927726;
424 quadPoints.resize(8);
425 quadPoints[0] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0, (1.0 + qdpt) * hz_ / 2.0);
426 quadPoints[1] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0, (1.0 + qdpt) * hz_ / 2.0);
427 quadPoints[2] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0, (1.0 + qdpt) * hz_ / 2.0);
428 quadPoints[3] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0, (1.0 + qdpt) * hz_ / 2.0);
429 quadPoints[4] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0, (1.0 - qdpt) * hz_ / 2.0);
430 quadPoints[5] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0, (1.0 - qdpt) * hz_ / 2.0);
431 quadPoints[6] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0, (1.0 - qdpt) * hz_ / 2.0);
432 quadPoints[7] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0, (1.0 - qdpt) * hz_ / 2.0);
433 quadWeights.resize(8);
434 quadWeights[0] = hx_ * hy_ * hz_ / 8.0;
435 quadWeights[1] = hx_ * hy_ * hz_ / 8.0;
436 quadWeights[2] = hx_ * hy_ * hz_ / 8.0;
437 quadWeights[3] = hx_ * hy_ * hz_ / 8.0;
438 quadWeights[4] = hx_ * hy_ * hz_ / 8.0;
439 quadWeights[5] = hx_ * hy_ * hz_ / 8.0;
440 quadWeights[6] = hx_ * hy_ * hz_ / 8.0;
441 quadWeights[7] = hx_ * hy_ * hz_ / 8.0;
442}

444template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
445void HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalBasis(Point& quadPoint, std::vector<double>& vals, std::vector<double>& dxs, std::vector<double>& dys, std::vector<double>& dzs) {
446 // For the reference element, evaluate each basis function and its derivatives at a particular quadrature point
447 int numBasis = (px_ + 1) * (py_ + 1) * (pz_ + 1);
448 vals.resize(numBasis);
449 dxs.resize(numBasis);
450 dys.resize(numBasis);
451 dzs.resize(numBasis);
452 double x = quadPoint.x;
453 double y = quadPoint.y;
454 double z = quadPoint.z;
455 double valx1 = (hx_ - x) / hx_;
456 double dx1 = -1 / hx_;
457 double valx2 = x / hx_;
458 double dx2 = 1 / hx_;
459 double valy1 = (hy_ - y) / hy_;
460 double dy1 = -1 / hy_;
461 double valy2 = y / hy_;
462 double dy2 = 1 / hy_;
463 double valz1 = (hz_ - z) / hz_;
464 double dz1 = -1 / hz_;
465 double valz2 = z / hz_;
466 double dz2 = 1 / hz_;
467 // linear function values, derivatives in x, y, and z directions
468 vals[0] = valx1 * valy1 * valz1;
469 dxs[0] = dx1 * valy1 * valz1;
470 dys[0] = valx1 * dy1 * valz1;
471 dzs[0] = valx1 * valy1 * dz1;
472 vals[1] = valx2 * valy1 * valz1;
473 dxs[1] = dx2 * valy1 * valz1;
474 dys[1] = valx2 * dy1 * valz1;
475 dzs[1] = valx2 * valy1 * dz1;
476 vals[2] = valx1 * valy2 * valz1;
477 dxs[2] = dx1 * valy2 * valz1;
478 dys[2] = valx1 * dy2 * valz1;
479 dzs[2] = valx1 * valy2 * dz1;
480 vals[3] = valx2 * valy2 * valz1;
481 dxs[3] = dx2 * valy2 * valz1;
482 dys[3] = valx2 * dy2 * valz1;
483 dzs[3] = valx2 * valy2 * dz1;
484 vals[4] = valx1 * valy1 * valz2;
485 dxs[4] = dx1 * valy1 * valz2;
486 dys[4] = valx1 * dy1 * valz2;
487 dzs[4] = valx1 * valy1 * dz2;
488 vals[5] = valx2 * valy1 * valz2;
489 dxs[5] = dx2 * valy1 * valz2;
490 dys[5] = valx2 * dy1 * valz2;
491 dzs[5] = valx2 * valy1 * dz2;

503void HelmholtzFEM3DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalStretch(double& shiftx, double& shifty, double& shiftz, const std::vector<Point>& quadPoints, std::vector<Scalar>& sx, std::vector<Scalar>& sy, std::vector<Scalar>& sz, std::vector<Scalar>& cs) {
504 // For the current element domain, evaluate the PML stretching functions at the quadrature points
505 sx.resize(quadPoints.size());
506 sy.resize(quadPoints.size());
507 sz.resize(quadPoints.size());
508 cs.resize(quadPoints.size());
509 for (unsigned int i = 0; i < quadPoints.size(); i++) {
510 double quadx = quadPoints[i].x;
511 double quady = quadPoints[i].y;
512 double quadz = quadPoints[i].z;
513 double curx = shiftx + quadx;
514 double cury = shifty + quady;
515 double curz = shiftz + quadz;
516 double sigx, sigy, sigz;
517 if (curx < LBx_) {
518 sigx = delta_ * pow((curx - LBx_) / PMLwidthx_, 2);
519 } else if (curx > RBx_) {
520 sigx = delta_ * pow((curx - RBx_) / PMLwidthx_, 2);
521 } else {
522 sigx = 0.0;
523 }
524 if (cury < LBy_) {
525 sigy = delta_ * pow((cury - LBy_) / PMLwidthy_, 2);
526 } else if (cury > RBy_) {
527 sigy = delta_ * pow((cury - RBy_) / PMLwidthy_, 2);
528 } else {
529 sigy = 0.0;
530 }
531 if (curz < LBz_) {
532 sigz = delta_ * pow((curz - LBz_) / PMLwidthz_, 2);
533 } else if (curz > RBz_) {
534 sigz = delta_ * pow((curz - RBz_) / PMLwidthz_, 2);
535 } else {
536 sigz = 0.0;
537 }
538 Scalar pmlx(1.0, sigx);
539 Scalar pmly(1.0, sigy);
540 Scalar pmlz(1.0, sigz);
541 Scalar speed = velocitymodel_.getVelocity(curx, cury, curz);
542 sx[i] = pmlx;
543 sy[i] = pmly;
544 sz[i] = pmlz;
545 cs[i] = speed;
546 }
