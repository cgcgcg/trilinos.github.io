27TEUCHOS_UNIT_TEST(PhalanxViewOfViews,OldImpl) {
28
29 const int num_cells = 10;
30 const int num_pts = 8;
31 const int num_equations = 32;
32
33 Kokkos::View<double***,mem_t> a("a",num_cells,num_pts,num_equations);
34 Kokkos::View<double***,mem_t> b("b",num_cells,num_pts,num_equations);
35 Kokkos::View<double***,mem_t> c("c",num_cells,num_pts,num_equations);
36 Kokkos::View<double***,mem_t> d("d",num_cells,num_pts,num_equations);
37
38 Kokkos::deep_copy(a,2.0);
39 Kokkos::deep_copy(b,3.0);
40 Kokkos::deep_copy(c,4.0);
41
42 {
43 using InnerView = Kokkos::View<double***,mem_t,Kokkos::MemoryTraits<Kokkos::Unmanaged>>;
44 constexpr int OuterViewRank = 2;
45 PHX::ViewOfViews<OuterViewRank,InnerView,mem_t> v_of_v("outer host",2,2);
46
47 v_of_v.addView(a,0,0);
48 v_of_v.addView(b,0,1);
49 v_of_v.addView(c,1,0);
50 v_of_v.addView(d,1,1);
51
52 v_of_v.syncHostToDevice();
53
54 {
55 auto v_dev = v_of_v.getViewDevice();
56 auto policy = Kokkos::MDRangePolicy<Kokkos::Rank<3>>({0,0,0},{num_cells,num_pts,num_equations});
57 Kokkos::parallel_for("view of view test",policy,KOKKOS_LAMBDA (const int cell,const int pt, const int eq) {
58 v_dev(1,1)(cell,pt,eq) = v_dev(0,0)(cell,pt,eq) + v_dev(0,1)(cell,pt,eq) + v_dev(1,0)(cell,pt,eq);
59 });
60 }
61
62 // Uncomment the line below to prove the ViewOfViews prevents
63 // device views from outliving host view. This line will cause a
64 // Kokkos::abort() and error message since v_dev above is still in
65 // scope when the ViewOfViews is destoryed.
66 // v_of_v = PHX::ViewOfViews<OuterViewRank,InnerView,mem_t>("outer host",2,2);
67 }
68
69 auto d_host = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(),d);
70
71 const auto tol = std::numeric_limits<double>::epsilon() * 100.0;
72 for (int cell=0; cell < num_cells; ++cell)
73 for (int pt=0; pt < num_pts; ++pt)
74 for (int eq=0; eq < num_equations; ++eq) {
75 TEST_FLOATING_EQUALITY(d_host(cell,pt,eq),9.0,tol);
76 }
77}

467// ********************************
468// Demonstrates an alternative path for ViewOfViews that uses a user
469// defined wrapper and the assignment operator on device to disable
470// the reference tracking.
471// ********************************
472
473// Force this test to always run without UVM.
474// For Cuda builds, ignore the default memory space in the Cuda
475// execution space since it can be set to UVM in Trilinos
476// configure. For non-Cuda builds, just use the default memory space
477// in the execution space.
478namespace Kokkos {
479 class Cuda;
480 class CudaSpace;
481}
482using DeviceExecutionSpace = Kokkos::DefaultExecutionSpace;
483using DeviceMemorySpace = std::conditional<std::is_same<DeviceExecutionSpace,Kokkos::Cuda>::value,
484 Kokkos::CudaSpace,
485 Kokkos::DefaultExecutionSpace::memory_space>::type;
486using view = Kokkos::View<double*,DeviceMemorySpace>;
487using view_host = view::host_mirror_type;
488
489class Wrapper {
490public:
491 view a_;
492 view b_;
493 KOKKOS_INLINE_FUNCTION
494 Wrapper() : a_(nullptr,0),b_(nullptr,0) {}
495
496 Wrapper(const view_host& a, const view_host& b)
497 : a_(a.label(),a.layout()),b_(b.label(),b.layout())
498 {
499 Kokkos::deep_copy(a_,a);
500 Kokkos::deep_copy(b_,b);
501 }
502
503 KOKKOS_DEFAULTED_FUNCTION
504 Wrapper& operator=(const Wrapper& src) = default;
505
506 KOKKOS_INLINE_FUNCTION
507 double multiply(int i) const {return a_(i) * b_(i);}
508};
