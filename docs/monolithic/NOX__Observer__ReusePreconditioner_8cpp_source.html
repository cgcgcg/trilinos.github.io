75void NOX::ObserverReusePreconditioner::runPreSolve(const NOX::Solver::Generic& solver)
76{
77 // First time through, grab the operator and tell the group that the
78 // user will control preconditioning.
79 if (precOperator_.is_null()) {
80 const auto& const_group = solver.getSolutionGroup();
81 auto& group = const_cast<NOX::Abstract::Group&>(const_group);
82 auto& thyra_group = dynamic_cast<NOX::Thyra::Group&>(group);
83
84 auto lows_factory = thyra_group.getLinearOpWithSolveFactory();
85 TEUCHOS_ASSERT(nonnull(lows_factory));
86
87 // Creating the preconditioner objects follows the logic of the
88 // NOX::Thyra::Group constructors for consistency.
89 precFactory_ = thyra_group.getPreconditionerFactory();
90 if (precFactory_.is_null()) {
91 precFactory_ = lows_factory->getPreconditionerFactory();
92 }
93
94 precOperator_ = thyra_group.getNonconstPreconditioner();
95 if (precOperator_.is_null()) {
96 if (nonnull(precFactory_)) {
97 precOperator_ = precFactory_->createPrec();
98 } else if (thyra_group.getModel()->createOutArgs().supports( ::Thyra::ModelEvaluatorBase::OUT_ARG_W_prec)) {
99 // precOperator_ = thyra_group.getModel()->create_W_prec();
100 }
101 }
102
103 thyra_group.takeControlOfPreconditionerUpdates(precOperator_);
104
105 TEUCHOS_TEST_FOR_EXCEPTION(precFactory_.is_null(),std::runtime_error,"ERROR: the ReusePreconditioner observer is registered but there is no preconditioner available!");
106 TEUCHOS_TEST_FOR_EXCEPTION(precOperator_.is_null(),std::runtime_error,"ERROR: the ReusePreconditioner observer is registered but there is no preconditioner available!");
107
108 // First time through, create the preconditioner
109 // this->updatePreconditioner(thyra_group);
110 }

113void NOX::ObserverReusePreconditioner::runPreIterate(const NOX::Solver::Generic& solver)
114{
115 TEUCHOS_ASSERT(this->isInitialized());
116
117 bool update_the_preconditioner = false;
118
119 if (update_at_start_of_nonlinear_solve_ && (solver.getNumIterations() == 0) ) {
120 update_the_preconditioner = true;
121 }
122
123 if (update_after_n_nonlinear_solves_ && (solver.getNumIterations() == 0) ) {
124 if (num_nonlinear_solves_count_ % num_nonlinear_solves_for_update_ == 0) {
125 update_the_preconditioner = true;
126 }
127 }
128
129 if (update_after_n_linear_iterations_) {
130 if (solver.getNumIterations() % num_linear_iterations_for_update_ == 0) {
131 update_the_preconditioner = true;
132 }
133 }
134
135 if (update_if_stalled_) {
136 const int num_linear_iterations = solver.getSolverStatistics()->linearSolve.lastLinearSolve_NumIterations;
137 const bool is_converged = solver.getSolverStatistics()->linearSolve.lastLinearSolve_Converged;
138
139 if (num_linear_iterations >= max_linear_iterations_for_stall_) {
140 ++stall_count_;
141 } else { // reset count on a successful linear solve
142 stall_count_ = 0;
143 }
144
145 if ( (!is_converged) || (stall_count_ >= max_count_for_stall_) )
146 update_the_preconditioner = true;
147 }
148
149 if (update_the_preconditioner) {
150 const auto& const_group = solver.getSolutionGroup();
151 auto& group = const_cast<NOX::Abstract::Group&>(const_group);
152 auto& thyraGroup = dynamic_cast<NOX::Thyra::Group&>(group);
153 this->updatePreconditioner(thyraGroup);
154 }
155}

177void NOX::ObserverReusePreconditioner::updatePreconditioner(NOX::Thyra::Group& group)
178{
179 TEUCHOS_ASSERT(this->isInitialized());
180
181 if (!group.isJacobian())
182 group.computeJacobian();
183
184 auto jacOperator = group.getScaledJacobianOperator();
185
186 ::Thyra::initializePrec<double>(*precFactory_,jacOperator,precOperator_.ptr());
187
188 group.unscaleJacobianOperator();
189
190 stall_count_ = 0;
191 iterations_since_last_update_ = 0;
192
193 ++update_preconditioner_count_;
194}
