139Thyra::SolveStatus<double> Thyra::NOXNonlinearSolver::
140solve(VectorBase<double> *x,
141 const SolveCriteria<double> * /* solveCriteria */,
142 VectorBase<double> *delta)
143{
144
145#ifdef ENABLE_NOX_THYRA_TIMERS
146 TEUCHOS_FUNC_TIME_MONITOR("Thyra::NOXNonlinearSolver::solve");
147#endif
148
149 TEUCHOS_ASSERT(nonnull(model_));
150 TEUCHOS_ASSERT(nonnull(param_list_));
151
152 NOX::Thyra::Vector initial_guess(Teuchos::rcp(x, false)); // View of x
153
154 if (Teuchos::is_null(solver_) || rebuild_solver_) {
155
156 rebuild_solver_ = false;
157
158 this->validateAndParseThyraGroupOptions(param_list_->sublist("Thyra Group Options"));
159
160 if (function_scaling_ != "None") {
161
162 if (function_scaling_ == "Row Sum")
163 this->setupRowSumScalingObjects();
164
165 TEUCHOS_ASSERT(nonnull(scaling_vector_));
166 }
167 else {
168 TEUCHOS_ASSERT(is_null(scaling_vector_));
169 }
170
171 right_scaling_vector_ = Teuchos::null;
172 inv_right_scaling_vector_ = Teuchos::null;
173 if(param_list_->isParameter("Right Scaling Vector")){
174 Teuchos::RCP<NOX::Abstract::Vector> abstract_vec = param_list_->get<RCP<NOX::Abstract::Vector> >("Right Scaling Vector");
175 right_scaling_vector_ = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(abstract_vec)->getThyraRCPVector();
176 }
177
178 if(param_list_->isParameter("Inverse Right Scaling Vector")){
179 Teuchos::RCP<NOX::Abstract::Vector> abstract_inv_vec = param_list_->get<RCP<NOX::Abstract::Vector> >("Inverse Right Scaling Vector");
180 inv_right_scaling_vector_ = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(abstract_inv_vec)->getThyraRCPVector();
181 }
182
183 if (is_null(user_defined_nox_group_)) {
184 if (is_null(precOp_))
185 nox_group_ = Teuchos::rcp(new NOX::Thyra::Group(initial_guess, model_, scaling_vector_, right_scaling_vector_, inv_right_scaling_vector_, rightScalingFirst_));
186 else {
187 auto lowsFactory = model_->get_W_factory();
188 auto linOp = model_->create_W_op();
189 nox_group_ = Teuchos::rcp(new NOX::Thyra::Group(initial_guess, model_, linOp, lowsFactory, precOp_, precFactory_, scaling_vector_, right_scaling_vector_, inv_right_scaling_vector_, rightScalingFirst_, updatePreconditioner_));
190 }
191 }
192 else
193 nox_group_ = user_defined_nox_group_;
194
195 if (use_base_point_)
196 this->getThyraGroupNonConst(nox_group_)->setBasePoint(this->basePoint_);
197
198 status_test_ = this->buildStatusTests(*param_list_);
199 solver_ = NOX::Solver::buildSolver(nox_group_, status_test_, param_list_);
200 }
201 else {
202 if (use_base_point_)
203 this->getThyraGroupNonConst(nox_group_)->setBasePoint(this->basePoint_);
204
205 const auto thyra_group = this->getThyraGroupConst(solver_->getSolutionGroupPtr());
206 auto nonconst_thyra_group = Teuchos::rcp_const_cast<NOX::Thyra::Group>(thyra_group);
207 nonconst_thyra_group->setX(initial_guess);
208
209 solver_->reset();
210 }
211
212 NOX::StatusTest::StatusType solvStatus = solver_->solve();
213
214 Thyra::SolveStatus<double> t_status;
215
216 if (solvStatus == NOX::StatusTest::Converged)
217 t_status.solveStatus = SOLVE_STATUS_CONVERGED;
218 else if (solvStatus == NOX::StatusTest::Unconverged)
219 t_status.solveStatus = SOLVE_STATUS_UNCONVERGED;
220 else if (solvStatus == NOX::StatusTest::Failed)
221 t_status.solveStatus = SOLVE_STATUS_UNCONVERGED;
222 else
223 t_status.solveStatus = SOLVE_STATUS_UNCONVERGED;
224
225 if (is_null(t_status.extraParameters))
226 t_status.extraParameters = Teuchos::parameterList("NOX Solver Status");
227
228 t_status.extraParameters->set("Number of Iterations",
229 solver_->getNumIterations());
230
231 // Get the solution and update
232 const Teuchos::RCP<const NOX::Abstract::Group> final_group = solver_->getSolutionGroupPtr();
233 const Teuchos::RCP<const NOX::Thyra::Group> final_thyra_group = this->getThyraGroupConst(final_group);
234 const ::Thyra::VectorBase<double>& new_x = *(final_thyra_group->get_current_x());
235
236 if (delta)
237 ::Thyra::V_StVpStV<double>(Teuchos::ptr(delta),1.0,new_x,-1.0,*x);
238
239 ::Thyra::assign(Teuchos::ptr(x), new_x);
240
241 return t_status;
242
243}

314Teuchos::RCP<NOX::StatusTest::Generic> Thyra::NOXNonlinearSolver::
315buildStatusTests(Teuchos::ParameterList& p)
316{
317 Teuchos::RCP<NOX::StatusTest::Generic> status_test;
318
319 NOX::Utils utils(p.sublist("Printing"));
320
321 if (p.isSublist("Status Tests")) {
322 status_test =
323 NOX::StatusTest::buildStatusTests(p.sublist("Status Tests"), utils);
324 }
325 else { // Default status test
326 Teuchos::RCP<NOX::StatusTest::NormF> absresid =
327 Teuchos::rcp(new NOX::StatusTest::NormF(1.0e-8));
328 Teuchos::RCP<NOX::StatusTest::NormWRMS> wrms =
329 Teuchos::rcp(new NOX::StatusTest::NormWRMS(1.0e-2, 1.0e-8));
330 Teuchos::RCP<NOX::StatusTest::Combo> converged =
331 Teuchos::rcp(new NOX::StatusTest::Combo(NOX::StatusTest::Combo::AND));
332 converged->addStatusTest(absresid);
333 converged->addStatusTest(wrms);
334 Teuchos::RCP<NOX::StatusTest::MaxIters> maxiters =
335 Teuchos::rcp(new NOX::StatusTest::MaxIters(20));
336 Teuchos::RCP<NOX::StatusTest::FiniteValue> fv =
337 Teuchos::rcp(new NOX::StatusTest::FiniteValue);
338 Teuchos::RCP<NOX::StatusTest::Combo> combo =
339 Teuchos::rcp(new NOX::StatusTest::Combo(NOX::StatusTest::Combo::OR));
340 combo->addStatusTest(fv);
341 combo->addStatusTest(converged);
342 combo->addStatusTest(maxiters);
343
344 status_test = combo;
345 }
346
347 return status_test;
348}

369void Thyra::NOXNonlinearSolver::
370validateAndParseThyraGroupOptions(Teuchos::ParameterList& thyra_group_options_sublist)
371{
372 using Teuchos::ParameterList;
373
374 ParameterList validParams;
375 {
376 validParams.set(
377 "Function Scaling", "None",
378 "Determines if function scaling of residual, Jacobian, etc. should be used.",
379 rcp(new Teuchos::StringValidator(
380 Teuchos::tuple<std::string>("None", "Row Sum", "User Defined"))));
381
382 validParams.set(
383 "Update Row Sum Scaling", "Before Each Nonlinear Solve",
384 "Determines if function scaling of residual, Jacobian, etc. should be used.",
385 rcp(new Teuchos::StringValidator(
386 Teuchos::tuple<std::string>("Before Each Nonlinear Solve","Before Each Nonlinear Iteration"))));
387
388 validParams.set<Teuchos::RCP< ::Thyra::VectorBase<double> > >("User Defined Scaling", Teuchos::null);
389 validParams.set<bool >("Do Right Scaling First", false);
390 }
391
392 thyra_group_options_sublist.validateParametersAndSetDefaults(validParams);
393
394 function_scaling_ = thyra_group_options_sublist.get<std::string>("Function Scaling");
395
396 if (function_scaling_ =="Row Sum")
397 do_row_sum_scaling_ = true;
398 else
399 do_row_sum_scaling_ = false;
400
401 std::string string_when_to_update = thyra_group_options_sublist.get<std::string>("Update Row Sum Scaling");
402 if (string_when_to_update == "Before Each Nonlinear Solve")
403 when_to_update_ = NOX::RowSumScaling::UpdateInvRowSumVectorAtBeginningOfSolve;
404 else if (string_when_to_update == "Before Each Nonlinear Iteration")
405 when_to_update_ = NOX::RowSumScaling::UpdateInvRowSumVectorAtBeginningOfIteration;
406
407 if (function_scaling_ =="User Defined")
408 scaling_vector_ = thyra_group_options_sublist.get<Teuchos::RCP< ::Thyra::VectorBase<double> > >("User Defined Scaling");
409
410 rightScalingFirst_ = thyra_group_options_sublist.get<bool>("Do Right Scaling First");
411
412}

415// ****************************************************************
416void Thyra::NOXNonlinearSolver::setupRowSumScalingObjects()
417{
418 using Teuchos::RCP;
419 using Teuchos::rcp;
420
421 scaling_vector_ = ::Thyra::createMember(model_->get_f_space());
422
423 ::Thyra::V_S(scaling_vector_.ptr(),1.0);
424
425 RCP<NOX::Abstract::PrePostOperator> row_sum_observer =
426 rcp(new NOX::RowSumScaling(scaling_vector_, when_to_update_));
427
428 Teuchos::ParameterList& nox_parameters = *param_list_;
429
430 if (nox_parameters.sublist("Solver Options").
431 isType< RCP<NOX::Abstract::PrePostOperator> >("User Defined Pre/Post Operator")) {
432
433 RCP<NOX::Abstract::PrePostOperator> user_observer =
434 nox_parameters.sublist("Solver Options").get< RCP<NOX::Abstract::PrePostOperator> >("User Defined Pre/Post Operator");
435
436 // NOTE: the row_sum_observer should be evalauted after any user
437 // oberservers to make sure that the jacobian is accurate. This
438 // is needed, for example, if we have a model evaluator decorator
439 // that adds extra input parameters to the model such as a
440 // predictor or previous time step solution to be used for
441 // semi-implicit models. The row sum would accidentally use the
442 // previous predicted value which would be bad.
443 RCP<NOX::PrePostOperatorVector> observer_vector = Teuchos::rcp(new NOX::PrePostOperatorVector);
444 observer_vector->pushBack(user_observer);
445 observer_vector->pushBack(row_sum_observer);
446
447 nox_parameters.sublist("Solver Options").set< RCP<NOX::Abstract::PrePostOperator> >("User Defined Pre/Post Operator", observer_vector);
448
449 }
450 else
451 nox_parameters.sublist("Solver Options").set< RCP<NOX::Abstract::PrePostOperator> >("User Defined Pre/Post Operator", row_sum_observer);
452
453
454 // Set the weighted merit function. Throw error if a user defined
455 // merit funciton is present.
456 // ETP 5/23/16 -- Commenting this out because the parameter list may have
457 // been reused from previous solves, and so the merit function that has been
458 // set below from previous solves may still be there.
459 //TEUCHOS_ASSERT( !(nox_parameters.sublist("Solver Options").isType<RCP<NOX::MeritFunction::Generic> >("User Defined Merit Function")));
460
461 RCP<NOX::MeritFunction::Generic> mf = rcp(new NOX::Thyra::WeightedMeritFunction(scaling_vector_));
462
463 nox_parameters.sublist("Solver Options").set<RCP<NOX::MeritFunction::Generic> >("User Defined Merit Function",mf);
464
465}
