105template <typename T, typename D = int, typename tol_t>
106void solve_and_check(const tol_t tol)
107{
108 klu_symbolic<T, D> *Symbolic;
109 klu_numeric <T, D> *Numeric;
110 klu_common <T, D> Common;
111
112 klu_defaults<T, D> (&Common);
113
115 auto Ax = get_Ax <T>();
116 const auto x = get_x <T>();
117 const auto x_tr = get_x_transpose<T>();
118
120 Symbolic = klu_analyze<T, D>(size, Ap, Ai, &Common);
121
123 Numeric = klu_factor<T, D>(Ap, Ai, Ax.data(), Symbolic, &Common);
124
126 {
127 auto b = get_b<T>();
128 klu_solve<T, D>(Symbolic, Numeric, size, 1, b.data(), &Common);
129 print_and_compare<T>(x, b, tol);
130 }
131
133 {
134 auto b = get_b<T>();
135 klu_tsolve<T, D>(Symbolic, Numeric, size, 1, b.data(), &Common);
136 print_and_compare<T>(x_tr, b, tol);
137 }
138
140 klu_free_symbolic<T, D>(&Symbolic, &Common);
141 klu_free_numeric <T, D>(&Numeric, &Common);
142}
