100 using Teuchos::ParameterList;
101 using Teuchos::parameterList;
102 using Teuchos::RCP;
103 using Teuchos::rcp;
104 typedef Belos::LinearProblem<ST, MV, OP > problem_type;
105 typedef Belos::SolverFactory<ST, MV, OP> solver_factory_type;
106 typedef Belos::SolverManager<ST, MV, OP> solver_type;
107 typedef Belos::MultiVecTraits<ST, MV> MVT;
108
109 // Set these in advance, so that if the Belos solver throws an
110 // exception for some reason, these will have sensible values.
111 converged = false;
112 numItersPerformed = 0;
113
114 TEUCHOS_TEST_FOR_EXCEPTION(A.is_null () || X.is_null () || B.is_null (),
115 std::invalid_argument, "solveWithBelos: The A, X, and B arguments must all "
116 "be nonnull.");
117 const int numColsB = MVT::GetNumberVecs (*B);
118 const int numColsX = MVT::GetNumberVecs (*X);
119 TEUCHOS_TEST_FOR_EXCEPTION(numColsB != numColsX, std::invalid_argument,
120 "solveWithBelos: X and B must have the same number of columns. X has "
121 << numColsX << " columns, but B has " << numColsB << " columns.");
122
123 RCP<ParameterList> belosParams = parameterList ();
124 belosParams->set ("Block Size", numColsB);
125 belosParams->set ("Maximum Iterations", maxNumIters);
126 belosParams->set ("Num Blocks", maxNumIters);
127 belosParams->set ("Convergence Tolerance", tol);
128 if (solverName == "GMRES") {
129 belosParams->set ("Orthogonalization", "ICGS");
130 belosParams->set ("maxNumOrthogPasses", 1);
131 }
132 belosParams->set ("Output Frequency", 10);
133 belosParams->set ("Output Style", 1);
134 belosParams->set ("Verbosity", 33);
135
136 RCP<problem_type> problem = rcp (new problem_type (A, X, B));
137 if (! M_left.is_null ()) {
138 problem->setLeftPrec (M_left);
139 }
140 if (! M_right.is_null ()) {
141 problem->setRightPrec (M_right);
142 }
143
144 // Create solver
145 RCP<solver_type> solver;
146 {
147 solver_factory_type factory;
148 solver = factory.create (solverName, belosParams);
149 }
150
151 // Enter "time step" loop -- we're really solving the same system repeatedly
152 converged = true;
153 numItersPerformed = 0;
154 for (int step = 0; step < num_steps; ++step) {
155 // Reset problem
156 const bool set = problem->setProblem ();
157 TEUCHOS_TEST_FOR_EXCEPTION(
158 ! set, std::runtime_error, "solveWithBelos: The "
159 "Belos::LinearProblem's setProblem() method returned false. "
160 "This probably indicates that there is something wrong with A, X, or B.");
161 solver->setProblem (problem);
162
163 // Solve
164 Belos::ReturnType result = solver->solve ();
165
166 converged = converged && (result == Belos::Converged);
167 numItersPerformed += solver->getNumIters ();
168 }
