84/* FLIP is a "negation about -1", and is used to mark an integer i that is
85 * normally non-negative. FLIP (AMESOS2_KLU2_EMPTY) is AMESOS2_KLU2_EMPTY. FLIP of a number > AMESOS2_KLU2_EMPTY
86 * is negative, and FLIP of a number < EMTPY is positive. FLIP (FLIP (i)) = i
87 * for all integers i. UNFLIP (i) is >= AMESOS2_KLU2_EMPTY. */
88#define AMESOS2_KLU2_EMPTY (-1)
89#define FLIP(i) (-(i)-2)
90#define UNFLIP(i) (((i) < AMESOS2_KLU2_EMPTY) ? FLIP (i) : (i))
