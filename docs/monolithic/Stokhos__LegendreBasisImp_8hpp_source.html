41template <typename ordinal_type, typename value_type>
42bool
43Stokhos::LegendreBasis<ordinal_type, value_type>::
44computeRecurrenceCoefficients(ordinal_type n,
45 Teuchos::Array<value_type>& theAlpha,
46 Teuchos::Array<value_type>& theBeta,
47 Teuchos::Array<value_type>& theDelta,
48 Teuchos::Array<value_type>& theGamma) const
49{
50 // Legendre 3 term recurrence:
51 // P_0(x) = 1
52 // P_1(x) = x
53 // P_i(x) = (2*i-1)/i*x*P_{i-1}(x) - (i-1)/i*P_{i-2}(x), i=2,3,...
54 theAlpha[0] = 0.0;
55 theBeta[0] = 1.0;
56 theDelta[0] = 1.0;
57 theGamma[0] = 1.0;
58 for (ordinal_type i=1; i<n; i++) {
59 theAlpha[i] = 0.0;
60 //theBeta[i] = value_type(i*i) / value_type((2*i-1)*(2*i+1));
61 theBeta[i] = value_type(i) / value_type(i+1);
62 theDelta[i] = value_type(2*i+1) / value_type(i+1);
63 theGamma[i] = 1.0;
64 }
65
66 return false;
67}
