10/*======================== basker ==============================================*/
11/* Finds the LU factorization of a sparse A such that A=L*U.
12 * Requires 2 workspaces. The integer workspace of size ancol+2*anrow and a
13 * double workspace of size 2*anrow. A is expected in compressed column form.
14 *
15 * The output L and U are also in compressed column form. The pointers for both
16 * L and U should be preallocated. The size of expected L and U should be passed
17 * to the method as lnnz and unnz. basker will return an error as the size of
18 * computed L and U exceeds the expected L and U.
19 *
20 */
21

23#include <stdlib.h>

28/* ==================== dfs function ============== */
29/* Does a depth first search of a trapezoidal L that is partly computed and
30 * partly equal to the identity. The result of the dfs is in the pattern.
31 */
32
33void BASKER(dfs)
34(
35 Int j,
36 Int Li [],
37 Int Lp [],
38 Int color [],
39 Int pattern [],
40 Int *top,
41 Int k,
42 Int pinv []
43)
44{
45 Int i, t, i1 ;
46 Int start, end ;
47
48 PRINT(("DFS : %d ***********************\n", j));
49 color[j] = 1 ;
50 t = pinv [j] ;
51
52 if ( t != -1 )
53 {
54 start = Lp[t] ;
55 end = Lp[t+1] ;
56 for ( i1 = start ; i1 < end ; i1++ )
57 {
58 i = Li[i1] ;
59 if ( color[i] == 0 )
60 {
61 BASKER(dfs)(i, Li, Lp, color, pattern, top, k, pinv) ;
62 }
63 }
64 }
65 pattern[--*top] = j ;
66 color[j] = 2 ;
67}

69void BASKER(dfs_iter)
70(
71 Int n,
72 Int j,
73 Int Li [],
74 Int Lp [],
75 Int color [],
76 Int pattern [], /* o/p */
77 Int *top, /* o/p */
78 Int k,
79 Int pinv [],
80 Int stack []
81)
82{
83 Int i, t, head, i1 ;
84 Int start, end, done, *store ;
85
86 store = stack + n ;
87 head = 0;
88 stack[head] = j;
89
90 while (head >= 0)
91 {
92 j = stack[head] ;
93 PRINT(("DFS : %d ***********************\n", j));
94 t = pinv [j] ;
95 if (color[j] == 0)
96 {
97 /* Seeing this column for first time */
98 color[j] = 1 ;
99 start = Lp[t] ;
100 }
101 else
102 {
103 BASKERASSERT (color[j] == 1) ; /* color cannot be 2 when we are here */
104 start = store[j];
105 }
106 done = 1;
107
108 if ( t != -1 )
109 {
110 end = Lp[t+1] ;
111 for ( i1 = start ; i1 < end ; i1++ )
112 {
113 i = Li[i1] ;
114 if ( color[i] == 0 )
115 {
116 stack[++head] = i;
117 store[j] = i1+1;
118 done = 0;
119 break;
120 }
121 }
122 /*if (i1 == end)
123 done = 1;*/
124 }
125 if (done)
126 {
127 pattern[--*top] = j ;
128 color[j] = 2 ;
129 head--; /* pop j */
130 }
131 }
132 PRINT(("Out of DFS : %d ***********************\n", j));
133
134}

136/* ==================== basker function ============== */
137Int BASKER(basker)
138(
139 Int Ap [],
140 Int Ai [],
141 double Ax [],
142 Int anrow,
143 Int ancol,
144 Int ws [],
145 double X [],
146 Int *Lp,
147 Int **Li_p,
148 double **Lx_p,
149 Int *Up,
150 Int **Ui_p,
151 double **Ux_p,
152 Int *llnnz_p,
153 Int *uunnz_p,
154 Int *pinv
155)
156{
157 Int i, j, k;
158 Int *tptr, *color, *pattern, *stack ;
159 Int *Li, *Ui ;
160 Int top, top1, maxindex, t, j1, j2, llnnz, uunnz ;
161 Int lnnz, unnz, xnnz, lcnt, ucnt;
162 Int cu_ltop, cu_utop ;
163 Int pp, p2, p ;
164 Int newsize;
165 double pivot, value, xj ;
166 double absv, maxv ;
167 double *Lx, *Ux ;
168
169 llnnz = *llnnz_p;
170 uunnz = *uunnz_p;
171 Li = *Li_p ;
172 Lx = *Lx_p ;
173 Ui = *Ui_p ;
174 Ux = *Ux_p ;
175
176 tptr = ws ;
177
178 color = ws ;
179 tptr += ancol ;
180
181 pattern = tptr ;
182 tptr += anrow ;
183
184 stack = tptr ;
185 tptr += 2*anrow ;
186
187 cu_ltop = 0 ;
188 cu_utop = 0 ;
189 top = ancol ;
190 top1 = ancol ;
191 lnnz = 0 ;
192 unnz = 0 ;
193
194 for (k = 0; k<ancol ; k++)
195 {
196 pinv[k] = -1 ;
197 }
198
199 /* for all the columns in A ... */
200 for (k = 0; k<ancol ; k++)
201 {
202 PRINT(("k = %d ****************** \n", k));
203 value = 0.0 ;
204 pivot = 0.0 ;
205 maxindex = -1 ;
206 j1 = 0 ;
207 j2 = 0 ;
208 lcnt = 0;
209 ucnt = 0;
210
211#ifdef DEBUG
212 BASKERASSERT( top == ancol ) ;
213
214 for ( i = 0 ; i < anrow ;i++)
215 {
216 BASKERASSERT( X[i] == 0 ) ;
217 }
218
219 for ( i = 0 ; i < ancol ;i++)
220 {
221 BASKERASSERT ( color[i] == 0 ) ;
222 }
223#endif
224
225 /* reachability for every non zero in Ak */
226 for (i = Ap[k] ; i < Ap[k+1] ; i++ )
227 {
228 j = Ai[i] ;
229 X [j] = Ax[i] ;
230
231 if ( color[j] == 0 )
232 {
233 /*BASKER(dfs) (j, Li, Lp, color, pattern, &top, k, pinv) ;*/
234 BASKER(dfs_iter) (anrow, j, Li, Lp, color, pattern, &top, k,
235 pinv, stack) ;
236 }
237 }
238
239
240 xnnz = ancol - top ;
241 BASKERASSERT ( xnnz <= anrow ) ;
242
243 /* Lx = b where x will be the column k in in L and U */
244 top1 = top ;
245 for ( pp = 0 ; pp < xnnz ; pp++ )
246 {
247 j = pattern[top1++] ;
248 color[j] = 0 ;
249 t = pinv[j] ;
250
251 if ( t != -1 )
252 {
253 xj = X [j] ;
254 p2 = Lp [t+1] ; /* TBV */
255 for (p = Lp [t]+1 ; p < p2 ; p++)
256 {
257 X [Li [p]] -= Lx [p] * xj ;
258 }
259 }
260 }
261
262 /* get the pivot */
263 maxv = 0.0 ;
264 for ( i = top ; i < anrow ;i++)
265 {
266 j = pattern[i] ;
267 t = pinv[j] ;
268 value = X[j] ;
269 absv = ( value < 0.0 ? -value : value ) ;
270
271 if ( t == -1 )
272 {
273 lcnt++;
274 if( absv > maxv)
275 {
276 maxv = absv ;
277 pivot = value ;
278 maxindex = j ;
279 }
280 }
281 }
282 ucnt = anrow - top - lcnt + 1;
283
284 if ( maxindex == -1 || pivot == 0 )
285 {
286 printf(" matrix is singular maxindex=%d pivot=%g\n", maxindex, pivot ) ;
287 return 1;
288 }
289
290 pinv[maxindex] = k ;
291 if (maxindex != k)
292 {
293 PRINT(("********* Permuting pivot %d as row %d\n", k, maxindex));
294 }
295
296
297 if (lnnz + lcnt >= llnnz)
298 {
299 /* reallocate space for L */
300 /* Note that there can be atmost anrow - top - 1 entries in L
301 * from the for loop below as P[maxindex] != -1.
302 * The index and value for maxindex entry is assigned outside
303 * the loop. */
304 newsize = llnnz * 1.1 + 2 * anrow + 1;
305 PRINT(("Reallocating L oldsize=%d, newsize=%d\n", llnnz, newsize));
306 Li = BASKERREALLOC(Li, newsize * sizeof(Int));
307 if (!Li)
308 {
309 printf("Cannot allocate memory\n");
310 return 1;
311 }
312 Lx = BASKERREALLOC(Lx, newsize * sizeof(double));
313 if (!Lx)
314 {
315 printf("Cannot allocate memory\n");
316 return 1;
317 }
318 llnnz = newsize;
319 }
320
321 if (unnz + ucnt >= uunnz)
322 {
323 /* reallocate space for U */
324 newsize = uunnz * 1.1 + 2 * anrow + 1;
325 PRINT(("Reallocating L oldsize=%d, newsize=%d\n", uunnz, newsize));
326 Ui = BASKERREALLOC(Ui, newsize * sizeof(Int));
327 if (!Ui)
328 {
329 printf("Cannot allocate memory\n");
330 return 1;
331 }
332 Ux = BASKERREALLOC(Ux, newsize * sizeof(double));
333 if (!Ux)
334 {
335 printf("Cannot allocate memory\n");
336 return 1;
337 }
338 uunnz = newsize;
339 }
340
341 /* L(k,k) = 1 */
342 BASKERASSERT(lnnz < llnnz);
343 Li[lnnz] = maxindex ;
344 Lx[lnnz] = 1.0 ;
345 lnnz++;
346
347 for ( i = top ; i < anrow ;i++ )
348 {
349 j = pattern[i] ;
350 t = pinv[j] ;
351
352 /* chk for numerical cancellation */
353 if ( X[j] != 0 )
354 {
355 if ( t != -1 )
356 {
357 if ( unnz >= uunnz )
358 {
359 printf ("basker : Insufficient memory for U %d %d \n", unnz, uunnz);
360 return 1;
361 }
362 /* BASKERASSERT(unnz < uunnz ) ; */
363 Ui[unnz] = pinv[j] ;
364 Ux[unnz] = X[j] ;
365 unnz++ ;
366 }
367 else if ( t == -1 )
368 {
369 if ( lnnz >= llnnz )
370 {
371 printf ("basker : Insufficient memory for L \n");
372 return 1;
373 }
374 BASKERASSERT(lnnz < llnnz ) ;
375 /*printf("I am assigning Li[%d]=%d Lx[%d]=%g t=%d, j =%d\n", lnnz, j, lnnz, X[j]/pivot, t, j) ;*/
376 Li[lnnz] = j ;
377 Lx[lnnz] = X[j]/pivot ;
378 lnnz++;
379 }
380 }
381 X[j] = 0 ;
382 }
383
384 xnnz = 0;
385 top = ancol ;
386
387 Lp[k] = cu_ltop ;
388 Lp[k+1] = lnnz ;
389 cu_ltop = lnnz ;
390
391 Up[k] = cu_utop ;
392 Up[k+1] = unnz ;
393 cu_utop = unnz ;
394
395 }
396
397#ifdef DEBUG
398 for (k = 0; k<lnnz ; k++)
399 {
400 printf("L[%d]=%g", k, Lx[k]);
401 }
402 printf("\n");
403
404 for (k = 0; k<lnnz ; k++)
405 {
406 printf("Li[%d]=%d", k, Li[k]);
407 }
408 printf("\n");
409#endif
410
411 *llnnz_p = llnnz;
412 *uunnz_p = uunnz;
413 *Li_p = Li ;
414 *Lx_p = Lx ;
415 *Ui_p = Ui ;
416 *Ux_p = Ux ;
417
418 return 0 ;
419}
