27 template <typename FieldType>
28 class MemoryBinder {
29 FieldType* ptr_;
30 public:
31 MemoryBinder(FieldType* f) : ptr_(f) {}
32 MemoryBinder(const MemoryBinder& ) = default;
33 MemoryBinder& operator=(const MemoryBinder& ) = default;
34 MemoryBinder(MemoryBinder&& ) = default;
35 MemoryBinder& operator=(MemoryBinder&& ) = default;
36
37 // Use SFINAE to select this for non-Kokkos::View (i.e. Field and MDField).
38 template<typename T=FieldType>
39 typename std::enable_if<!Kokkos::is_view<T>::value,void>::type
40 operator()(const std::any& f) { ptr_->setFieldData(f); }
41
42 // Use SFINAE to select this for Kokkos::View.
43 template<typename T=FieldType>
44 typename std::enable_if<Kokkos::is_view<T>::value,void>::type
45 operator()(const std::any& f)
46 {
47 // std::any object is always the non-const data type. To
48 // correctly cast the any object to the Kokkos::View, need to
49 // pull the const off the scalar type if this MDField has a
50 // const scalar type.
51 using non_const_view = Kokkos::View<typename FieldType::non_const_data_type,typename FieldType::array_layout,PHX::Device>;
52 try {
53 non_const_view tmp = std::any_cast<non_const_view>(f);
54 *ptr_ = tmp;
55 }
56 catch (std::exception& ) {
57 std::cout << "\n\nERROR in MemoryBinder using std::any_cast. Tried to cast a field "
58 << "to the type:\n \"" << Teuchos::demangleName(typeid(non_const_view).name())
59 << "\"\nfrom a std::any object containing a field of type:\n \""
60 << Teuchos::demangleName(f.type().name()) << "\"." << std::endl;
61 throw;
62 }
63 }
64 };
