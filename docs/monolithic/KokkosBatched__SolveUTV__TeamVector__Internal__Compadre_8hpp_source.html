28 // Attention!: This is a fork of TeamVectorSolveUTV_Internal from
29 // KokkosKernels that takes advantage of all non-square matrices
30 // having a RHS that is diagonal.
31 //
32 // This prevents having to actually form U'*B which would be very
33 // large if m >> n for the A matrix (result is m x m)

37 template<typename MemberType,
38 typename ValueType,
39 typename IntType>
40 KOKKOS_INLINE_FUNCTION
41 static int
42 invoke(const MemberType &member,
43 const int matrix_rank,
44 const int m, const int n, const int nrhs,
45 const ValueType * U, const int us0, const int us1,
46 const ValueType * T, const int ts0, const int ts1,
47 const ValueType * V, const int vs0, const int vs1,
48 const IntType * p, const int ps0,
49 /* */ ValueType * B, const int bs0, const int bs1,
50 /* */ ValueType * X, const int xs0, const int xs1,
51 /* */ ValueType * w, ValueType * wq,
52 const bool implicit_RHS) {
53
54 typedef ValueType value_type;
55
56 const value_type one(1), zero(0);
57
58 value_type * W = w;
59 value_type * WQ = wq;
60 const int ws0 = nrhs, ws1=1; // only works with w layout right
61
62 bool do_print = false;
63 if (do_print) {
64 Kokkos::single(Kokkos::PerTeam(member), [&] () {
65 using Kokkos::printf;
66 printf("size is: %d %d %d %d\n", matrix_rank, m, n, nrhs);
67 printf("U, us1, us0: %d %d\n", us1, us0);
68 printf("T, ts0, ts1: %d %d\n", ts0, ts1);
69 printf("B, bs0, bs1: %d %d\n", bs0, bs1);
70 printf("W, ws0, ws1: %d %d\n", ws0, ws1);
71 printf("B=zeros(%d,%d);\n", m, nrhs);
72 for (int i=0; i<m; ++i) {
73 for (int j=0; j<nrhs; ++j) {
74 printf("B(%d,%d)= %f;\n", i+1,j+1,B[i*bs0+j*bs1]);
75 }
76 }
77 });
78 }
79
80 if (matrix_rank < n) {
81 // U is m x matrix_rank
82 // T is matrix_rank x matrix_rank
83 // V is matrix_rank x n
84 // W = U^T B
85 if (!implicit_RHS) { // LU case
86 TeamVectorGemmInternal<Algo::Gemm::Unblocked>
87 ::invoke(member,
88 matrix_rank, nrhs, m,
89 one,
90 U, us1, us0,
91 B, bs0, bs1,
92 zero,
93 W, ws0, ws1);
94 } else {
95 Kokkos::parallel_for
96 (Kokkos::ThreadVectorRange(member,nrhs),
97 [&](const int &j) {
98 Kokkos::parallel_for
99 (Kokkos::TeamThreadRange(member,matrix_rank),
100 [&](const int &i) {
101 W[i*ws0+j*ws1] = U[j*us0+i*us1] * B[j];
102 });
103
104 });
105
106 }
107 member.team_barrier();
108
109 if (do_print) {
110 Kokkos::single(Kokkos::PerTeam(member), [&] () {
111 using Kokkos::printf;
112 printf("W=zeros(%d,%d);\n", m, nrhs);
113 for (int i=0; i<m; ++i) {

122 }
123 }
124 });
125 }

169 });
170 }
171 } else {
172 if (!implicit_RHS) { // LU case
174 TeamVectorGemmInternal<Algo::Gemm::Unblocked>
175 ::invoke(member,
176 matrix_rank, nrhs, m,
177 one,
178 U, us1, us0,

273 }
274 }
275 });
276 }
277
278 return 0;
279 }
