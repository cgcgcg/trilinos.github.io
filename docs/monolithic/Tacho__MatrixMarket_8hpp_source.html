113 template <typename DeviceType>
114 static int read(const std::string &filename, CrsMatrixBase<ValueType, DeviceType> &A,
115 const ordinal_type mm_base = 1, const ordinal_type sanitize = 0, const ordinal_type verbose = 0) {
116 static_assert(Kokkos::Impl::MemorySpaceAccess<Kokkos::HostSpace, typename DeviceType::memory_space>::assignable,
117 "DeviceType is not assignable from HostSpace");
118
119 Kokkos::Timer timer;
120
121 timer.reset();
122
123 std::ifstream file;
124 file.open(filename);
125 if (file.good()) {
126 std::cout << "Read matrix from " << filename << std::endl;
127 } else {
128 std::cout << std::endl
129 << "Failed to open the matrix file: " << filename
130 << std::endl << std::endl;
131 return -1;
132 }
133
134 // reading mm header
135 ordinal_type m, n;
136 size_type nnz, nnz_input;
137 bool symmetry = false, hermitian = false, cmplx = false;
138 {
139 std::string header;
140 std::getline(file, header);
141 while (file.good()) {
142 char c = file.peek();
143 if (c == '%' || c == '\n') {
144 file.ignore(256, '\n');
145 continue;
146 }
147 break;
148 }
149 std::transform(header.begin(), header.end(), header.begin(), ::tolower);
150 symmetry = (header.find("symmetric") != std::string::npos || header.find("hermitian") != std::string::npos);
151
152 hermitian = (header.find("hermitian") != std::string::npos);
153
154 cmplx = (header.find("complex") != std::string::npos);
155
156 file >> m >> n >> nnz;
157 }
158
159 // read data into coo format
160 typedef ValueType value_type;
161 typedef Coo<value_type> ijv_type;
162 std::vector<ijv_type> mm;
163 {
164 std::vector<ijv_type> mm_org;
165 mm_org.reserve(nnz * (symmetry ? 2 : 1));
166 for (size_type i = 0; i < nnz; ++i) {
167 ordinal_type row, col;
168 value_type val;
169
170 if (file.eof()) {
171 std::cout << " ERROR: Reached the end of file before nnz (invalid nnz?)" << std::endl << std::endl;
172 return -1;
173 }
174 impl_read_value_from_file(file, cmplx, row, col, val);
175
176 row -= mm_base;
177 col -= mm_base;
178 mm_org.push_back(ijv_type(row, col, val));
179
180 if (symmetry && row != col) {
181 value_type conj_val;
182 impl_conj_val(val, conj_val);
183 mm_org.push_back(ijv_type(col, row, hermitian ? conj_val : val));
184 }
185 }
186 std::sort(mm_org.begin(), mm_org.end(), std::less<ijv_type>());
187
188 // update nnz (this is the nnz from input matrix)
189 nnz = mm_org.size();
190 nnz_input = nnz;
191
192 // copy to mm
193 mm.reserve(nnz);
194 if (sanitize) {
195 for (size_type i = 0; i < nnz; ++i) {
196 bool is_zero;
197 impl_is_zero(mm_org[i].val, is_zero);
198 if (!is_zero)
199 mm.push_back(mm_org[i]);
200 }
201 nnz = mm.size();
202 } else {
203 for (size_type i = 0; i < nnz; ++i)
204 mm.push_back(mm_org[i]);
205 nnz = mm.size();
206 }
207 }
208
209 // change mm to crs
210 Kokkos::View<size_type *, DeviceType> ap("ap", m + 1);
211 Kokkos::View<ordinal_type *, DeviceType> aj("aj", nnz);
212 Kokkos::View<value_type *, DeviceType> ax("ax", nnz);
213 {
214 ordinal_type icnt = 0;
215 size_type jcnt = 0;
216 ijv_type prev = mm[0];
217
218 ap[icnt++] = 0;
219 aj[jcnt] = prev.j;
220 ax[jcnt++] = prev.val;
221
222 for (auto it = (mm.begin() + 1); it < (mm.end()); ++it) {
223 const ijv_type aij = (*it);
224
225 if (aij.i != prev.i)
226 ap[icnt++] = jcnt;
227
228 if (aij == prev) {
229 aj[jcnt] = aij.j;
230 ax[jcnt] += aij.val;
231 } else {
232 aj[jcnt] = aij.j;
233 ax[jcnt++] = aij.val;
234 }
235 prev = aij;
236 }
237 ap[icnt++] = jcnt;
238 nnz = jcnt;
239 }
240
241 // create crs matrix view
242 A.clear();
243 A.setExternalMatrix(m, n, nnz, ap, aj, ax);
244
245 const double t = timer.seconds();
246 if (verbose) {
247 printf("Summary: MatrixMarket\n");
248 printf("=====================\n");
249 printf(" File: %s\n", filename.c_str());
250 printf(" Time\n");
251 printf(" time for reading A: %10.6f s\n", t);
252 printf("\n");
253 printf(" Sparse Matrix (%s) \n", (symmetry ? "symmetric" : "non-symmetric"));
254 printf(" number of rows: %10d\n", m);
255 printf(" number of cols: %10d\n", n);
256 printf(" number of nonzeros from input: %10d\n", ordinal_type(nnz_input));
257 printf(" number of nonzeros after sanitized: %10d\n", ordinal_type(nnz));
258 printf("\n");
259 }
260
261 return 0;
262 }

265 template <typename DeviceType>
266 static void write(std::ofstream &file, const CrsMatrixBase<ValueType, DeviceType> &A,
267 const int uplo = 0, // 0 - all, 1 - upper, 2 - lower
268 const std::string comment = "%% Tacho::MatrixMarket::Export") {
269 static_assert(Kokkos::Impl::MemorySpaceAccess<Kokkos::HostSpace, typename DeviceType::memory_space>::assignable,
270 "DeviceType is not assignable from HostSpace");
271
272 typedef ValueType value_type;
273 constexpr bool is_complex = (std::is_same<value_type, Kokkos::complex<float>>::value ||
274 std::is_same<value_type, Kokkos::complex<double>>::value);
275
276 std::streamsize prec = file.precision();
277 file.precision(16);
278 file << std::scientific;
279
280 {
281 file << "%%MatrixMarket matrix coordinate " << (is_complex ? "complex " : "real ") << std::endl;
282 file << comment << std::endl;
283 }
284 // cnt nnz
285 size_type nnz = 0;
286 {
287 for (ordinal_type i = 0; i < A.NumRows(); ++i) {
288 const size_type jbegin = A.RowPtrBegin(i), jend = A.RowPtrEnd(i);
289 for (size_type j = jbegin; j < jend; ++j) {
290 const auto aj = A.Col(j);
291 if (uplo == 1 && i <= aj)
292 ++nnz;
293 if (uplo == 2 && i >= aj)
294 ++nnz;
295 if (uplo == 0)
296 ++nnz;
297 }
298 }
299 file << A.NumRows() << " " << A.NumCols() << " " << nnz << std::endl;
300 }
301
302 const int w = 10;
303 {
304 for (ordinal_type i = 0; i < A.NumRows(); ++i) {
305 const size_type jbegin = A.RowPtrBegin(i), jend = A.RowPtrEnd(i);
306 for (size_type j = jbegin; j < jend; ++j) {
307 const auto aj = A.Col(j);
308 bool flag = false;
309 if (uplo == 1 && i <= aj)
310 flag = true;
311 if (uplo == 2 && i >= aj)
312 flag = true;
313 if (uplo == 0)
314 flag = true;
315 if (flag) {
316 value_type val = A.Value(j);
317 impl_write_value_to_file(file, i + 1, aj + 1, val, w);
318 }
319 }
320 }
321 }
322
323 file.unsetf(std::ios::scientific);
324 file.precision(prec);
325 }

342 {

346 char c = file.peek();

354 if ( m != ordinal_type(B.extent(0))) {
355 std::cout << std::endl

357 << std::endl << std::endl;
358 return -1;
359 }
360 Kokkos::resize(B, m, n);
361 }
362
363 // reading numerical values
364 ValueType val;
365 auto hB = Kokkos::create_mirror_view(B);
366 for (ordinal_type j = 0; j < n; j++) {
367 for (ordinal_type i = 0; i < m; i++) {
368 file >> val;
369 hB(i,j) = val;
370 }
371 }
372 Kokkos::deep_copy(B, hB);
373 if (verbose) {
374 printf("Summary: MatrixMarket\n");
375 printf("=====================\n");
376 printf(" File: %s\n", filename.c_str());
377 printf(" number of rows: %10d\n", m);
378 printf(" number of cols: %10d\n", n);
379 printf("\n");
380 }
