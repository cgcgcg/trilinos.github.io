37void
38Stokhos::MPBlockDiagonalPreconditioner::
39setupPreconditioner(const Teuchos::RCP<Stokhos::BlockDiagonalOperator>& mp_op,
40 const Epetra_Vector& x)
41{
42 TEUCHOS_TEST_FOR_EXCEPTION(prec_factory == Teuchos::null, std::logic_error,
43 "Error! setupPreconditioner() cannot be called when " <<
44 "prec_factory is null!" << std::endl);
45
46 Teuchos::RCP<Stokhos::ProductContainer<Epetra_Operator> > mp_ops =
47 mp_op->getMPOps();
48 for (int i=0; i<num_mp_blocks; i++)
49 block_precs[i] = prec_factory->compute(mp_ops->getCoeffPtr(i));
50 if (num_mp_blocks > 0) {
51 label = std::string("Stokhos MP Block Diagonal Preconditioner:\n") +
52 std::string(" ***** ") +
53 std::string(block_precs[0]->Label());
54 }
55}

68int
69Stokhos::MPBlockDiagonalPreconditioner::
70Apply(const Epetra_MultiVector& Input, Epetra_MultiVector& Result) const
71{
72 EpetraExt::BlockMultiVector mp_input(View, *base_map, Input);
73 EpetraExt::BlockMultiVector mp_result(View, *base_map, Result);
74 for (int i=0; i<num_mp_blocks; i++) {
75 block_precs[i]->Apply(*(mp_input.GetBlock(i)), *(mp_result.GetBlock(i)));
76 }
77
78 return 0;
79}

95double
96Stokhos::MPBlockDiagonalPreconditioner::
97NormInf() const
98{
99 double product_nrm = 0.0;
100 for (int i=0; i<num_mp_blocks; i++) {
101 double nrm = block_precs[i]->NormInf();
102 if (nrm > product_nrm)
103 product_nrm = nrm;
104 }
105
106 return product_nrm;
107}
