85 *----------------------------------------------------------------------*/
86bool NLNML::NLNML_CoarseLevelNoxInterface::computeF(
87 const Epetra_Vector& x, Epetra_Vector& F,
88 const FillType fillFlag)
89{
90 bool err;
91 if (!Level())
92 {
93 err = fineinterface_->computeF(x,F,fillFlag);
94 if (err==false)
95 {
96 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::computeF:\n"
97 << "**ERR**: call to fine-userinterface returned false on level " << level_ << "\n"
98 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
99 }
100 }
101 else
102 {
103 RefCountPtr<Epetra_Vector> Ffine = rcp(new Epetra_Vector(fineinterface_->getGraph()->RowMap(),false));
104 RefCountPtr<Epetra_Vector> xfine = rcp(prolong_this_to_fine(x));
105 err = fineinterface_->computeF(*xfine,*Ffine,fillFlag);
106 if (err==false)
107 {
108 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::computeF:\n"
109 << "**ERR**: call to fine-userinterface returned false on level " << level_ << "\n"
110 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
111 }
112 RefCountPtr<Epetra_Vector> Fcoarse = rcp(restrict_fine_to_this(*Ffine));
113 F.Scale(1.0,*Fcoarse);
114 }
115 if (isFAS())
116 F.Update(-1.0,*fxbar_,1.0,*fbar_,1.0);
117 return err;
118}

120/*----------------------------------------------------------------------*
121 | restrict from fine to this level (public) m.gee 3/06|
122 *----------------------------------------------------------------------*/
123Epetra_Vector* NLNML::NLNML_CoarseLevelNoxInterface::restrict_fine_to_this(
124 const Epetra_Vector& xfine)
125{
126 if (!Level())
127 {
128 Epetra_Vector* xcoarse = new Epetra_Vector(xfine);
129 return xcoarse;
130 }
131 else
132 {
133 // Note that the GIDs in xfine match those of the fineinterface and
134 // might be different from those in P_[1]->OperatorRangeMap().
135 // The LIDs and the map match, so we have to copy xfine to xfineP
136 // using LIDs
137 Epetra_Vector* xfineP = wvec_[0].get(); // RangeMap of P_[1]
138 if (xfine.MyLength() != xfineP->MyLength() || xfine.GlobalLength() != xfineP->GlobalLength())
139 {
140 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::restrict_fine_to_this:\n"
141 << "**ERR**: mismatch in dimension of xfine and xfineP\n"
142 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
143 }
144 const int mylength = xfine.MyLength();
145 for (int i=0; i<mylength; i++)
146 (*xfineP)[i] = xfine[i];
147
148 // loop from the finest level to this level and
149 // apply series of restrictions (that is transposed prolongations)
150 Epetra_Vector* fvec = xfineP;
151 for (int i=0; i<Level()-1; i++)
152 {
153 Epetra_Vector* cvec = wvec_[i+1].get();
154 (*P_)[i+1]->Multiply(true,*fvec,*cvec);
155 fvec = cvec;
156 }
157 Epetra_Vector* out = new Epetra_Vector((*P_)[Level()]->OperatorDomainMap(),false);
158 (*P_)[Level()]->Multiply(true,*fvec,*out);
159 return out;
160 }
161}

163/*----------------------------------------------------------------------*
164 | prolongate from this level to fine (public) m.gee 3/06|
165 *----------------------------------------------------------------------*/
166Epetra_Vector* NLNML::NLNML_CoarseLevelNoxInterface::prolong_this_to_fine(
167 const Epetra_Vector& xcoarse)
168{
169 if (!Level())
170 return new Epetra_Vector(xcoarse);
171 else
172 {
173 Epetra_Vector* cvec = const_cast<Epetra_Vector*>(&xcoarse);
174 for (int i=Level(); i>0; i--)
175 {
176 // allocate a vector matching level i-1
177 Epetra_Vector* fvec = wvec_[i-1].get();
178 // multiply
179 (*P_)[i]->Multiply(false,*cvec,*fvec);
180 cvec = fvec;
181 }
182 // Note that the GIDs in cvec do NOT match those of the fineinterface as
183 // they match the P_[1]->RangeMap.
184 // The LIDs match, so we have to copy cvec to xfine_fineinterface
185 // using LIDs
186 Epetra_Vector* xfine_fineinterface = new Epetra_Vector(fineinterface_->getGraph()->RowMap(),false);
187 if (cvec->MyLength() != xfine_fineinterface->MyLength() ||
188 cvec->GlobalLength() != xfine_fineinterface->GlobalLength())
189 {
190 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::prolong_this_to_fine:\n"
191 << "**ERR**: mismatch in dimension of cvec and xfine_fineinterface\n"
192 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
193 }
194 const int mylength = cvec->MyLength();
195 for (int i=0; i<mylength; i++)
196 (*xfine_fineinterface)[i] = (*cvec)[i];
197 return xfine_fineinterface;
198 }
199}

201/*----------------------------------------------------------------------*
202 | restrict from this to next coarser level (public) m.gee 3/06|
203 *----------------------------------------------------------------------*/
204Epetra_Vector* NLNML::NLNML_CoarseLevelNoxInterface::restrict_to_next_coarser_level(
205 Epetra_Vector* thisvec,
206 int current, int next)
207{
208 Epetra_Vector* xfineP = 0;
209 if (current==0)
210 {
211 xfineP = new Epetra_Vector((*P_)[1]->RowMap(),false);
212 if (thisvec->MyLength() != xfineP->MyLength() || thisvec->GlobalLength() != xfineP->GlobalLength())
213 {
214 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::restrict_to_next_coarser_level:\n"
215 << "**ERR**: mismatch in dimension of thisvec and xfineP\n"
216 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
217 }
218 const int mylength = thisvec->MyLength();
219 for (int i=0; i<mylength; i++)
220 (*xfineP)[i] = (*thisvec)[i];
221 }
222 else
223 {
224 xfineP = thisvec;
225 }
226 Epetra_Vector* cvec = new Epetra_Vector((*P_)[next]->OperatorDomainMap(),false);
227 (*P_)[next]->Multiply(true,*xfineP,*cvec);
228 if (current==0) delete xfineP;
229 return cvec;
230}

233/*----------------------------------------------------------------------*
234 | prolongate from next coarser level to this level (public) m.gee 3/06|
235 *----------------------------------------------------------------------*/
236Epetra_Vector* NLNML::NLNML_CoarseLevelNoxInterface::prolong_to_this_level(
237 Epetra_Vector* coarsevec,
238 int current, int next)
239{
240 Epetra_Vector* fvec = new Epetra_Vector((*P_)[next]->OperatorRangeMap(),false);
241 (*P_)[next]->Multiply(false,*coarsevec,*fvec);
242 if (current==0)
243 {
244 Epetra_Vector* fine = new Epetra_Vector(fineinterface_->getMap(),false);
245 if (fvec->MyLength() != fine->MyLength() || fvec->GlobalLength() != fine->GlobalLength())
246 {
247 cout << "**ERR**: NLNML::NLNML_CoarseLevelNoxInterface::prolong_to_this_level:\n"
248 << "**ERR**: mismatch in dimension of fvec and fine\n"
249 << "**ERR**: file/line: " << __FILE__ << "/" << __LINE__ << "\n"; throw -1;
250 }
251 const int mylength = fvec->MyLength();
252 for (int i=0; i<mylength; ++i)
253 (*fine)[i] = (*fvec)[i];
254 delete fvec;
255 return fine;
256 }
257 else
258 return fvec;
259}

270/*----------------------------------------------------------------------*
271 | make application apply all constraints (public) m.gee 3/06|
272 *----------------------------------------------------------------------*/
273void NLNML::NLNML_CoarseLevelNoxInterface::ApplyAllConstraints(
274 Epetra_Vector& gradient)
275{
276 if (!Level())
277 fineinterface_->ApplyAllConstraints(gradient,0);
278 else
279 {
280 RefCountPtr<Epetra_Vector> gradientfine = rcp(prolong_this_to_fine(gradient));
281 fineinterface_->ApplyAllConstraints(*gradientfine,Level());
282 RefCountPtr<Epetra_Vector> gradientcoarse = rcp(restrict_fine_to_this(*gradientfine));
283 gradient.Scale(1.0,*gradientcoarse);
284 }
285 return;
286}
