22template<typename Traits, typename FactoryTraits>
23Teuchos::RCP< std::vector< Teuchos::RCP<PHX::Evaluator_TemplateManager<Traits> > > >
24PHX::EvaluatorFactory<Traits, FactoryTraits>::
25buildEvaluators(const std::map<std::string, Teuchos::RCP<Teuchos::ParameterList> >& data)
26{
27 Teuchos::RCP< std::vector< Teuchos::RCP<PHX::Evaluator_TemplateManager<Traits> > > > vector_tm =
28 Teuchos::rcp(new std::vector< Teuchos::RCP<Evaluator_TemplateManager<Traits> > >);
29
30 std::map<std::string, Teuchos::RCP<Teuchos::ParameterList> >::const_iterator
31 it = data.begin();
32
33 for (; it != data.end(); ++it) {
34
35 Teuchos::RCP< PHX::Evaluator_TemplateManager<Traits> > tm =
36 Teuchos::rcp(new PHX::Evaluator_TemplateManager<Traits>);
37
38 Teuchos::RCP<Teuchos::ParameterList> p = it->second;
39
40 bool found_type = false;
41 int object_type = p->get<int>("Type");
42 static const int size = Sacado::mpl::size<typename FactoryTraits::EvaluatorTypes>::value;
43 Sacado::mpl::for_each_no_kokkos< Sacado::mpl::range_c<int,0,size> >( UFO<Traits,FactoryTraits>(object_type, p, tm, found_type) );
44
45 if (!found_type) {
46 std::ostringstream msg;
47 msg << "Unable to find model in EvaluatorFactory for "
48 << it->first
49 << ". Please make sure you have set a valid integer "
50 << "for \"Type\" in the parameter list!";
51 TEUCHOS_TEST_FOR_EXCEPTION(!found_type, std::logic_error, msg.str());
52 }
53
54 vector_tm->push_back(tm);
55
56 }
57
58 return vector_tm;
59
60}

62template<typename Traits>
63void PHX::registerEvaluators(const Teuchos::RCP< std::vector< Teuchos::RCP<PHX::Evaluator_TemplateManager<Traits> > > >& providers, PHX::FieldManager<Traits>& fm)
64{
65 // Loop over each provider template manager
66 typename std::vector< Teuchos::RCP<Evaluator_TemplateManager<Traits> > >::iterator
67 tm = providers->begin();
68 for (; tm != providers->end(); ++tm) {
69
70 // Loop over Evaluation Types
71 typename PHX::FieldManager<Traits>::iterator vmit = fm.begin();
72 typename Evaluator_TemplateManager<Traits>::iterator vpit =
73 (*tm)->begin();
74 for (; vpit != (*tm)->end(); ++vpit) {
75 Teuchos::RCP<PHX::Evaluator<Traits> > vp =
76 Teuchos::rcp_dynamic_cast<PHX::Evaluator<Traits> >(vpit.rcp());
77 fm.registerEvaluator(vmit, vp);
78 ++vmit;
79 }
80 }

82}
