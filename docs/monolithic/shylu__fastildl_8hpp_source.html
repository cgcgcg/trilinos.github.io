189 void initialize()
190 {
191 Kokkos::Timer timer;
192 //Here we have to allocate memory for L, U.
193 if((level > 0) && (guessFlag != 0))
194 {
195 initGuessPrec->initialize();
196 }
197
198 //symbolicILUAndInit(); //Note the diagonal scaling is now embedded in this routine.
199 symbolicILU();
200 #ifdef SHYLU_DEBUG
201 //initialize L, U, A patterns
202 MemoryPrimeFunctorN<Ordinal, Scalar, ExecSpace> copyFunc1(aRowMap, lRowMap, lRowMap, diagElems);
203
204 //Make sure all memory resides on the device
205 ExecSpace().fence();
206 Kokkos::parallel_for(RangePolicy(0, nRows), copyFunc1);
207
208 //Note that the following is a temporary measure
209 //to ensure that memory resides on the device.
210 Ordinal nnzL = lRowMap[nRows];
211 Ordinal nnzA = aRowMap[nRows];
212 MemoryPrimeFunctorNnzCoo<Ordinal, Scalar, ExecSpace> copyFunc2(aColIdx, aRowIdx, aVal);
213 MemoryPrimeFunctorNnzCsr<Ordinal, Scalar, ExecSpace> copyFunc3(lColIdx, lVal);
214
215 ExecSpace().fence();
216 Kokkos::parallel_for(RangePolicy(0, nRows), copyFunc1);
217 Kokkos::parallel_for(RangePolicy(0, nnzA), copyFunc2);
218 Kokkos::parallel_for(RangePolicy(0, nnzL), copyFunc3);

220 double t = timer.seconds();
221 #ifdef FASTILDL_DEBUG_OUTPUT
222 std::cout << "Symbolic phase complete." << std::endl;
223 std::cout << "Init time: "<< t << "s" << std::endl;

242 void transposeL()
243 {
244 auto ltRowMap_ = Kokkos::create_mirror_view(ltRowMap);
245 auto ltColIdx_ = Kokkos::create_mirror_view(ltColIdx);
246 auto ltVal_ = Kokkos::create_mirror_view(ltVal);
247
248 //Count the elements in each row of Lt
249 auto temp = OrdinalArrayHost("temp", nRows + 1);
250 auto rowPtrs = OrdinalArrayHost("rowPtrs", nRows);
251 for (Ordinal i = 0; i <= nRows; i++)
252 {
253 temp[i] = 0;
254 }
255 for (Ordinal i = 0; i < nRows; i++)
256 {
257 for (Ordinal k = lRowMap_[i]; k < lRowMap_[i+1]; k++)
258 {
259 temp[lColIdx_[k]+1]++;
260
261 }
262 }
263 //Perform an add scan to get the row map for
264 //the transpose
265 for (Ordinal i = 0; i <= nRows; i++)
266 {
267 ltRowMap_[i] = temp[i];
268 }
269 for (Ordinal i = 1; i <= nRows; i++)
270 {
271 ltRowMap_[i] += ltRowMap_[i-1];
272 }
273 //Set the row pointers to their initial places;
274 for (Ordinal i = 0; i < nRows; i++)
275 {
276 rowPtrs[i] = ltRowMap_[i];
277 }
278 //Copy the data
279 Kokkos::deep_copy(lVal_, lVal);
280 for (Ordinal i = 0; i < nRows; i++)
281 {
282 for (Ordinal k = lRowMap_[i]; k < lRowMap_[i+1]; k++)
283 {
284 Ordinal row = lColIdx_[k];
285 Scalar value = lVal_[k];
286 ltVal_[rowPtrs[row]] = value;
287 ltColIdx_[rowPtrs[row]] = i;
288 rowPtrs[row]++;
289 assert(rowPtrs[row] <= ltRowMap_[row + 1]);
290 }
291 }
292 Kokkos::deep_copy(ltRowMap, ltRowMap_);
293 Kokkos::deep_copy(ltColIdx, ltColIdx_);
294 Kokkos::deep_copy(ltVal, ltVal_);
295 }

297 void symbolicILU()
298 {
299 using std::vector;
300 using std::cout;
301 using std::stable_sort;
302 using std::sort;
303 OrdinalArrayMirror ia = aRowMapHost;
304 OrdinalArrayMirror ja = aColIdxHost;
305 int n = nRows;
306 int nzl = aRowMapHost[nRows] * (level + 2);
307 int nzu = aRowMapHost[nRows] * (level + 2);
308 Ordinal i;
309 int levfill = level;
310 vector<int> lnklst(n);
311 vector<int> curlev(n);
312 vector<int> levels(nzu);
313 vector<int> iwork(n);
314 vector<int> ial(nzl);
315 vector<int> jal(nzl);
316 vector<int> iau(nzu);
317 vector<int> jau(nzu);
318
319 int knzl = 0;
320 int knzu = 0;
321
322 ial[0] = 0;
323 iau[0] = 0;
324
325 for (i=0; i<n; i++)
326 {
327 int first, next, j;
328
329 /* copy column indices of row into workspace and sort them */
330
331 int len = ia[i+1] - ia[i];
332 next = 0;
333 for (j=ia[i]; j<ia[i+1]; j++)
334 iwork[next++] = ja[j];
335 // shell_sort(len, iwork);
336 //stable_sort(iwork.begin(), iwork.begin() + len);
337 sort(iwork.begin(), iwork.begin() + len);
338
339 /* construct implied linked list for row */
340
341 first = iwork[0];
342 curlev[first] = 0;
343
344 for (j=0; j<=len-2; j++)
345 {
346 lnklst[iwork[j]] = iwork[j+1];
347 curlev[iwork[j]] = 0;
348 }
349
350 lnklst[iwork[len-1]] = n;
351 curlev[iwork[len-1]] = 0;
352
353 /* merge with rows in U */
354
355 next = first;
356 while (next < i)
357 {
358 int oldlst = next;
359 int nxtlst = lnklst[next];
360 int row = next;
361 int ii;
362
363 /* scan row */
364
365 for (ii=iau[row]+1; ii<iau[row+1]; /*nop*/)
366 {
367 if (jau[ii] < nxtlst)
368 {
369 /* new fill-in */
370 int newlev = curlev[row] + levels[ii] + 1;
371 if (newlev <= levfill)
372 {
373 lnklst[oldlst] = jau[ii];
374 lnklst[jau[ii]] = nxtlst;
375 oldlst = jau[ii];
376 curlev[jau[ii]] = newlev;
377 }
378 ii++;
379 }
380 else if (jau[ii] == nxtlst)
381 {
382 int newlev;
383 oldlst = nxtlst;
384 nxtlst = lnklst[oldlst];
385 newlev = curlev[row] + levels[ii] + 1;
386 //curlev[jau[ii]] = MIN(curlev[jau[ii]], newlev);
387 if (curlev[jau[ii]] > newlev)
388 {
389 curlev[jau[ii]] = newlev;
390 }
391 ii++;
392 }
393 else /* (jau[ii] > nxtlst) */
394 {
395 oldlst = nxtlst;
396 nxtlst = lnklst[oldlst];
397 }
398 }
399 next = lnklst[next];
400 }
401
402 /* gather the pattern into L and U */
403
404 next = first;
405 while (next < i)
406 {
407 assert(knzl < nzl);
408 jal[knzl++] = next;
409 if (knzl >= nzl)
410 {
411 //(*nzl)++;
412 nzl += nRows;
413 jal.resize(nzl);
414 }
415 //jal.push_back(next);
416 // knzl++;
417 next = lnklst[next];
418 }
419 ial[i+1] = knzl;
420 assert(next == i);
421
422 while (next < n)
423 {
424 assert(knzu < nzu);
425 levels[knzu] = curlev[next];
426 jau[knzu++] = next;
427 if (knzu >= nzu)
428 {
429 //(*nzu)++;
430 nzu += nzu + nRows;
431 jau.resize(nzu);
432 levels.resize(nzu);
433 }
434 //jau.push_back(next);
435 //knzu++;
436 next = lnklst[next];
437 }
438 iau[i+1] = knzu;
439 }
440
441 #ifdef FASTILDL_DEBUG_OUTPUT
442 std::cout << "knzl =" << knzl;
443 std::cout << "knzu =" << knzu;
444 #endif
445
446 nzl = knzl;
447 nzu = knzu;
448
449 #ifdef FASTILDL_DEBUG_OUTPUT
450 std::cout << "ILU: nnz = "<< knzl + knzu << std::endl;
451 std::cout << "Actual nnz for ILU: " << nzl + nzu << std::endl;
452 #endif
453 //Initialize the A matrix that is to be used in the computation
454 aRowMap = OrdinalArray("aRowMap", nRows + 1);
455 aColIdx = OrdinalArray("aColIdx", knzl + knzu);
456 aRowIdx = OrdinalArray("aRowIds", knzl + knzu);
457 aRowMap_ = Kokkos::create_mirror_view(aRowMap);
458 aColIdx_ = Kokkos::create_mirror_view(aColIdx);
459 aRowIdx_ = Kokkos::create_mirror_view(aRowIdx);
460
461 Ordinal aRowPtr = 0;
462 aRowMap_[0] = 0;
463 for (i = 0; i < nRows; i++)
464 {
465 #ifdef FASTILDL_DEBUG_OUTPUT
466 std::cout << "***row:" << i << std::endl;
467 #endif
468 for(Ordinal k = ial[i]; k < ial[i+1]; k++)
469 {
470 #ifdef FASTILDL_DEBUG_OUTPUT
471 std::cout << "jal[k]=" << jal[k] << std::endl;
472 #endif
473 aColIdx_[aRowPtr] = jal[k];
474 aRowIdx_[aRowPtr] = i;
475 aRowPtr++;
476 }
477 for(Ordinal k = iau[i]; k < iau[i+1]; k++)
478 {
479 aColIdx_[aRowPtr] = jau[k];
480 aRowIdx_[aRowPtr] = i;
481 aRowPtr++;
482 }
483 aRowMap_[i+1] = aRowPtr;
484 }
485 Kokkos::deep_copy(aRowMap, aRowMap_);
486 Kokkos::deep_copy(aColIdx, aColIdx_);
487 Kokkos::deep_copy(aRowIdx, aRowIdx_);
488
489 //Now allocate memory for L and U.
490 //
491 lRowMap = OrdinalArray("lRowMap", nRows + 1);
492 ltRowMap = OrdinalArray("ltRowMap", nRows + 1);
493 countL();
494 //Allocate memory and initialize pattern for L, U (transpose).
495 lColIdx = OrdinalArray("lColIdx", lRowMap_[nRows]);
496 ltColIdx = OrdinalArray("ltColIdx", lRowMap_[nRows]);
497 #ifdef FASTILDL_DEBUG_OUTPUT
498 std::cout << "nnz L = " << lRowMap[nRows] << std::endl;
499 #endif
500 }

630 void applyDiagonalScaling()
631 {
632 int anext = 0;
633#if KOKKOS_VERSION >= 40799
634 const Real one = KokkosKernels::ArithTraits<Real>::one();
635#else
636 const Real one = Kokkos::ArithTraits<Real>::one();
637#endif
638 //First fill Aj and extract the diagonal scaling factors
639 //Use diag array to store scaling factors since
640 //it gets set to the correct value by findFactorPattern anyway.
641 auto diagFact_ = Kokkos::create_mirror_view(diagFact);
642 for (int i = 0; i < nRows; i++)
643 {
644 for(int k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
645 {
646 aRowIdx_[anext++] = i;
647 if (aColIdx_[k] == i)
648 {
649 diagFact_[i] = one/(RTS::sqrt(STS::abs(aVal_[k])));
650 #ifdef FASTILDL_DEBUG_OUTPUT
651 std::cout << "diagFact["<<i<<"]="<<aVal[k]<<std::endl;
652 #endif
653 }
654 }
655 }
656
657 //Now go through each element of A and apply the scaling
658 int row;
659 int col;
660 Real sc1, sc2;
661 for (int i = 0; i < nRows; i++)
662 {
663 for (int k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
664 {
665 row = aRowIdx_[k];
666 col = aColIdx_[k];
667
668 sc1 = diagFact_[row];
669 sc2 = diagFact_[col];
670 aVal_[k] = aVal_[k]*sc1*sc2;
671 }
672 }
673 Kokkos::deep_copy(diagFact, diagFact_);
674 }

676 void applyManteuffelShift()
677 {
678#if KOKKOS_VERSION >= 40799
679 const Scalar one = KokkosKernels::ArithTraits<Scalar>::one();
680#else
681 const Scalar one = Kokkos::ArithTraits<Scalar>::one();
682#endif
683 //Scalar shift = 0.05;
684 for (Ordinal i = 0; i < nRows; i++)
685 {
686 for (Ordinal k = aRowMap_[i]; k < aRowMap_[i+1]; k++)
687 {
688 Ordinal row = i;
689 Ordinal col = aColIdx_[k];
690 if (row != col)
691 {
692 aVal_[k] = (one/(one + shift))*aVal_[k];
693 }
694 }
695 }
