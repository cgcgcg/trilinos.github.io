42#if defined(TACHO_HAVE_METIS)

74 if constexpr (std::is_same_v<ordering_type, long>) {

92void GraphTools_Metis::reorder(const ordinal_type verbose) {
93 Kokkos::Timer timer;
94 double t_metis = 0;
95
96 if (_algo == 0) {
97 for (ordinal_type i = 0; i < _nvts; ++i) {
98 _perm(i) = i;
99 _peri(i) = i;
100 }
101 } else if (_algo == 1) {
102 int ierr = 0;
103 double amd_info[TRILINOS_AMD_INFO];
104
105 timer.reset();
106 ierr = GraphTools_Metis::amd_order(_nvts, _xadj.data(), _adjncy.data(), _perm_t.data(), NULL, amd_info);
107 t_metis = timer.seconds();
108
109 for (idx_t i = 0; i < _nvts; ++i) {
110 _perm(i) = _perm_t(i);
111 _peri(_perm(i)) = i;
112 }
113
114 // ierr != TRILINOS_AMD_OK && ierr != TRILINOS_AMD_OK_BUT_JUMBLED
115 TACHO_TEST_FOR_EXCEPTION(ierr < TRILINOS_AMD_OK, std::runtime_error, "Failed in trilinos_amd");
116 } else {
117#if defined(TACHO_HAVE_METIS)
118 int ierr = 0;
119
120 idx_t *xadj = (idx_t *)_xadj.data();
121 idx_t *adjncy = (idx_t *)_adjncy.data();
122 idx_t *vwgt = (idx_t *)_vwgt.data();
123
124 idx_t *perm = (idx_t *)_perm_t.data();
125 idx_t *peri = (idx_t *)_peri_t.data();
126
127 timer.reset();
128 ierr = METIS_NodeND(&_nvts, xadj, adjncy, vwgt, _options, perm, peri);
129 t_metis = timer.seconds();
130
131 for (idx_t i = 0; i < _nvts; ++i) {
132 _perm(i) = _perm_t(i);
133 _peri(i) = _peri_t(i);
134 }
135
136 TACHO_TEST_FOR_EXCEPTION(ierr != METIS_OK, std::runtime_error, "Failed in METIS_NodeND");
137#else
138 TACHO_TEST_FOR_EXCEPTION(true, std::runtime_error, "METIS is not enabled");
139#endif
140 }
141 _is_ordered = true;
142
143 if (verbose) {
144 if (_algo == 0) {
145 printf("Summary: GraphTools (Natural)\n");
146 printf("=============================\n");
147 } else if (_algo == 1) {
148 printf("Summary: GraphTools (AMD)\n");
149 printf("=========================\n");
150 } else {
151 printf("Summary: GraphTools (Metis)\n");
152 printf("===========================\n");
153 }
154
155 switch (verbose) {
156 case 1: {
157 printf(" Time\n");
158 printf(" time for reordering: %10.6f s\n", t_metis);
159 printf("\n");
160 }
161 }
162 }
163}
