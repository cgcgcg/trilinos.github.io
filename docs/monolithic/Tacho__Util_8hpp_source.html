150template <typename MemberType>
151KOKKOS_FORCEINLINE_FUNCTION static void clear(MemberType &member, char *buf, size_type bufsize) {
152 KOKKOS_IF_ON_HOST(( memset(buf, 0, bufsize); ))
153 KOKKOS_IF_ON_DEVICE((
154 const ordinal_type team_index_range = (bufsize / CudaVectorSize) + (bufsize % CudaVectorSize > 0);
155 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, team_index_range), [&](const int &idx) {
156 const int ioff = idx * CudaVectorSize;
157 const int itmp = bufsize - ioff;
158 const int icnt = itmp > CudaVectorSize ? CudaVectorSize : itmp;
159 Kokkos::parallel_for(Kokkos::ThreadVectorRange(member, icnt), [&](const int &ii) {
160 const int i = ioff + ii;
161 buf[i] = 0;
162 });
163 });
164 ))
165}

167template <typename T1, typename T2, typename CompareType>
168KOKKOS_INLINE_FUNCTION static T1 *lower_bound(T1 *first, T1 *last, const T2 &val, CompareType compare) {
169 T1 *it;
170 ordinal_type step = 0, count = last - first;
171 while (count > 0) {
172 it = first;
173 step = count / 2;
174 it += step;
175 if (compare(*it, val)) {
176 first = ++it;
177 count -= step + 1;
178 } else {
179 count = step;
180 }
181 }
182 return first;
183}

185template <size_t BufSize, typename SpaceType = Kokkos::DefaultExecutionSpace> struct Flush {
186 typedef double value_type;
187
188 // flush a large host buffer
189 Kokkos::View<value_type *, SpaceType> _buf;
190 Flush() : _buf("Flush::buf", BufSize / sizeof(double)) { Kokkos::deep_copy(_buf, 1); }
191
192 KOKKOS_INLINE_FUNCTION
193 void init(value_type &update) { update = 0; }
194
195 KOKKOS_INLINE_FUNCTION
196 void join(volatile value_type &update, const volatile value_type &input) { update = update + input; }
197
198 KOKKOS_INLINE_FUNCTION
199 void operator()(const int i, value_type &update) const { update += _buf[i]; }
200
201 void run() {
202 double sum = 0;
203 Kokkos::parallel_reduce(Kokkos::RangePolicy<SpaceType>(0, BufSize / sizeof(double)), *this, sum);
204 SpaceType().fence();
205 FILE *fp = fopen("/dev/null", "w");
206 fprintf(fp, "%f\n", sum);
207 fclose(fp);
208 }
