21 template <class S, class LO, class GO, class NO>
22 LO get_mp_vector_size(
23 const Teuchos::RCP<const Tpetra::CrsMatrix<Sacado::MP::Vector<S>, LO, GO, NO > >& A = Teuchos::null,
24 const Teuchos::RCP<Tpetra::MultiVector<Sacado::MP::Vector<S>, LO, GO, NO > >& X = Teuchos::null,
25 const Teuchos::RCP<const Tpetra::MultiVector<Sacado::MP::Vector<S>, LO, GO, NO > >& B = Teuchos::null)
26 {
27 if (A != Teuchos::null) {
28 return Kokkos::dimension_scalar(A->getLocalValuesDevice(Tpetra::Access::ReadOnly));
29 }
30 else if (X != Teuchos::null) {
31 return Kokkos::dimension_scalar(X->getLocalViewDevice(Tpetra::Access::ReadOnly));
32 }
33 else if (B != Teuchos::null) {
34 return Kokkos::dimension_scalar(B->getLocalViewDevice(Tpetra::Access::ReadOnly));
35 }
36 return 0;
37 }

478 apply(Teuchos::RCP<const Matrix> A,
479 Teuchos::RCP<Vector> X,
480 Teuchos::RCP<const Vector> B ) {
481 ctassert<
482 std::is_same_v<
483 typename MatrixTraits<Matrix>::scalar_t,
484 typename MultiVecAdapter<Vector>::scalar_t
485 >
486 > same_scalar_assertion;
487 (void)same_scalar_assertion; // This stops the compiler from warning about unused declared variables

489 // If our assertion did not fail, then create and return a new solver
490 return Teuchos::rcp( new MPVectorSolverAdapter<ST,LO,GO,NO,ConcreteSolver>(A, X, B) );
491 }
