267 template <typename T>
268 Taylor<T> diff(const Taylor<T>& x, int n = 1) {
269 const int d = x.degree();
270 if (n <= 0)
271 return x;
272 else if (n > d) {
273 Taylor<T> y(0);
274 return y;
275 }
276 Taylor<T> y(d-n);
277 int c = 1;
278 for (int i=1; i<=n; ++i)
279 c *= i;
280 for (int i=n; i<=d; ++i) {
281 y.fastAccessCoeff(i-n) = x.fastAccessCoeff(i) * T(c);
282 c = (c / (i-n+1)) * (i+1);
283 }
284 return y;
285 }
