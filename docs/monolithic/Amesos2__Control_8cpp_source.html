54void Control::setControlParameters(
55 const Teuchos::RCP<Teuchos::ParameterList> & parameterList )
56{
57 // Only check for boolean "Transpose" parameters. If they are of some other
58 // type, we will let the solver interfaces deal with them.
59 if( parameterList->isType<bool>("Transpose") ){
60 useTranspose_ = parameterList->get<bool>("Transpose");
61 }
62
63 if( parameterList->isType<bool>("Iterative refinement") ){
64 useIterRefine_ = parameterList->get<bool>("Iterative refinement");
65 }
66 if( parameterList->isType<int>("Number of iterative refinements") ){
67 maxNumIterRefines_ = parameterList->get<int>("Number of iterative refinements");
68 }
69 if( parameterList->isType<bool>("Verboes for iterative refinement") ){
70 verboseIterRefine_ = parameterList->get<bool>("Verboes for iterative refinement");
71 }
72
73 // Add this value to all diagonal elements which are structurally
74 // non-zero. No change is made to non-zero structure of the matrix.
75 if( parameterList->isParameter("AddToDiag") ){
76 addToDiag_ = parameterList->get<std::string>("AddToDiag");
77 }
78
79 // Add zero to diagonal if diagonal element is not present.
80 // - not supported for matrices which are missing elements from the diagonal. See bug #1928 for discussion
81 if( parameterList->isParameter("AddZeroToDiag") ){
82 addZeroToDiag_ = parameterList->get<bool>("AddZeroToDiag");
83 }
84
85 // Matrix property, defined internally in Amesos2::Mumps as an integer,
86 // whose value can be:
87 // - 0 : general unsymmetric matrix;
88 // - 1 : SPD;
89 // - 2 : general symmetric matrix.
90 if( parameterList->isParameter("MatrixProperty") ) {
91 std::string MatrixProperty;
92 MatrixProperty = parameterList->get<std::string>("MatrixProperty");
93 if( MatrixProperty == "general" )
94 matrixProperty_ = 0;
95 else if( MatrixProperty == "SPD" )
96 matrixProperty_ = 1;
97 else if( MatrixProperty == "symmetric" )
98 matrixProperty_ = 2;
99 else {
100 std::ostringstream oss;
101 oss << "Amesos2 : ERROR" << std::endl
102 << "Amesos2 : matrixProperty value not recognized ("
103 << MatrixProperty << ")" << std::endl;
104 TEUCHOS_TEST_FOR_EXCEPTION(true,
105 std::invalid_argument,
106 oss.str());
107 }
108
109 // If true, the Amesos2 class should reindex the matrix to
110 // standard indexing (i.e. 0-(n-1)). At present, only
111 // Amesos2::Klu supports this option.
112 if( parameterList->isParameter("Reindex") ){
113 reindex_ = parameterList->get<bool>("Reindex");
114 }
115 }
116}
