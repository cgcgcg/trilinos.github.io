34template<typename Scalar>
35void MatrixFreeJacobianOperator<Scalar>::
36setBaseEvaluationToRawThyra(const Teuchos::RCP<const ::Thyra::VectorBase<Scalar> >& x_base,
37 const Teuchos::RCP<const ::Thyra::VectorBase<Scalar> >& f_base,
38 const Teuchos::RCP< ::Thyra::ModelEvaluator<Scalar> > model)

42 TEUCHOS_ASSERT(nonnull(f_base));
43 TEUCHOS_ASSERT(nonnull(model));
44 x_base_ = x_base;
45 f_base_ = f_base;
46 x_perturb_ = x_base_->space()->createMember();
47 f_perturb_ = f_base_->space()->createMember();

49 if (difference_type_ == Centered)
50 f2_perturb_ = f_base_->space()->createMember();
51
52 model_ = model;
53 base_evaluation_type_ = RawThyra;
54}

56template<typename Scalar>
57void MatrixFreeJacobianOperator<Scalar>::
58setBaseEvaluationToNOXGroup(const Teuchos::RCP<const NOX::Abstract::Group>& base_group)
59{
60 TEUCHOS_ASSERT(setup_called_);
61 TEUCHOS_ASSERT(nonnull(base_group));
62 base_group_ = Teuchos::rcp_dynamic_cast<const NOX::Thyra::Group>(base_group,true);
63 x_base_ = Teuchos::rcp_dynamic_cast<const NOX::Thyra::Vector>(base_group_->getXPtr(),true)->getThyraRCPVector();
64 f_base_ = Teuchos::rcp_dynamic_cast<const NOX::Thyra::Vector>(base_group_->getFPtr(),true)->getThyraRCPVector();
65 x_perturb_ = ::Thyra::createMember(x_base_->space());
66 f_perturb_ = ::Thyra::createMember(f_base_->space());
67
68 if (difference_type_ == Centered)
69 f2_perturb_ = ::Thyra::createMember(f_base_->space());
70
71 model_ = base_group_->getModel();
72 base_evaluation_type_ = NoxGroup;
73}

164template<typename Scalar>
165void
166MatrixFreeJacobianOperator<Scalar>::applyImpl(const ::Thyra::EOpTransp M_trans,
167 const ::Thyra::MultiVectorBase< Scalar > &thyra_mv_y,
168 const Teuchos::Ptr< ::Thyra::MultiVectorBase< Scalar > > &thyra_mv_u,
169 const Scalar alpha,
170 const Scalar beta) const
171{
172 TEUCHOS_ASSERT(setup_called_);
173
174 // Use a directional derivative to approximate u = Jy
175
176 /*
177 * delta = scalar perturbation
178 * x = solution vector used to evaluate f
179 * f = function evaluation (RHS)
180 * y = vector that J is applied to
181 *
182 * f(x + delta * y) - f(x)
183 * u = Jy = -----------------------
184 * delta
185 */
186 TEUCHOS_ASSERT(thyra_mv_y.domain()->dim() == 1);
187 TEUCHOS_ASSERT(thyra_mv_u->domain()->dim() == 1);
188 const Teuchos::RCP<const ::Thyra::VectorBase<Scalar> > y_ptr = thyra_mv_y.col(0);
189 const Teuchos::RCP< ::Thyra::VectorBase<Scalar> > u_ptr = thyra_mv_u->col(0);
190 const ::Thyra::VectorBase<Scalar>& y = *y_ptr;
191 ::Thyra::VectorBase<Scalar>& u = *u_ptr;
192 typename Teuchos::ScalarTraits<Scalar>::magnitudeType norm_2_x = ::Thyra::norm(*x_base_);
193 typename Teuchos::ScalarTraits<Scalar>::magnitudeType norm_2_y = ::Thyra::norm(y);
194
195 // Make sure the y-norm is not zero, otherwise we can get an inf perturbation from divide by zero
196 if (norm_2_y == 0.0) {
197 norm_2_y = 1.0;
198 ::Thyra::assign(Teuchos::ptrFromRef(u),0.0);
199 return;
200 }
201
202 if (perturbation_type_ == SalingerLOCA) {
203 delta_ = lambda_ * (lambda_ + norm_2_x / norm_2_y);
204 }
205 else if (perturbation_type_ == KelleySalingerPawlowski) {
206 Scalar inner_prod_x_y = ::Thyra::inner(*x_base_,y);
207
208 if (inner_prod_x_y==Teuchos::ScalarTraits<Scalar>::zero())
209 inner_prod_x_y = 1.0e-12;
210
211 delta_ = lambda_ * (1.0e-12 / lambda_ + std::fabs(inner_prod_x_y) / (norm_2_y * norm_2_y) ) * inner_prod_x_y / std::fabs(inner_prod_x_y);
212 }
213 else if (perturbation_type_ == KnollKeyes) {
214 delta_ = lambda_ * ::Thyra::norm_1(*x_base_) / (Teuchos::as<double>(x_base_->space()->dim()) * norm_2_y) + lambda_;
215 }
216 else {
217 delta_ = user_defined_delta_;
218 }
219
220 // perturbed solution: x_p = delta * y + x_0
221 ::Thyra::V_StVpV(x_perturb_.ptr(),delta_,y,*x_base_);
222
223 if (is_null(in_args_)) {
224 in_args_ = Teuchos::rcp(new ::Thyra::ModelEvaluatorBase::InArgs<Scalar>(model_->createInArgs()));
225 in_args_->setArgs(model_->getNominalValues());
226 }
227
228 in_args_->set_x(x_perturb_);
229
230 if (is_null(out_args_))
231 out_args_ = Teuchos::rcp(new ::Thyra::ModelEvaluatorBase::OutArgs<Scalar>(model_->createOutArgs()));
232
233 out_args_->set_f( ::Thyra::ModelEvaluatorBase::Evaluation< ::Thyra::VectorBase<Scalar> >(f_perturb_, ::Thyra::ModelEvaluatorBase::EVAL_TYPE_APPROX_DERIV) );
234
235 // f_p = f(delta * y + x)
236 model_->evalModel(*in_args_,*out_args_);
237
238 // to be safe, remove arguments
239 in_args_->set_x(Teuchos::null);
240 out_args_->set_f(Teuchos::null);
241
242 Scalar inv_delta = Teuchos::ScalarTraits<Scalar>::one() / delta_;
243
244 if ( difference_type_ == Centered ) {
245 // perturbed solution: x_p = -delta * y + x_0
246 ::Thyra::V_StVpV(x_perturb_.ptr(),-delta_,y,*x_base_);
247
248 out_args_->set_f(f2_perturb_);
249
250 // f_p2 = f(-delta * y + x)
251 model_->evalModel(*in_args_,*out_args_);
252
253 // to be safe, remove arguments
254 in_args_->set_x(Teuchos::null);
255 out_args_->set_f(Teuchos::null);
256
257 ::Thyra::V_StVpStV(ptrFromRef(u),inv_delta,*f_perturb_,-inv_delta,*f2_perturb_);
258 }
259 else {
260 ::Thyra::V_StVpStV(ptrFromRef(u),inv_delta,*f_perturb_,-inv_delta,*f_base_);
261 }
262
263}
