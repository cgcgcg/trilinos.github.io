115 // Functors to convert to CRS format
116 // from row-major
117 template <typename MemberType>
118 KOKKOS_INLINE_FUNCTION void operator()(const ExtractPtrTag &, const MemberType &member) const {
119 const ordinal_type id = member.league_rank();
120 const ordinal_type p = _pbeg + id;
121
122 const value_type zero(0);
123 const ordinal_type sid = (p == _pend ? 0 : _level_sids(p));
124 const ordinal_type mode = (p == _pend ? 0 : _compute_mode(sid));
125 if (mode == 0) {
126 // extract this supernode
127 const auto &s = _info.supernodes(sid);
128 const ordinal_type offm = (p == _pend ? _m : s.row_begin);
129 #define TACHO_INSERT_DIAGONALS
130 #ifdef TACHO_INSERT_DIAGONALS
131 // last row of previous supernode
132 ordinal_type row_id = 0;
133 if (p > _pbeg) {
134 const ordinal_type prev_sid = _level_sids(p-1);
135 const auto &prev_s = _info.supernodes(prev_sid);
136 row_id = prev_s.row_begin + prev_s.m;
137 }
138 // add diagonal entry
139 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, offm-row_id),
140 [&](const int& i) { _rowptr[row_id+i+1] = 1; });
141 #endif
142 if (p < _pend) {
143 if (s.m > 0) {
144 // extract this supernode
145 // stored by row, but checking for nonzereo (instead of just taking all s.n nonzereos)
146 value_type *aptr = s.u_buf;
147 UnmanagedViewType<value_type_matrix> AT(aptr, s.m, s.n);
148 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, s.m),
149 [&](const int& i) {
150 _rowptr[1+i+offm] = 0;
151 for (ordinal_type j = 0; j < s.n; j++) {
152 if (AT(i,j) != zero) {
153 _rowptr[1+i+offm] ++;
154 }
155 }
156 });
157 }
158 }
159 }
160 }

162 template <typename MemberType>
163 KOKKOS_INLINE_FUNCTION void operator()(const ExtractValTag &, const MemberType &member) const {
164 const ordinal_type id = member.league_rank();
165 const ordinal_type p = _pbeg + id;
166
167 const value_type one (1);
168 const value_type zero(0);
169 const ordinal_type sid = (p == _pend ? 0 : _level_sids(p));
170 const ordinal_type mode = (p == _pend ? 0 : _compute_mode(sid));
171 if (mode == 0) {
172 // extract this supernode
173 const auto &s = _info.supernodes(sid);
174 const ordinal_type offm = (p == _pend ? _m : s.row_begin);
175 const ordinal_type offn = (p == _pend ? 0 : s.gid_col_begin);
176 #ifdef TACHO_INSERT_DIAGONALS
177 // last row of previous supernode
178 ordinal_type row_id = 0;
179 if (p > _pbeg) {
180 const ordinal_type prev_sid = _level_sids(p-1);
181 const auto &prev_s = _info.supernodes(prev_sid);
182 row_id = prev_s.row_begin + prev_s.m;
183 }
184 // insert diagonals for the missing rows between previous and this block
185 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, offm-row_id),
186 [&](const int& i) {
187 int nnz = _rowptr[row_id+i];
188 _colind[nnz] = row_id+i;
189 _nzvals[nnz] = one;
190 _rowptr[row_id+i]++;
191 });
192 #endif
193 if (p < _pend) {
194 if (s.m > 0) {
195 // extract this supernode
196 value_type *aptr = s.u_buf;
197 UnmanagedViewType<value_type_matrix> AT(aptr, s.m, s.n);
198 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, s.m),
199 [&](const int& i) {
200 // diagonal block
201 ordinal_type j = i;
202 if (_unit_diag) {
203 // one on diagonal
204 int nnz = _rowptr[i+offm];
205 _colind[nnz] = j+offm;
206 _nzvals[nnz] = one;
207 _rowptr[i+offm] ++;
208 // save diagonal
209 _d[i+offm] = AT(i,i);
210 // move to next entry
211 j ++;
212 }
213 for (; j < s.m; j++) {
214 if (AT(i,j) != zero) {
215 int nnz = _rowptr[i+offm];
216 _colind[nnz] = j+offm;
217 _nzvals[nnz] = AT(i,j);
218 _rowptr[i+offm] ++;
219 }
220 }
221 // off-diagonal blocksa
222 j = s.m;
223 for (ordinal_type blk_id = s.sid_col_begin + 1; blk_id < s.sid_col_end - 1; blk_id++) {
224 for (ordinal_type k = _info.sid_block_colidx(blk_id).second; k < _info.sid_block_colidx(blk_id + 1).second; k++) {
225 if (AT(i,j) != zero) {
226 int nnz = _rowptr[i+offm];
227 _colind[nnz] = _info.gid_colidx(k+offn);
228 _nzvals[nnz] = AT(i,j);
229 _rowptr[i+offm] ++;
230 }
231 j++;
232 }
233 }
234 });
235 }
236 }
237 }

267 #endif
268 if (p < _pend) {
269 // extract this supernode (AL is stored by col)
270 if (s.m > 0) {
271 value_type *aptr = s.l_buf;
272 UnmanagedViewType<value_type_matrix> AL(aptr, s.n, s.m);
273 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, s.m),
274 [&](const int& j) {
275 // first extract diagnal block (each thread extract row in parallel)
276 for (ordinal_type i = 0; i < s.m; i++) {
277 if (AL(i,j) != zero) {
278 Kokkos::atomic_add(&(_rowptr[1+i+offm]), 1);
279 }
280 }
281 // off-diagonals (each thread extract col, needing atomic-add)
282 ordinal_type i = s.m;
283 for (ordinal_type blk_id = s.sid_col_begin + 1; blk_id < s.sid_col_end - 1; blk_id++) {
284 for (ordinal_type k = _info.sid_block_colidx(blk_id).second; k < _info.sid_block_colidx(blk_id + 1).second; k++) {
285 if (AL(i, j) != zero) {
286 ordinal_type gid_i = _info.gid_colidx(k+offn);
287 Kokkos::atomic_add(&(_rowptr[1+gid_i]), 1);
288 }
289 i++;
290 }
291 }
292 });

313 // last row of previous supernode
314 ordinal_type row_id = 0;
315 if (p > _pbeg) {
316 const ordinal_type prev_sid = _level_sids(p-1);
317 const auto &prev_s = _info.supernodes(prev_sid);
318 row_id = prev_s.row_begin + prev_s.m;
319 }
320 // add diagonal entry
321 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, offm-row_id),
322 [&](const int& i) {
323 ordinal_type nnz = Kokkos::atomic_fetch_add(&(_rowptr[row_id+i]), 1);
324 _colind[nnz] = row_id+i;
325 _nzvals[nnz] = one;
326 });
327 #endif
328 if (p < _pend) {
329 // extract this supernode
330 // stored by col
331 if (s.m > 0) {
332 bool no_perm = s.do_not_apply_pivots;
333 ConstUnmanagedViewType<ordinal_type_array> perm(_piv.data() + 4 * offm + 2 * s.m, s.m);
334
335 value_type *aptr = s.l_buf;
336 UnmanagedViewType<value_type_matrix> AL(aptr, s.n, s.m);
337 Kokkos::parallel_for(Kokkos::TeamThreadRange(member, s.m),
338 [&](const int& j) {
339 ordinal_type gid_j = (no_perm ? j+offm : perm(j)+offm);
340 // diagnal block
341 for (ordinal_type i = 0; i < s.m; i++) {
342 if (_unit_diag && i == j) {
343 // set diagonal to be one
344 ordinal_type nnz = Kokkos::atomic_fetch_add(&(_rowptr[offm+i]), 1);
345 _colind[nnz] = gid_j;
346 _nzvals[nnz] = one;
347 // save diagonal
348 _d[gid_j] = AL(i,j);
349 }
350 else if (AL(i,j) != zero) {
351 ordinal_type nnz = Kokkos::atomic_fetch_add(&(_rowptr[offm+i]), 1);
352 _colind[nnz] = gid_j;
353 _nzvals[nnz] = AL(i,j);
354 }
355 }
356 // off-diagonals (each thread extract col, needing atomic-add)
357 ordinal_type i = s.m;
358 for (ordinal_type blk_id = s.sid_col_begin + 1; blk_id < s.sid_col_end - 1; blk_id++) {
359 for (ordinal_type k = _info.sid_block_colidx(blk_id).second; k < _info.sid_block_colidx(blk_id + 1).second; k++) {
360 if (AL(i, j) != zero) {
361 ordinal_type gid_i = _info.gid_colidx(k+offn);
362 ordinal_type nnz = Kokkos::atomic_fetch_add(&(_rowptr[gid_i]), 1);
363 _colind[nnz] = gid_j;
364 _nzvals[nnz] = AL(i,j);
365 }
