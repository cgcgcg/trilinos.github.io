44// Create a new timer with the given name if it hasn't already been
45// created, else get the previously created timer with that name.
46RCP<Time> getTimer (const std::string& timerName) {
47 RCP<Time> timer =
48 TimeMonitor::lookupCounter (timerName);
49 if (timer.is_null ()) {
50 timer = TimeMonitor::getNewCounter (timerName);
51 }
52 return timer;
53}

55void
56benchmarkTpetraImport (ArrayView<const GO> srcGlobalElts,
57 ArrayView<const GO> destGlobalElts,
58 const GO indexBase,
59 RCP<const Comm<int> > comm,
60 const int numMapCreateTrials,
61 const int numImportCreateTrials,
62 const int numVectorCreateTrials,
63 const int numImportExecTrials)
64{
65 typedef Tpetra::Import<LO, GO, NT> import_type;
66 typedef Tpetra::Map<LO, GO, NT> map_type;
67 typedef Tpetra::Vector<ST, LO, GO, NT> vector_type;
68 const global_size_t INVALID = Teuchos::OrdinalTraits<global_size_t>::invalid ();
69
70 TEUCHOS_TEST_FOR_EXCEPTION(
71 numMapCreateTrials < 1 && numImportCreateTrials > 0, std::invalid_argument,
72 "numMapCreateTrials must be > 0 if numImportCreateTrials > 0.");
73 TEUCHOS_TEST_FOR_EXCEPTION(
74 numImportCreateTrials < 1 && numImportExecTrials > 0, std::invalid_argument,
75 "numImportCreateTrials must be > 0 if numImportExecTrials > 0.");
76 TEUCHOS_TEST_FOR_EXCEPTION(
77 numVectorCreateTrials < 1 && numMapCreateTrials > 0, std::invalid_argument,
78 "numVectorCreateTrials must be > 0 if numMapCreateTrials > 0.");
79 TEUCHOS_TEST_FOR_EXCEPTION(
80 numVectorCreateTrials < 1 && numImportExecTrials > 0, std::invalid_argument,
81 "numVectorCreateTrials must be > 0 if numImportExecTrials > 0.");
82
83 RCP<Time> mapCreateTimer = getTimer ("Tpetra: Map: Create");
84 RCP<Time> importCreateTimer = getTimer ("Tpetra: Import: Create");
85 RCP<Time> vectorCreateTimer = getTimer ("Tpetra: Vector: Create");
86 RCP<Time> importExecTimer = getTimer ("Tpetra: Import: Execute");
87
88 RCP<map_type> srcMap, destMap;
89 {
90 TimeMonitor timeMon (*mapCreateTimer);
91 for (int k = 0; k < numMapCreateTrials; ++k) {
92 srcMap = rcp (new map_type (INVALID, srcGlobalElts, indexBase, comm));
93 destMap = rcp (new map_type (INVALID, destGlobalElts, indexBase, comm));
94 }
95 }
96 RCP<import_type> import;
97 {
98 TimeMonitor timeMon (*importCreateTimer);
99 for (int k = 0; k < numImportCreateTrials; ++k) {
100 import = rcp (new import_type (srcMap, destMap));
101 }
102 }
103 RCP<vector_type> srcVec, destVec;
104 {
105 TimeMonitor timeMon (*vectorCreateTimer);
106 // This benchmarks both vector creation and vector destruction.
107 for (int k = 0; k < numVectorCreateTrials; ++k) {
108 srcVec = rcp (new vector_type (srcMap));
109 destVec = rcp (new vector_type (destMap));
110 }
111 }
112 {
113 TimeMonitor timeMon (*importExecTimer);
114 for (int k = 0; k < numImportExecTrials; ++k) {
115 destVec->doImport (*srcVec, *import, Tpetra::ADD);
116 }
117 }
118}

187void
188createGidLists (Array<GO>& srcGlobalElts,
189 Array<GO>& destGlobalElts,
190 const int numProcs,
191 const int myRank,
192 const int numEltsPerProc,
193 const GO indexBase)
194{
195 const int overlap = (myRank == 0 || myRank == numProcs-1) ? 1 : 2;
196
197 srcGlobalElts.resize (numEltsPerProc);
198 destGlobalElts.resize (numEltsPerProc + overlap);
199
200 const GO myStartSrcGid = indexBase + myRank * numEltsPerProc;
201 const GO myEndSrcGid = myStartSrcGid + numEltsPerProc - 1; // inclusive
202
203 // Put the GIDs in reverse order, to simulate a noncontiguously
204 // ordered Map. Neither Epetra nor Tpetra optimize for this case.
205 for (int k = 0; k < numEltsPerProc; ++k) {
206 const GO curGid = myStartSrcGid + as<GO> (k);
207 TEUCHOS_TEST_FOR_EXCEPTION(
208 curGid < indexBase, std::logic_error,
209 "curGid = " << curGid << " < indexBase = " << indexBase << ".");
210 srcGlobalElts[numEltsPerProc - k - 1] = curGid;
211 destGlobalElts[numEltsPerProc - k - 1] = curGid;
212 }
213
214 // 1-D Poisson overlap.
215 if (myRank == 0) {
216 const GO rightGid = myEndSrcGid + 1;
217 TEUCHOS_TEST_FOR_EXCEPTION(
218 rightGid < indexBase, std::logic_error,
219 "At left proc, right boundary GID = " << rightGid
220 << " < indexBase = " << indexBase << ".");
221 destGlobalElts[numEltsPerProc] = rightGid;
222 } else if (myRank == numProcs-1) {
223 const GO leftGid = myStartSrcGid - 1;
224 TEUCHOS_TEST_FOR_EXCEPTION(
225 leftGid < indexBase, std::logic_error,
226 "At right proc, left boundary GID = " << leftGid
227 << " < indexBase = " << indexBase << ".");
228 destGlobalElts[numEltsPerProc] = leftGid;
229 } else {
230 const GO leftGid = myStartSrcGid - 1;
231 TEUCHOS_TEST_FOR_EXCEPTION(
232 leftGid < indexBase, std::logic_error,
233 "At middle proc, left boundary GID = " << leftGid
234 << " < indexBase = " << indexBase << ".");
235 destGlobalElts[numEltsPerProc] = leftGid;
236
237 const GO rightGid = myEndSrcGid + 1;
238 TEUCHOS_TEST_FOR_EXCEPTION(
239 rightGid < indexBase, std::logic_error,
240 "At middle proc, right boundary GID = " << rightGid
241 << " < indexBase = " << indexBase << ".");
242 destGlobalElts[numEltsPerProc+1] = rightGid;
243 }
244}

247int main (int argc, char* argv[]) {
248 using std::cout;
249 using std::endl;
250 Teuchos::oblackholestream blackHole;
251 Teuchos::GlobalMPISession mpiSession (&argc, &argv, &blackHole);
252
253 RCP<const Teuchos::Comm<int> > tpetraComm;
254#ifdef EPETRA_MPI
255 Epetra_MpiComm epetraComm (MPI_COMM_WORLD);
256 tpetraComm = rcp (new Teuchos::MpiComm<int> (MPI_COMM_WORLD));
257#else
258 Epetra_SerialComm epetraComm;
259 tpetraComm = rcp (new Teuchos::SerialComm<int>);
260#endif // EPETRA_MPI
261
262 const int numProcs = tpetraComm->getSize ();
263 const int myRank = tpetraComm->getRank ();
264 const int indexBase = 1; // of interest to Sierra
265
266 // Benchmark parameters
267 int numEltsPerProc = 100000;
268 int numTrials = 100;
269 bool runEpetra = true;
270 bool runTpetra = true;
271
272 CommandLineProcessor cmdp;
273 cmdp.setOption ("numEltsPerProc", &numEltsPerProc, "Number of global indices "
274 "owned by each process");
275 cmdp.setOption ("numTrials", &numTrials, "Number of times to repeat each "
276 "operation in a timing loop");
277 cmdp.setOption ("runEpetra", "noEpetra", &runEpetra,
278 "Whether to run the Epetra benchmark");
279 cmdp.setOption ("runTpetra", "noTpetra", &runTpetra,
280 "Whether to run the Tpetra benchmark");
281 const CommandLineProcessor::EParseCommandLineReturn parseResult =
282 cmdp.parse (argc, argv);
283 if (parseResult == CommandLineProcessor::PARSE_HELP_PRINTED) {
284 // The user specified --help at the command line to print help
285 // with command-line arguments. We printed help already, so quit
286 // with a happy return code.
287 return EXIT_SUCCESS;
288 } else {
289 TEUCHOS_TEST_FOR_EXCEPTION(
290 parseResult != CommandLineProcessor::PARSE_SUCCESSFUL,
291 std::invalid_argument, "Failed to parse command-line arguments.");
292 TEUCHOS_TEST_FOR_EXCEPTION(
293 numTrials < 0, std::invalid_argument, "numTrials must be nonnegative.");
294 TEUCHOS_TEST_FOR_EXCEPTION(
295 numEltsPerProc < 0, std::invalid_argument,
296 "numEltsPerProc must be nonnegative.");
297 }
298
299 // Derived benchmark parameters
300 const int numMapCreateTrials = numTrials;
301 const int numImportCreateTrials = numTrials;
302 const int numVectorCreateTrials = numTrials;
303 const int numImportExecTrials = numTrials;
304
305 if (myRank == 0) {
306 cout << endl << "---" << endl
307 << "Command-line options:" << endl
308 << " numEltsPerProc: " << numEltsPerProc << endl
309 << " numTrials: " << numTrials << endl
310 << " runEpetra: " << (runEpetra ? "true" : "false") << endl
311 << " runTpetra: " << (runTpetra ? "true" : "false") << endl
312 << endl;
313 }
314
315 // Run the benchmark
316 Array<GO> srcGlobalElts, destGlobalElts;
317 createGidLists (srcGlobalElts, destGlobalElts, numProcs, myRank,
318 numEltsPerProc, indexBase);
319 if (runEpetra) {
320 benchmarkEpetraImport (srcGlobalElts, destGlobalElts, indexBase, epetraComm,
321 numMapCreateTrials, numImportCreateTrials,
322 numVectorCreateTrials, numImportExecTrials);
323 }
324 if (runTpetra) {
325 benchmarkTpetraImport (srcGlobalElts, destGlobalElts, indexBase, tpetraComm,
326 numMapCreateTrials, numImportCreateTrials,
327 numVectorCreateTrials, numImportExecTrials);
328 }
329 TimeMonitor::report (tpetraComm.ptr (), std::cout);
330 return EXIT_SUCCESS;
331}
