430void ILU::computeSetup() {
431#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
432 TEUCHOS_FUNC_TIME_MONITOR("ILU::computeSetup");
433#endif
434 typedef Teuchos::ScalarTraits<Scalar> ST;
435 typedef Teuchos::OrdinalTraits<LocalOrdinal> LOT;
436 LocalOrdinal izero = LOT::zero();
437 Scalar zero = ST::zero();
438
439 L_ = rcp(new Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>(Copy, Graph().getL_Graph()));
440 U_ = rcp(new Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>(Copy, Graph().getU_Graph()));
441 D_ = rcp(new Tpetra::Vector<Scalar, LocalOrdinal, GlobalOrdinal, Node>(Graph().getL_Graph().getRowMap()));
442 TEUCHOS_TEST_FOR_EXCEPTION(((L_.get() == 0) || (U_.get() == 0) || (D_.get() == 0)), std::runtime_error,
443 "Ifpack2::ILU::computeSetup(): Local column value of " << k << " for user matrix at local row " << i << " and index " << j << " less than zero on node " << getGraph().getL_Graph().getRowMap().getComm()->getRank());
444
445 // Get Maximun Row length
446 LocalOrdinal maxNumEntries = getMatrix().getMaxNumEntries();
447
448 // Set L range map and U domain map
449 U_DomainMap_ = &(Matrix().OperatorDomainMap());
450 L_RangeMap_ = &(Matrix().OperatorRangeMap());
451
452 // this is the old InitAllValues()
453 LocalOrdinal i, j;
454 LocalOrdinal numIn, numL, numU;
455 bool diagFound;
456 LocalOrdinal numNonzeroDiags = izero;
457
458 vector<LocalOrdinal> InI(MaxNumEntries); // Allocate temp space
459 vector<LocalOrdinal> LI(MaxNumEntries);
460 vector<LocalOrdinal> UI(MaxNumEntries);
461 vector<Scalar> InV(MaxNumEntries);
462 vector<Scalar> LV(MaxNumEntries);
463 vector<Scalar> UV(MaxNumEntries);
464
465 bool ReplaceValues = (L_->getStaticGraph() || L_->getIndicesAreLocal()); // Check if values should be inserted or replaced
466
467 if (ReplaceValues) {
468 L_->putScalar(zero); // Zero out L and U matrices
469 U_->putScalar(zero);
470 }
471
472 D_->putScalar(zero); // Set diagonal values to zero
473 Scalar* DV;
474 D_->extractView(&DV); // Get view of diagonal
475
476 // First we copy the user's matrix into L and U, regardless of fill level
477
478 for (i = 0; i < getNumMyRows(); i++) {
479 getMatrix().extractMyRowCopy(i, maxNumEntries, numIn, &InV[0], &InI[0]); // Get Values and Indices
480
481 // Split into L and U (we don't assume that indices are ordered).
482
483 numL = izero;
484 numU = izero;
485 diagFound = false;
486
487 for (j = 0; j < numIn; j++) {
488 int k = InI[j];
489
490 TEUCHOS_TEST_FOR_EXCEPTION((k < 0 || k > getNumMyRows()), std::runtime_error,
491 "Ifpack2::ILU::computeSetup(): Local column value of " << k
492 << " for user matrix at local row " << i << " and index " << j << " out of column range "
493 << izero << " and " << getNumMyRows() << " on node " << getGraph().getL_Graph().getRowMap().getComm()->getRank());
494 if (k == i) {
495 diagFound = true;
496 if (InV[j] >= zero)
497 DV[i] += Rthresh_ * InV[j] + Athresh_; // Store perturbed diagonal in Tpetra::Vector D_
498 else
499 DV[i] += Rthresh_ * InV[j] - Athresh_; // Store perturbed diagonal in Tpetra::Vector D_
500
501 }
502
503 else if (k < i) {
504 LI[NumL] = k;
505 LV[NumL] = InV[j];
506 NumL++;
507 } else { // if (k<NumMyRows())
508 UI[NumU] = k;
509 UV[NumU] = InV[j];
510 NumU++;
511 }
512 }
513
514 // Check in things for this row of L and U
515
516 if (DiagFound)
517 numNonzeroDiags++;
518 else
519 DV[i] = athresh_;
520
521 if (numL) {
522 if (replaceValues)
523 L_->replaceMyValues(i, numL, &LV[0], &LI[0]);
524 else
525 L_->insertMyValues(i, numL, &LV[0], &LI[0]);
526 }
527
528 if (numU) {
529 if (replaceValues)
530 U_->replaceMyValues(i, numU, &UV[0], &UI[0]);
531 else
532 U_->insertMyValues(i, numU, &UV[0], &UI[0]);
533 }
534 }
535
536 if (!replaceValues) {
537 // The domain of L and the range of U are exactly their own row maps (there is no communication).
538 // The domain of U and the range of L must be the same as those of the original matrix,
539 // However if the original matrix is a VbrMatrix, these two latter maps are translation from
540 // a block map to a point map.
541 L_->fillComplete((L_->getRowMatrixColMap()), *L_RangeMap_);
542 U_->fillComplete(*U_DomainMap_, U_->getRowMatrixRowMap());
543 }
544
545 // At this point L and U have the values of A in the structure of L and U, and diagonal vector D
546 GlobalOrdinal totalNonzeroDiags = Teuchos::OrdinalTraits<GlobalOrdinal>::zero();
547 GlobalOrdinal temp = (GlobalOrdinal)numNonzeroDiags;
548 getGraph().getL_Graph().getRowMap().getComm().SumAll(&numNonzeroDiags, &totalNonzeroDiags, 1));
549 numMyDiagonals_ = numNonzeroDiags;
550
551 TEUCHOS_TEST_FOR_EXCEPTION(numNonzeroDiags != getNumMyRows(), std::runtime_error,
552 "Ifpack2::ILU::computeSetup(): Number of nonzero diagonal values " << numNonzeroDiags << " for factorization is less than the number of rows " << getNumMyRows() << " in factor node " << getGraph().getL_Graph().getRowMap().getComm()->getRank()); // Diagonals are not right, warn user
553}

555//==========================================================================
556int ILU::Initialize() {
557#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
558 TEUCHOS_FUNC_TIME_MONITOR("ILU::Initialize");
559#endif
560
561 Time_.ResetStartTime();
562 IsInitialized_ = false;
563
564 // reset this object
565 Destroy();
566
567 Tpetra::CrsMatrix* CrsMatrix;
568 CrsMatrix = dynamic_cast<Tpetra::CrsMatrix*>(&*A_);
569 if (CrsMatrix == 0) {
570 // this means that we have to create
571 // the graph from a given Tpetra::RowMatrix. Note
572 // that at this point we are ignoring any possible
573 // graph coming from VBR matrices.
574 int size = A_->MaxNumEntries();
575 CrsGraph_ = rcp(new Tpetra::CrsGraph(Copy, A_->RowMatrixRowMap(), size));
576 if (CrsGraph_.get() == 0)
577 IFPACK2_CHK_ERR(-5); // memory allocation error
578
579 vector<int> Indices(size);
580 vector<double> Values(size);
581
582 // extract each row at-a-time, and insert it into
583 // the graph, ignore all off-process entries
584 for (int i = 0; i < A_->NumMyRows(); ++i) {
585 int NumEntries;
586 int GlobalRow = A_->RowMatrixRowMap().GID(i);
587 IFPACK2_CHK_ERR(A_->ExtractMyRowCopy(i, size, NumEntries,
588 &Values[0], &Indices[0]));
589 // convert to global indices
590 for (int j = 0; j < NumEntries; ++j) {
591 Indices[j] = A_->RowMatrixColMap().GID(Indices[j]);
592 }
593 IFPACK2_CHK_ERR(CrsGraph_->InsertGlobalIndices(GlobalRow, NumEntries,
594 &Indices[0]));
595 }
596
597 IFPACK2_CHK_ERR(CrsGraph_->FillComplete(A_->RowMatrixRowMap(),
598 A_->RowMatrixRowMap()));
599
600 // always overlap zero, wider overlap will be handled
601 // by the AdditiveSchwarz preconditioner.
602 Graph_ = rcp(new Ifpack2_IlukGraph(*CrsGraph_, LevelOfFill_, 0));
603
604 } else {
605 // see comment above for the overlap.
606 Graph_ = rcp(new Ifpack2_IlukGraph(CrsMatrix->Graph(), LevelOfFill_, 0));
607 }
608
609 if (Graph_.get() == 0)
610 IFPACK2_CHK_ERR(-5); // memory allocation error
611 IFPACK2_CHK_ERR(Graph_->ConstructFilledGraph());
612
613 IsInitialized_ = true;
614 NumInitialize_++;
615 InitializeTime_ += Time_.ElapsedTime();
616
617 return (0);
618}

620//==========================================================================
621int ILU::Compute() {
622#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
623 TEUCHOS_FUNC_TIME_MONITOR("ILU::Compute");
624#endif
625
626 if (!IsInitialized())
627 IFPACK2_CHK_ERR(Initialize());
628
629 Time_.ResetStartTime();
630 IsComputed_ = false;
631
632 // convert Matrix() into L and U factors.
633 IFPACK2_CHK_ERR(ComputeSetup());
634
635 // MinMachNum should be officially defined, for now pick something a little
636 // bigger than IEEE underflow value
637
638 double MinDiagonalValue = Tpetra::MinDouble;
639 double MaxDiagonalValue = 1.0 / MinDiagonalValue;
640
641 int ierr = 0;
642 int i, j, k;
643 int *LI, *UI;
644 double *LV, *UV;
645 int NumIn, NumL, NumU;
646
647 // Get Maximun Row length
648 int MaxNumEntries = L_->MaxNumEntries() + U_->MaxNumEntries() + 1;
649
650 vector<int> InI(MaxNumEntries + 1); // Allocate temp space, pad by one to
651 vector<double> InV(MaxNumEntries + 1); // to avoid debugger complaints for pathological cases
652 vector<int> colflag(NumMyCols());
653
654 double* DV;
655 ierr = D_->ExtractView(&DV); // Get view of diagonal
656
657 int current_madds = 0; // We will count multiply-add as they happen
658
659 // =========================== //
660 // Now start the factorization //
661 // =========================== //
662
663 // Need some integer workspace and pointers
664 int NumUU;
665 int* UUI;
666 double* UUV;
667 for (j = 0; j < NumMyCols(); ++j) colflag[j] = -1;
668
669 for (i = 0; i < NumMyRows(); ++i) {
670 // Fill InV, InI with current row of L, D and U combined
671
672 NumIn = MaxNumEntries;
673 IFPACK2_CHK_ERR(L_->ExtractMyRowCopy(i, NumIn, NumL, &InV[0], &InI[0]));
674 LV = &InV[0];
675 LI = &InI[0];
676
677 InV[NumL] = DV[i]; // Put in diagonal
678 InI[NumL] = i;
679
680 IFPACK2_CHK_ERR(U_->ExtractMyRowCopy(i, NumIn - NumL - 1, NumU, &InV[NumL + 1], &InI[NumL + 1]));
681 NumIn = NumL + NumU + 1;
682 UV = &InV[NumL + 1];
683 UI = &InI[NumL + 1];
684
685 // Set column flags
686 for (j = 0; j < NumIn; j++) colflag[InI[j]] = j;
687
688 double diagmod = 0.0; // Off-diagonal accumulator
689
690 for (int jj = 0; jj < NumL; jj++) {
691 j = InI[jj];
692 double multiplier = InV[jj]; // current_mults++;
693
694 InV[jj] *= DV[j];
695
696 IFPACK2_CHK_ERR(U_->ExtractMyRowView(j, NumUU, UUV, UUI)); // View of row above
697
698 if (RelaxValue_ == 0.0) {
699 for (k = 0; k < NumUU; k++) {
700 int kk = colflag[UUI[k]];
701 if (kk > -1) {
702 InV[kk] -= multiplier * UUV[k];
703 current_madds++;
704 }
705 }
706 } else {
707 for (k = 0; k < NumUU; k++) {
708 int kk = colflag[UUI[k]];
709 if (kk > -1)
710 InV[kk] -= multiplier * UUV[k];
711 else
712 diagmod -= multiplier * UUV[k];
713 current_madds++;
714 }
715 }
716 }
717 if (NumL) {
718 IFPACK2_CHK_ERR(L_->ReplaceMyValues(i, NumL, LV, LI)); // Replace current row of L
719 }
720
721 DV[i] = InV[NumL]; // Extract Diagonal value
722
723 if (RelaxValue_ != 0.0) {
724 DV[i] += RelaxValue_ * diagmod; // Add off diagonal modifications
725 // current_madds++;
726 }
727
728 if (fabs(DV[i]) > MaxDiagonalValue) {
729 if (DV[i] < 0)
730 DV[i] = -MinDiagonalValue;
731 else
732 DV[i] = MinDiagonalValue;
733 } else
734 DV[i] = 1.0 / DV[i]; // Invert diagonal value
735
736 for (j = 0; j < NumU; j++) UV[j] *= DV[i]; // Scale U by inverse of diagonal
737
738 if (NumU) {
739 IFPACK2_CHK_ERR(U_->ReplaceMyValues(i, NumU, UV, UI)); // Replace current row of L and U
740 }
741
742 // Reset column flags
743 for (j = 0; j < NumIn; j++) colflag[InI[j]] = -1;
744 }
745
746 // Validate that the L and U factors are actually lower and upper triangular
747
748 if (!L_->LowerTriangular())
749 IFPACK2_CHK_ERR(-4);
750 if (!U_->UpperTriangular())
751 IFPACK2_CHK_ERR(-4);
752
753 // Add up flops
754
755 double current_flops = 2 * current_madds;
756 double total_flops = 0;
757
758 IFPACK2_CHK_ERR(Graph().L_Graph().RowMap().Comm().SumAll(&current_flops, &total_flops, 1)); // Get total madds across all PEs
759
760 // Now count the rest
761 total_flops += (double)L_->NumGlobalNonzeros(); // Accounts for multiplier above
762 total_flops += (double)D_->GlobalLength(); // Accounts for reciprocal of diagonal
763 if (RelaxValue_ != 0.0) total_flops += 2 * (double)D_->GlobalLength(); // Accounts for relax update of diag
764
765 // add to this object's counter
766 ComputeFlops_ += total_flops;
767
768 IsComputed_ = true;
769 NumCompute_++;
770 ComputeTime_ += Time_.ElapsedTime();
771
772 return (ierr);
773}

776// This function finds Y such that LDU Y = X or U(trans) D L(trans) Y = X for multiple RHS
777int ILU::Solve(bool Trans, const Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& X,
778 Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& Y) const {
779#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
780 TEUCHOS_FUNC_TIME_MONITOR("ILU::ApplyInverse - Solve");
781#endif
782
783 // in this function the overlap is always zero
784 bool Upper = true;
785 bool Lower = false;
786 bool UnitDiagonal = true;
787
788 if (!Trans) {
789 IFPACK2_CHK_ERR(L_->Solve(Lower, Trans, UnitDiagonal, X, Y));
790 // y = D*y (D_ has inverse of diagonal)
791 IFPACK2_CHK_ERR(Y.Multiply(1.0, *D_, Y, 0.0));
792 // Solve Uy = y
793 IFPACK2_CHK_ERR(U_->Solve(Upper, Trans, UnitDiagonal, Y, Y));
794 } else {
795 // Solve Uy = y
796 IFPACK2_CHK_ERR(U_->Solve(Upper, Trans, UnitDiagonal, X, Y));
797 // y = D*y (D_ has inverse of diagonal)
798 IFPACK2_CHK_ERR(Y.Multiply(1.0, *D_, Y, 0.0));
799 IFPACK2_CHK_ERR(L_->Solve(Lower, Trans, UnitDiagonal, Y, Y));
800 }
801
802 return (0);
803}

805//=============================================================================
806// This function finds X such that LDU Y = X or U(trans) D L(trans) Y = X for multiple RHS
807int ILU::Multiply(bool Trans, const Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& X,
808 Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& Y) const {
809#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
810 TEUCHOS_FUNC_TIME_MONITOR("ILU::Multiply");
811#endif
812
813 if (!IsComputed())
814 IFPACK2_CHK_ERR(-3);
815
816 if (!Trans) {
817 IFPACK2_CHK_ERR(U_->Multiply(Trans, X, Y));
818 // Y1 = Y1 + X1 (account for implicit unit diagonal)
819 IFPACK2_CHK_ERR(Y.Update(1.0, X, 1.0));
820 // y = D*y (D_ has inverse of diagonal)
821 IFPACK2_CHK_ERR(Y.ReciprocalMultiply(1.0, *D_, Y, 0.0));
822 Tpetra::MultiVector Y1temp(Y); // Need a temp copy of Y1
823 IFPACK2_CHK_ERR(L_->Multiply(Trans, Y1temp, Y));
824 // (account for implicit unit diagonal)
825 IFPACK2_CHK_ERR(Y.Update(1.0, Y1temp, 1.0));
826 } else {
827 IFPACK2_CHK_ERR(L_->Multiply(Trans, X, Y));
828 // Y1 = Y1 + X1 (account for implicit unit diagonal)
829 IFPACK2_CHK_ERR(Y.Update(1.0, X, 1.0));
830 // y = D*y (D_ has inverse of diagonal)
831 IFPACK2_CHK_ERR(Y.ReciprocalMultiply(1.0, *D_, Y, 0.0));
832 Tpetra::MultiVector Y1temp(Y); // Need a temp copy of Y1
833 IFPACK2_CHK_ERR(U_->Multiply(Trans, Y1temp, Y));
834 // (account for implicit unit diagonal)
835 IFPACK2_CHK_ERR(Y.Update(1.0, Y1temp, 1.0));
836 }
837
838 return (0);
839}

841//=============================================================================
842// This function finds X such that LDU Y = X or U(trans) D L(trans) Y = X for multiple RHS
843int ILU::ApplyInverse(const Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& X,
844 Tpetra::MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node>& Y) const {
845#ifdef ENABLE_IFPACK2_ILU_TEUCHOS_TIMERS
846 TEUCHOS_FUNC_TIME_MONITOR("ILU::ApplyInverse");
847#endif
848
849 if (!IsComputed())
850 IFPACK2_CHK_ERR(-3);
851
852 if (X.NumVectors() != Y.NumVectors())
853 IFPACK2_CHK_ERR(-2);
854
855 Time_.ResetStartTime();
856
857 // AztecOO gives X and Y pointing to the same memory location,
858 // need to create an auxiliary vector, Xcopy
859 Teuchos::RCP<const Tpetra::MultiVector> Xcopy;
860 if (X.Pointers()[0] == Y.Pointers()[0])
861 Xcopy = Teuchos::rcp(new Tpetra::MultiVector(X));
862 else
863 Xcopy = Teuchos::rcp(&X, false);
864
865 IFPACK2_CHK_ERR(Solve(ILU::UseTranspose(), *Xcopy, Y));
866
867 // approx is the number of nonzeros in L and U
868 ApplyInverseFlops_ += X.NumVectors() * 4 *
869 (L_->NumGlobalNonzeros() + U_->NumGlobalNonzeros());
870
871 ++NumApplyInverse_;
872 ApplyInverseTime_ += Time_.ElapsedTime();
873
874 return (0);
875}
