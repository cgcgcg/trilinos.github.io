73void NOX::ObserverPrint::printStep(const NOX::Solver::Generic& solver)
74{
75 const auto& stats = *solver.getSolverStatistics();
76 auto& os = os_->out();
77 auto original_flags = os.flags();
78 const int precision = 6;
79
80 if (offset_ > 0) {
81 for (size_t i=0; i < offset_; ++i) {
82 os << " ";
83 }
84 }
85
86 os.width(5);
87 os.setf(std::ios::left);
88 os << stats.numNonlinearIterations;
89
90 os.width(13);
91 if (solver.getStatus() == NOX::StatusTest::Unconverged)
92 os << "Unconverged";
93 else if (solver.getStatus() == NOX::StatusTest::Converged)
94 os << "Converged!";
95 else if (solver.getStatus() == NOX::StatusTest::Failed)
96 os << "Failed!";
97
98 os.width(14);
99 os.precision(precision);
100 os.setf(std::ios::left|std::ios::scientific);
101 auto& grp = solver.getSolutionGroup();
102 if (!grp.isF())
103 const_cast<NOX::Abstract::Group&>(grp).computeF();
104 os << grp.getNormF();
105
106 const auto* is_linesearch = dynamic_cast<const NOX::Solver::LineSearchBased*>(&solver);
107 if (is_linesearch) {
108 os.width(14);
109 os.setf(std::ios::left|std::ios::scientific);
110 os.precision(precision);
111 os << is_linesearch->getStepSize();
112 }
113
114 os.width(11);
115 os.setf(std::ios::left);
116 os.precision(precision);
117 os << stats.linearSolve.lastLinearSolve_NumIterations;
118
119 os.width(14);
120 os.setf(std::ios::left|std::ios::scientific);
121 os.precision(precision);
122 os << stats.linearSolve.lastLinearSolve_AchievedTolerance;
123
124 os << std::endl;
125
126 os.flags(original_flags);
127}
