37void
38Stokhos::MPMeanBasedPreconditioner::
39setupPreconditioner(const Teuchos::RCP<Stokhos::BlockDiagonalOperator>& mp_op,
40 const Epetra_Vector& x)
41{
42 TEUCHOS_TEST_FOR_EXCEPTION(prec_factory == Teuchos::null, std::logic_error,
43 "Error! setupPreconditioner() cannot be called when " <<
44 "prec_factory is null!" << std::endl);
45
46 Teuchos::RCP<Stokhos::ProductContainer<Epetra_Operator> > mp_ops =
47 mp_op->getMPOps();
48 mean_prec = prec_factory->compute(mp_ops->getCoeffPtr(0));
49 if (num_mp_blocks > 0) {
50 label = std::string("Stokhos MP Mean Preconditioner:\n") +
51 std::string(" ***** ") +
52 std::string(mean_prec->Label());
53 }
54}

66int
67Stokhos::MPMeanBasedPreconditioner::
68Apply(const Epetra_MultiVector& Input, Epetra_MultiVector& Result) const
69{
70 EpetraExt::BlockMultiVector mp_input(View, *base_map, Input);
71 EpetraExt::BlockMultiVector mp_result(View, *base_map, Result);
72 for (int i=0; i<num_mp_blocks; i++) {
73 mean_prec->Apply(*(mp_input.GetBlock(i)), *(mp_result.GetBlock(i)));
74 }
75
76 return 0;
77}
