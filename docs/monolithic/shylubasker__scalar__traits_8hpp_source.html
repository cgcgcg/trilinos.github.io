74 ComplexT ret ;
75 cr = (Teuchos::ScalarTraits<ComplexT>::real(c)) ;
76 ci = (Teuchos::ScalarTraits<ComplexT>::imag(c)) ;
77 if (MY_SCALAR_ABS (cr) >= MY_SCALAR_ABS (ci))
78 {
79 r = ci / cr ;
80 den = cr + r * ci ;
81 ret = std::complex<T>(1.0 / den, -r / den) ;
82 }
83 else
84 {
85 r = cr / ci ;
86 den = r * cr + ci ;
87 ret = std::complex<T>(r / den, -1.0 / den) ;
88 }
89 return ret;

93 {

96

98 bi = (Teuchos::ScalarTraits<ComplexT>::imag(b)) ;

102 {
103 r = bi / br ;
104 den = br + r * bi ;

109 r = br / bi ;
