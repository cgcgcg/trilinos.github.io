67 template <typename ViewTypeA, typename ViewTypeP, typename ViewTypeD>
68 inline static int modify(const ViewTypeA &A, const ViewTypeP &P, const ViewTypeD &D) {
69 int r_val = 0;
70
71 static constexpr bool runOnHost = run_tacho_on_host_v<typename ViewTypeA::execution_space>;
72
73 if constexpr(runOnHost) {
74 typedef typename ViewTypeA::non_const_value_type value_type;
75
76 static_assert(ViewTypeA::rank == 2, "A is not rank 2 view.");
77 static_assert(ViewTypeP::rank == 1, "P is not rank 1 view.");
78 static_assert(ViewTypeD::rank == 2, "D is not rank 2 view.");
79
80 TACHO_TEST_FOR_EXCEPTION(D.extent(0) < A.extent(0), std::runtime_error, "D extent(0) is smaller than A extent(0).");
81 TACHO_TEST_FOR_EXCEPTION(D.extent(1) != 2, std::runtime_error, "D is supposed to store 2x2 blocks .");
82 TACHO_TEST_FOR_EXCEPTION(P.extent(0) < 4 * A.extent(0), std::runtime_error, "P should be 4*A.extent(0) .");
83
84 const ordinal_type m = A.extent(0);
85 if (m > 0) {
86 value_type *KOKKOS_RESTRICT Aptr = A.data();
87 ordinal_type *KOKKOS_RESTRICT ipiv = P.data(), *KOKKOS_RESTRICT fpiv = ipiv + m, *KOKKOS_RESTRICT perm = fpiv + m,
88 *KOKKOS_RESTRICT peri = perm + m;
89
90 const value_type one(1), zero(0);
91 for (ordinal_type i = 0; i < m; ++i)
92 perm[i] = i;
93 for (ordinal_type i = 0; i < m; ++i) {
94 if (ipiv[i] < 0) {
95 {
96 // first pivot
97 ipiv[i] = 0;
98 fpiv[i] = 0;
99
100 D(i, 0) = A(i, i);
101 D(i, 1) = A(i + 1, i);
102 A(i, i) = one;
103 }
104 {
105 // second pivot
106 i++;
107 const ordinal_type fla_pivot = -ipiv[i] - i - 1;
108 fpiv[i] = fla_pivot;
109 if (fla_pivot) {
110 value_type *KOKKOS_RESTRICT src = Aptr + i;
111 value_type *KOKKOS_RESTRICT tgt = src + fla_pivot;
112 for (ordinal_type j = 0; j < (i - 1); ++j) {
113 const ordinal_type idx = j * m;
114 swap(src[idx], tgt[idx]);
115 }
116 }
117
118 D(i, 0) = A(i, i - 1);
119 D(i, 1) = A(i, i);
120 A(i, i - 1) = zero;
121 A(i, i) = one;
122 }
123 } else {
124 const ordinal_type fla_pivot = ipiv[i] - i - 1;
125 fpiv[i] = fla_pivot;
126 if (fla_pivot) {
127 value_type *src = Aptr + i;
128 value_type *tgt = src + fla_pivot;
129 for (ordinal_type j = 0; j < i; ++j) {
130 const ordinal_type idx = j * m;
131 swap(src[idx], tgt[idx]);
132 }
133 }
134
135 D(i, 0) = A(i, i);
136 A(i, i) = one;
137 }
138
140 if (fpiv[i]) {
141 const ordinal_type pidx = i + fpiv[i];
142 swap(perm[i], perm[pidx]);
143 }
144 }
145 for (ordinal_type i = 0; i < m; ++i)
146 peri[perm[i]] = i;
147 }
148 } else {
149 TACHO_TEST_FOR_ABORT(true, ">> This function is only allowed in host space.");
150 }
151 return r_val;
152 }
