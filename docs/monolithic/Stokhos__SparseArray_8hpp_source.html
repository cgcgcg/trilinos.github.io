30 template <typename ordinal_type, typename val_type>
31 struct SparseArray {
32
34 Teuchos::Array<ordinal_type> indices;
35
37 Teuchos::Array<val_type> values;
38
40 ordinal_type size() const { return indices.size(); }
41
43 void resize(ordinal_type sz) {
44 indices.resize(sz);
45 values.resize(sz);
46 }
47
48 typedef typename Teuchos::Array<ordinal_type>::const_iterator index_const_iterator;
49 typedef typename Teuchos::Array<val_type>::const_iterator value_const_iterator;
50 typedef SparseArrayIterator<index_const_iterator, value_const_iterator> const_iterator;
51 typedef SparseArrayReverseIterator<index_const_iterator, value_const_iterator> const_reverse_iterator;
52
54 const_iterator begin() const {
55 return const_iterator(indices.begin(), values.begin());
56 }
57
59 const_iterator end() const {
60 return const_iterator(indices.end(), values.end());
61 }
62
64 const_reverse_iterator rbegin() const {
65 return const_reverse_iterator(end());
66 }
67
69 const_reverse_iterator rend() const {
70 return const_reverse_iterator(begin());
71 }
72
74
77 const_iterator find(ordinal_type i) const {
78 const_iterator it = std::lower_bound(begin(), end(), i);
79 bool found = it != end() && *it == i;
80 if (found)
81 return it;
82 else
83 return end();
84 }
