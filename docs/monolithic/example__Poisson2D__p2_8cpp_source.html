1217

1220 }

1225 cellCounter++;
1226
1227 } // end cell loop
1228
1229 } // end loop over worksets

1236#else

1238 H1errTot = H1err;
1239 LinferrTot = Linferr;
1240#endif
1241
1242

1343/************ Source Term (RHS) ****************/
1344template<typename Scalar>
1345const Scalar sourceTerm(Scalar& x, Scalar& y){
1346
1347 Scalar u;
1348 Scalar grad_u[2];
1349 Scalar flux[2] = {0.0, 0.0};
1350 Scalar material[2][2];
1351 Scalar f = 0.;
1352
1353 // Indicate the independent variables
1354 x.diff(0,2);
1355 y.diff(1,2);
1356
1357 // Get exact solution and its gradient
1358 u = exactSolution(x, y);
1359 exactSolutionGrad(grad_u, x, y);
1360
1361 // Get material tensor
1362 materialTensor<Scalar>(material, x, y);
1363
1364 // Compute total flux = (A.grad u)
1365 for(int i = 0; i < 2; i++){
1366
1367 // Add diffusive flux
1368 for(int j = 0; j < 2; j++){
1369 flux[i] += material[i][j]*grad_u[j];
1370 }
1371 }
1372
1373 // Compute source term (right hand side): f = -div(A.grad u)
1374 f = -(flux[0].dx(0) + flux[1].dx(1));
1375
1376 return f;
1377}
