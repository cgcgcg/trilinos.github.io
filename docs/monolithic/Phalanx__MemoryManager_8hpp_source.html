30 public:
31 KokkosViewSizeFunctor(const PHX::FieldTag& tag,
32 const std::vector<PHX::index_size_type>& extended_dimensions,
33 std::size_t& size) :
34 tag_(tag),
35 extended_dimensions_(extended_dimensions),
36 size_(size) {}
37
38 template <typename ScalarT>
39 void operator()(ScalarT t) const
40 {
41 // To get the padded sizes from kokkos view, we actually have to
42 // allocate the view. Inefficient, but this only happens during
43 // setup. See issue: https://github.com/kokkos/kokkos/issues/2182
44 if (tag_.dataTypeInfo() == typeid(t)) {
45 using LT = PHX::DataLayout::KokkosLayoutType;
46 const auto layout = tag_.dataLayout().kokkosLayout();
47 if (layout == LT::Default)
48 size_ = PHX::getAllocationSize<ScalarT,typename PHX::DevLayout<ScalarT>::type,PHX::Device>(tag_,extended_dimensions_);
49 else if (layout == LT::Left)
50 size_ = PHX::getAllocationSize<ScalarT,Kokkos::LayoutLeft,PHX::Device>(tag_,extended_dimensions_);
51 else
52 size_ = PHX::getAllocationSize<ScalarT,Kokkos::LayoutRight,PHX::Device>(tag_,extended_dimensions_);
53 }
54 }
55 };
