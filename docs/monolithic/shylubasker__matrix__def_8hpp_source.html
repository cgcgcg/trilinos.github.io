507 template <class Int, class Entry, class Exe_Space>
508 BASKER_INLINE
509 void BaskerMatrix<Int,Entry,Exe_Space>::convert2D
510 (
511 BASKER_MATRIX &M,
512 BASKER_BOOL alloc,
513 Int kid,
514 bool keep_zeros
515 )
516 {
517 if(nnz == 0)
518 {
519 for(Int i = 0; i < ncol+1; i++)
520 {
521 col_ptr(i) = 0;
522 }
523
524 MALLOC_INT_1DARRAY(row_idx, 1);
525 row_idx(0) = (Int) 0;
526 MALLOC_ENTRY_1DARRAY(val, 1);
527 val(0) = (Entry) 0;
528 return;
529 }
530
531 //info();
532 //We could check some flag ??
533 //We assume a pre-scan has already happened
534 if(alloc == BASKER_TRUE)
535 {
536 //printf("ALLOC\n");
537 if(nnz > 0)
538 {
539 BASKER_ASSERT(nnz > 0, "matrix row nnz 2");
540 MALLOC_INT_1DARRAY(row_idx, nnz);
541 }
542 else if(nnz == 0)
543 {
544 BASKER_ASSERT((nnz+1)>0, "matrix row nnz 3");
545 MALLOC_INT_1DARRAY(row_idx, nnz+1);
546 }
547 }
548
549 if(alloc == BASKER_TRUE)
550 {
551 if(nnz > 0)
552 {
553 BASKER_ASSERT(nnz > 0, "matrix nnz 4");
554 MALLOC_ENTRY_1DARRAY(val, nnz);
555 }
556 else if(nnz == 0)
557 {
558 BASKER_ASSERT((nnz+1) > 0, "matrix nnz 5");
559 MALLOC_ENTRY_1DARRAY(val, nnz+1);
560 }
561 }
562
563 const Entry zero(0.0);
564
565 anorm = abs(zero);
566 Int temp_count = 0;
567 for(Int k = scol; k < scol+ncol; ++k)
568 {
569 //note col_ptr[k-scol] contains the starting index (by find_2D_convert)
570 // i.e., start of the diagonal block for U, or start of the first off-diagonal block for L
571 if(col_ptr(k-scol) == BASKER_MAX_IDX)
572 {
573 col_ptr(k-scol) = temp_count;
574 continue;
575 }
576
577 Int skipped_count = 0;
578 Int kept_count = 0;
579 Mag anorm_k (0.0);
580 for(Int i = col_ptr(k-scol); i < M.col_ptr(k+1); i++)
581 {
582 Int j = M.row_idx(i);
583 if(j >= srow+nrow)
584 {
585 if (!keep_zeros && (kept_count == 0 && skipped_count > 0)) {
586 // if all were zero, then add the last entry to avoid empty column.
587 row_idx(temp_count) = M.row_idx(i-1)-srow;
588 val(temp_count) = M.val(i-1);
589 anorm_k += abs(M.val(i-1));
590 temp_count++;
591 }
592 break;
593 }
594
595 if(j < srow)
596 {
597 std::cout << std::endl
598 << "BaskerMatrix::convert2D(kid = " << kid
599 << "): j " << j
600 << " srow: " << srow
601 << " scol: " << scol
602 << " nrow: " << nrow
603 << " ncol: " << ncol
604 << " with k: " << k
605 << " idx: " << i
606 << std::endl << std::endl;
607 char error_msg[100];
608 sprintf(error_msg, " ERROR: j is less than srow (j=%d, srow=%d, kid=%d)",(int)j,(int)srow,(int)kid);
609 BASKER_ASSERT(0 == 1, error_msg);
610 }
611 if (keep_zeros || M.val(i) != zero)
612 {
613 row_idx(temp_count) = j-srow;
614 val(temp_count) = M.val(i);
615
616 anorm_k += abs(M.val(i));
617 temp_count++;
618 kept_count ++;
619 } else {
620 skipped_count ++;
621 }
622 }
623 anorm = (anorm > anorm_k ? anorm : anorm_k);
624 col_ptr(k-scol) = temp_count;
625 }
626
627 //NO!!1
628 //Slide over the col counts
629 for(Int i = ncol; i > 0; i--)
630 {
631 col_ptr(i) = col_ptr(i-1);
632 }
633 col_ptr(0) = (Int) 0;
634
635 }//end convert2d(Matrix)
