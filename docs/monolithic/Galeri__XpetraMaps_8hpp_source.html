81template <class LocalOrdinal, class GlobalOrdinal, class Map>
82RCP<Map> CreateMap(const std::string& mapType, const Teuchos::RCP<const Teuchos::Comm<int> >& comm, Teuchos::ParameterList& list) {
83 GlobalOrdinal n = -1, nx = -1, ny = -1, nz = -1, mx = -1, my = -1, mz = -1;
84
85 // Get matrix dimensions
86 if (list.isParameter("n")) {
87 if (list.isType<int>("n"))
88 n = Teuchos::as<GlobalOrdinal>(list.get<int>("n"));
89 else
90 n = list.get<GlobalOrdinal>("n");
91 }
92 if (list.isParameter("nx")) {
93 if (list.isType<int>("nx"))
94 nx = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
95 else
96 nx = list.get<GlobalOrdinal>("nx");
97 }
98 if (list.isParameter("ny")) {
99 if (list.isType<int>("ny"))
100 ny = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
101 else
102 ny = list.get<GlobalOrdinal>("ny");
103 }
104 if (list.isParameter("nz")) {
105 if (list.isType<int>("nz"))
106 nz = Teuchos::as<GlobalOrdinal>(list.get<int>("nz"));
107 else
108 nz = list.get<GlobalOrdinal>("nz");
109 }
110 if (list.isParameter("mx")) {
111 if (list.isType<int>("mx"))
112 mx = Teuchos::as<GlobalOrdinal>(list.get<int>("mx"));
113 else
114 mx = list.get<GlobalOrdinal>("mx");
115 }
116 if (list.isParameter("my")) {
117 if (list.isType<int>("my"))
118 my = Teuchos::as<GlobalOrdinal>(list.get<int>("my"));
119 else
120 my = list.get<GlobalOrdinal>("my");
121 }
122 if (list.isParameter("mz")) {
123 if (list.isType<int>("mz"))
124 mz = Teuchos::as<GlobalOrdinal>(list.get<int>("mz"));
125 else
126 mz = list.get<GlobalOrdinal>("mz");
127 }
128
129 if (mapType == "Cartesian1D") {
130 if (nx == -1) {
131 if (n <= 0)
132 throw Exception(__FILE__, __LINE__, "If nx is not set, then n must be set");
133
134 nx = n;
135 list.set("nx", nx);
136 }
137 list.set("n", nx);
138
139 if (mx == -1) {
140 mx = Teuchos::as<GlobalOrdinal>(comm->getSize());
141 list.set("mx", mx);
142 }
143
144 return Maps::Cartesian1D<LocalOrdinal, GlobalOrdinal, Map>(comm, nx, mx, list);
145
146 } else if (mapType == "Cartesian2D") {
147 if (nx == -1 || ny == -1) {
148 if (n <= 0)
149 throw Exception(__FILE__, __LINE__, "If nx or ny are not set, then n must be set");
150
151 nx = ny = Teuchos::as<GlobalOrdinal>(sqrt(Teuchos::as<double>(n)));
152 if (nx * ny != n)
153 throw Exception(__FILE__, __LINE__,
154 "The number of global elements (n) must be",
155 "a perfect square, otherwise set nx and ny");
156
157 list.set("nx", nx);
158 list.set("ny", ny);
159 }
160 list.set("n", nx * ny);
161
162 if (mx == -1 || my == -1) {
163 // Simple method to try to find mx and my such that mx*my = NumProc
164 mx = Teuchos::as<GlobalOrdinal>(sqrt(Teuchos::as<double>(comm->getSize()) + .001));
165 my = comm->getSize() / mx;
166
167 while ((mx * my) != comm->getSize())
168 my = comm->getSize() / (--mx);
169
170 list.set("mx", mx);
171 list.set("my", my);
172 }
173
174 return Maps::Cartesian2D<LocalOrdinal, GlobalOrdinal, Map>(comm, nx, ny, mx, my, list);
175
176 } else if (mapType == "Cartesian3D") {
177 if (nx == -1 || ny == -1 || nz == -1) {
178 if (n <= 0)
179 throw Exception(__FILE__, __LINE__, "If nx or ny or nz are not set, then n must be set");
180
181 nx = ny = nz = Teuchos::as<GlobalOrdinal>(pow(Teuchos::as<double>(n), 0.333334));
182 if (nx * ny * nz != n)
183 throw Exception(__FILE__, __LINE__,
184 "The number of global elements n (" +
185 toString(n) + ") must be",
186 "a perfect cube, otherwise set nx, ny and nz");
187
188 list.set("nx", nx);
189 list.set("ny", ny);
190 list.set("nz", nz);
191 }
192 list.set("n", nx * ny * nz);
193
194 if (mx == -1 || my == -1 || mz == -1) {
195 mx = my = mz = Teuchos::as<GlobalOrdinal>(pow(Teuchos::as<double>(comm->getSize()), 0.333334));
196
197 if (mx * my * mz != comm->getSize()) {
198 // Simple method to find a set of processor assignments
199 mx = my = mz = 1;
200
201 int ProcTemp = comm->getSize();
202 GlobalOrdinal factors[50];
203 for (GlobalOrdinal jj = 0; jj < 50; jj++) factors[jj] = 0;
204 for (GlobalOrdinal jj = 2; jj < 50; jj++) {
205 bool flag = true;
206 while (flag) {
207 GlobalOrdinal temp = ProcTemp / jj;
208 if (temp * jj == ProcTemp) {
209 factors[jj]++;
210 ProcTemp = temp;
211
212 } else {
213 flag = false;
214 }
215 }
216 }
217 mx = ProcTemp;
218 for (GlobalOrdinal jj = 50 - 1; jj > 0; jj--) {
219 while (factors[jj] != 0) {
220 if ((mx <= my) && (mx <= mz))
221 mx = mx * jj;
222 else if ((my <= mx) && (my <= mz))
223 my = my * jj;
224 else
225 mz = mz * jj;
226 factors[jj]--;
227 }
228 }
229 }
230 list.set("mx", mx);
231 list.set("my", my);
232 list.set("mz", mz);
233
234 } else {
235 if (mx * my * mz != comm->getSize())
236 throw Exception(__FILE__, __LINE__,
237 "mx * my * mz != number of processes!",
238 "mx = " + toString(mx) + ", my = " + toString(my) + ", mz = " + toString(mz));
239 }
240
241 return Maps::Cartesian3D<LocalOrdinal, GlobalOrdinal, Map>(comm, nx, ny, nz, mx, my, mz, list);
242
243 } else {
244 throw Exception(__FILE__, __LINE__,
245 "`mapType' has incorrect value (" + mapType + ")",
246 "in input to function CreateMap()",
247 "Check the documentation for a list of valid choices");
248 }
249} // CreateMap()
