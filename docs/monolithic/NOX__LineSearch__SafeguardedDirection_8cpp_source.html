43bool NOX::LineSearch::SafeguardedDirection::
44reset(const Teuchos::RCP<NOX::GlobalData>& gd,
45 Teuchos::ParameterList& params)
46{
47 globalDataPtr_ = gd;
48 print_.reset(gd->getUtils());
49 counter_ = &gd->getNonConstSolverStatistics()->lineSearch;
50 paramsPtr_ = &params;
51
52 Teuchos::ParameterList& p = params.sublist("Safeguarded Direction");
53 p.validateParametersAndSetDefaults(*getValidParameters());
54
55 userLimits_ = p.get<Teuchos::RCP<NOX::Abstract::Vector> >("Update Limit Vector");
56 useCounter_ = p.get<bool>("Use Counters");
57
58 TEUCHOS_TEST_FOR_EXCEPTION(is_null(userLimits_), std::runtime_error,
59 "Error: The line search NOX::LineSearch::SafeguardedDirection requires the user to supply the \"Update Limit Vector\" parameter of type NOX::Abstract::Vector for in the parameter list.");
60
61 if (is_null(limitDifference_))
62 limitDifference_ = userLimits_->clone(NOX::ShapeCopy);
63
64 if (is_null(scaledUpdate_))
65 scaledUpdate_ = userLimits_->clone(NOX::ShapeCopy);
66
67 // Set up counter
68 if (useCounter_)
69 counter_->reset();
70
71 return true;
72}

74bool NOX::LineSearch::SafeguardedDirection::compute(Abstract::Group& newGrp,
75 double& step,
76 const Abstract::Vector& dir,
77 const Solver::Generic& /* s */)
78{
79 printOpeningRemarks();
80
81 if (useCounter_) {
82 counter_->incrementNumLineSearches();
83 counter_->incrementNumNonTrivialLineSearches();
84 }
85
86 // Limit individual entries
87 *scaledUpdate_ = dir;
88 Teuchos::RCP< ::Thyra::VectorBase<double> > thyraUserLimits = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(userLimits_,true)->getThyraRCPVector();
89 Teuchos::RCP< ::Thyra::VectorBase<double> > thyraScaledUpdate = Teuchos::rcp_dynamic_cast<NOX::Thyra::Vector>(scaledUpdate_,true)->getThyraRCPVector();
90 ::Thyra::ele_wise_min_swap(*thyraUserLimits,thyraScaledUpdate.ptr());
91
92 // Now compute an equivalent "step" for this limiting (used by
93 // status test to make sure no limiting is going on at the converged
94 // solution)
95 limitDifference_->update(1.0,dir,-1.0,*scaledUpdate_,0.0);
96 double dirNorm = dir.norm(NOX::Abstract::Vector::TwoNorm);
97 // Protect against divide by zero corner case (for a perfectly converged solution)
98 if (dirNorm < 100.0 * Teuchos::ScalarTraits<double>::eps())
99 step = 1.0;
100 else
101 step = 1.0 - limitDifference_->norm(NOX::Abstract::Vector::TwoNorm) / dir.norm(NOX::Abstract::Vector::TwoNorm);
102
103 if (print_.isPrintType(NOX::Utils::Details)) {
104 print_.out () << "userLimits_:" << std::endl;
105 userLimits_->print(print_.out());
106 print_.out () << "scaledUpdate_:" << std::endl;
107 scaledUpdate_->print(print_.out());
108 print_.out () << "dir:" << std::endl;
109 dir.print(print_.out());
110 print_.out () << "Old Solution:" << std::endl;
111 newGrp.getX().print(print_.out());
112 }
113
114 if (print_.isPrintType(NOX::Utils::InnerIteration)) {
115 print_.out () << " approximated equivalent step = " << step << std::endl;
116 }
117
118 newGrp.computeX(newGrp,*scaledUpdate_,1.0);
119
120 if (print_.isPrintType(NOX::Utils::Details)) {
121 print_.out () << "New Solution:" << std::endl;
122 newGrp.getX().print(print_.out());
123 }
124
125 if (useCounter_)
126 counter_->setValues(*paramsPtr_);
127
128 return true;
129}
