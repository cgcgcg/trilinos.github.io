77 DirichletBC_ = DIR_ALL;
78 if (this->list_.get("left boundary", "Dirichlet") == "Neumann") DirichletBC_ ^= DIR_LEFT;
79 if (this->list_.get("right boundary", "Dirichlet") == "Neumann") DirichletBC_ ^= DIR_RIGHT;

82 if (this->list_.get("front boundary", "Dirichlet") == "Neumann") DirichletBC_ ^= DIR_FRONT;
83 if (this->list_.get("back boundary", "Dirichlet") == "Neumann") DirichletBC_ ^= DIR_BACK;
84 }
