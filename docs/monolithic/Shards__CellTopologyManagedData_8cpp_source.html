144CellTopologyManagedData::CellTopologyManagedData(
145 const std::string & name_,
146 const unsigned vertex_count_ ,
147 const unsigned node_count_,
148 const std::vector< const CellTopologyData * > & edges ,
149 const std::vector< unsigned > & edge_node_map ,
150 const CellTopologyData * pbase )
151 : m_name(name_),
152 m_subcell(),
153 m_node_map()
154{
155
156 // Compute size of the edge map & check edge homogeneity (suffices to compare nodes per edge)
157 const unsigned edge_count_ = edges.size();
158 unsigned edge_map_size = 0 ;
159 unsigned node_count_edge0 = edges[0]->node_count;
160 bool edge_homogeneity = true;
161
162 for ( unsigned i = 0 ; i < edge_count_ ; ++i ) {
163 edge_map_size += edges[i]->node_count ;
164 if(node_count_edge0 != edges[i]->node_count ) edge_homogeneity = false;
165 }
166
167 const bool error_base = pbase && (
168 pbase->base != pbase ||
169 pbase->vertex_count != pbase->node_count ||
170 pbase->vertex_count != vertex_count_ ||
171 pbase->edge_count != edges.size() );
172
173 const bool error_base_self = ! pbase && ( vertex_count_ != node_count_ );
174
175 const bool error_edge = edge_map_size != edge_node_map.size();
176
177 if ( error_base || error_base_self || error_edge ) {
178 // Throw an error
179 }
180
181 m_subcell.resize( 1 + edges.size() );
182 m_subcell[ edge_count_ ].topology = this ; // This subcell is the cell itself, will be assigned to subcell[2]
183 m_subcell[ edge_count_ ].node = index_identity_array();
184
185 m_node_map.resize( edge_map_size );
186
187 for ( unsigned i = 0 ; i < edge_map_size ; ++i ) {
188 m_node_map[i] = edge_node_map[i];
189 }
190
191 edge_map_size = 0 ;
192 for ( unsigned i = 0 ; i < edge_count_ ; ++i ) {
193 m_subcell[i].topology = edges[i] ;
194 m_subcell[i].node = & m_node_map[ edge_map_size ] ;
195 edge_map_size += edges[i]->node_count ;
196 }
197
198 pbase = (pbase == NULL ? this : pbase) ; // PBB 12-03-08
199 name = m_name.c_str();
200 key = cellTopologyKey( 2, 0, edge_count_, vertex_count_, node_count_ );
201 dimension = 2 ;
202 vertex_count = vertex_count_ ;
203 node_count = node_count_ ; // PBB 12-03-08
204 edge_count = edge_count_ ;
205 side_count = 0 ;
206 subcell_homogeneity[0] = 1 ;
207 subcell_homogeneity[1] = edge_homogeneity ;
208 subcell_homogeneity[2] = 0 ;
209 subcell_homogeneity[3] = 0 ;
210 subcell_count[0] = node_count_ ;
211 subcell_count[1] = edge_count_ ;
212 subcell_count[2] = 1 ; // PBB 12-03-08
213 subcell_count[3] = 0 ;
214 subcell[0] = subcell_nodes_array();
215 subcell[1] = & m_subcell[0] ;
216 subcell[2] = & m_subcell[ edge_count_ ] ;
217 subcell[3] = NULL ;
218 side = subcell[1] ;
219 edge = subcell[1] ;
220}

225CellTopologyManagedData::CellTopologyManagedData(
226 const std::string & name_,
227 const unsigned vertex_count_,
228 const unsigned node_count_,
229 const std::vector< const CellTopologyData * > & edges ,
230 const std::vector< unsigned > & edge_node_map ,
231 const std::vector< const CellTopologyData * > & faces ,
232 const std::vector< unsigned > & face_node_map ,
233 const CellTopologyData * pbase )
234 : m_name(name_),
235 m_subcell(),
236 m_node_map()
237
238{
239 const unsigned edge_count_ = edges.size();
240 unsigned edge_map_size = 0 ;
241 unsigned node_count_edge0 = edges[0]->node_count;
242 bool edge_homogeneity = true;
243
244 // Compute size of the edge map & check edge homogeneity (suffices to compare nodes per edge)
245 for ( unsigned i = 0 ; i < edge_count_ ; ++i ) {
246 edge_map_size += edges[i]->node_count ;
247 if(node_count_edge0 != edges[i]->node_count ) edge_homogeneity = false;
248 }
249
250 // Compute size of the face map & check face homogeneity (to do)
251 const unsigned face_count_ = faces.size();
252 unsigned face_map_size = 0 ;
253 for ( unsigned i = 0 ; i < face_count_ ; ++i ) {
254 face_map_size += faces[i]->node_count ;
255 }
256
257 // Set error flags for base, edges and faces
258 const bool error_base = pbase && (pbase->base != pbase ||
259 pbase->vertex_count != pbase->node_count ||
260 pbase->vertex_count != vertex_count_ ||
261 pbase->edge_count != edges.size() ||
262 pbase->side_count != faces.size() );
263
264 const bool error_base_self = ! pbase && ( vertex_count_ != node_count_ );
265
266 const bool error_edge = edge_map_size != edge_node_map.size();
267
268 const bool error_face = face_map_size != face_node_map.size();
269
270 if ( error_base || error_base_self || error_edge || error_face) {
271 // Throw an error
272 }
273
274 // Flat array for the 1,2,3-subcells of the custom cell.
275 m_subcell.resize( 1 + edges.size() + faces.size() );
276
277 m_subcell[ edge_count_ + face_count_].topology = this ; // The last subcell is the cell itself & will be assigned to subcell[3]
278 m_subcell[ edge_count_ + face_count_].node = index_identity_array();
279
280 // Flat array with edge nodes followed by face nodes (edges and faces can be non-homogeneous)
281 m_node_map.resize( edge_map_size + face_map_size);
282
283 // Copy edge nodes followed by face nodes
284 for ( unsigned i = 0 ; i < edge_map_size ; ++i ) {
285 m_node_map[i] = edge_node_map[i];
286 }
287 for ( unsigned i = 0 ; i < face_map_size ; ++i ) {
288 m_node_map[edge_map_size + i] = face_node_map[i];
289 }
290
291
292 // Copy edge topologies & list of nodes for each edge to m_subcell:
293 edge_map_size = 0 ;
294 for ( unsigned i = 0 ; i < edge_count_ ; ++i ) {
295 m_subcell[i].topology = edges[i] ;
296 m_subcell[i].node = & m_node_map[ edge_map_size ] ;
297 edge_map_size += edges[i]->node_count ;
298 }
299
300 // Copy face topologies & list of nodes for each face to m_subcell:
301 face_map_size = 0;
302 for ( unsigned i = 0 ; i < face_count_ ; ++i ) {
303 m_subcell[edge_count_ + i].topology = faces[i] ;
304 m_subcell[edge_count_ + i].node = & m_node_map[ edge_map_size + face_map_size ] ;
305 face_map_size += faces[i]->node_count ;
306 }
307
308 // Fill CellTopologyData with custom cell data: default base is the custom cell itself
309 pbase = (pbase == NULL ? this : pbase) ;
310 name = m_name.c_str();
311 key = cellTopologyKey( 3,
312 face_count_,
313 edge_count_,
314 vertex_count_,
315 node_count_ );
316 dimension = 3 ;
317 vertex_count = vertex_count_ ;
318 node_count = node_count_ ;
319 edge_count = edge_count_ ;
320 side_count = face_count_ ;
321 subcell_homogeneity[0] = 1 ;
322 subcell_homogeneity[1] = edge_homogeneity ;
323 subcell_homogeneity[2] = 0 ;
324 subcell_homogeneity[3] = 0 ;
325 subcell_count[0] = node_count_ ;
326 subcell_count[1] = edge_count_ ;
327 subcell_count[2] = face_count_ ;
328 subcell_count[3] = 1 ;
329 subcell[0] = subcell_nodes_array();
330 subcell[1] = & m_subcell[0] ;
331 subcell[2] = & m_subcell[ edge_count_ ] ;
332 subcell[3] = & m_subcell[ edge_count_ + face_count_] ;
333 side = subcell[2] ;
334 edge = subcell[1] ;
335}
