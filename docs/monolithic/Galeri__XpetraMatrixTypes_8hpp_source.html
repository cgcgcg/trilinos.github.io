77template <typename GlobalOrdinal>
78KOKKOS_FORCEINLINE_FUNCTION void GetNeighboursCartesian2dKokkos(const GlobalOrdinal i,
79 const GlobalOrdinal nx, const GlobalOrdinal ny,
80 GlobalOrdinal& left, GlobalOrdinal& right,
81 GlobalOrdinal& lower, GlobalOrdinal& upper,
82 const GlobalOrdinal INVALID) {
83 GlobalOrdinal ix, iy;
84 ix = i % nx;
85 iy = (i - ix) / nx;
86
87 if (ix == 0)
88 left = INVALID;
89 else
90 left = i - 1;
91 if (ix == nx - 1)
92 right = INVALID;
93 else
94 right = i + 1;
95 if (iy == 0)
96 lower = INVALID;
97 else
98 lower = i - nx;
99 if (iy == ny - 1)
100 upper = INVALID;
101 else
102 upper = i + nx;
103}

105template <typename GlobalOrdinal>
106KOKKOS_FORCEINLINE_FUNCTION void GetNeighboursCartesian3dKokkos(const GlobalOrdinal i,
107 const GlobalOrdinal nx, const GlobalOrdinal ny, const GlobalOrdinal nz,
108 GlobalOrdinal& left, GlobalOrdinal& right,
109 GlobalOrdinal& front, GlobalOrdinal& back,
110 GlobalOrdinal& bottom, GlobalOrdinal& top,
111 const GlobalOrdinal INVALID) {
112 GlobalOrdinal ixy, iz;
113 ixy = i % (nx * ny);
114
115 iz = (i - ixy) / (nx * ny);
116
117 if (iz == 0)
118 bottom = INVALID;
119 else
120 bottom = i - nx * ny;
121 if (iz == nz - 1)
122 top = INVALID;
123 else
124 top = i + nx * ny;
125
126 GetNeighboursCartesian2dKokkos(ixy, nx, ny, left, right, front, back, INVALID);
127
128 if (left != INVALID) left += iz * (nx * ny);
129 if (right != INVALID) right += iz * (nx * ny);
130 if (front != INVALID) front += iz * (nx * ny);
131 if (back != INVALID) back += iz * (nx * ny);
132}

134// This macro is used when we enter count up the number of nonzero entries per row.
135#define Galeri_processEntry(entry) \
136 { \
137 if (entry != INVALID) { \
138 ++partial_nnz; \
139 if (is_final) { \
140 if (lclMap.getLocalElement(entry) == INVALID) { \
141 off_rank_indices.insert(entry); \
142 } \
143 } \
144 } \
145 }
146
147// This macro is used when we enter values in the matrix.
148// We insert in a way that guarantees for rows to be sorted.
149#define Galeri_enterValue(rowStart, column_index, value) \
150 { \
151 if (column_index != INVALID) { \
152 auto clid = lclColMap.getLocalElement(column_index); \
153 auto K = entryPtr; \
154 while ((rowStart + 1 <= K) && (clid < colidx(K - 1))) { \
155 colidx(K) = colidx(K - 1); \
156 values(K) = values(K - 1); \
157 --K; \
158 } \
159 colidx(K) = clid; \
160 values(K) = value; \
161 if (i != column_index) \
162 offDiagonalSum += value; \
163 ++entryPtr; \
164 } \
165 }

711 KOKKOS_FORCEINLINE_FUNCTION
712 void GetNeighbours(const GlobalOrdinal i,
713 StencilIndices stencil_indices,
714 bool& isDirichlet) const {
715 GlobalOrdinal& below = stencil_indices[LOW][MID][MID];
716 GlobalOrdinal& front = stencil_indices[MID][LOW][MID];
717 GlobalOrdinal& left = stencil_indices[MID][MID][LOW];
718 GlobalOrdinal& right = stencil_indices[MID][MID][HIGH];
719 GlobalOrdinal& back = stencil_indices[MID][HIGH][MID];
720 GlobalOrdinal& above = stencil_indices[HIGH][MID][MID];
721
722 GetNeighboursCartesian3dKokkos(i, nx, ny, nz, left, right, front, back, below, above, INVALID);
723
724 stencil_indices[LOW][LOW][LOW] = below - nx - 1;
725 stencil_indices[LOW][LOW][MID] = below - nx;
726 stencil_indices[LOW][LOW][HIGH] = below - nx + 1;
727 stencil_indices[LOW][MID][LOW] = below - 1;
728 stencil_indices[LOW][MID][MID] = below;
729 stencil_indices[LOW][MID][HIGH] = below + 1;
730 stencil_indices[LOW][HIGH][LOW] = below + nx - 1;
731 stencil_indices[LOW][HIGH][MID] = below + nx;
732 stencil_indices[LOW][HIGH][HIGH] = below + nx + 1;
733
734 stencil_indices[MID][LOW][LOW] = front - 1;
735 stencil_indices[MID][LOW][MID] = front;
736 stencil_indices[MID][LOW][HIGH] = front + 1;
737 stencil_indices[MID][MID][LOW] = left;
738 stencil_indices[MID][MID][MID] = i;
739 stencil_indices[MID][MID][HIGH] = right;
740 stencil_indices[MID][HIGH][LOW] = back - 1;
741 stencil_indices[MID][HIGH][MID] = back;
742 stencil_indices[MID][HIGH][HIGH] = back + 1;
743
744 stencil_indices[HIGH][LOW][LOW] = above - nx - 1;
745 stencil_indices[HIGH][LOW][MID] = above - nx;
746 stencil_indices[HIGH][LOW][HIGH] = above - nx + 1;
747 stencil_indices[HIGH][MID][LOW] = above - 1;
748 stencil_indices[HIGH][MID][MID] = above;
749 stencil_indices[HIGH][MID][HIGH] = above + 1;
750 stencil_indices[HIGH][HIGH][LOW] = above + nx - 1;
751 stencil_indices[HIGH][HIGH][MID] = above + nx;
752 stencil_indices[HIGH][HIGH][HIGH] = above + nx + 1;
753
754 if (left == INVALID) {
755 for (size_t k0 = 0; k0 < 3; ++k0)
756 for (size_t k1 = 0; k1 < 3; ++k1)
757 stencil_indices[k0][k1][LOW] = INVALID;
758 }
759 if (right == INVALID) {
760 for (size_t k0 = 0; k0 < 3; ++k0)
761 for (size_t k1 = 0; k1 < 3; ++k1)
762 stencil_indices[k0][k1][HIGH] = INVALID;
763 }
764 if (front == INVALID) {
765 for (size_t k0 = 0; k0 < 3; ++k0)
766 for (size_t k2 = 0; k2 < 3; ++k2)
767 stencil_indices[k0][LOW][k2] = INVALID;
768 }
769 if (back == INVALID) {
770 for (size_t k0 = 0; k0 < 3; ++k0)
771 for (size_t k2 = 0; k2 < 3; ++k2)
772 stencil_indices[k0][HIGH][k2] = INVALID;
773 }
774 if (below == INVALID) {
775 for (size_t k1 = 0; k1 < 3; ++k1)
776 for (size_t k2 = 0; k2 < 3; ++k2)
777 stencil_indices[LOW][k1][k2] = INVALID;
778 }
779 if (above == INVALID) {
780 for (size_t k1 = 0; k1 < 3; ++k1)
781 for (size_t k2 = 0; k2 < 3; ++k2)
782 stencil_indices[HIGH][k1][k2] = INVALID;
783 }
784
785 isDirichlet = (left == INVALID && (DirichletBC & DIR_LEFT)) ||
786 (right == INVALID && (DirichletBC & DIR_RIGHT)) ||
787 (below == INVALID && (DirichletBC & DIR_BOTTOM)) ||
788 (above == INVALID && (DirichletBC & DIR_TOP)) ||
789 (front == INVALID && (DirichletBC & DIR_FRONT)) ||
790 (back == INVALID && (DirichletBC & DIR_BACK));
791 }

793 KOKKOS_FORCEINLINE_FUNCTION
794 bool IsBoundary(const GlobalOrdinal i) const {
795 GlobalOrdinal ix = i % nx;
796 GlobalOrdinal ixy = i % (nx * ny);
797 GlobalOrdinal iy = (ixy - ix) / nx;
798 GlobalOrdinal iz = (i - ixy) / (nx * ny);
799 return (ix == 0 || ix == nx - 1 || iy == 0 || iy == ny - 1 || iz == 0 || iz == nz - 1);
800 }

897 const GlobalOrdinal nx, // note: nx unused
898 const Scalar a, const Scalar b, const Scalar c) {
899 using Teuchos::RCP;
900 using Teuchos::rcp;
901 using Teuchos::TimeMonitor;
902
903 Teuchos::RCP<Matrix> mtx = MatrixTraits<Map, Matrix>::Build(map, 3);
904
905 LocalOrdinal NumMyElements = map->getLocalNumElements();
906 Teuchos::ArrayView<const GlobalOrdinal> MyGlobalElements = map->getLocalElementList();
907 GlobalOrdinal indexBase = map->getIndexBase();
908
909 Teuchos::RCP<const Teuchos::Comm<int>> comm = map->getComm();

912
913 GlobalOrdinal NumEntries;

917
918 comm->barrier();

920 // c a b

926 // off-diagonal for first row
927 Indices[0] = 1 + indexBase;
928 Values[0] = c;
929 NumEntries = 1;
930
931 } else if (MyGlobalElements[i] == NumGlobalElements + indexBase - 1) {
932 // off-diagonal for last row
933 Indices[0] = NumGlobalElements - 2 + indexBase;
934 Values[0] = b;
935 NumEntries = 1;
936
937 } else {
938 // off-diagonal for internal row
939 Indices[0] = MyGlobalElements[i] - 1;
940 Values[0] = b;
941 Indices[1] = MyGlobalElements[i] + 1;
942 Values[1] = c;
943 NumEntries = 2;
944 }
945
946 // put the off-diagonal entries
947 // Xpetra wants ArrayViews (sigh)
948 Teuchos::ArrayView<Scalar> av(&Values[0], NumEntries);
949 Teuchos::ArrayView<GlobalOrdinal> iv(&Indices[0], NumEntries);
950 mtx->insertGlobalValues(MyGlobalElements[i], iv, av);
951
952 // Put in the diagonal entry
953 mtx->insertGlobalValues(MyGlobalElements[i],
954 Teuchos::tuple<GlobalOrdinal>(MyGlobalElements[i]),
955 Teuchos::tuple<Scalar>(a));
956 }
957 }
958
959 {
960 Teuchos::RCP<TimeMonitor> tm = rcp(new TimeMonitor(*TimeMonitor::getNewTimer("Galeri: Laplace 1D fillComplete")));
961 mtx->fillComplete();
962 }

970template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix>
971Teuchos::RCP<Matrix>
972Cross2D(const Teuchos::RCP<const Map>& map,
973 const GlobalOrdinal nx, const GlobalOrdinal ny,
974 const Scalar a, const Scalar b, const Scalar c,
975 const Scalar d, const Scalar e,
976 const DirBC DirichletBC = 0, const bool keepBCs = false) {
977 using Teuchos::RCP;
978 using Teuchos::rcp;
979 using Teuchos::TimeMonitor;
980
981 LocalOrdinal nnz = 5;
982
983 RCP<Matrix> mtx = MatrixTraits<Map, Matrix>::Build(map, nnz);
984
985 LocalOrdinal numMyElements = map->getLocalNumElements();
986 GlobalOrdinal indexBase = map->getIndexBase();
987
988 Teuchos::ArrayView<const GlobalOrdinal> myGlobalElements = map->getLocalElementList();
989
990 GlobalOrdinal center, left, right, lower, upper;
991 std::vector<Scalar> vals(nnz);
992 std::vector<GlobalOrdinal> inds(nnz);
993
994 // e
995 // b a c
996 // d
997 {
998 Teuchos::RCP<TimeMonitor> tm = rcp(new TimeMonitor(*TimeMonitor::getNewTimer("Galeri: Laplace 2D Generation")));
999 for (LocalOrdinal i = 0; i < numMyElements; ++i) {
1000 size_t n = 0;
1001
1002 center = myGlobalElements[i] - indexBase;
1003 GetNeighboursCartesian2d(center, nx, ny, left, right, lower, upper);
1004
1005 bool isDirichlet = (left == -1 && (DirichletBC & DIR_LEFT)) ||
1006 (right == -1 && (DirichletBC & DIR_RIGHT)) ||
1007 (lower == -1 && (DirichletBC & DIR_BOTTOM)) ||
1008 (upper == -1 && (DirichletBC & DIR_TOP));
1009
1010 if (isDirichlet && keepBCs) {
1011 // Dirichlet unknown we want to keep
1012 inds[n] = center;
1013 vals[n++] = Teuchos::ScalarTraits<Scalar>::one();
1014
1015 } else {
1016 // The Neumann b.c. are treated in a sane way. The Dirichlet b.c., however, are treated
1017 // insane when the option keepBCs=false. Speicifically, in this case we don't want to keep
1018 // Dirichlet b.c., but that would result in inconsistency between the map and the number of
1019 // degrees of freedom, plus the problem with GIDs. Therefore, we virtually expand domain by
1020 // one node in the direction of the Dirichlet b.c., and then assume that that node was
1021 // not kept. But we use an old GIDs. So yes, that's weird.
1022
1023 if (left != -1) {
1024 inds[n] = left;
1025 vals[n++] = b;
1026 }
1027 if (right != -1) {
1028 inds[n] = right;
1029 vals[n++] = c;
1030 }
1031 if (lower != -1) {
1032 inds[n] = lower;
1033 vals[n++] = d;
1034 }
1035 if (upper != -1) {
1036 inds[n] = upper;
1037 vals[n++] = e;
1038 }
1039
1040 // diagonal
1041 Scalar z = a;
1042 if (IsBoundary2d(center, nx, ny) && !isDirichlet) {
1043 // Neumann boundary unknown (diagonal = sum of all offdiagonal)
1044 z = Teuchos::ScalarTraits<Scalar>::zero();
1045 for (size_t j = 0; j < n; j++)
1046 z -= vals[j];
1047 }
1048 inds[n] = center;
1049 vals[n++] = z;
1050 }
1051
1052 for (size_t j = 0; j < n; j++)
1053 inds[j] += indexBase;
1054
1055 Teuchos::ArrayView<GlobalOrdinal> iv(&inds[0], n);
1056 Teuchos::ArrayView<Scalar> av(&vals[0], n);
1057 mtx->insertGlobalValues(myGlobalElements[i], iv, av);
1058 }
1059 }
1060
1061 {
1062 Teuchos::RCP<TimeMonitor> tm = rcp(new TimeMonitor(*TimeMonitor::getNewTimer("Galeri: Laplace 2D FillComplete")));
1063 mtx->fillComplete();
1064 }
1065
1066 return mtx;
1067}

1506template <typename Matrix, typename Map, typename Stencil>
1507Teuchos::RCP<Matrix>
1508StencilMatrixKokkos(const Teuchos::RCP<const Map>& map,
1509 Stencil& stencil,
1510 const std::string& matLabel) {
1511 using Teuchos::RCP;
1512 using Teuchos::rcp;
1513 using Teuchos::TimeMonitor;
1514 using Scalar = typename Matrix::scalar_type;
1515 using LocalOrdinal = typename Map::local_ordinal_type;
1516 using GlobalOrdinal = typename Map::global_ordinal_type;
1517 using Node = typename Map::node_type;
1518 using local_matrix_type = typename ::Xpetra::Matrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>::local_matrix_type;
1519 using rowptr_type = typename local_matrix_type::row_map_type::non_const_type;
1520 using colidx_type = typename local_matrix_type::index_type::non_const_type;
1521 using values_type = typename local_matrix_type::values_type::non_const_type;
1522 using tpetra_map = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
1523 using exec_space = typename Node::execution_space;
1524 using memory_space = typename Node::memory_space;
1525 using range_type = Kokkos::RangePolicy<LocalOrdinal, exec_space>;
1526
1527 // We perform the assembly in 3 steps:
1528 //
1529 // 1) We count the number of entries per row and collect the off-rank entries in a hashmap.
1530 // 2) We construct the column map from the row map and the hashmap.
1531 // 3)_We allocate indices and values and enter the values.
1532
1533 Teuchos::RCP<TimeMonitor> tm = rcp(new TimeMonitor(*TimeMonitor::getNewTimer("Galeri: " + matLabel + " generation")));
1534
1536 // Step 1 - rowptr
1537
1538 LocalOrdinal numMyElements = map->getLocalNumElements();
1539 // The hash map that collects all off-rank entries that we encounter.
1540 // We need to provide sufficient capacity so that no insertions fail.
1541 stencil.off_rank_indices = Kokkos::UnorderedMap<GlobalOrdinal, void, exec_space>(std::max(numMyElements, 100));
1542
1543 auto lclMap = map->getLocalMap();
1544 // The row offsets of the matrix, initialized to zero.
1545 auto rowptr = rowptr_type("rowptr", numMyElements + 1);
1546
1547 // We loop over the rows and over the stencil.
1548 // We count entries of row i using rowptr(i+2) and accumulate the results.
1549 //
1550 // E.g if we have a matrix with three rows with 2, 3 and 4 entries respectively, rowptr will look like
1551 // [0 0 2 5]
1552 // and myNNZ = 2+3+4 = 9.
1553 //
1554 // Note that this is not yet the correct rowptr for the matrix which looks like
1555 // [0 2 5 9]
1556 //
1557 // The reason why we are not constructing the correct rowptr right away is that we will use
1558 // rowptr(i+1) as the offset for filling the colidx and values views.
1559 LocalOrdinal myNNZ = 0; // The number of nonzero entries on this rank.
1560 Kokkos::parallel_scan(
1561 "Galeri::" + matLabel + "::Graph", range_type(0, numMyElements),
1562 KOKKOS_LAMBDA(const LocalOrdinal i, LocalOrdinal& partial_nnz,
1563 const bool is_final) {
1564 stencil.CountRowNNZ(i, partial_nnz, is_final);
1565 if (is_final) {
1566 if (i + 2 < numMyElements + 1)
1567 rowptr(i + 2) = partial_nnz;
1568 }
1569 },
1570 myNNZ);
1571 // We check that no hashmap insertions failed.
1572 // If this turns out to not be enough we can bump up the capacity.
1573 TEUCHOS_ASSERT(!stencil.off_rank_indices.failed_insert());
1574
1576 // Step 2 - Column map
1577
1578 auto columnMap_entries = Kokkos::View<GlobalOrdinal*, memory_space>("columnMap_entries", numMyElements + stencil.off_rank_indices.size());
1579 // Copy over on-rank entries from rowmap
1580 Kokkos::deep_copy(Kokkos::subview(columnMap_entries, Kokkos::make_pair(0, numMyElements)),
1581 map->getMyGlobalIndicesDevice());
1582 // Set off-rank entries from hashmap.
1583 // This is the Kokkos recommended way of iterating, see
1584 // https://kokkos.org/kokkos-core-wiki/API/containers/Unordered-Map.html#iteration
1585 Kokkos::parallel_scan(
1586 "Galeri::" + matLabel + "::Graph", range_type(0, stencil.off_rank_indices.capacity()), KOKKOS_LAMBDA(const LocalOrdinal i, size_t& pos, const bool is_final) {
1587 if (stencil.off_rank_indices.valid_at(i)) {
1588 if (is_final) {
1589 auto key = stencil.off_rank_indices.key_at(i);
1590 columnMap_entries(numMyElements + pos) = key;
1591 }
1592 ++pos;
1593 }
1594 });
1595 // We no longer need the hashmap.
1596 stencil.off_rank_indices.clear();
1597 stencil.off_rank_indices.rehash(0);
1598
1599 {
1600 // Sort the off-rank entries by PID
1601 auto remoteGIDs = Kokkos::subview(columnMap_entries, Kokkos::make_pair(numMyElements, columnMap_entries.extent_int(0)));
1602 auto remotePIDs = Kokkos::View<int*, memory_space>("remotePIDs", remoteGIDs.extent(0));
1603 {
1604 auto remoteGIDs_h = Kokkos::create_mirror_view(remoteGIDs);
1605 auto remotePIDs_h = Kokkos::create_mirror_view(remotePIDs);
1606 Kokkos::deep_copy(remoteGIDs_h, remoteGIDs);
1607 Teuchos::ArrayView<GlobalOrdinal> remoteGIDs_av(remoteGIDs_h.data(), remoteGIDs_h.extent(0));
1608 Teuchos::ArrayView<int> remotePIDs_av(remotePIDs_h.data(), remotePIDs_h.extent(0));
1609 auto ret = map->getRemoteIndexList(remoteGIDs_av, remotePIDs_av);
1610 if constexpr (std::is_same_v<Map, tpetra_map>) {
1611 TEUCHOS_ASSERT(ret == Tpetra::AllIDsPresent);
1612 } else {
1613 TEUCHOS_ASSERT(ret == ::Xpetra::AllIDsPresent);
1614 }
1615 Kokkos::deep_copy(remotePIDs, remotePIDs_h);
1616 }
1617 {
1618 typename decltype(remoteGIDs)::execution_space exec;
1619 Kokkos::Experimental::sort_by_key(exec, remotePIDs, remoteGIDs);
1620 }
1621 }
1622
1623 RCP<Map> ghosted_map;
1624 if constexpr (std::is_same_v<Map, tpetra_map>) {
1625 ghosted_map = rcp(new Map(Teuchos::OrdinalTraits<GlobalOrdinal>::invalid(),
1626 columnMap_entries,
1627 map->getIndexBase(),
1628 map->getComm()));
1629 } else {
1630 ghosted_map = ::Xpetra::MapFactory<typename Matrix::local_ordinal_type, typename Matrix::global_ordinal_type, typename Matrix::node_type>::Build(map->lib(),
1631 Teuchos::OrdinalTraits<GlobalOrdinal>::invalid(),
1632 columnMap_entries,
1633 map->getIndexBase(),
1634 map->getComm());
1635 }
1636
1638 // Step 3 - Fill
1639
1640 stencil.lclColMap = ghosted_map->getLocalMap();
1641 stencil.colidx = colidx_type("colidx", myNNZ);
1642 stencil.values = values_type("values", myNNZ);
1643
1644 // Loop over rows and stencil and fill the matrix.
1645 // To enter values in row i we use rowptr(i+1) as offset.
1646 //
1647 // Using the same example as above, after this step rowptr is
1648 // [0 2 5 9]
1649 Kokkos::parallel_for(
1650 "Galeri::" + matLabel + "::fill", range_type(0, numMyElements),
1651 KOKKOS_LAMBDA(const LocalOrdinal i) {
1652 stencil.EnterValues(i, rowptr(i + 1));
1653 });
1654
1655 auto lclA = local_matrix_type(matLabel, numMyElements, ghosted_map->getLocalNumElements(), myNNZ, stencil.values, rowptr, stencil.colidx);
1656 auto mtx = MatrixTraits<Map, Matrix>::Build(lclA, map, ghosted_map, map, map);
1657
1658#ifdef HAVE_GALERI_DEBUG
1659 // Checks to run in debug mode:
1660 // - local CRS graph is sorted by row
1661 // - imports can be aliased to target MV
1662 // - remote entries of column map are sorted by PID so that transfers are faster
1663
1664 TEUCHOS_ASSERT(::KokkosSparse::isCrsGraphSorted(lclA.graph.row_map,
1665 lclA.graph.entries));
1666
1667 if constexpr (std::is_same_v<Map, tpetra_map>) {
1668 auto import = mtx->getCrsGraph()->getImporter();
1669 if (!import.is_null()) {
1670 TEUCHOS_ASSERT(import->areRemoteLIDsContiguous());
1671
1672 auto distor = import->getDistributor();
1673 TEUCHOS_ASSERT(distor.getPlan().getIndicesTo().is_null());
1674 }
1675 } else if constexpr (std::is_same_v<Matrix, ::Xpetra::TpetraCrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>>) {
1676 auto import = mtx->getCrsGraph()->getImporter();
1677 if (!import.is_null()) {
1678 auto xp_tp_import = Teuchos::rcp_dynamic_cast<const ::Xpetra::TpetraImport<LocalOrdinal, GlobalOrdinal, Node>>(import, true);
1679 auto tp_import = xp_tp_import->getTpetra_Import();
1680 TEUCHOS_ASSERT(tp_import->areRemoteLIDsContiguous());
1681
1682 auto distor = tp_import->getDistributor();
1683 TEUCHOS_ASSERT(distor.getPlan().getIndicesTo().is_null());
1684 }
1685 } else {
1686 auto import = mtx->getCrsMatrix()->getCrsGraph()->getImporter();
1687 if (!import.is_null()) {
1688 auto xp_tp_import = Teuchos::rcp_dynamic_cast<const ::Xpetra::TpetraImport<LocalOrdinal, GlobalOrdinal, Node>>(import, true);
1689 auto tp_import = xp_tp_import->getTpetra_Import();
1690 const bool areRemoteLIDsContiguous = tp_import->areRemoteLIDsContiguous();
1691 TEUCHOS_ASSERT(areRemoteLIDsContiguous);
1692
1693 auto distor = tp_import->getDistributor();
1694 const bool fastPath = distor.getPlan().getIndicesTo().is_null();
1695 TEUCHOS_ASSERT(fastPath);
1696 }
1697 }
1698#endif // HAVE_GALERI_DEBUG
1699
1700 return mtx;
1701}
