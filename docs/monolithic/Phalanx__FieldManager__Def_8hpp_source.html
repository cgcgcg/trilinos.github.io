94// **************************************************************
95template<typename Traits>
96template<typename EvalT, typename DataT, typename Layout>
97inline
98void PHX::FieldManager<Traits>::
99getFieldData(const PHX::FieldTag& ft, Kokkos::View<DataT,Layout,PHX::Device>& f)
100{
101 std::any a = m_eval_containers.template
102 getAsObject<EvalT>()->getFieldData(ft);
103
104 // std::any object is always the non-const data type. To
105 // correctly cast the any object to the Kokkos::View, need to
106 // pull the const off the scalar type if this MDField has a
107 // const scalar type.
108 typedef PHX::View<typename Kokkos::View<DataT,Layout,PHX::Device>::non_const_data_type> non_const_view;
109 try {
110 non_const_view tmp = std::any_cast<non_const_view>(a);
111 f = tmp;
112 }
113 catch (std::exception& ) {
114 std::cout << "\n\nError in FieldManager::getFieldData() using std::any_cast. Tried to cast a field "
115 << "\" to a type of \"" << Teuchos::demangleName(typeid(non_const_view).name())
116 << "\" from a std::any object containing a type of \""
117 << Teuchos::demangleName(a.type().name()) << "\"." << std::endl;
118 throw;
119 }
120}

145// **************************************************************
146template<typename Traits>
147template<typename EvalT, typename DataT, typename Layout>
148inline
149void PHX::FieldManager<Traits>::
150setUnmanagedField(const PHX::FieldTag& ft, Kokkos::View<DataT,Layout,PHX::Device>& f,
151 const bool cleanup_output)
152{
153 // Make sure field data type is not const. We always store static
154 // non-const views so that we know how to cast back from an any
155 // object.
156 typedef typename Kokkos::View<DataT,Layout,PHX::Device>::value_type value_type;
157 typedef typename Kokkos::View<DataT,Layout,PHX::Device>::non_const_value_type non_const_value_type;
158 static_assert(std::is_same<value_type,non_const_value_type>::value, "FieldManager::setUnmanagedField(FieldTag, View) - DataT must be non-const!");
159
160 std::any any_f(f);
161 m_eval_containers.template getAsObject<EvalT>()->setUnmanagedField(ft,any_f,cleanup_output);
162}
