30 //Local Lower Sparse Tri-Solve
31 template <class Int, class Entry, class Exe_Space>
32 BASKER_INLINE
33 int local_lower_tri_solve
34 (
35 Int kid,
36 BASKER_MATRIX &L,
37 INT_1DARRAY &pinv,
38 const ENTRY_2DARRAY &x,
39 const ENTRY_2DARRAY &b
40 )
41 {
42 //over all rhs
43 //for(Int i=0; i < b.size(); i++)
44 for(Int i = 0; i < 1; i++)
45 {
46 //over all columns
47 for(Int k = 0; k < L.ncol; k++)
48 {
49
50 if(L.val[L.col_ptr[k]] == (Entry) 0)
51 {
52 printf("kid: %d L found zero pivot in column: %d \n",
53 kid, k+L.scol);
54 return -1;
55 }
56
57 #ifdef BASKER_DEBUG_SOLVE
58 printf("kid b(1) = %f \n", b[i][1]);
59 printf("kid: %d %d x(%d) = %f b(%d) = %f\n",
60 kid, i , k+L.scol, x[i][k+L.scol], k+L.scol, b[i][k+L.scol]);
61 #endif
62
63 x[i][k+L.scol] = b[i][k+L.scol] / L.val[L.col_ptr[k]];
64
65 //Over each nnz in column
66 for(Int j = L.col_ptr[k]+1; j < L.col_ptr[k+1]; j++)
67 {
68 Int loc = pinv[L.row_idx[j]];
69
70 #ifdef BASKER_DEBUG_SOLVE_RHS_LOWER
71 printf("kid %d L update b: %d %d %f value: %f %f \n",
72 kid, i,loc, b[i][loc], L.val[j], x[i][k+L.scol]);
73 #endif
74
75 //Add Atomic: come back... only do atomics for shared vals
76 Kokkos::atomic_fetch_sub(&(b[i][loc]),
77 L.val[j]*x[i][k+L.scol]);
78
79 }//end overall nnz in column
80 }//end over all columns
81 }//end over all rhs
82
83 return 0;
84 }//end local_lower_tri_solve

86 //Local Upper Sparse Tri-Solve
87 template <class Int, class Entry, class Exe_Space>
88 BASKER_INLINE
89 int local_upper_tri_solve
90 (
91 Int kid,
92 BASKER_MATRIX U,
93 INT_1DARRAY pinv,
94 const ENTRY_2DARRAY x,
95 const ENTRY_2DARRAY b
96 )
97 {
98 //over all rhs
99 //for(Int i=0; i < b.size(); i++)
100 for(Int i = 0; i < 1; i++)
101 {
102 //over all columns
103 for(Int k = U.ncol-1; k > 0; k--)
104 {
105
106 if(U.val[U.col_ptr[k+1]-1] == (Entry) 0)
107 {
108 printf("kid: %d U found zero pivot in column: %d \n",
109 kid, k+U.scol);
110 }
111
112 #ifdef BASKER_DEBUG_SOLVE_RHS_UPPER
113 printf("kid: %d x(%d) = %f b(%d) = %f U.val: %f \n",
114 kid, k+U.scol, x[i][k+U.scol],
115 k+U.scol, b[i][k+U.scol],
116 U.val[U.col_ptr[k+1]-1]);
117 #endif
118
119 x[i][k+U.scol] = b[i][k+U.scol] / U.val[U.col_ptr[k+1]-1];
120
121 //Over each nnz in column
122 for(Int j = U.col_ptr[k+1]-2; j >= U.col_ptr[k]; j--)
123 {
124 Int loc = U.row_idx[j];
125 //No need for atomic, only over own blk
126 b[i][loc] = b[i][loc] - U.val[j]*x[i][k+U.scol];
127 }//end overall nnz in column
128
129 }//end over all columns
130
131 x[i][U.scol] = b[i][U.scol]/U.val[0];
132
133 }//end over all rhs
134
135 return 0;
136 }//end local_upper_tri_solve()

141 int local_spmv_update
142 (
143 Int kid,
144 BASKER_MATRIX &A,
145 INT_1DARRAY &pinv,
146 const ENTRY_2DARRAY &inX,
147 const ENTRY_2DARRAY &outB,
148 int *info
149 )
150 {
151 if(outB.size() != inX.size())
152 {return -1;}
153
154 //over each vector (can be moved to inside for better performance
155 //for(Int i=0; i < inX.size(); i++)
156 for(Int i = 0; i < 1; i++)
157 {
158 //over each column
159 for(Int k = 0; k < A.ncol; k++)
160 {
161 //over each nnz in a column
162 for(Int j=A.col_ptr[k]; j<A.col_ptr[k+1]; j++)
163 {
164 #ifdef BASKER_DEBUG_SOLVE_RHS_UPPER
165 printf("kid: %d, b(%d) = %f x(%d) = %f Aval: %f \n",
166 kid, A.row_idx[j], outB[i][A.row_idx[j]],
167 A.scol+k, inX[i][A.scol+k], A.val[j]);
168 #endif
169
170 outB[i][A.row_idx[j]] -= A.val[j]*inX[i][A.scol+k];
171 }//end over all nnz in a column
172 }//end over each column
173 }//end over each vector
174
175 return 0;
176 }//end local_spmv
