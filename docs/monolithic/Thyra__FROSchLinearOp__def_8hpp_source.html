77 template <class SC, class LO, class GO, class NO>
78 bool FROSchLinearOp<SC,LO,GO,NO>::opSupportedImpl(EOpTransp M_trans) const
79 {
80 if (is_null(xpetraOperator_))
81 return false;
82
83 if (M_trans == NOTRANS)
84 return true;
85
86 if (M_trans == CONJ) {
87 // For non-complex scalars, CONJ is always supported since it is equivalent to NO_TRANS.
88 // For complex scalars, Xpetra does not support conjugation without transposition.
89 return !ScalarTraits<SC>::isComplex;
90 }
91
92 return xpetraOperator_->hasTransposeApply();
93 }

95 template <class SC, class LO, class GO, class NO>
96 void FROSchLinearOp<SC,LO,GO,NO>::applyImpl(const EOpTransp M_trans,
97 const MultiVectorBase<SC> &X_in,
98 const Ptr<MultiVectorBase<SC> > &Y_inout,
99 const SC alpha,
100 const SC beta) const
101 {
102 FROSCH_ASSERT(getConstXpetraOperator()!=null,"XpetraLinearOp::applyImpl: internal Operator is null.");
103 RCP< const Comm<int> > comm = getConstXpetraOperator()->getRangeMap()->getComm();
104 //Transform to Xpetra MultiVector
105 RCP<MultiVector<SC,LO,GO,NO> > xY;
106
107 ETransp transp = NO_TRANS;
108 switch (M_trans) {
109 case NOTRANS: transp = NO_TRANS; break;
110 case TRANS: transp = Teuchos::TRANS; break;
111 case CONJTRANS: transp = CONJ_TRANS; break;
112 default: FROSCH_ASSERT(false,"Thyra::XpetraLinearOp::apply. Unknown value for M_trans. Only NOTRANS, TRANS and CONJTRANS are supported.");
113 }
114 if (bIsTpetra_) {
115 // Convert input vector to Xpetra
116 const RCP<const Tpetra::MultiVector<SC,LO,GO,NO> > xTpMultVec = Thyra::TpetraOperatorVectorExtraction<SC,LO,GO,NO>::getConstTpetraMultiVector(rcpFromRef(X_in));
117 TEUCHOS_TEST_FOR_EXCEPT(is_null(xTpMultVec));
118 RCP<Tpetra::MultiVector<SC,LO,GO,NO> > tpNonConstMultVec = rcp_const_cast<Tpetra::MultiVector<SC,LO,GO,NO> >(xTpMultVec);
119 TEUCHOS_TEST_FOR_EXCEPT(is_null(tpNonConstMultVec));
120 const RCP<const Xpetra::MultiVector<SC,LO,GO,NO> > xX = rcp(new Xpetra::TpetraMultiVector<SC,LO,GO,NO>(tpNonConstMultVec));
121 TEUCHOS_TEST_FOR_EXCEPT(is_null(xX));
122
123 // Convert output vector to Xpetra
124 const RCP<Tpetra::MultiVector<SC,LO,GO,NO> > yTpMultVec = Thyra::TpetraOperatorVectorExtraction<SC,LO,GO,NO>::getTpetraMultiVector(rcpFromPtr(Y_inout));
125 TEUCHOS_TEST_FOR_EXCEPT(is_null(yTpMultVec));
126 xY = rcp(new Xpetra::TpetraMultiVector<SC,LO,GO,NO>(yTpMultVec));
127 TEUCHOS_TEST_FOR_EXCEPT(is_null(xY));
128
129 // Apply operator
130 xpetraOperator_->apply(*xX, *xY, transp, alpha, beta);

132 } else {
