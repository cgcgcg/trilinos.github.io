25template <typename S>
26class InnerProductSpaceTraits< Sacado::UQ::PCE<S> > {
27public:
28 typedef Sacado::UQ::PCE<S> val_type;
29
30 typedef typename val_type::value_type base_value_type;
31 typedef typename val_type::ordinal_type ordinal_type;
32 typedef InnerProductSpaceTraits<base_value_type> BIT;
33 typedef typename BIT::dot_type base_dot_type;
34
35#if KOKKOS_VERSION >= 40799
36 typedef typename KokkosKernels::ArithTraits<val_type>::mag_type mag_type;
37#else
38 typedef typename Kokkos::ArithTraits<val_type>::mag_type mag_type;
39#endif
40 typedef base_dot_type dot_type;
41
42 static KOKKOS_FORCEINLINE_FUNCTION
43 mag_type norm (const val_type& x) {
44 //return ArithTraits<val_type>::abs (x);
45 const ordinal_type sz = x.size();
46 mag_type nrm = mag_type(0);
47 for (ordinal_type i=0; i<sz; ++i) {
48 const mag_type n = BIT::norm( x.fastAccessCoeff(i) );
49 nrm += n*n;
50 }
51 return std::sqrt(nrm);
52 }
53
54 static KOKKOS_FORCEINLINE_FUNCTION
55 dot_type dot (const val_type& x, const val_type& y) {
56 const ordinal_type xsz = x.size();
57 const ordinal_type ysz = y.size();
58 const ordinal_type sz = xsz > ysz ? xsz : ysz;
59
60 dot_type r = dot_type(0);
61 if (x.hasFastAccess(sz) && y.hasFastAccess(sz))
62 for (ordinal_type i=0; i<sz; ++i)
63 r += BIT::dot( x.fastAccessCoeff(i), y.fastAccessCoeff(i) );
64 else
65 for (ordinal_type i=0; i<sz; ++i)
66 r += BIT::dot( x.coeff(i), y.coeff(i) );
67
68 return r;
69 }
70
71};
