24template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
25class Elasticity2DProblem : public Problem<Map, Matrix, MultiVector> {
26 public:
27 using RealValuedMultiVector = typename Problem<Map, Matrix, MultiVector>::RealValuedMultiVector;
28 Elasticity2DProblem(Teuchos::ParameterList& list, const Teuchos::RCP<const Map>& map)
29 : Problem<Map, Matrix, MultiVector>(list, map) {
30 E = list.get("E", Teuchos::as<typename Teuchos::ScalarTraits<Scalar>::magnitudeType>(1e9));
31 nu = list.get("nu", Teuchos::as<typename Teuchos::ScalarTraits<Scalar>::magnitudeType>(0.25));
32
33 nx_ = -1;
34 ny_ = -1;
35
36 if (list.isParameter("nx")) {
37 if (list.isType<int>("nx"))
38 nx_ = Teuchos::as<GlobalOrdinal>(list.get<int>("nx"));
39 else
40 nx_ = list.get<GlobalOrdinal>("nx");
41 }
42 if (list.isParameter("ny")) {
43 if (list.isType<int>("ny"))
44 ny_ = Teuchos::as<GlobalOrdinal>(list.get<int>("ny"));
45 else
46 ny_ = list.get<GlobalOrdinal>("ny");
47 }
48
49 nDim_ = 2;
50 double one = 1.0;
51 stretch.push_back(list.get("stretchx", one));
52 stretch.push_back(list.get("stretchy", one));
53
54 // NOTE: -1 is because galeri counts points, not elements
55 dims.push_back(nx_ - 1);
56 dims.push_back(ny_ - 1);
57
58 TEUCHOS_TEST_FOR_EXCEPTION(nx_ <= 0 || ny_ <= 0, std::logic_error, "nx and ny must be positive");
59 mode_ = list.get<std::string>("mode", "plane stress");
60 }
61
62 Teuchos::RCP<Matrix> BuildMatrix();
63 Teuchos::RCP<MultiVector> BuildNullspace();
64 Teuchos::RCP<RealValuedMultiVector> BuildCoords();
65
66 private:
67 typedef Scalar SC;
68 typedef LocalOrdinal LO;
69 typedef GlobalOrdinal GO;
70
71 struct Point {
72 SC x, y, z;
73
74 Point() { z = Teuchos::ScalarTraits<SC>::zero(); }
75 Point(SC x_, SC y_, SC z_ = Teuchos::ScalarTraits<SC>::zero())
76 : x(x_)
77 , y(y_)
78 , z(z_) {}
79 };
80
81 GlobalOrdinal nx_, ny_, nz_;
82 size_t nDim_;
83 std::vector<GO> dims;
84 // NOTE: nodes correspond to a local subdomain nodes. I have to construct overlapped subdomains because
85 // InsertGlobalValues in Epetra does not support inserting into rows owned by other processor
86 std::vector<Point> nodes_;
87 std::vector<std::vector<LO> > elements_;
88 std::vector<GO> local2Global_;
89
90 std::vector<char> dirichlet_;
91
92 typename Teuchos::ScalarTraits<Scalar>::magnitudeType E, nu;
93 std::vector<Scalar> stretch;
94 std::string mode_;
95
96#if KOKKOS_VERSION >= 40799
97 using impl_scalar_type = typename KokkosKernels::ArithTraits<SC>::val_type;
98#else
99 using impl_scalar_type = typename Kokkos::ArithTraits<SC>::val_type;
100#endif
101#if KOKKOS_VERSION >= 40799
102 using KAT = KokkosKernels::ArithTraits<impl_scalar_type>;
103#else
104 using KAT = Kokkos::ArithTraits<impl_scalar_type>;
105#endif
106 using Memory2D = Kokkos::View<impl_scalar_type**, Kokkos::HostSpace>;

108 void EvalD(const std::vector<Point>& refPoints, Point& gaussPoint, Memory2D S);

115template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
116Teuchos::RCP<Matrix> Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
117 BuildMesh();
118
119 const size_t numDofPerNode = 2;
120 const size_t numNodesPerElem = 4;
121 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
122
123 TEUCHOS_TEST_FOR_EXCEPTION(elements_[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
124
125 // Material constant
126 SC t = 1;
127
128 // Material matrix
129 RCP<Memory2D> D(new Memory2D);
130 BuildMaterialMatrix(*D);
131
132 // Reference element, and reference Gauss points
133 size_t numRefPoints, numGaussPoints;
134 std::vector<Point> refPoints, gaussPoints;
135 BuildReferencePoints(numRefPoints, refPoints, numGaussPoints, gaussPoints);
136
137 // Evaluate the B matrix for the reference element
138 size_t sDim = 6;
139 size_t bDim = 4;
140 std::vector<Memory2D> Bs(numGaussPoints);
141 std::vector<Memory2D> Ss(numGaussPoints);
142
143 for (size_t j = 0; j < numGaussPoints; j++) {
144 Memory2D& S = Ss[j];
145 S = Memory2D("S", Teuchos::as<LO>(sDim), Teuchos::as<LO>(nDim_));
146 EvalD(refPoints, gaussPoints[j], S);
147
148 Memory2D& B = Bs[j];
149 B = Memory2D("B", Teuchos::as<LO>(bDim), numDofPerElem);
150
151 for (size_t k = 0; k < numNodesPerElem; k++) {
152 B(0, numDofPerNode * k + 0) = S(k, 0);
153 B(1, numDofPerNode * k + 0) = S(k, 1);
154 B(2, numDofPerNode * k + 1) = S(k, 0);
155 B(3, numDofPerNode * k + 1) = S(k, 1);
156 }
157 }
158
159 // Construct reordering matrix (see 6.2-9 from Cook)
160 Memory2D R("R", D->extent(0), Teuchos::as<LO>(bDim));
161 R(0, 0) = R(1, 3) = R(2, 1) = R(2, 2) = 1;
162
163 this->A_ = MatrixTraits<Map, Matrix>::Build(this->Map_, 8 * numNodesPerElem);
164 this->A_->setObjectLabel(this->getObjectLabel());
165
166 auto one = KAT::one(), zero = KAT::zero();
167 Memory2D prevKE("prevKE", numDofPerElem, numDofPerElem), prevElementNodes("prevElementNodes", numNodesPerElem, Teuchos::as<LO>(nDim_)); // cache
168 for (size_t i = 0; i < elements_.size(); i++) {
169 // Select nodes subvector
170 Memory2D elementNodes("elementNodes", numNodesPerElem, Teuchos::as<LO>(nDim_));
171 std::vector<LO>& elemNodes = elements_[i];
172 for (size_t j = 0; j < numNodesPerElem; j++) {
173 elementNodes(j, 0) = nodes_[elemNodes[j]].x;
174 elementNodes(j, 1) = nodes_[elemNodes[j]].y;
175 }
176
177 // Check if element is a translation of the previous element
178 auto xMove = elementNodes(0, 0) - prevElementNodes(0, 0), yMove = elementNodes(0, 1) - prevElementNodes(0, 1);
179 typename KAT::magnitudeType eps = 1e-15; // coordinate comparison criteria
180 bool recompute = false;
181 {
182 size_t j = 0;
183 for (j = 0; j < numNodesPerElem; j++)
184 if (KAT::magnitude(elementNodes(j, 0) - (prevElementNodes(j, 0) + xMove)) > eps ||
185 KAT::magnitude(elementNodes(j, 1) - (prevElementNodes(j, 1) + yMove)) > eps)
186 break;
187 if (j != numNodesPerElem)
188 recompute = true;
189 }
190
191 Memory2D KE("KE", numDofPerElem, numDofPerElem);
192 Kokkos::View<SC**, Kokkos::HostSpace> KE2("KE2", KE.extent(0), KE.extent(1));
193 if (recompute == false) {
194 // If an element has the same form as previous element, reuse stiffness matrix
195 Kokkos::deep_copy(KE, prevKE);
196 } else {
197 // Evaluate new stiffness matrix for the element
198 Memory2D K0("K0", D->extent(0), numDofPerElem);
199 for (size_t j = 0; j < numGaussPoints; j++) {
200 Memory2D& B = Bs[j];
201 Memory2D& S = Ss[j];
202
203 Memory2D JAC("JAC", Teuchos::as<LO>(nDim_), Teuchos::as<LO>(nDim_));
204
205 for (size_t p = 0; p < nDim_; p++)
206 for (size_t q = 0; q < nDim_; q++) {
207 JAC(p, q) = zero;
208
209 for (size_t k = 0; k < numNodesPerElem; k++)
210 JAC(p, q) += S(k, p) * elementNodes(k, q);
211 }
212
213 auto detJ = JAC(0, 0) * JAC(1, 1) - JAC(0, 1) * JAC(1, 0);
214
215 // J2 = inv([JAC zeros(2); zeros(2) JAC])
216 Memory2D J2("J2", Teuchos::as<LO>(nDim_ * nDim_), Teuchos::as<LO>(nDim_ * nDim_));
217 J2(0, 0) = J2(2, 2) = JAC(1, 1) / detJ;
218 J2(0, 1) = J2(2, 3) = -JAC(0, 1) / detJ;
219 J2(1, 0) = J2(3, 2) = -JAC(1, 0) / detJ;
220 J2(1, 1) = J2(3, 3) = JAC(0, 0) / detJ;
221
222 Memory2D B2("B2", J2.extent(0), B.extent(1));
223 KokkosBlas::gemm("N", "N", one, J2, B, zero, B2);
224
225 // KE = KE + t * J2B' * D * J2B * detJ
226 Memory2D J2B("J2B", R.extent(0), B2.extent(1));
227 KokkosBlas::gemm("N", "N", one, R, B2, zero, J2B);
228 KokkosBlas::gemm("N", "N", one, *D, J2B, zero, K0);
229 KokkosBlas::gemm("T", "N", t * detJ, J2B, K0, one, KE);
230 }
231
232 // Cache the matrix and nodes
233 Kokkos::deep_copy(prevKE, KE);
234 Kokkos::deep_copy(prevElementNodes, elementNodes);
235 }
236
237 Teuchos::Array<GO> elemDofs(numDofPerElem);
238 for (size_t j = 0; j < numNodesPerElem; j++) { // FIXME: this may be inconsistent with the map
239 elemDofs[numDofPerNode * j + 0] = local2Global_[elemNodes[j]] * numDofPerNode;
240 elemDofs[numDofPerNode * j + 1] = elemDofs[numDofPerNode * j + 0] + 1;
241 }
242
243 // Deal with Dirichlet nodes
244 bool isDirichlet = false;
245 for (size_t j = 0; j < numNodesPerElem; j++)
246 if (dirichlet_[elemNodes[j]])
247 isDirichlet = true;
248
249 if (isDirichlet) {
250 bool keepBCs = this->list_.get("keepBCs", false);
251 if (keepBCs) {
252 // Simple case: keep Dirichlet DOF
253 // We rewrite rows and columns corresponding to Dirichlet DOF with zeros
254 // The diagonal elements corresponding to Dirichlet DOF are set to 1.
255 for (size_t j = 0; j < numNodesPerElem; j++)
256 if (dirichlet_[elemNodes[j]]) {
257 LO j0 = numDofPerNode * j + 0;
258 LO j1 = numDofPerNode * j + 1;
259
260 for (size_t k = 0; k < numDofPerElem; k++)
261 KE(j0, k) = KE(k, j0) = KE(j1, k) = KE(k, j1) = zero;
262 KE(j0, j0) = KE(j1, j1) = one;
263 }
264 } else {
265 // Complex case: get rid of Dirichlet DOF
266 // The case is complex because if we simply reduce the size of the matrix, it would become inconsistent
267 // with maps. So, instead, we modify values of the boundary cells as if we had an additional cell close
268 // to the boundary. For instance, if we have a following cell
269 // D--.
270 // | |
271 // D--D
272 // we multiply all D-D connections by 2 and multiply diagonals corresponding to D by 2 or 4, depending
273 // whether it is connected to 1 or 2 other D DOFs.
274 for (size_t j = 0; j < numNodesPerElem; j++)
275 if (dirichlet_[elemNodes[j]]) {
276 LO j0 = numDofPerNode * j, j1 = j0 + 1;
277
278 for (size_t k = 0; k < numNodesPerElem; k++)
279 if ((j == k) || ((j + k) & 0x1)) {
280 // Nodes j and k are connected by an edge, or j == k
281 LO k0 = numDofPerNode * k, k1 = k0 + 1;
282 SC f = Teuchos::as<SC>(pow(2, Teuchos::as<int>(std::min(dirichlet_[elemNodes[j]], dirichlet_[elemNodes[k]]))));
283
284 KE(j0, k0) *= f;
285 KE(j0, k1) *= f;
286 KE(j1, k0) *= f;
287 KE(j1, k1) *= f;
288 }
289 }
290 }
291 }
292
293 if constexpr (!std::is_same_v<SC, impl_scalar_type>)
294 Kokkos::deep_copy(KE2, KE);
295
296 // Insert KE into the global matrix
297 // NOTE: KE is symmetric, therefore it does not matter that it is in the CSC format
298 for (size_t j = 0; j < numDofPerElem; j++)
299 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
300 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
301 if constexpr (std::is_same_v<SC, impl_scalar_type>) {
302 auto vals = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
303 this->A_->insertGlobalValues(elemDofs[j], inds, vals);
304 } else {
305 auto vals = Kokkos::Compat::getArrayView(Kokkos::subview(KE2, j, Kokkos::ALL()));
306 this->A_->insertGlobalValues(elemDofs[j], inds, vals);
307 }
308 }
309 }
310 this->A_->fillComplete();
311
312 return this->A_;
313}

315template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
316RCP<typename Problem<Map, Matrix, MultiVector>::RealValuedMultiVector>
317Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildCoords() {
318 // FIXME: map here is an extended map, with multiple DOF per node
319 // as we cannot construct a single DOF map in Problem, we repeat the coords
320 this->Coords_ = MultiVectorTraits<Map, RealValuedMultiVector>::Build(this->Map_, nDim_);
321
322 typedef typename RealValuedMultiVector::scalar_type real_type;
323
324 Teuchos::ArrayRCP<real_type> x = this->Coords_->getDataNonConst(0);
325 Teuchos::ArrayRCP<real_type> y = this->Coords_->getDataNonConst(1);
326
327 Teuchos::ArrayView<const GO> GIDs = this->Map_->getLocalElementList();
328
329 // NOTE: coordinates vector local ordering is consistent with that of the
330 // matrix map, as it is constructed by going through GIDs and translating
331 // those.
332 const typename KAT::magnitudeType hx = KAT::magnitude(stretch[0]),
333 hy = KAT::magnitude(stretch[1]);
334 for (GO p = 0; p < GIDs.size(); p += 2) { // FIXME: we assume that DOF for the same node are label consequently
335 GlobalOrdinal ind = GIDs[p] >> 1;
336 size_t i = ind % nx_, j = ind / nx_;
337
338 x[p] = x[p + 1] = (i + 1) * hx;
339 y[p] = y[p + 1] = (j + 1) * hy;
340 }
341
342 return this->Coords_;
343}

345template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
346RCP<MultiVector> Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildNullspace() {
347 const int numVectors = 3;
348 this->Nullspace_ = MultiVectorTraits<Map, MultiVector>::Build(this->Map_, numVectors);
349
350 typedef typename RealValuedMultiVector::scalar_type real_type;
351
352 if (this->Coords_ == Teuchos::null)
353 BuildCoords();
354
355 // Teuchos::ArrayView<const GO> GIDs = this->Map_->getLocalElementList();
356
357 size_t numDofs = this->Map_->getLocalNumElements();
358 Teuchos::ArrayRCP<real_type> x = this->Coords_->getDataNonConst(0);
359 Teuchos::ArrayRCP<real_type> y = this->Coords_->getDataNonConst(1);
360
361 auto one = KAT::one();
362
363 // NOTE: nullspace local ordering is consistent with that of the matrix
364 // map, as it inherits ordering from coordinates, which is consistent.
365
366 // Translations
367 Teuchos::ArrayRCP<SC> T0 = this->Nullspace_->getDataNonConst(0), T1 = this->Nullspace_->getDataNonConst(1);
368 for (size_t i = 0; i < numDofs; i += nDim_) {
369 T0[i] = one;
370 T1[i + 1] = one;
371 }
372
373 // Calculate center
374 real_type cx = this->Coords_->getVector(0)->meanValue();
375 real_type cy = this->Coords_->getVector(1)->meanValue();
376
377 // Rotations
378 Teuchos::ArrayRCP<SC> R0 = this->Nullspace_->getDataNonConst(2);
379 for (size_t i = 0; i < numDofs; i += nDim_) {
380 // Rotate in Y-Z Plane (around Z axis): [ -y; x]
381 R0[i + 0] = -(y[i] - cy);
382 R0[i + 1] = (x[i] - cx);
383 }
384
385 // Equalize norms of all vectors to that of the first one
386 // We do not normalize them as a vector of ones seems nice
387 Teuchos::Array<typename KAT::magnitudeType> norms2(numVectors);
388 this->Nullspace_->norm2(norms2);
389 Teuchos::Array<SC> norms2scalar(numVectors);
390 for (int i = 0; i < numVectors; i++)
391 norms2scalar[i] = norms2[0] / norms2[i];
392 this->Nullspace_->scale(norms2scalar);
393
394 return this->Nullspace_;
395}

397template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
398void Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMesh() {
399 using Teuchos::as;
400
401 const typename KAT::magnitudeType hx = KAT::magnitude(stretch[0]),
402 hy = KAT::magnitude(stretch[1]);
403
404 GO myPID = this->Map_->getComm()->getRank();
405 GO const& one = 1;
406 GO mx = (this->list_).get("mx", one), my = (this->list_).get("my", one);
407
408 GO startx, starty, endx, endy;
409 Utils::getSubdomainData(dims[0], mx, myPID % mx, startx, endx);
410 Utils::getSubdomainData(dims[1], my, myPID / mx, starty, endy);
411
412 LO nx = as<LO>(endx - startx), ny = as<LO>(endy - starty);
413
414 // Expand subdomain to do overlap
415 if (startx > 0) {
416 nx++;
417 startx--;
418 }
419 if (starty > 0) {
420 ny++;
421 starty--;
422 }
423 if (startx + nx < dims[0]) {
424 nx++;
425 }
426 if (starty + ny < dims[1]) {
427 ny++;
428 }
429
430 nodes_.resize((nx + 1) * (ny + 1));
431 local2Global_.resize((nx + 1) * (ny + 1));
432 dirichlet_.resize((nx + 1) * (ny + 1));
433 elements_.resize(nx * ny);
434
435#define NODE(i, j) ((j) * (nx + 1) + (i))
436#define CELL(i, j) ((j)*nx + (i))
437 // NOTE: the fact that local ordering here is not consistent with that of
438 // the matrix map does not matter. The two things that matter are:
439 // local2Global_ assigns to a correct GID, and nodes_ contain correct
440 // coordinates
441 for (LO j = 0; j <= ny; j++)
442 for (LO i = 0; i <= nx; i++) {
443 GO ii = startx + i, jj = starty + j;
444 LO nodeID = NODE(i, j);
445 nodes_[nodeID] = Point((ii + 1) * hx, (jj + 1) * hy);
446 local2Global_[nodeID] = jj * nx_ + ii;
447
448 if (ii == 0 && (this->DirichletBC_ & DIR_LEFT)) dirichlet_[nodeID]++;
449 if (ii == nx_ && (this->DirichletBC_ & DIR_RIGHT)) dirichlet_[nodeID]++;
450 if (jj == 0 && (this->DirichletBC_ & DIR_BOTTOM)) dirichlet_[nodeID]++;
451 if (jj == ny_ && (this->DirichletBC_ & DIR_TOP)) dirichlet_[nodeID]++;
452 }
453
454 for (LO j = 0; j < ny; j++)
455 for (LO i = 0; i < nx; i++) {
456 std::vector<LO>& element = elements_[CELL(i, j)];
457 element.resize(4);
458 element[0] = NODE(i, j);
459 element[1] = NODE(i + 1, j);
460 element[2] = NODE(i + 1, j + 1);
461 element[3] = NODE(i, j + 1);
462 }
463#undef NODE
464#undef CELL
465}

467template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
468void Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMaterialMatrix(Memory2D& D) {
469 D = Memory2D("D", 3, 3);
470
471 if (!strcmp(mode_.c_str(), "plane stress")) {
472 typename Teuchos::ScalarTraits<SC>::magnitudeType c = E / (1 - nu * nu);
473 D(0, 0) = c;
474 D(0, 1) = c * nu;
475 D(1, 0) = c * nu;
476 D(1, 1) = c;
477 D(2, 2) = c * (1 - nu) / 2;
478
479 } else if (!strcmp(mode_.c_str(), "plane strain")) {
480 typename Teuchos::ScalarTraits<SC>::magnitudeType c = E / (1 + nu) / (1 - 2 * nu);
481 D(0, 0) = c * (1 - nu);
482 D(0, 1) = c * nu;
483 D(1, 0) = c * nu;
484 D(1, 1) = c * (1 - nu);
485 D(2, 2) = c * (1 - 2 * nu) / 2;
486
487 } else {
488 TEUCHOS_TEST_FOR_EXCEPTION(true, std::logic_error, "Unknown material model for 2D");
489 }
490}

492template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
493void Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildReferencePoints(size_t& numRefPoints, std::vector<Point>& refPoints, size_t& numGaussPoints, std::vector<Point>& gaussPoints) {
494 numRefPoints = 4;
495 numGaussPoints = 4;
496 refPoints.resize(numRefPoints);
497 gaussPoints.resize(numGaussPoints);
498
499 refPoints[0] = Point(-1, -1);
500 refPoints[1] = Point(1, -1);
501 refPoints[2] = Point(1, 1);
502 refPoints[3] = Point(-1, 1);
503
504 // Gauss points (reference)
505 SC sq3 = 1.0 / sqrt(3);
506 gaussPoints[0] = Point(sq3, sq3);
507 gaussPoints[1] = Point(sq3, -sq3);
508 gaussPoints[2] = Point(-sq3, sq3);
509 gaussPoints[3] = Point(-sq3, -sq3);
510}

512template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
513void Elasticity2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalD(const std::vector<Point>& refPoints, Point& gaussPoint, Memory2D S) {
514 const Scalar one = Teuchos::ScalarTraits<Scalar>::one();
515 const Scalar two = one + one;
516 const Scalar quarter = one / (two + two);
517 for (size_t j = 0; j < refPoints.size(); j++) {
518 S(j, 0) = refPoints[j].x * (one + refPoints[j].y * gaussPoint.y) * quarter;
519 S(j, 1) = (one + gaussPoint.x * refPoints[j].x) * refPoints[j].y * quarter;
520 }
521 S(4, 0) = -two * gaussPoint.x;
522 S(5, 0) = 0.0;
523 S(4, 1) = 0.0;
524 S(5, 1) = -two * gaussPoint.y;
525}
