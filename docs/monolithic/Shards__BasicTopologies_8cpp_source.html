21enum { MAXIMUM_INDICES = 256 };
22
23const unsigned * index_identity_array()
24{
25 static unsigned self[ MAXIMUM_INDICES ];
26 static int init = 1 ;
27
28 if ( init ) {
29 for ( unsigned i = 0 ; i < MAXIMUM_INDICES ; ++i ) { self[i] = i ; }
30 init = 0 ;
31 }
32
33 return self ;
34}
35
36const Subcell * subcell_nodes_array()
37{
38 static Subcell self[ MAXIMUM_INDICES ];
39 static int init = 1 ;
40
41 if ( init ) {
42
43 const CellTopologyData * const top = getCellTopologyData<Node>();
44 const unsigned * const ident = index_identity_array();
45
46 for ( int i = 0 ; i < MAXIMUM_INDICES ; ++i ) {
47 self[i].topology = top ;
48 self[i].node = ident + i ;
49 }
50
51 init = 0 ;
52 }
53
54 return self ;
55}

102template< class TList , class IList , unsigned N >
103struct SubcellValue ;
104
105template< class TList , class IList >
106struct SubcellValue<TList,IList,0>
107{ static void assign( Subcell * ) {} };
108
109template< class TList , class IList , unsigned N >
110struct SubcellValue {
111 static void assign( Subcell * s )
112 {
113 enum { I = N - 1 };
114 SubcellValue<TList,IList,I>::assign( s );
115 s[I].topology = getCellTopologyData< typename TypeListAt<TList,I>::type >();
116 s[I].node = index_list( typename TypeListAt<IList,I>::type() );
117 }
118};
119
120template< class TList , class IList >
121struct SubcellArray {
122
123 enum { N = TypeListLength<TList>::value };
124
125 Subcell array[ N ];
126
127 SubcellArray() { SubcellValue<TList,IList,N>::assign( array ); }
128};

132template< class IList , class PList, unsigned N >
133struct PermutationValue ;
134
135template< class IList , class PList>
136struct PermutationValue<IList,PList,0>
137{ static void assign( Permutation * , Permutation * ) {} };
138
139template< class IList , class PList, unsigned N >
140struct PermutationValue {
141 static void assign( Permutation * forward , Permutation * inverse )
142 {
143 enum { I = N - 1 };
144 typedef typename TypeListAt<IList,I>::type ForwardType ;
145 enum { polarity = IndexListAt<PList,I>::value };
146 typedef typename IndexListInverse< ForwardType >::type InverseType ;
147 PermutationValue<IList,PList,I>::assign( forward , inverse );
148 forward[I].node = index_list( ForwardType() );
149 forward[I].polarity = polarity;
150 inverse[I].node = index_list( InverseType() );
151 inverse[I].polarity = polarity;
152 }
153};
154
155template< class IList , class PList,
156 unsigned NPerm = TypeListLength< IList >::value >
157struct PermutationArray ;
158
159template< class IList , class PList>
160struct PermutationArray< IList , PList, 0 > {
161 Permutation * forward ;
162 Permutation * inverse ;
163 PermutationArray() : forward(NULL), inverse(NULL) {}
164};
165
166template< class IList , class PList, unsigned NPerm >
167struct PermutationArray {
168
169 enum { N = NPerm };
170
171 Permutation forward[ N ];
172 Permutation inverse[ N ];
173
174 PermutationArray()
175 { PermutationValue< IList , PList, N >::assign( forward , inverse ); }
176};

182template< unsigned Number_Vertex ,
183 unsigned Number_Node ,
184 class EdgeList ,
185 class EdgeMaps ,
186 class FaceList ,
187 class FaceMaps >
188struct Descriptor<
189 CellTopologyTraits< 3 , Number_Vertex , Number_Node ,
190 EdgeList , EdgeMaps , FaceList , FaceMaps ,
191 TypeListEnd > >
192{
193 typedef CellTopologyTraits< 3 , Number_Vertex , Number_Node ,
194 EdgeList , EdgeMaps ,
195 FaceList , FaceMaps ,
196 TypeListEnd > Traits ;
197
198 typedef SubcellArray< EdgeList , EdgeMaps > EdgeArray ;
199 typedef SubcellArray< FaceList , FaceMaps > FaceArray ;
200
201 EdgeArray edges ;
202 FaceArray faces ;
203
204 Subcell self ;
205
206 CellTopologyData top ;
207
208 Descriptor( const CellTopologyData * base , const char * name )
209 : edges(), faces()
210 {
211 typedef typename Traits::template subcell<0> subcell_0 ;
212 typedef typename Traits::template subcell<1> subcell_1 ;
213 typedef typename Traits::template subcell<2> subcell_2 ;
214 typedef typename Traits::template subcell<3> subcell_3 ;
215
216 self.topology = & top ;
217 self.node = index_identity_array();
218
219 top.base = base ? base : & top ;
220 top.name = name ;
221 top.key = Traits::key ;
222 top.dimension = 3 ;
223 top.vertex_count = Number_Vertex ;
224 top.node_count = Number_Node ;
225 top.edge_count = EdgeArray::N ;
226 top.side_count = Traits::side_count ;
227 top.subcell_homogeneity[0] = subcell_0::homogeneity ;
228 top.subcell_homogeneity[1] = subcell_1::homogeneity ;
229 top.subcell_homogeneity[2] = subcell_2::homogeneity ;
230 top.subcell_homogeneity[3] = subcell_3::homogeneity ;
231 top.subcell_count[0] = Number_Node ;
232 top.subcell_count[1] = EdgeArray::N ;
233 top.subcell_count[2] = FaceArray::N ;
234 top.subcell_count[3] = 1 ;
235 top.subcell[0] = subcell_nodes_array();
236 top.subcell[1] = edges.array ;
237 top.subcell[2] = faces.array ;
238 top.subcell[3] = & self ;
239 top.side = faces.array ;
240 top.edge = edges.array ;
241 };
242};

367template< unsigned Number_Node , unsigned Number_Vertex>
368struct Descriptor<
369 CellTopologyTraits< 0 , Number_Node , Number_Vertex ,
370 TypeListEnd , TypeListEnd ,
371 TypeListEnd , TypeListEnd ,
372 TypeListEnd > >
373{
374 // Two cases: Nodes=Vertices=0 for topo=Node, and Nodes=Vertices=1 for topo=Particle
375 static_assert (Number_Node==0 || Number_Node==1,
376 "Invalid number of nodes for 0-dimensional topology.");
377 static_assert (Number_Node==Number_Vertex,
378 "Incompatible number of nodes/vertices for 0-dimensional topology.");
379
380 typedef CellTopologyTraits< 0 , Number_Node , Number_Vertex ,
381 TypeListEnd , TypeListEnd ,
382 TypeListEnd , TypeListEnd ,
383 TypeListEnd > Traits ;
384
385 Subcell self ;
386
387 CellTopologyData top ;
388
389 Descriptor( const CellTopologyData * base , const char * name )
390 {
391 self.topology = & top ;
392 self.node = index_identity_array();
393
394 top.base = base ? base : & top ;
395 top.name = name ;
396 top.key = Traits::key ;
397 top.dimension = 0 ;
398 top.vertex_count = Number_Vertex ;
399 top.node_count = Number_Node ;
400 top.edge_count = 0 ;
401 top.side_count = 0 ;
402 top.permutation_count = 0 ;
403 top.subcell_homogeneity[0] = true ;
404 top.subcell_homogeneity[1] = false ;
405 top.subcell_homogeneity[2] = false ;
406 top.subcell_homogeneity[3] = false ;
407 top.subcell_count[0] = 1 ;
408 top.subcell_count[1] = 0 ;
409 top.subcell_count[2] = 0 ;
410 top.subcell_count[3] = 0 ;
411 top.subcell[0] = & self ;
412 top.subcell[1] = NULL ;
413 top.subcell[2] = NULL ;
414 top.subcell[3] = NULL ;
415 top.side = NULL ;
416 top.edge = NULL ;
417 top.permutation = NULL ;
418 top.permutation_inverse = NULL ;
419 };
420};

426std::ostream & operator << ( std::ostream & s , const CellTopologyData & t )
427{
428 s << t.name ;
429 s << " { D = " << t.dimension ;
430 s << " , NV = " << t.vertex_count ;
431 s << " , K = 0x" << std::hex << t.key << std::dec ;
432 s << std::endl ;
433
434 for ( unsigned d = 0 ; d < 4 ; ++d ) {
435 for ( unsigned i = 0 ; i < t.subcell_count[d] ; ++i ) {
436
437 const Subcell & sub = t.subcell[d][i] ;
438
439 s << " subcell[" << d << "][" << i << "] = { " ;
440
441 s << sub.topology->name ;
442 s << " ," ;
443 for ( unsigned j = 0 ; j < sub.topology->node_count ; ++j ) {
444 s << " " << sub.node[j] ;
445 }
446 s << " }" << std::endl ;
447 }
448 }
449
450 s << "}" << std::endl ;
451 return s ;
452}
