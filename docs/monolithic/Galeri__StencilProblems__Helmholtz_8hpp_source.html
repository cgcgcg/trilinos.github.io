75template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
76Teuchos::RCP<Matrix> Helmholtz2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
77 int model = this->list_.get("model", 0);
78 GlobalOrdinal nx = this->list_.get("nx", (GlobalOrdinal)-1);
79 GlobalOrdinal ny = this->list_.get("ny", (GlobalOrdinal)-1);
80 double h = this->list_.get("h", 1.0);
81 double delta = this->list_.get("delta", 1.0);
82 int PMLx_left = this->list_.get("PMLx_left", 0);
83 int PMLx_right = this->list_.get("PMLx_right", 0);
84 int PMLy_left = this->list_.get("PMLy_left", 0);
85 int PMLy_right = this->list_.get("PMLy_right", 0);
86 double omega = this->list_.get("omega", 2.0 * M_PI);
87 double shift = this->list_.get("shift", 0.5);
88 Scalar cpxshift(1.0, shift);
89
90 if (nx == -1 || ny == -1) {
91 GlobalOrdinal n = this->Map_->getGlobalNumElements();
92 nx = (GlobalOrdinal)sqrt((double)n);
93 ny = nx;
94 TEUCHOS_TEST_FOR_EXCEPTION(nx * ny != n, std::logic_error, "You need to specify nx and ny.");
95 }
96
97 this->A_ = Cross2D_Helmholtz<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix>(this->Map_, nx, ny, h, delta,
98 PMLx_left, PMLx_right,
99 PMLy_left, PMLy_right,
100 omega, cpxshift, model);
101 this->A_->setObjectLabel(this->getObjectLabel());
102 return this->A_;
103}
