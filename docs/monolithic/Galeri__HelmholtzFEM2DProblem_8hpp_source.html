26template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
27class HelmholtzFEM2DProblem : public Problem_Helmholtz<Map, Matrix, MultiVector> {
28 public:
29 HelmholtzFEM2DProblem(Teuchos::ParameterList& list, const Teuchos::RCP<const Map>& map)
30 : Problem_Helmholtz<Map, Matrix, MultiVector>(list, map) {
31 // parameters
32 hx_ = list.get("stretchx", 1.0);
33 hy_ = list.get("stretchy", 1.0);
34 px_ = list.get("px", 1);
35 py_ = list.get("py", 1);
36 if (list.isType<int>("nx"))
37 nx_ = Teuchos::as<GO>(list.get("nx", -1));
38 else
39 nx_ = list.get<GO>("nx", -1);
40 if (list.isType<int>("ny"))
41 ny_ = Teuchos::as<GO>(list.get("ny", -1));
42 else
43 ny_ = list.get<GO>("ny", -1);
44 if (list.isType<int>("mx"))
45 mx_ = Teuchos::as<GO>(list.get("mx", -1));
46 else
47 mx_ = list.get<GO>("mx", -1);
48 if (list.isType<int>("my"))
49 my_ = Teuchos::as<GO>(list.get("my", -1));
50 else
51 my_ = list.get<GO>("my", -1);
52
53 omega_ = list.get("omega", 2.0 * M_PI);
54 shift_ = list.get("shift", 0.0);
55 delta_ = list.get("delta", 2.0);
56 model_ = list.get("model", 0);
57 PMLx_left = list.get("PMLx_left", 0);
58 PMLx_right = list.get("PMLx_right", 0);
59 PMLy_left = list.get("PMLy_left", 0);
60 PMLy_right = list.get("PMLy_right", 0);
61
62 // calculate info
63 Dx_ = hx_ * nx_;
64 Dy_ = hy_ * ny_;
65 LBx_ = PMLx_left * hx_;
66 RBx_ = Dx_ - PMLx_right * hx_;
67 LBy_ = PMLy_left * hy_;
68 RBy_ = Dy_ - PMLy_right * hy_;
69 PMLwidthx_ = std::max(LBx_, Dx_ - RBx_);
70 if (PMLwidthx_ == 0) {
71 PMLwidthx_ = 1.0;
72 }
73 PMLwidthy_ = std::max(LBy_, Dy_ - RBy_);
74 if (PMLwidthy_ == 0) {
75 PMLwidthy_ = 1.0;
76 }
77 nDim = 2;
78
79 // velocity model
80 velocitymodel_.setDim(nDim);
81 velocitymodel_.setModel(model_);
82
83 // currently AMG doesn't suport higher orders. set to linears
84 px_ = 1;
85 py_ = 1;
86
87 // NOTE: galeri counts points, not elements
88 dims.push_back(nx_ * px_ - 1);
89 dims.push_back(ny_ * py_ - 1);
90 TEUCHOS_TEST_FOR_EXCEPTION(nx_ <= 0 || ny_ <= 0, std::logic_error, "nx and ny must be positive");
91 }

122 int PMLx_left, PMLx_right;
123 int PMLy_left, PMLy_right;
124 int PMLz_left, PMLz_right;

139template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
140Teuchos::RCP<Matrix> HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrix() {
141 BuildMesh();
142
143 const size_t numDofPerNode = 1;
144 const size_t numNodesPerElem = (px_ + 1) * (py_ + 1);
145 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
146
147 TEUCHOS_TEST_FOR_EXCEPTION(elements[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
148
149 // Compute quadrature points/rules
150 std::vector<Point> quadPoints;
151 std::vector<double> quadWeights;
152 BuildPoints(quadPoints, quadWeights);
153
154 // Compute basis function values and derivatives at each quadrature point
155 std::vector<std::vector<double> > vals, dxs, dys;
156 vals.resize(quadPoints.size());
157 dxs.resize(quadPoints.size());
158 dys.resize(quadPoints.size());
159 for (unsigned int i = 0; i < quadPoints.size(); i++) {
160 EvalBasis(quadPoints[i], vals[i], dxs[i], dys[i]);
161 }
162
163 this->A_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * numDofPerNode);
164 SC cpxshift(1.0, shift_);
165
166 // iterate over elements
167 for (size_t i = 0; i < elements.size(); i++) {
168 Memory2D KE("KE", numDofPerElem, numDofPerElem);
169
170 // element domain is [shiftx,shiftx+hx] x [shifty,shifty+hy]
171 std::vector<LO>& elemNodes = elements[i];
172 double shiftx = nodes[elemNodes[0]].x;
173 double shifty = nodes[elemNodes[0]].y;
174
175 // evaluate PML values at quadrature points for this element
176 std::vector<SC> stretchx, stretchy, cs;
177 EvalStretch(shiftx, shifty, quadPoints, stretchx, stretchy, cs);
178
179 // Evaluate the stiffness matrix for the element
180 for (size_t j = 0; j < quadPoints.size(); j++) {
181 Scalar sx = stretchx[j];
182 Scalar sy = stretchy[j];
183 Scalar cc = cs[j] * cs[j];
184 double qdwt = quadWeights[j];
185 std::vector<double> curvals = vals[j];
186 std::vector<double> curdxs = dxs[j];
187 std::vector<double> curdys = dys[j];
188 Scalar mass = qdwt * cpxshift * omega_ * omega_ * sx * sy / cc;
189 Scalar pml1 = qdwt * sy / sx;
190 Scalar pml2 = qdwt * sx / sy;
191 for (unsigned int m = 0; m < numDofPerElem; m++) {
192 for (unsigned int n = 0; n < numDofPerElem; n++) {
193 KE(m, n) += pml1 * curdxs[m] * curdxs[n] + pml2 * curdys[m] * curdys[n] - mass * curvals[m] * curvals[n];
194 }
195 }
196 }
197
198 Teuchos::Array<GO> elemDofs(numDofPerElem);
199 for (size_t j = 0; j < numDofPerElem; j++) {
200 elemDofs[j] = local2Global_[elemNodes[j]];
201 if (local2Global_[elemNodes[j]] > 9999) {
202 std::cout << "index " << local2Global_[elemNodes[j]] << std::endl;
203 }
204 }
205
206 // Insert KE into the global matrix
207 for (size_t j = 0; j < numDofPerElem; j++)
208 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
209 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
210 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
211 this->A_->insertGlobalValues(elemDofs[j], inds, values);
212 }
213 }
214
215 this->A_->fillComplete();
216 return this->A_;
217}

220std::pair<Teuchos::RCP<Matrix>, Teuchos::RCP<Matrix> > HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMatrices() {
221 BuildMesh();
222
223 const size_t numDofPerNode = 1;
224 const size_t numNodesPerElem = (px_ + 1) * (py_ + 1);
225 const size_t numDofPerElem = numNodesPerElem * numDofPerNode;
226
227 TEUCHOS_TEST_FOR_EXCEPTION(elements[0].size() != numNodesPerElem, std::logic_error, "Incorrect number of element vertices");
228
229 // Compute quadrature points/rules
230 std::vector<Point> quadPoints;
231 std::vector<double> quadWeights;
232 BuildPoints(quadPoints, quadWeights);
233
234 // Compute basis function values and derivatives at each quadrature point
235 std::vector<std::vector<double> > vals, dxs, dys;
236 vals.resize(quadPoints.size());
237 dxs.resize(quadPoints.size());
238 dys.resize(quadPoints.size());
239 for (unsigned int i = 0; i < quadPoints.size(); i++) {
240 EvalBasis(quadPoints[i], vals[i], dxs[i], dys[i]);
241 }
242
243 this->K_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * numDofPerNode);
244 this->M_ = MatrixTraits<Map, Matrix>::Build(this->Map_, (2 * px_ + 1) * (2 * py_ + 1) * numDofPerNode);
245
246 // iterate over elements
247 for (size_t i = 0; i < elements.size(); i++) {
248 Memory2D KE("KE", numDofPerElem, numDofPerElem);
249 Memory2D ME("ME", numDofPerElem, numDofPerElem);
250
251 // element domain is [shiftx,shiftx+hx] x [shifty,shifty+hy]
252 std::vector<LO>& elemNodes = elements[i];
253 double shiftx = nodes[elemNodes[0]].x;
254 double shifty = nodes[elemNodes[0]].y;
255
256 // evaluate PML values at quadrature points for this element
257 std::vector<SC> stretchx, stretchy, cs;
258 EvalStretch(shiftx, shifty, quadPoints, stretchx, stretchy, cs);
259
260 // Evaluate the stiffness matrix for the element
261 for (size_t j = 0; j < quadPoints.size(); j++) {
262 Scalar sx = stretchx[j];
263 Scalar sy = stretchy[j];
264 Scalar cc = cs[j] * cs[j];
265 double qdwt = quadWeights[j];
266 std::vector<double> curvals = vals[j];
267 std::vector<double> curdxs = dxs[j];
268 std::vector<double> curdys = dys[j];
269 Scalar mass = qdwt * sx * sy / cc;
270 Scalar pml1 = qdwt * sy / sx;
271 Scalar pml2 = qdwt * sx / sy;
272 for (unsigned int m = 0; m < numDofPerElem; m++) {
273 for (unsigned int n = 0; n < numDofPerElem; n++) {
274 KE(m, n) += pml1 * curdxs[m] * curdxs[n] + pml2 * curdys[m] * curdys[n];
275 ME(m, n) += mass * curvals[m] * curvals[n];
276 }
277 }
278 }
279
280 Teuchos::Array<GO> elemDofs(numDofPerElem);
281 for (size_t j = 0; j < numDofPerElem; j++) {
282 elemDofs[j] = local2Global_[elemNodes[j]];
283 if (local2Global_[elemNodes[j]] > 9999) {
284 std::cout << "index " << local2Global_[elemNodes[j]] << std::endl;
285 }
286 }
287
288 // Insert KE and ME into the global matrices
289 for (size_t j = 0; j < numDofPerElem; j++) {
290 if (this->Map_->isNodeGlobalElement(elemDofs[j])) {
291 {
292 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
293 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(KE, j, Kokkos::ALL()));
294 this->K_->insertGlobalValues(elemDofs[j], inds, values);
295 }
296 {
297 auto inds = Kokkos::Compat::getConstArrayView(elemDofs);
298 auto values = Kokkos::Compat::getArrayView(Kokkos::subview(ME, j, Kokkos::ALL()));
299 this->M_->insertGlobalValues(elemDofs[j], inds, values);
300 }
301 }
302 }
303 }
304
305 this->K_->fillComplete();
306 this->M_->fillComplete();
307 std::pair<Teuchos::RCP<Matrix>, Teuchos::RCP<Matrix> > system;
308 system = std::make_pair(this->K_, this->M_);
309 return system;
310}

312template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
313void HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildMesh() {
314 GO myPID = this->Map_->getComm()->getRank();
315 GO mySize = this->Map_->getComm()->getSize();
316 GO mx = mx_, my = my_;
317 if (mx * my != mySize) {
318 int squareRoot = std::max(1, (int)std::floor(sqrt((double)mySize)));
319 my = squareRoot;
320 while (mySize % my != 0)
321 ++my;
322 mx = mySize / my;
323 }
324
325 GO shiftx, shifty, endx, endy;
326 Utils::getSubdomainData(dims[0], mx, myPID % mx, shiftx, endx);
327 Utils::getSubdomainData(dims[1], my, myPID / mx, shifty, endy);
328
329 GO nx = endx - shiftx, ny = endy - shifty;
330
331 // Expand subdomain to do overlap
332 if (shiftx > 0) {
333 nx++;
334 shiftx--;
335 }
336 if (shifty > 0) {
337 ny++;
338 shifty--;
339 }
340 if (shiftx + nx < dims[0]) {
341 nx++;
342 }
343 if (shifty + ny < dims[1]) {
344 ny++;
345 }
346
347 nodes.resize((nx + 1) * (ny + 1));
348 local2Global_.resize((nx + 1) * (ny + 1));
349 elements.resize(nx * ny);
350
351#define NODE(i, j) ((j) * (nx + 1) + (i))
352#define CELL(i, j) ((j)*nx + (i))
353 for (int j = 0; j <= ny; j++)
354 for (int i = 0; i <= nx; i++) {
355 int ii = shiftx + i, jj = shifty + j;
356 int nodeID = NODE(i, j);
357 nodes[nodeID] = Point((ii + 1) * hx_, (jj + 1) * hy_);
358 local2Global_[nodeID] = jj * nx_ + ii;
359 }
360
361 for (int j = 0; j < ny; j++)
362 for (int i = 0; i < nx; i++) {
363 std::vector<int>& element = elements[CELL(i, j)];
364 element.resize(4);
365 element[0] = NODE(i, j);
366 element[1] = NODE(i + 1, j);
367 element[2] = NODE(i, j + 1);
368 element[3] = NODE(i + 1, j + 1);
369 }
370#undef NODE
371#undef CELL
372}

374template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
375void HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::BuildPoints(std::vector<Point>& quadPoints, std::vector<double>& quadWeights) {
376 // Use nonstandard quadrature for Helmholtz from the following paper:
377 // M. Ainsworth, H.A. Wajid. Optimally blended spectral-finite element scheme for wave propagation
378 // and nonstandard reduced integration. SIAM J. Numer. Analysis, 48(1): 346-371, 2010.
379
380 // Make reference domain [0,hx_] x [0,hy_].
381 double qdpt = 0.816496580927726;
382 quadPoints.resize(4);
383 quadPoints[0] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0);
384 quadPoints[1] = Point((1.0 + qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0);
385 quadPoints[2] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 + qdpt) * hy_ / 2.0);
386 quadPoints[3] = Point((1.0 - qdpt) * hx_ / 2.0, (1.0 - qdpt) * hy_ / 2.0);
387 quadWeights.resize(4);
388 quadWeights[0] = hx_ * hy_ / 4.0;
389 quadWeights[1] = hx_ * hy_ / 4.0;
390 quadWeights[2] = hx_ * hy_ / 4.0;
391 quadWeights[3] = hx_ * hy_ / 4.0;
392}

394template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
395void HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalBasis(Point& quadPoint, std::vector<double>& vals, std::vector<double>& dxs, std::vector<double>& dys) {
396 // For the reference element, evaluate each basis function and its derivatives at a particular quadrature point
397 int numBasis = (px_ + 1) * (py_ + 1);
398 vals.resize(numBasis);
399 dxs.resize(numBasis);
400 dys.resize(numBasis);
401 double x = quadPoint.x;
402 double y = quadPoint.y;
403 double valx1 = (hx_ - x) / hx_;
404 double dx1 = -1 / hx_;
405 double valx2 = x / hx_;
406 double dx2 = 1 / hx_;
407 double valy1 = (hy_ - y) / hy_;
408 double dy1 = -1 / hy_;
409 double valy2 = y / hy_;
410 double dy2 = 1 / hy_;
411 // linear function values, derivatives in x, y directions
412 vals[0] = valx1 * valy1;
413 dxs[0] = dx1 * valy1;
414 dys[0] = valx1 * dy1;
415 vals[1] = valx2 * valy1;
416 dxs[1] = dx2 * valy1;
417 dys[1] = valx2 * dy1;
418 vals[2] = valx1 * valy2;
419 dxs[2] = dx1 * valy2;
420 dys[2] = valx1 * dy2;
421 vals[3] = valx2 * valy2;
422 dxs[3] = dx2 * valy2;
423 dys[3] = valx2 * dy2;
424}

426template <typename Scalar, typename LocalOrdinal, typename GlobalOrdinal, typename Map, typename Matrix, typename MultiVector>
427void HelmholtzFEM2DProblem<Scalar, LocalOrdinal, GlobalOrdinal, Map, Matrix, MultiVector>::EvalStretch(double& shiftx, double& shifty, const std::vector<Point>& quadPoints, std::vector<Scalar>& sx, std::vector<Scalar>& sy, std::vector<Scalar>& cs) {
428 // For the current element domain, evaluate the PML stretching functions at the quadrature points
429 sx.resize(quadPoints.size());
430 sy.resize(quadPoints.size());
431 cs.resize(quadPoints.size());
432 for (unsigned int i = 0; i < quadPoints.size(); i++) {
433 double quadx = quadPoints[i].x;
434 double quady = quadPoints[i].y;
435 double curx = shiftx + quadx;
436 double cury = shifty + quady;
437 double sigx, sigy;
438 if (curx < LBx_) {
439 sigx = delta_ * pow((curx - LBx_) / PMLwidthx_, 2);
440 } else if (curx > RBx_) {
441 sigx = delta_ * pow((curx - RBx_) / PMLwidthx_, 2);
442 } else {
443 sigx = 0.0;
444 }
445 if (cury < LBy_) {
446 sigy = delta_ * pow((cury - LBy_) / PMLwidthy_, 2);
447 } else if (cury > RBy_) {
448 sigy = delta_ * pow((cury - RBy_) / PMLwidthy_, 2);
449 } else {
450 sigy = 0.0;
451 }
452 Scalar pmlx(1.0, sigx);
453 Scalar pmly(1.0, sigy);
454 Scalar speed = velocitymodel_.getVelocity(curx, cury, 0.0);
455 sx[i] = pmlx;
456 sy[i] = pmly;
457 cs[i] = speed;
458 }
459}
