44template <typename ordinal_type, typename value_type>
45ordinal_type
46Stokhos::SchurPreconditioner<ordinal_type,value_type>::
47size (ordinal_type n, ordinal_type lm) const
48{
49 //n is the polynomial order and lm is the number of random variables
50 // return (fact(n+lm)/(fact(n)*fact(lm)));
51 ordinal_type min;
52 if (n == 0 )
53 return 1;
54 else {
55 if (n<=lm){
56 min = n;
57 }
58 else {
59 min = lm;
60 }
61
62 ordinal_type num = n+lm;
63 for (ordinal_type i=1; i<=min-1; i++)
64 num = num*(n+lm-i);
65 return num/fact(min);
66 }
67}

69template <typename ordinal_type, typename value_type>
70ordinal_type
71Stokhos::SchurPreconditioner<ordinal_type,value_type>::
72ApplyInverse(const Teuchos::SerialDenseMatrix<ordinal_type, value_type>& Input,
73 Teuchos::SerialDenseMatrix<ordinal_type, value_type>& U,
74 const ordinal_type n) const
75{
76 //p: polynomial order; m: number of random variables; n: level to stop and form actual Schur Complement; diag=0: Use only diagonal of block D
77 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Resid(Teuchos::Copy, Input);
78 ordinal_type ret;
79 ordinal_type lmin;
80 if (n<=1)
81 lmin=1;
82 else
83 lmin=n;
84
85 Teuchos::SerialDenseSolver<ordinal_type, value_type> solver;
86 for (ordinal_type l=p; l>=lmin; l--){
87 ordinal_type c=size(l,m);
88 ordinal_type s = size(l-1,m);
89
90 //split residual
91 Teuchos::SerialDenseMatrix<ordinal_type, value_type> rminus(Teuchos::View, Resid, s, 1);
92 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r(Teuchos::Copy, Resid, c-s, 1, s, 0);
93
94 //Compute pre-correction
95 Teuchos::SerialDenseMatrix<ordinal_type, value_type> B(Teuchos::View, K, s, c-s, 0, s);
96 Teuchos::SerialDenseMatrix<ordinal_type, value_type> D(Teuchos::View, K, c-s, c-s, s,s);
97
98 //Computing inv(D)r
99 if (diag == 0){
100 //For D diagonal
101 for (ordinal_type i=0; i<c-s; i++)
102 r(i,0)=r(i,0)/D(i,i);
103
104 }
105 else{
106 //For full D block
107 Teuchos::RCP< Teuchos::SerialDenseMatrix<ordinal_type, value_type> > DD, RR;
108 DD = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (Teuchos::Copy, D));
109
110 RR = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (r));
111
112
113 // Setup solver
114 solver.setMatrix(DD);
115 solver.setVectors(RR, RR);
116 //Solve D*r=r
117 if (solver.shouldEquilibrate()) {
118 solver.factorWithEquilibration(true);
119 solver.equilibrateMatrix();
120 }
121 solver.solve();
122
123
124 for (ordinal_type i=0; i<c-s; i++){
125 r(i,0)=(*RR)(i,0);
126 }
127
128 }
129
130 Teuchos::SerialDenseMatrix<ordinal_type, value_type> rm(Teuchos::Copy, rminus);
131 ret = rm.multiply(Teuchos::NO_TRANS,Teuchos::NO_TRANS, -1.0, B, r, 1.0);
132 TEUCHOS_ASSERT(ret == 0);
133
134 //set r(l-1)=g(l-1)
135 if (l>lmin){
136 for (ordinal_type i=0; i<s; i++)
137 Resid(i,0)=rm(i,0);
138 }
139 else {
140 //For l=1, solve A0u0=g0
141 if (n<1){
142 U(0,0)=rm(0,0)/K(0,0);
143 }
144 //Compute the Schur completement
145 else {
146 for (ordinal_type i=0; i<s; i++)
147 Resid(i,0)=rm(i,0);
148
149 Teuchos::SerialDenseMatrix<ordinal_type, value_type> S(Teuchos::Copy, K, s, s);
150 Teuchos::SerialDenseMatrix<ordinal_type, value_type> BinvD(s,c-s);
151 for (ordinal_type i=0; i<c-s; i++)
152 for (ordinal_type j=0; j<s; j++)
153 BinvD(j,i)=B(j,i)/D(i,i);
154
155 S.multiply(Teuchos::NO_TRANS,Teuchos::TRANS, -1.0, BinvD, B, 1.0);
156 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Ul(Teuchos::View, U, s, 1);
157 Teuchos::RCP< Teuchos::SerialDenseMatrix<ordinal_type, value_type> > SS, UU, RR;
158 SS = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (S));
159 UU = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (Teuchos::Copy, Ul));
160 RR = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (rm));
161 //Setup solver
162 Teuchos::SerialDenseSolver<ordinal_type, value_type> solver2;
163 solver2.setMatrix(SS);
164 solver2.setVectors(UU, RR);
165 //Solve S*u=rm
166 if (solver2.shouldEquilibrate()) {
167 solver2.factorWithEquilibration(true);
168 solver2.equilibrateMatrix();
169 }
170 solver2.solve();
171
172 for (ordinal_type i=0; i<s; i++)
173 U(i,0)=(*UU)(i,0);
174 }
175 }
176 }
177
178 for (ordinal_type l=lmin; l<=p; l++){
179 //compute post-correction
180 ordinal_type c = size(l,m);
181 ordinal_type s = size(l-1,m);
182
183 Teuchos::SerialDenseMatrix<ordinal_type, value_type> B(Teuchos::View, K, s, c-s, 0, s);
184 Teuchos::SerialDenseMatrix<ordinal_type, value_type> D(Teuchos::Copy, K, c-s, c-s, s,s);
185 //split residual
186 Teuchos::SerialDenseMatrix<ordinal_type, value_type> r(Teuchos::Copy, Resid, c-s, 1, s, 0);
187 //Get first s values of U
188 Teuchos::SerialDenseMatrix<ordinal_type, value_type> Ul(Teuchos::View, U, s, 1);
189 ret = r.multiply(Teuchos::TRANS,Teuchos::NO_TRANS, -1.0, B, Ul, 1.0);
190 if (diag == 0) {
191 //For diag D
192 //Concatenate U
193 for (ordinal_type i=s; i<c; i++)
194 U(i,0)=r(-s+i,0)/D(-s+i,-s+i);
195
196 }
197 else {
198 //For full block D
199
200 Teuchos::RCP< Teuchos::SerialDenseMatrix<ordinal_type, value_type> > DD, RR;
201 DD = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (D));
202 RR = Teuchos::rcp(new Teuchos::SerialDenseMatrix<ordinal_type, value_type> (r));
203 // Setup solver
204 solver.setMatrix(DD);
205 solver.setVectors(RR, RR);
206 //Solve D*r=r
207 if (solver.shouldEquilibrate()) {
208 solver.factorWithEquilibration(true);
209 solver.equilibrateMatrix();
210 }
211 solver.solve();
212 for (ordinal_type i=s; i<c; i++)
213 U(i,0)=(*RR)(-s+i,0);
214 }
215
216
217 }
218
219 return 0;
220}
