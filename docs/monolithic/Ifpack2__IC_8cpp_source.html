95int Ifpack2_IC::ComputeSetup() {
96 // (re)allocate memory for ICT factors
97 U_ = rcp(new Tpetra_CrsMatrix(Copy, Matrix().RowMatrixRowMap(),
98 Matrix().RowMatrixRowMap(), 0));
99 D_ = rcp(new Tpetra_Vector(Matrix().RowMatrixRowMap()));
100
101 if (U_.get() == 0 || D_.get() == 0)
102 IFPACK2_CHK_ERR(-5); // memory allocation error
103
104 int ierr = 0;
105 int i, j;
106 int NumIn, NumL, NumU;
107 bool DiagFound;
108 int NumNonzeroDiags = 0;
109
110 // Get Maximun Row length
111 int MaxNumEntries = Matrix().MaxNumEntries();
112
113 vector<int> InI(MaxNumEntries); // Allocate temp space
114 vector<int> UI(MaxNumEntries);
115 vector<double> InV(MaxNumEntries);
116 vector<double> UV(MaxNumEntries);
117
118 double *DV;
119 ierr = D_->ExtractView(&DV); // Get view of diagonal
120
121 // First we copy the user's matrix into diagonal vector and U, regardless of fill level
122
123 int NumRows = Matrix().NumMyRows();
124
125 for (i = 0; i < NumRows; i++) {
126 Matrix().ExtractMyRowCopy(i, MaxNumEntries, NumIn, &InV[0], &InI[0]); // Get Values and Indices
127
128 // Split into L and U (we don't assume that indices are ordered).
129 NumL = 0;
130 NumU = 0;
131 DiagFound = false;
132
133 for (j = 0; j < NumIn; j++) {
134 int k = InI[j];
135
136 if (k == i) {
137 DiagFound = true;
138 // Store perturbed diagonal in Tpetra_Vector D_
139 DV[i] += Rthresh_ * InV[j] + EPETRA_SGN(InV[j]) * Athresh_;
140 }
141
142 else if (k < 0)
143 return (-1); // Out of range
144 else if (i < k && k < NumRows) {
145 UI[NumU] = k;
146 UV[NumU] = InV[j];
147 NumU++;
148 }
149 }
150
151 // Check in things for this row of L and U
152
153 if (DiagFound) NumNonzeroDiags++;
154 if (NumU) U_->InsertMyValues(i, NumU, &UV[0], &UI[0]);
155 }
156
157 U_->FillComplete(Matrix().OperatorDomainMap(),
158 Matrix().OperatorRangeMap());
159
160 int ierr1 = 0;
161 if (NumNonzeroDiags < U_->NumMyRows()) ierr1 = 1;
162 Matrix().Comm().MaxAll(&ierr1, &ierr, 1);
163 IFPACK2_CHK_ERR(ierr);
164
165 return (0);
166}

168//==========================================================================
169int Ifpack2_IC::Compute() {
170 if (!IsInitialized())
171 IFPACK2_CHK_ERR(Initialize());
172
173 IsComputed_ = false;
174
175 // copy matrix into L and U.
176 IFPACK2_CHK_ERR(ComputeSetup());
177
178 int i;
179
180 int m, n, nz, Nrhs, ldrhs, ldlhs;
181 int *ptr = 0, *ind;
182 double *val, *rhs, *lhs;
183
184 int ierr = Tpetra_Util_ExtractHbData(U_.get(), 0, 0, m, n, nz, ptr, ind,
185 val, Nrhs, rhs, ldrhs, lhs, ldlhs);
186 if (ierr < 0)
187 IFPACK2_CHK_ERR(ierr);
188
189 Ifpack2_AIJMatrix *Aict;
190 if (Aict_ == 0) {
191 Aict = new Ifpack2_AIJMatrix;
192 Aict_ = (void *)Aict;
193 } else
194 Aict = (Ifpack2_AIJMatrix *)Aict_;
195 Ifpack2_AIJMatrix *Lict;
196 if (Lict_ == 0) {
197 Lict = new Ifpack2_AIJMatrix;
198 Lict_ = (void *)Lict;
199 } else {
200 Lict = (Ifpack2_AIJMatrix *)Lict_;
201 Ifpack2_AIJMatrix_dealloc(Lict); // Delete storage, crout_ict will allocate it again.
202 }
203 if (Ldiag_ != 0) delete[] Ldiag_; // Delete storage, crout_ict will allocate it again.
204 Aict->val = val;
205 Aict->col = ind;
206 Aict->ptr = ptr;
207 double *DV;
208 EPETRA_CHK_ERR(D_->ExtractView(&DV)); // Get view of diagonal
209
210 crout_ict(m, Aict, DV, Droptol_, Lfil_, Lict, &Ldiag_);
211
212 // Get rid of unnecessary data
213 delete[] ptr;
214
215 // Create Epetra View of L from crout_ict
216 U_ = rcp(new Tpetra_CrsMatrix(View, A_->RowMatrixRowMap(), A_->RowMatrixRowMap(), 0));
217 D_ = rcp(new Tpetra_Vector(View, A_->RowMatrixRowMap(), Ldiag_));
218
219 ptr = Lict->ptr;
220 ind = Lict->col;
221 val = Lict->val;
222
223 for (i = 0; i < m; i++) {
224 int NumEntries = ptr[i + 1] - ptr[i];
225 int *Indices = ind + ptr[i];
226 double *Values = val + ptr[i];
227 U_->InsertMyValues(i, NumEntries, Values, Indices);
228 }
229
230 U_->FillComplete(A_->OperatorDomainMap(), A_->OperatorRangeMap());
231 D_->Reciprocal(*D_); // Put reciprocal of diagonal in this vector
232
233 double current_flops = 2 * nz; // Just an estimate
234 double total_flops = 0;
235
236 A_->Comm().SumAll(&current_flops, &total_flops, 1); // Get total madds across all PEs
237
238 ComputeFlops_ += total_flops;
239 // Now count the rest. NOTE: those flops are *global*
240 ComputeFlops_ += (double)U_->NumGlobalNonzeros(); // Accounts for multiplier above
241 ComputeFlops_ += (double)D_->GlobalLength(); // Accounts for reciprocal of diagonal
242
243 IsComputed_ = true;
244
245 return (0);
246}

249// This function finds Y such that LDU Y = X or U(trans) D L(trans) Y = X for multiple RHS
250int Ifpack2_IC::ApplyInverse(const Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node> &X,
251 Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node> &Y) const {
252 if (!IsComputed())
253 IFPACK2_CHK_ERR(-3); // compute preconditioner first
254
255 if (X.NumVectors() != Y.NumVectors())
256 IFPACK2_CHK_ERR(-2); // Return error: X and Y not the same size
257
258 bool Upper = true;
259 bool UnitDiagonal = true;
260
261 // AztecOO gives X and Y pointing to the same memory location,
262 // need to create an auxiliary vector, Xcopy
263 RefCountPtr<const Tpetra_MultiVector> Xcopy;
264 if (X.Pointers()[0] == Y.Pointers()[0])
265 Xcopy = rcp(new Tpetra_MultiVector(X));
266 else
267 Xcopy = rcp(&X, false);
268
269 U_->Solve(Upper, true, UnitDiagonal, *Xcopy, Y);
270 Y.Multiply(1.0, *D_, Y, 0.0); // y = D*y (D_ has inverse of diagonal)
271 U_->Solve(Upper, false, UnitDiagonal, Y, Y); // Solve Uy = y
272
273 ++NumApplyInverse_;
274 ApplyInverseFlops_ += 4.0 * U_->NumGlobalNonzeros();
275 ApplyInverseFlops_ += D_->GlobalLength();
276 return (0);
277}

280// This function finds X such that LDU Y = X or U(trans) D L(trans) Y = X for multiple RHS
281int Ifpack2_IC::Apply(const Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node> &X,
282 Tpetra_MultiVector<Scalar, LocalOrdinal, GlobalOrdinal, Node> &Y) const {
283 if (X.NumVectors() != Y.NumVectors())
284 IFPACK2_CHK_ERR(-2); // Return error: X and Y not the same size
285
286 Tpetra_MultiVector *X1 = (Tpetra_MultiVector *)&X;
287 Tpetra_MultiVector *Y1 = (Tpetra_MultiVector *)&Y;
288
289 U_->Multiply(false, *X1, *Y1);
290 Y1->Update(1.0, *X1, 1.0); // Y1 = Y1 + X1 (account for implicit unit diagonal)
291 Y1->ReciprocalMultiply(1.0, *D_, *Y1, 0.0); // y = D*y (D_ has inverse of diagonal)
292 Tpetra_MultiVector Y1temp(*Y1); // Need a temp copy of Y1
293 U_->Multiply(true, Y1temp, *Y1);
294 Y1->Update(1.0, Y1temp, 1.0); // (account for implicit unit diagonal)
295 return (0);
296}
