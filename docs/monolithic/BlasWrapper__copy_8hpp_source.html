32copy (const XMV& X, const YMV& Y)

34 static_assert (Kokkos::is_view<XMV>::value, "BlasWrapper::copy: "
35 "X is not a Kokkos::View.");
36 static_assert (Kokkos::is_view<YMV>::value, "BlasWrapper::copy: "
37 "Y is not a Kokkos::View.");
38 static_assert (std::is_same<typename YMV::value_type,
39 typename YMV::non_const_value_type>::value,
40 "BlasWrapper::copy: Y is const. "
41 "It must be nonconst, because it is an output argument "
42 "(we have to be able to write to its entries).");
43 static_assert (int(XMV::rank) == int(YMV::rank), "BlasWrapper::copy: "
44 "X and Y must have the same rank.");
45 static_assert (YMV::rank == 1 || YMV::rank == 2, "BlasWrapper::copy: "
46 "YMV and XMV must either have rank 1 or rank 2.");
47
48 // Check compatibility of dimensions at run time.
49 if (X.extent(0) != Y.extent(0) ||
50 X.extent(1) != Y.extent(1)) {
51 std::ostringstream os;
52 os << "BlasWrapper::copy (MV): Dimensions of Y and X do not match: "
53 << "Y: " << Y.extent(0) << " x " << Y.extent(1)
54 << ", X: " << X.extent(0) << " x " << X.extent(1);
55 throw std::runtime_error (os.str ());
56 }
57
58 // Create unmanaged versions of the input Views. RMV and XMV may be
59 // rank 1 or rank 2.
60 typedef Kokkos::View<
61 typename std::conditional<
62 YMV::rank == 1,
63 typename YMV::non_const_value_type*,
64 typename YMV::non_const_value_type** >::type,
65 typename KokkosKernels::Impl::GetUnifiedLayout<YMV>::array_layout,
66 typename YMV::device_type,
67 Kokkos::MemoryTraits<Kokkos::Unmanaged> > YMV_Internal;
68 typedef Kokkos::View<
69 typename std::conditional<
70 XMV::rank == 1,
71 typename XMV::const_value_type*,
72 typename XMV::const_value_type** >::type,
73 typename KokkosKernels::Impl::GetUnifiedLayout<XMV>::array_layout,
74 typename XMV::device_type,
75 Kokkos::MemoryTraits<Kokkos::Unmanaged> > XMV_Internal;
76
77 YMV_Internal Y_internal = Y;
78 XMV_Internal X_internal = X;
79
80 Impl::Copy<XMV_Internal, YMV_Internal>::copy (X_internal, Y_internal);
81}
