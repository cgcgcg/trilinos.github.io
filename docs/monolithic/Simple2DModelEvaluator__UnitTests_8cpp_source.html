53TEUCHOS_UNIT_TEST_TEMPLATE_1_DECL( Simple2DModelEvaluator, eval, Scalar )
54{
55 using Teuchos::as; using Teuchos::rcp_dynamic_cast;
56 typedef Teuchos::ScalarTraits<Scalar> ST;
57 typedef typename ST::magnitudeType ScalarMag;
58
59 out << "A) Create the Simple2DModelEvaluator and objects ...\n";
60
61 RCP<Simple2DModelEvaluator<Scalar> > model = simple2DModelEvaluator<Scalar>();
62
63 ModelEvaluatorBase::InArgs<Scalar> in_args = model->getNominalValues();
64 ModelEvaluatorBase::OutArgs<Scalar> out_args = model->createOutArgs();
65
66 const RCP<const VectorSpaceBase<Scalar> > f_space = model->get_f_space();
67 const RCP<VectorBase<Scalar> > f = createMember(f_space);
68 const RCP<LinearOpBase<Scalar> > W_op = model->create_W_op();
69 const RCP<PreconditionerBase<Scalar> > W_prec = model->create_W_prec();
70
71 out << "B) Get the concrete output objects ...\n";
72
73 const RCP<SimpleDenseLinearOp<Scalar> > W_sdlo =
74 rcp_dynamic_cast<SimpleDenseLinearOp<Scalar> >(W_op, true);
75 const RCP<MultiVectorBase<Scalar> > W_mv = W_sdlo->getNonconstMultiVector();
76
77 const RCP<SimpleDenseLinearOp<Scalar> > W_prec_sdlo =
78 rcp_dynamic_cast<SimpleDenseLinearOp<Scalar> >(W_prec->getNonconstUnspecifiedPrecOp(), true);
79 const RCP<MultiVectorBase<Scalar> > W_prec_mv = W_prec_sdlo->getNonconstMultiVector();
80
81 out << "C) Fill the output objects with the wrong values ...\n";
82
83 const Scalar threethree = 3.3;
84 V_S(f.ptr(), threethree);
85 assign(W_mv.ptr(), threethree);
86 assign(W_prec_mv.ptr(), threethree);
87
88 out << "D) Make sure all output objects are initialized to the wrong values ...\n";
89
90 const ScalarMag tol = as<ScalarMag>(10.0) * ST::eps();
91
92 {
93 const ConstDetachedVectorView<Scalar> f_dv(f);
94 TEST_FLOATING_EQUALITY(f_dv[0], threethree, tol);
95 TEST_FLOATING_EQUALITY(f_dv[1], threethree, tol);
96
97 const ConstDetachedMultiVectorView<Scalar> W_dv(*W_mv);
98 TEST_FLOATING_EQUALITY(W_dv(0,0), threethree, tol);
99 TEST_FLOATING_EQUALITY(W_dv(0,1), threethree, tol);
100 TEST_FLOATING_EQUALITY(W_dv(1,0), threethree, tol);
101 TEST_FLOATING_EQUALITY(W_dv(1,1), threethree, tol);
102
103 const ConstDetachedMultiVectorView<Scalar> W_prec_dv(*W_mv);
104 TEST_FLOATING_EQUALITY(W_prec_dv(0,0), threethree, tol);
105 TEST_FLOATING_EQUALITY(W_prec_dv(0,1), threethree, tol);
106 TEST_FLOATING_EQUALITY(W_prec_dv(1,0), threethree, tol);
107 TEST_FLOATING_EQUALITY(W_prec_dv(1,1), threethree, tol);
108 }
109
110 out << "E) Evaluate the output objects ...\n";
111
112 out_args.set_f(f);
113 out_args.set_W_op(W_op);
114 out_args.set_W_prec(W_prec);
115
116 model->evalModel(in_args, out_args);
117
118 out << "F) Check the values of the output objects ...\n";
119
120 // Based on nominalValue settings x0=1, x1=1, p0=2, p1=0, d=10
121
122 const ScalarMag zero = ST::zero();
123
124 {
125 const ConstDetachedVectorView<Scalar> f_dv(f);
126 TEST_FLOATING_EQUALITY(f_dv[0], as<Scalar>(1.0+1.0*1.0-2.0), tol);
127 TEST_FLOATING_EQUALITY(f_dv[0], as<Scalar>(10.0*(1.0*1.0-1.0-0.0)), tol);
128
129 const ConstDetachedMultiVectorView<Scalar> W_dv(*W_mv);
130 TEST_FLOATING_EQUALITY(W_dv(0,0), as<Scalar>(1.0), tol);
131 TEST_FLOATING_EQUALITY(W_dv(0,1), as<Scalar>(2.0), tol);
132 TEST_FLOATING_EQUALITY(W_dv(1,0), as<Scalar>(10.0 * 2.0 * 1.0), tol);
133 TEST_FLOATING_EQUALITY(W_dv(1,1), as<Scalar>(-10.0), tol);
134
135 const ConstDetachedMultiVectorView<Scalar> W_prec_dv(*W_prec_mv);
136 TEST_FLOATING_EQUALITY(W_prec_dv(0,0), as<Scalar>(1.0), tol);
137 TEST_FLOATING_EQUALITY(W_prec_dv(0,1), zero, tol);
138 TEST_FLOATING_EQUALITY(W_prec_dv(1,0), zero, tol);
139 TEST_FLOATING_EQUALITY(W_prec_dv(1,1), as<Scalar>(-0.1), tol);
140 }
141
142}
