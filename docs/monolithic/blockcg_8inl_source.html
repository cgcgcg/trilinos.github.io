69template <class LinearOperator,
70 class Vector,
71 class Monitor,
72 class Preconditioner>
73void blockcg(LinearOperator& A,
74 Vector& x,
75 Vector& b,
76 Monitor& monitor,
77 Preconditioner& M)
78{
79 CUSP_PROFILE_SCOPED();
80
81 typedef typename LinearOperator::value_type ValueType;
82 typedef typename LinearOperator::memory_space MemorySpace;
83 typedef typename LinearOperator::index_type IndexType;
84 typedef typename Vector::orientation Orientation;
85
86 assert(A.num_rows == A.num_cols); // sanity check
87
88 const size_t N = A.num_rows;
89
90 const size_t numRHS = b.num_cols;
91
92 // allocate workspace
93 cusp::array2d<ValueType,MemorySpace,Orientation> y(N, numRHS, 0);
94 cusp::array2d<ValueType,MemorySpace,Orientation> r(N, numRHS, 0);
95 cusp::array2d<ValueType,MemorySpace,Orientation> z(N, numRHS, 0);
96
97 cusp::array1d<ValueType,MemorySpace> rTz(numRHS);
98 cusp::array1d<ValueType,MemorySpace> rTz_old(numRHS);
99 cusp::array1d<ValueType,MemorySpace> pAp(numRHS);
100
101 cusp::array1d<ValueType,MemorySpace> alpha(numRHS);
102 cusp::array1d<ValueType,MemorySpace> beta(numRHS);
103
104 cusp::array2d<ValueType,MemorySpace,Orientation> p(N, numRHS,0);
105 cusp::array2d<ValueType,MemorySpace,Orientation> tmp(N, numRHS,0);
106
107 //Compute the initial alpha and (r^T)z
108 // y <- Ax
109 cusp::OVmultiply(A, x, y);
110 // r <- b - A*x
111 cusp::axpby_array(ValueType(1), b, ValueType(-1), y, r);
112
113 cusp::axpby_array(ValueType(1.0), b, ValueType(-1.0), y, p);
114 // z <- M*r
115 cusp::MVmultiply(M, r, z);
116
117 // p <- z
118 cusp::copy(z, p);
119 // rz = <r^H, z>
120 //compute vector rHz where the components are the individual dot-products rHz[i] = r[i]^T * z[i] wherer[i] is the i-th column of r
121 cusp::MVdot(r,z,rTz);
122
123 //Iterate until moniter isfinished (if convg tol reached or max iters reached)
124 while (!monitor.finished(r))
125 {
126 // y <- Ap
127 cusp::OVmultiply(A, p, y);
128
129 //pAp <- <p, Ap>
130 cusp::MVdot(p, y, pAp);
131
132 // Compute alpha <- <r,z>/<p,y> = (r^T)*z / (p^T)*A*p
133 for (IndexType i = 0; i < numRHS; i++){
134 alpha[i] = rTz[i] / pAp[i];
135 }
136
137 //Update the solution
138 // x <- x + p *alpha
139
140 cusp::MVmultiply(p, alpha, tmp);
141 cusp::axpby_array(ValueType(1.0), x, ValueType(1.0), tmp, x);
142
143 //Save the denomiator of beta before residual is updated to rHz_old
144 cusp::copy(rTz, rTz_old);
145
146 //Compute the new residual
147 // r <- r - y * alpha = r - Ax * alpha
148 cusp::MVmultiply(y, alpha, tmp);
149 cusp::axpby_array(ValueType(1.0), r, ValueType(-1.0), tmp, r);
150
151
152 //if precond
153 // z <- M*r
154 cusp::MVmultiply(M, r, z);
155 //Compute beta <- new <r, z> / old <r, z>
156 // rTz = <r, z>
157 cusp::MVdot(r, z, rTz);
158 for (IndexType i = 0; i < numRHS; i++){
159 beta[i] = rTz[i] / rTz_old[i];
160 }
161 cusp::MVmultiply(p, beta, p);
162 cusp::axpby_array(ValueType(1.0), z, ValueType(1.0), p, p);
163
164 ++monitor;
165 }
166}
