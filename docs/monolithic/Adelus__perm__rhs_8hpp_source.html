26 template<class HandleType, class ZViewType, class PViewType>
27 inline
28 void permute_rhs(HandleType& ahandle, ZViewType& RHS, PViewType& permute) {
29 using value_type = typename ZViewType::value_type;
30 using execution_space = typename ZViewType::device_type::execution_space ;
31 using memory_space = typename ZViewType::device_type::memory_space ;
32 using ViewVectorType = Kokkos::View<value_type*, Kokkos::LayoutLeft, memory_space>;
33#ifdef ADELUS_HOST_PINNED_MEM_MPI
34 #if defined(KOKKOS_ENABLE_CUDA)
35 using ViewVectorHostPinnType = Kokkos::View<value_type*, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace>;//CudaHostPinnedSpace
36 #elif defined(KOKKOS_ENABLE_HIP)
37 using ViewVectorHostPinnType = Kokkos::View<value_type*, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace>;//HIPHostPinnedSpace
38 #endif
39#endif
40
41 MPI_Comm col_comm = ahandle.get_col_comm();
42 int myrow = ahandle.get_myrow();
43 int nprocs_col = ahandle.get_nprocs_col();
44 int nrows_matrix = ahandle.get_nrows_matrix();
45
46 int pivot_row, k_row;
47 ViewVectorType tmpr( "tmpr", RHS.extent(1) );
48 ViewVectorType tmps( "tmps", RHS.extent(1) );
49#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
50 ViewVectorHostPinnType h_tmpr( "h_tmpr", RHS.extent(1) );
51 ViewVectorHostPinnType h_tmps( "h_tmps", RHS.extent(1) );
52#endif
53
54 MPI_Status msgstatus;
55
56 //TODO: try this later
57 //MPI_Datatype strided_vec_type;
58 //int strided_vec_nblocks = RHS.extent(1);
59 //int strided_vec_blocklen = 1;
60 //int strided_vec_stride = RHS.extent(0);
61 //MPI_Type_vector( strided_vec_nblocks, strided_vec_blocklen, strided_vec_stride,
62 // ADELUS_MPI_DATA_TYPE, &strided_vec_type);
63 //MPI_Type_commit(&strided_vec_type);
64
65#ifdef GET_TIMING
66 double permuterhstime,t1;
67
68 t1 = MPI_Wtime();
69#endif
70
71 for (int k=0;k<=nrows_matrix-2;k++) {
72 k_row=k%nprocs_col;
73
74 if (ahandle.get_my_rhs() > 0) {
75 if (myrow==k_row) pivot_row = static_cast<int>(permute(k/nprocs_col));
76 MPI_Bcast(&pivot_row,1,MPI_INT,k_row,col_comm);
77 int pivot_row_pid = pivot_row%nprocs_col;
78
79 if (k != pivot_row) {
80 if (k_row == pivot_row_pid) {//pivot row is in the same rank
81 if (myrow == k_row) {
82 int curr_lrid = k/nprocs_col;
83 int piv_lrid = pivot_row/nprocs_col;
84 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
85 value_type tmp = RHS(curr_lrid,i);
86 RHS(curr_lrid,i) = RHS(piv_lrid,i);
87 RHS(piv_lrid,i) = tmp;
88 });
89 Kokkos::fence();
90 }
91 }
92 else {//pivot row is is a different rank
93 if (myrow == k_row) {
94 int curr_lrid = k/nprocs_col;
95 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
96 tmps(i) = RHS(curr_lrid,i);
97 });
98
99#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
100 Kokkos::deep_copy(h_tmps,tmps);
101 MPI_Send(reinterpret_cast<char *>(h_tmps.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,pivot_row_pid,2,col_comm);
102 MPI_Recv(reinterpret_cast<char *>(h_tmpr.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,pivot_row_pid,3,col_comm,&msgstatus);
103 Kokkos::deep_copy(tmpr,h_tmpr);
104#else //GPU-aware MPI
105 Kokkos::fence();
106
107 MPI_Send(reinterpret_cast<char *>(tmps.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,pivot_row_pid,2,col_comm);
108 MPI_Recv(reinterpret_cast<char *>(tmpr.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,pivot_row_pid,3,col_comm,&msgstatus);
109#endif
110
111 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
112 RHS(curr_lrid,i) = tmpr(i);
113 });
114 Kokkos::fence();
115 }
116 if (myrow == pivot_row_pid) {
117 int piv_lrid = pivot_row/nprocs_col;
118 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
119 tmps(i) = RHS(piv_lrid,i);
120 });
121
122#if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
123 Kokkos::deep_copy(h_tmps,tmps);
124 MPI_Recv(reinterpret_cast<char *>(h_tmpr.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,k_row,2,col_comm,&msgstatus);
125 MPI_Send(reinterpret_cast<char *>(h_tmps.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,k_row,3,col_comm);
126 Kokkos::deep_copy(tmpr,h_tmpr);
127#else // GPU-aware MPI
128 Kokkos::fence();
129
130 MPI_Recv(reinterpret_cast<char *>(tmpr.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,k_row,2,col_comm,&msgstatus);
131 MPI_Send(reinterpret_cast<char *>(tmps.data()),RHS.extent(1)*sizeof(value_type),MPI_CHAR,k_row,3,col_comm);
132#endif
133
134 Kokkos::parallel_for(Kokkos::RangePolicy<execution_space>(0,RHS.extent(1)), KOKKOS_LAMBDA (const int i) {
135 RHS(piv_lrid,i) = tmpr(i);
136 });
137 Kokkos::fence();
138 }
139 }//End of pivot row is is a different rank
140 }// End of if (k != pivot_row)
141
142 }// End of if (my_num_rhs > 0)
143
144 }// End of for (k=0;k<=nrows_matrix-2;k++)
145
146#ifdef GET_TIMING
147 permuterhstime = MPI_Wtime()-t1;
148
149 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
150 "Time to permute rhs", &permuterhstime);
151#endif
152 }// End of function permute_rhs
