428 /*************************************************************************************************
429 * *
430 * DISCRETIZATION SETUP: get inputs, generate mesh, choose cell topology, cubature and basis *
431 * *
432 ************************************************************************************************/
433
434 // *********************** Read XML file with mesh specifications *******************************
435
436 Teuchos::ParameterList inputMeshList;
437 std::string meshInput;
438 Teuchos::ParameterList inputSolverList;
439 int verbose = 1;
440
441 getInputArguments(inputMeshList, meshInput, inputSolverList, argc, argv, Comm);
442
443
444 // ************************************** GENERATE MESH *****************************************
445
446 // Arrays for mesh data: will be resized to required dimensions by the Pamgen wrapper
447 FieldContainer<int128> globalNodeOrdinalsFC;
448 FieldContainer<int128> localCellToNodeFC;
449 FieldContainer<double> localNodeCoordsFC;
450 FieldContainer<int> localNodeOnBoundaryFC;
451 FieldContainer<bool> localNodeIsMineFC;
452
453 // Generate Pamgen mesh and return mesh data in Field Containers
454 getPamgenMesh<double>(localNodeCoordsFC, localCellToNodeFC, localNodeOnBoundaryFC,
455 localNodeIsMineFC, globalNodeOrdinalsFC, meshInput,
456 procRank, numProcs, Comm,
457 Time, msg, verbose);
458
459 // ********************************* GET PHYSICS INFORMATION ************************************
460 // Read physics info from mesh xml file if it's there. Otherwise do nothing.
461 if(inputMeshList.isSublist("Physics Input")){
462 Teuchos::ParameterList & physicsList=inputMeshList.sublist("Physics Input");
463 g_advection[0]=physicsList.get("x advection",0.0);
464 g_advection[1]=physicsList.get("y advection",0.0);
465 g_advection[2]=physicsList.get("z advection",0.0);
466 }
467
468 // *********************************** DEFINE CELL TOPOLOGY *************************************
469
470 // All cells in the Pamgen grid are of the Hexahedron<8> type.
471 ShardsCellTopology cellType(shards::getCellTopologyData<shards::Hexahedron<8> >() );
472
473 int numNodesPerCell = cellType.getNodeCount();
474 int spaceDim = cellType.getDimension();
475
476
477 // ************************************* DEFINE CUBATURE ****************************************
478
479 if (MyPID == 0) std::cout << "Getting cubature ... \n\n";
480
481 // Define cubature of the specified degree for the cellType
482 DefaultCubatureFactory<double> cubFactory;
483 int cubDegree = 2;
484 Teuchos::RCP<Cubature<double> > cellCubature = cubFactory.create(cellType, cubDegree);
485
486 int cubDim = cellCubature -> getDimension();
487 int numCubPoints = cellCubature -> getNumPoints();
488
489 // Get numerical integration points and weights
490 FieldContainer<double> cubPoints (numCubPoints, cubDim);
491 FieldContainer<double> cubWeights(numCubPoints);
492
493 cellCubature -> getCubature(cubPoints, cubWeights);
494
495
496 // *************************************** DEFINE BASIS ****************************************
497
498 if (MyPID == 0) std::cout << "Getting basis ... \n\n";
499
500 // Define basis
501 Basis_HGRAD_HEX_C1_FEM<double, FieldContainer<double> > HGB;
502 int numHGBFields = HGB.getCardinality();
503
504 FieldContainer<double> HGBValues(numHGBFields, numCubPoints);
505 FieldContainer<double> HGBGrads (numHGBFields, numCubPoints, spaceDim);
506
507 // Evaluate basis values and gradients at cubature points on the reference Hex cell
508 HGB.getValues(HGBValues, cubPoints, OPERATOR_VALUE);
509 HGB.getValues(HGBGrads, cubPoints, OPERATOR_GRAD);
510
511
512 /*************************************************************************************************
513 * *
514 * GLOBAL MATRIX PROBLEM SETUP: generate Epetra map & define global Epetra arrays *
515 * *
516 ************************************************************************************************/
517
518 // Count number of myDofs: for the lowest-order nodal FEM = number of myNodes
519 int numMyDofs = 0;
520 int numMyNodes = 0;
521 int numLocalNodes = localNodeCoordsFC.dimension(0);
522
523 for(int localNode = 0; localNode < numLocalNodes; localNode++) {
524 if(localNodeIsMineFC[localNode]) numMyNodes++;
525 }
526 numMyDofs = numMyNodes;
527
528 // Build a list of the global Dof ordinals of myDofs: for the lowest-order nodal FEM = global node ordinals
529 int *myGlobalDofs = new int[numMyDofs];
530 int myGlobalDofOrdinal = 0;
531 for(int localNode = 0; localNode < numLocalNodes; localNode++){
532 if(localNodeIsMineFC[localNode]){
533 myGlobalDofs[myGlobalDofOrdinal] = (int)globalNodeOrdinalsFC[localNode]; // NTS: will need to switch back to int128
534 myGlobalDofOrdinal++;
535 }
536 }
537
538 // Define global Epetra arrays
539 Epetra_Map globalMapG (-1, numMyDofs, myGlobalDofs, 0, Comm);
540 Epetra_FECrsMatrix globalOperatorMatrix (Copy, globalMapG, 5*numHGBFields);
541 Epetra_FEVector globalSourceTermVector(globalMapG);
542
543
544 /*************************************************************************************************
545 * *
546 * DIRICHLET BOUNDARY CONDITIONS SETUP: count myDofs that are BV and set BVs into a map *
547 * *
548 ************************************************************************************************/
549
550 // Map, keyed by local node ordinal, with the bdry values at all local nodes that are on the bdry
551 std::map<int, double> localBoundaryValuesMap;
552
553 // Number of myDofs that are also boundary values: for lowest-order FEM = number of myNodes that are on the bdry
554 int numMyBoundaryValueDofs = 0;
555
556
557 // Set the map with the local boundary values & count myDofs that are boundary values
558 for(int localNode = 0; localNode < numLocalNodes; localNode++){
559
560 // If the localNode is on the boundary compute and set the exact solution value into the map
561 if (localNodeOnBoundaryFC(localNode) ){
562 double x = localNodeCoordsFC(localNode, 0);
563 double y = localNodeCoordsFC(localNode, 1);
564 double z = localNodeCoordsFC(localNode, 2);
565 localBoundaryValuesMap[localNode] = exactSolution(x, y, z);
566
567 // If this local node is also "mine", i.e., owned, increment counter for myDofs that are bdry values
568 if(localNodeIsMineFC[localNode]){
569 numMyBoundaryValueDofs++;
570 }
571 }
572 }
573
574 // Lookup table for ML_Epetra::Apply_OAZToMatrix method: myBoundaryValueDofs[myBvOrdinal] = myDofOrdinal
575 int * myBoundaryValueDofs = new int [numMyBoundaryValueDofs];
576 int myBvOrdinal = 0;
577 int myDofOrdinal = 0;
578
579 // Count the bdry values and set their associated myDofOrdinals into the lookup table
580 for (int localNode = 0; localNode < numLocalNodes; localNode++){
581 if (localNodeIsMineFC[localNode]){
582 if (localNodeOnBoundaryFC(localNode)){
583 myBoundaryValueDofs[myBvOrdinal] = myDofOrdinal;
584 myBvOrdinal++;
585 }
586 myDofOrdinal++;
587 }
588 }
589
590
591 /*************************************************************************************************
592 * *
593 * ASSEMBLY OF WORKSET CONTRIBUTIONS TO MATRIX AND RIGHT HAND SIDE: workset loop *
594 * *
595 ************************************************************************************************/
596
597 // Define desired workset size and count how many worksets there are on this processor's mesh block
598 int numLocalGridCells = localCellToNodeFC.dimension(0);
599 int desiredWorksetSize = numLocalGridCells; // change to desired workset size!
600 //int desiredWorksetSize = 100; // change to desired workset size!
601 int numWorksets = numLocalGridCells/desiredWorksetSize;
602
603 // When numLocalGridCells is not divisible by desiredWorksetSize, increase workset count by 1
604 if(numWorksets*desiredWorksetSize < numLocalGridCells) numWorksets += 1;
605
606 if (MyPID == 0) {
607 std::cout << "Building discretization matrix and right hand side... \n\n";
608 std::cout << "\tDesired workset size: " << desiredWorksetSize <<"\n";
609 std::cout << "\tNumber of worksets: " << numWorksets <<"\n\n";
610 Time.ResetStartTime();
611 }
612
613
614 // ************************************** Loop over worksets *************************************
615
616 for(int workset = 0; workset < numWorksets; workset++){
617
618 // Compute cell numbers where the workset starts and ends
619 int worksetSize = 0;
620 int worksetBegin = (workset + 0)*desiredWorksetSize;
621 int worksetEnd = (workset + 1)*desiredWorksetSize;
622
623 // When numLocalGridCells is not divisible by desiredWorksetSize, the last workset ends at numLocalGridCells
624 worksetEnd = (worksetEnd <= numLocalGridCells) ? worksetEnd : numLocalGridCells;
625
626 // Now we know the actual workset size and can allocate the array for the cell nodes
627 worksetSize = worksetEnd - worksetBegin;
628 FieldContainer<double> cellWorkset(worksetSize, numNodesPerCell, spaceDim);
629
630 // Copy cell nodes to the cell workset
631 int cellCounter = 0;
632 for(int cell = worksetBegin; cell < worksetEnd; cell++){
633 for (int node = 0; node < numNodesPerCell; node++) {
634 cellWorkset(cellCounter, node, 0) = localNodeCoordsFC( localCellToNodeFC(cell, node), 0);
635 cellWorkset(cellCounter, node, 1) = localNodeCoordsFC( localCellToNodeFC(cell, node), 1);
636 cellWorkset(cellCounter, node, 2) = localNodeCoordsFC( localCellToNodeFC(cell, node), 2);
637 }
638 cellCounter++;
639 }
640
641
642 /*************************************************************************************************
643 * Allocate arrays *
644 ************************************************************************************************/
645
646 // Containers for Jacobians, integration measure & cubature points in workset cells
647 FieldContainer<double> worksetJacobian (worksetSize, numCubPoints, spaceDim, spaceDim);
648 FieldContainer<double> worksetJacobInv (worksetSize, numCubPoints, spaceDim, spaceDim);
649 FieldContainer<double> worksetJacobDet (worksetSize, numCubPoints);
650 FieldContainer<double> worksetCubWeights(worksetSize, numCubPoints);
651 FieldContainer<double> worksetCubPoints (worksetSize, numCubPoints, cubDim);
652
653 // Containers for basis values transformed to workset cells and them multiplied by cubature weights
654 FieldContainer<double> worksetHGBValues (worksetSize, numHGBFields, numCubPoints);
655 FieldContainer<double> worksetHGBValuesWeighted(worksetSize, numHGBFields, numCubPoints);
656 FieldContainer<double> worksetHGBGrads (worksetSize, numHGBFields, numCubPoints, spaceDim);
657 FieldContainer<double> worksetHGBGradsWeighted (worksetSize, numHGBFields, numCubPoints, spaceDim);
658
659 // Containers for diffusive & advective fluxes & non-conservative adv. term and reactive terms
660 FieldContainer<double> worksetDiffusiveFlux(worksetSize, numHGBFields, numCubPoints, spaceDim);
661 FieldContainer<double> worksetAdvectiveFlux(worksetSize, numHGBFields, numCubPoints, spaceDim);
662 FieldContainer<double> worksetAdvectiveTerm(worksetSize, numHGBFields, numCubPoints);
663 FieldContainer<double> worksetReactiveTerm (worksetSize, numHGBFields, numCubPoints);
664
665 // Containers for diffusion, advection and reaction values. Require user-defined functions
666 FieldContainer<double> worksetDiffusion (worksetSize, numCubPoints, spaceDim, spaceDim);
667 FieldContainer<double> worksetAdvection (worksetSize, numCubPoints, spaceDim);
668 FieldContainer<double> worksetReaction (worksetSize, numCubPoints);
669 FieldContainer<double> worksetSourceTerm(worksetSize, numCubPoints);
670
671 // Containers for workset contributions to the discretization matrix and the right hand side
672 FieldContainer<double> worksetDiffusionMatrix (worksetSize, numHGBFields, numHGBFields);
673 FieldContainer<double> worksetAdvectionMatrix (worksetSize, numHGBFields, numHGBFields);
674 FieldContainer<double> worksetReactionMatrix (worksetSize, numHGBFields, numHGBFields);
675 FieldContainer<double> worksetSourceTermVector(worksetSize, numHGBFields);
676
677
678 if(MyPID == 0) {
679 std::cout << msg << "Allocate arrays = " << Time.ElapsedTime() << " sec."<< endl;
680 Time.ResetStartTime();
681 }
682
683
684
685
686 /*************************************************************************************************
687 * Reference frame computation: Jacobian data for each cell in the workset *
688 ************************************************************************************************/
689
690 // Compute cell Jacobians, their inverses and their determinants
691 IntrepidCTools::setJacobian (worksetJacobian, cubPoints, cellWorkset, cellType);
692 IntrepidCTools::setJacobianInv(worksetJacobInv, worksetJacobian );
693 IntrepidCTools::setJacobianDet(worksetJacobDet, worksetJacobian );
694
695
696 /*************************************************************************************************
697 * Physical frame computation: user-defined diffusion, advection, reaction and source terms *
698 ************************************************************************************************/
699
700 // Map cubature points to workset cells & compute diffusion, adection, reaction and source terms
701 IntrepidCTools::mapToPhysicalFrame (worksetCubPoints, cubPoints, cellWorkset, cellType);
702 evaluateDiffusionTensor (worksetDiffusion, worksetCubPoints);
703 evaluateAdvectiveVector (worksetAdvection, worksetCubPoints);
704 evaluateReactionCoefficient (worksetReaction, worksetCubPoints);
705
706 if(MyPID == 0) {
707 std::cout << msg << "tabulate problem data " << Time.ElapsedTime() << " sec."<< endl;
708 Time.ResetStartTime();
709 }
710
711
712 evaluateSourceTerm (worksetSourceTerm, worksetCubPoints);
713
714
715 if(MyPID == 0) {
716 std::cout << msg << "Compute RHS " << Time.ElapsedTime() << " sec."<< endl;
717 Time.ResetStartTime();
718 }
719
720
721
722 /*************************************************************************************************
723 * Transform basis functions and their gradients and compute integration measure *
724 ************************************************************************************************/
725
726 // Transform basis values to physical frame:
727 IntrepidFSTools::HGRADtransformVALUE<double>(worksetHGBValues, // clones basis values (u)
728 HGBValues);
729
730 // Transform basis gradients to physical frame:
731 IntrepidFSTools::HGRADtransformGRAD<double>(worksetHGBGrads, // DF^{-T}(grad u)
732 worksetJacobInv, HGBGrads);
733
734 // Compute integration measure for workset cells:
735 IntrepidFSTools::computeCellMeasure<double>(worksetCubWeights, // Det(DF)*w = J*w
736 worksetJacobDet, cubWeights);
737
738 // Multiply transformed (workset) values with weighted measure
739 IntrepidFSTools::multiplyMeasure<double>(worksetHGBValuesWeighted, // (u)*J*w
740 worksetCubWeights, worksetHGBValues);
741
742 // Multiply transformed (workset) gradients with weighted measure
743 IntrepidFSTools::multiplyMeasure<double>(worksetHGBGradsWeighted, // DF^{-T}(grad u)*J*w
744 worksetCubWeights, worksetHGBGrads);
745
746
747 if(MyPID == 0) {
748 std::cout << msg << "Transform time = " << Time.ElapsedTime() << " sec."<< endl;
749 Time.ResetStartTime();
750 }
751
752
753 /*************************************************************************************************
754 * Diffusion term: DF^{-T}(grad u)*K*DF^{-T}(grad u)*J*w *
755 ************************************************************************************************/
756
757 // Compute the diffusive flux:
758 IntrepidFSTools::tensorMultiplyDataField<double>(worksetDiffusiveFlux, // K*(DF^{-T}(grad u)
759 worksetDiffusion,
760 worksetHGBGrads);
761
762 // Integrate to compute workset diffusion contribution to global matrix:
763 IntrepidFSTools::integrate<double>(worksetDiffusionMatrix, // (DF^{-T}(grad u)*J*w)*(K*DF^{-T}(grad u))
764 worksetHGBGradsWeighted,
765 worksetDiffusiveFlux, COMP_BLAS);
766
767
768
769 if(MyPID == 0) {
770 std::cout << msg << "diffusion time = " << Time.ElapsedTime() << " sec."<< endl;
771 Time.ResetStartTime();
772 }
773
774
775
776 /*************************************************************************************************
777 * Advection term: b*DF^{-T}(grad u)*(u)*J*w requires div b = 0 *
778 ************************************************************************************************/
779
780 // Compute the advective term in non-conservative form: requires div b = 0!
781 IntrepidFSTools::dotMultiplyDataField<double>(worksetAdvectiveTerm, // b*(DF^{-T}*(grad u)
782 worksetAdvection,
783 worksetHGBGrads);
784
785 // Integrate to compute workset advection contribution to global matrix:
786 IntrepidFSTools::integrate<double>(worksetAdvectionMatrix, // b*(DF^{-T}*(grad u)(u)*J*w
787 worksetAdvectiveTerm,
788 worksetHGBValuesWeighted, COMP_BLAS);
789
790
791
792 if(MyPID == 0) {
793 std::cout << msg << "advection time = " << Time.ElapsedTime() << " sec."<< endl;
794 Time.ResetStartTime();
795 }
796
797
798
799 /*************************************************************************************************

855 for(int cell = worksetBegin; cell < worksetEnd; cell++){

879 globalOperatorMatrix.InsertGlobalValues(1, &globalRow, 1, &globalCol, &operatorMatrixContribution);

1107#ifdef HAVE_MPI

1109#else
1110 double uh1 = femCoefficients.Values()[rowIndex];
1111#endif
1112 uApprox += uh1*uhGValsTrans(0,i,nPt);
1113 graduApprox1 += uh1*uhGradsTrans(0,i,nPt,0);
1114 graduApprox2 += uh1*uhGradsTrans(0,i,nPt,1);
1115 graduApprox3 += uh1*uhGradsTrans(0,i,nPt,2);

1118 // REMOVE
1119 //std::cout << " exact solution = " << uExact << "\n";
1120 //std::cout << " FEM solution = " << uApprox << "\n\n";
1121
1122
1123
1124 // evaluate the error at cubature points
1125 Linferr = max(Linferr, abs(uExact - uApprox));
1126
1127 L2errElem+=(uExact - uApprox)*(uExact - uApprox)*worksetCubWeightsE(0,nPt);
1128 H1errElem+=((graduExact1 - graduApprox1)*(graduExact1 - graduApprox1))
1129 *worksetCubWeightsE(0,nPt);
1130 H1errElem+=((graduExact2 - graduApprox2)*(graduExact2 - graduApprox2))

1162 // std::cout.copyfmt(oldFormatState);

1258template<typename Scalar>
1259const Scalar sourceTerm(Scalar& x, Scalar& y, Scalar& z){
1260
1261 Scalar u;
1262 Scalar grad_u[3];
1263 Scalar flux[3] = {0.0, 0.0, 0.0};
1264 Scalar diffusion[3][3];
1265 Scalar advection[3];
1266 Scalar f = 0.;
1267
1268 // Indicate the independent variables
1269 x.diff(0,3);
1270 y.diff(1,3);
1271 z.diff(2,3);
1272
1273 // Get exact solution and its gradient
1274 u = exactSolution(x, y, z);
1275 exactSolutionGrad(grad_u, x, y, z);
1276
1277 // Get diffusion tensor and advective vector
1278 diffusionTensor<Scalar>(diffusion, x, y, z);
1279 advectiveVector<Scalar>(advection, x, y, z);
1280
1281 // Compute total flux = (K.grad u - b.u)
1282 for(int i = 0; i < 3; i++){
1283
1284 // Initialize by the advective flux
1285 flux[i] = -advection[i]*u;
1286 //flux[i] = 0.;
1287
1288 // Add diffusive flux
1289 for(int j = 0; j < 3; j++){
1290 flux[i] += diffusion[i][j]*grad_u[j];
1291 }
1292 }
1293
1294 // Compute source term (right hand side): f = -div(flux) + reactionCoeff*u
1295 f = -(flux[0].dx(0) + flux[1].dx(1) + flux[2].dx(2)) + reactionTerm<Scalar>(x, y, z)*u;
1296
1297 return f;
1298}

1462// Test ML

1476 Epetra_MultiVector* lhs = Problem.GetLHS();
1477 Epetra_MultiVector* rhs = Problem.GetRHS();

1479 Epetra_Time Time(A.Comm());

1482 // call ML and AztecOO //
1483 // =================== //

1493 solver.Iterate(500, 1e-10);

1495 delete MLPrec;

1497 uh = *lhs;

1499 // ==================================================== //
1500 // compute difference between exact solution and ML one //
1501 // ==================================================== //
1502 double d = 0.0, d_tot = 0.0;
1503 for( int i=0 ; i<lhs->Map().NumMyElements() ; ++i )
1504 d += ((*lhs)[0][i] - xexact[0][i]) * ((*lhs)[0][i] - xexact[0][i]);
1505
1506 A.Comm().SumAll(&d,&d_tot,1);
1507
1508 // ================== //
1509 // compute ||Ax - b|| //
1510 // ================== //
1511 double Norm;
1512 Epetra_Vector Ax(rhs->Map());
1513 A.Multiply(false, *lhs, Ax);
1514 Ax.Update(1.0, *rhs, -1.0);
1515 Ax.Norm2(&Norm);

1518
1519 if (A.Comm().MyPID() == 0) {

1527 TotalErrorResidual += Norm;
