26int mm_is_valid(MM_typecode matcode)
27{
28 if (!mm_is_matrix(matcode)) return 0;
29 if (mm_is_dense(matcode) && mm_is_pattern(matcode)) return 0;
30 if (mm_is_real(matcode) && mm_is_hermitian(matcode)) return 0;
31 if (mm_is_pattern(matcode) && (mm_is_hermitian(matcode) ||
32 mm_is_skew(matcode))) return 0;
33 return 1;
34}

36int mm_read_banner(FILE *f, MM_typecode *matcode)
37{
38 char line[MM_MAX_LINE_LENGTH];
39 char banner[MM_MAX_TOKEN_LENGTH];
40 char mtx[MM_MAX_TOKEN_LENGTH];
41 char crd[MM_MAX_TOKEN_LENGTH];
42 char data_type[MM_MAX_TOKEN_LENGTH];
43 char storage_scheme[MM_MAX_TOKEN_LENGTH];
44 char *p;
45
46
47 mm_clear_typecode(matcode);
48
49 if (fgets(line, MM_MAX_LINE_LENGTH, f) == NULL)
50 return MM_PREMATURE_EOF;
51
52 if (sscanf(line, "%s %s %s %s %s", banner, mtx, crd, data_type,
53 storage_scheme) != 5)
54 return MM_PREMATURE_EOF;
55
56 for (p=mtx; *p!='\0'; *p=tolower(*p),p++); /* convert to lower case */
57 for (p=crd; *p!='\0'; *p=tolower(*p),p++);
58 for (p=data_type; *p!='\0'; *p=tolower(*p),p++);
59 for (p=storage_scheme; *p!='\0'; *p=tolower(*p),p++);
60
61 /* check for banner */
62 if (strncmp(banner, MatrixMarketBanner, strlen(MatrixMarketBanner)) != 0)
63 return MM_NO_HEADER;
64
65 /* first field should be "mtx" */
66 if (strcmp(mtx, MM_MTX_STR) != 0)
67 return MM_UNSUPPORTED_TYPE;
68 mm_set_matrix(matcode);
69
70
71 /* second field describes whether this is a sparse matrix (in coordinate
72 storgae) or a dense array */
73
74
75 if (strcmp(crd, MM_SPARSE_STR) == 0)
76 mm_set_sparse(matcode);
77 else
78 if (strcmp(crd, MM_DENSE_STR) == 0)
79 mm_set_dense(matcode);
80 else
81 return MM_UNSUPPORTED_TYPE;
82
83
84 /* third field */
85
86 if (strcmp(data_type, MM_REAL_STR) == 0)
87 mm_set_real(matcode);
88 else
89 if (strcmp(data_type, MM_COMPLEX_STR) == 0)
90 mm_set_complex(matcode);
91 else
92 if (strcmp(data_type, MM_PATTERN_STR) == 0)
93 mm_set_pattern(matcode);
94 else
95 if (strcmp(data_type, MM_INT_STR) == 0)
96 mm_set_integer(matcode);
97 else
98 return MM_UNSUPPORTED_TYPE;
99
100
101 /* fourth field */
102
103 if (strcmp(storage_scheme, MM_GENERAL_STR) == 0)
104 mm_set_general(matcode);
105 else
106 if (strcmp(storage_scheme, MM_SYMM_STR) == 0)
107 mm_set_symmetric(matcode);
108 else
109 if (strcmp(storage_scheme, MM_HERM_STR) == 0)
110 mm_set_hermitian(matcode);
111 else
112 if (strcmp(storage_scheme, MM_SKEW_STR) == 0)
113 mm_set_skew(matcode);
114 else
115 return MM_UNSUPPORTED_TYPE;
116
117
118 return 0;
119}

129int mm_read_mtx_crd_size(FILE *f, size_t *M, size_t *N, size_t *nz )
130{
131 char line[MM_MAX_LINE_LENGTH];
132 int num_items_read;
133
134 /* set return null parameter values, in case we exit with errors */
135 *M = *N = *nz = 0;
136
137 /* now continue scanning until you reach the end-of-comments */
138 do
139 {
140 if (fgets(line,MM_MAX_LINE_LENGTH,f) == NULL)
141 return MM_PREMATURE_EOF;
142 }while (line[0] == '%');
143
144 /* line[] is either blank or has M,N, nz */
145 if (sscanf(line, "%lu %lu %lu", M, N, nz) == 3)
146 return 0;
147
148 else
149 do
150 {
151 num_items_read = fscanf(f, "%lu %lu %lu", M, N, nz);
152 if (num_items_read == EOF) return MM_PREMATURE_EOF;
153 }
154 while (num_items_read != 3);
155
156 return 0;
157}

160int mm_read_mtx_array_size(FILE *f, int *M, int *N)
161{
162 char line[MM_MAX_LINE_LENGTH];
163 int num_items_read;
164 /* set return null parameter values, in case we exit with errors */
165 *M = *N = 0;
166
167 /* now continue scanning until you reach the end-of-comments */
168 do
169 {
170 if (fgets(line,MM_MAX_LINE_LENGTH,f) == NULL)
171 return MM_PREMATURE_EOF;
172 }while (line[0] == '%');
173
174 /* line[] is either blank or has M,N, nz */
175 if (sscanf(line, "%d %d", M, N) == 2)
176 return 0;
177
178 else /* we have a blank line */
179 do
180 {
181 num_items_read = fscanf(f, "%d %d", M, N);
182 if (num_items_read == EOF) return MM_PREMATURE_EOF;
183 }
184 while (num_items_read != 2);
185
186 return 0;
187}

205int mm_read_mtx_crd_data(FILE *f, int M, int N, int nz, int I[], int J[],

207{

211 for (i=0; i<nz; i++)
212 if (fscanf(f, "%d %d %lg %lg", &I[i], &J[i], &val[2*i], &val[2*i+1])
213 != 4) return MM_PREMATURE_EOF;
214 }
215 else if (mm_is_real(matcode))
216 {
217 for (i=0; i<nz; i++)
218 {

220 != 3) return MM_PREMATURE_EOF;
221
222 }
223 }
224
225 else if (mm_is_pattern(matcode))
226 {
227 for (i=0; i<nz; i++)
228 if (fscanf(f, "%d %d", &I[i], &J[i])
229 != 2) return MM_PREMATURE_EOF;
230 }
231 else
232 return MM_UNSUPPORTED_TYPE;
233

279int mm_write_mtx_crd(char fname[], int M, int N, int nz, int I[], int J[],
280 double val[], MM_typecode matcode)
281{
282 FILE *f;
283 int i;
284
285 if (strcmp(fname, "stdout") == 0)
286 f = stdout;
287 else
288 if ((f = fopen(fname, "w")) == NULL)
289 return MM_COULD_NOT_WRITE_FILE;
290
291 /* print banner followed by typecode */
292 fprintf(f, "%s ", MatrixMarketBanner);
293 fprintf(f, "%s\n", mm_typecode_to_str(matcode));
294
295 /* print matrix sizes and nonzeros */
296 fprintf(f, "%d %d %d\n", M, N, nz);
297
298 /* print values */
299 if (mm_is_pattern(matcode))
300 for (i=0; i<nz; i++)
301 fprintf(f, "%d %d\n", I[i], J[i]);
302 else
303 if (mm_is_real(matcode))
304 for (i=0; i<nz; i++)
305 fprintf(f, "%d %d %20.16g\n", I[i], J[i], val[i]);
306 else
307 if (mm_is_complex(matcode))
308 for (i=0; i<nz; i++)
309 fprintf(f, "%d %d %20.16g %20.16g\n", I[i], J[i], val[2*i],
310 val[2*i+1]);
311 else
312 {
313 if (f != stdout) fclose(f);
314 return MM_UNSUPPORTED_TYPE;
315 }
316
317 if (f !=stdout) fclose(f);
318
319 return 0;
320}

335char *mm_typecode_to_str(MM_typecode matcode)
336{
337 char buffer[MM_MAX_LINE_LENGTH];
338 char *types[4];
339 char *mm_strdup(const char *);
340
341 /* check for MTX type */
342 if (mm_is_matrix(matcode))
343 types[0] = MM_MTX_STR;
344 else
345 return NULL;
346
347 /* check for CRD or ARR matrix */
348 if (mm_is_sparse(matcode))
349 types[1] = MM_SPARSE_STR;
350 else
351 if (mm_is_dense(matcode))
352 types[1] = MM_DENSE_STR;
353 else
354 return NULL;
355
356 /* check for element data type */
357 if (mm_is_real(matcode))
358 types[2] = MM_REAL_STR;
359 else
360 if (mm_is_complex(matcode))
361 types[2] = MM_COMPLEX_STR;
362 else
363 if (mm_is_pattern(matcode))
364 types[2] = MM_PATTERN_STR;
365 else
366 if (mm_is_integer(matcode))
367 types[2] = MM_INT_STR;
368 else
369 return NULL;
370
371
372 /* check for symmetry type */
373 if (mm_is_general(matcode))
374 types[3] = MM_GENERAL_STR;
375 else
376 if (mm_is_symmetric(matcode))
377 types[3] = MM_SYMM_STR;
378 else
379 if (mm_is_hermitian(matcode))
380 types[3] = MM_HERM_STR;
381 else
382 if (mm_is_skew(matcode))
383 types[3] = MM_SKEW_STR;
384 else
385 return NULL;
386
387 sprintf(buffer,"%s %s %s %s", types[0], types[1], types[2], types[3]);
388 return mm_strdup(buffer);
389
