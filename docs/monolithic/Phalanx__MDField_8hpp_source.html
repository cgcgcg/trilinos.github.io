618 template<int R> void setFieldData(ViewSpecialization<R>,const std::any& a)
619 {
620#if defined(PHX_DEBUG)
621 TEUCHOS_TEST_FOR_EXCEPTION(m_host_data == nullptr, std::logic_error, hostDataErrorMsg());
622 TEUCHOS_TEST_FOR_EXCEPTION(m_host_data->m_tag.is_null(), std::logic_error, fieldTagErrorMsg());
623 m_data_set = true;
624#endif
625
626 // std::any object is always the non-const data type. To correctly
627 // cast the any object to the Kokkos::View, need to pull the const
628 // off the scalar type if this MDField has a const scalar type.
629 using non_const_view = PHX::View<typename array_type::non_const_data_type>;
630 try {
631 non_const_view tmp = std::any_cast<non_const_view>(a);
632 m_view = tmp;
633 }
634 catch (std::exception& ) {
635 std::cout << "\n\nError in compiletime PHX::MDField::setFieldData() in std::any_cast. Tried to cast the field \""
636 << this->fieldTag().name() << "\" with the identifier \"" << this->fieldTag().identifier()
637 << "\" to a type of \"" << Teuchos::demangleName(typeid(non_const_view).name())
638 << "\" from a std::any object containing a type of \""
639 << Teuchos::demangleName(a.type().name()) << "\"." << std::endl;
640 throw;
641 }
642 }
643
644 void setFieldData(ViewSpecialization<0>,const std::any& a)
645 {
646#if defined(PHX_DEBUG)
647 TEUCHOS_TEST_FOR_EXCEPTION(m_host_data == nullptr, std::logic_error, hostDataErrorMsg());
648 TEUCHOS_TEST_FOR_EXCEPTION(m_host_data->m_tag.is_null(), std::logic_error, fieldTagErrorMsg());
649 m_data_set = true;
650#endif
651
652 m_host_data->m_any_holder.set(a);
653
654 using NonConstDataT = typename std::remove_const<Scalar>::type;
655 try {
656 if (m_host_data->m_tag->dataLayout().rank() == 1)
657 m_view = std::any_cast<Kokkos::View<NonConstDataT*,layout_type,device_type>>(a);
658 else if (m_host_data->m_tag->dataLayout().rank() == 2)
659 m_view = std::any_cast<Kokkos::View<NonConstDataT**,layout_type,device_type>>(a);
660 else if (m_host_data->m_tag->dataLayout().rank() == 3)
661 m_view = std::any_cast<Kokkos::View<NonConstDataT***,layout_type,device_type>>(a);
662 else if (m_host_data->m_tag->dataLayout().rank() == 4)
663 m_view = std::any_cast<Kokkos::View<NonConstDataT****,layout_type,device_type>>(a);
664 else if (m_host_data->m_tag->dataLayout().rank() == 5)
665 m_view = std::any_cast<Kokkos::View<NonConstDataT*****,layout_type,device_type>>(a);
666 else if (m_host_data->m_tag->dataLayout().rank() == 6)
667 m_view = std::any_cast<Kokkos::View<NonConstDataT******,layout_type,device_type>>(a);
668 else if (m_host_data->m_tag->dataLayout().rank() == 7)
669 m_view = std::any_cast<Kokkos::View<NonConstDataT*******,layout_type,device_type>>(a);
670 else {
671 throw std::runtime_error("ERROR - PHX::MDField::setFieldData (DynRank) - Invalid rank!");
672 }
673 }
674 catch (std::exception& ) {
675 //std::string type_cast_name = Teuchos::demangleName(typeid(non_const_view).name());
676 std::string type_cast_name = "???";
677 std::cout << "\n\nError in runtime PHX::MDField::setFieldData() in std::any_cast. Tried to cast the field \""
678 << this->fieldTag().name() << "\" with the identifier \"" << this->fieldTag().identifier()
679 << "\" to a type of \"" << type_cast_name
680 << "\" from a std::any object containing a type of \""
681 << Teuchos::demangleName(a.type().name()) << "\"." << std::endl;
682 throw;
683 }
684 }

686 // Functor for mpl::for_each for printing dimensions
687 struct AddString {
688 std::vector<std::string>& names;
689 AddString(std::vector<std::string>& extent_names) : names(extent_names) {}
690 template<typename Extent> void operator()(Extent ) const
691 {names.push_back(PHX::print<Extent>());}

694 template<int R> void print(ViewSpecialization<R>, std::ostream& os, bool printValues) const
695 {
696 std::vector<std::string> dim_names;
697 using extent_vector = typename RankCount<Props...>::vector_type;
698 Sacado::mpl::for_each_no_kokkos<extent_vector>(AddString(dim_names));
699
700 os << "MDField<";
701
702 int count = 0;
703 for (auto i=dim_names.rbegin(); i < dim_names.rend(); ++i,++count) {
704 if (count > 0)
705 os << ",";
706 os << std::string(*i);
707 }
