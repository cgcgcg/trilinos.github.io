55 template <typename SrcType>
56 SACADO_INLINE_FUNCTION
57 static void assign_equal(DstType& dst, const SrcType& x)
58 {
59 const int xsz = x.size();
60
61 if (xsz != dst.size())
62 dst.resizeAndZero(xsz);
63
64 const int sz = dst.size();
65
66 // For ViewStorage, the resize above may not in fact resize the
67 // derivative array, so it is possible that sz != xsz at this point.
68 // The only valid use case here is sz > xsz == 0, so we use sz in the
69 // assignment below
70
71 if (sz) {
72 if (x.hasFastAccess()) {
73 SACADO_FAD_DERIV_LOOP(i,sz)
74 dst.fastAccessDx(i) = x.fastAccessDx(i);
75 }
76 else
77 SACADO_FAD_DERIV_LOOP(i,sz)
78 dst.fastAccessDx(i) = x.dx(i);
79 }
80
81 dst.val() = x.val();
82 }

85 template <typename SrcType>
86 SACADO_INLINE_FUNCTION
87 static void assign_plus_equal(DstType& dst, const SrcType& x)
88 {
89 const int xsz = x.size(), sz = dst.size();
90
91#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__ )
92 if ((xsz != sz) && (xsz != 0) && (sz != 0))
93 throw "Fad Error: Attempt to assign with incompatible sizes";
94#endif
95
96 if (xsz) {
97 if (sz) {
98 if (x.hasFastAccess())
99 SACADO_FAD_DERIV_LOOP(i,sz)
100 dst.fastAccessDx(i) += x.fastAccessDx(i);
101 else
102 for (int i=0; i<sz; ++i)
103 dst.fastAccessDx(i) += x.dx(i);
104 }
105 else {
106 dst.resizeAndZero(xsz);
107 if (x.hasFastAccess())
108 SACADO_FAD_DERIV_LOOP(i,xsz)
109 dst.fastAccessDx(i) = x.fastAccessDx(i);
110 else
111 SACADO_FAD_DERIV_LOOP(i,xsz)
112 dst.fastAccessDx(i) = x.dx(i);
113 }
114 }
115
116 dst.val() += x.val();
117 }

121 SACADO_INLINE_FUNCTION
122 static void assign_minus_equal(DstType& dst, const SrcType& x)
123 {
124 const int xsz = x.size(), sz = dst.size();
125
126#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__ )
127 if ((xsz != sz) && (xsz != 0) && (sz != 0))
128 throw "Fad Error: Attempt to assign with incompatible sizes";

155 template <typename SrcType>
156 SACADO_INLINE_FUNCTION
157 static void assign_times_equal(DstType& dst, const SrcType& x)
158 {
159 const int xsz = x.size(), sz = dst.size();
160 const value_type xval = x.val();
161 const value_type v = dst.val();
162
163#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__ )
164 if ((xsz != sz) && (xsz != 0) && (sz != 0))
165 throw "Fad Error: Attempt to assign with incompatible sizes";
166#endif
167
168 if (xsz) {
169 if (sz) {
170 if (x.hasFastAccess())
171 SACADO_FAD_DERIV_LOOP(i,sz)
172 dst.fastAccessDx(i) = v*x.fastAccessDx(i) + dst.fastAccessDx(i)*xval;
173 else
174 SACADO_FAD_DERIV_LOOP(i,sz)
175 dst.fastAccessDx(i) = v*x.dx(i) + dst.fastAccessDx(i)*xval;
176 }
177 else {
178 dst.resizeAndZero(xsz);
179 if (x.hasFastAccess())
180 SACADO_FAD_DERIV_LOOP(i,xsz)
181 dst.fastAccessDx(i) = v*x.fastAccessDx(i);
182 else
183 SACADO_FAD_DERIV_LOOP(i,xsz)
184 dst.fastAccessDx(i) = v*x.dx(i);
185 }
186 }
187 else {
188 if (sz) {
189 SACADO_FAD_DERIV_LOOP(i,sz)
190 dst.fastAccessDx(i) *= xval;
191 }
192 }
193
194 dst.val() *= xval;
195 }

198 template <typename SrcType>
199 SACADO_INLINE_FUNCTION
200 static void assign_divide_equal(DstType& dst, const SrcType& x)
201 {
202 const int xsz = x.size(), sz = dst.size();
203 const value_type xval = x.val();
204 const value_type v = dst.val();
205
206#if defined(SACADO_DEBUG) && !defined(__CUDA_ARCH__ ) && !defined(__HIP_DEVICE_COMPILE__ )
207 if ((xsz != sz) && (xsz != 0) && (sz != 0))
208 throw "Fad Error: Attempt to assign with incompatible sizes";
209#endif
210
211 if (xsz) {
212 const value_type xval2 = xval*xval;
213 if (sz) {
214 if (x.hasFastAccess())
215 SACADO_FAD_DERIV_LOOP(i,sz)
216 dst.fastAccessDx(i) =
217 ( dst.fastAccessDx(i)*xval - v*x.fastAccessDx(i) ) / xval2;
218 else
219 SACADO_FAD_DERIV_LOOP(i,sz)
220 dst.fastAccessDx(i) =
221 ( dst.fastAccessDx(i)*xval - v*x.dx(i) ) / xval2;
222 }
223 else {
224 dst.resizeAndZero(xsz);
225 if (x.hasFastAccess())
226 SACADO_FAD_DERIV_LOOP(i,xsz)
227 dst.fastAccessDx(i) = - v*x.fastAccessDx(i) / xval2;
228 else
229 SACADO_FAD_DERIV_LOOP(i,xsz)
230 dst.fastAccessDx(i) = -v*x.dx(i) / xval2;
231 }
232 }
233 else {
234 if (sz) {
235 SACADO_FAD_DERIV_LOOP(i,sz)
236 dst.fastAccessDx(i) /= xval;
237 }
238 }
239
240 dst.val() /= xval;
241 }
