77 KOKKOS_INLINE_FUNCTION void solve_var0(MemberType &member, const supernode_type &s, value_type *bptr) const {
78 using TrsvAlgoType = typename TrsvAlgorithm::type;
79 using GemvAlgoType = typename GemvAlgorithm::type;
80
81 const value_type minus_one(-1), one(1);
82 {
83 const ordinal_type m = s.m, n = s.n, n_m = n - m;
84 if (m > 0) {
85 value_type *aptr = s.u_buf;
86 // solve
87 const UnmanagedViewType<value_type_matrix> ATL(aptr, m, m);
88 aptr += m * m;
89 const ordinal_type offm = s.row_begin;
90 const auto tT = Kokkos::subview(_t, range_type(offm, offm + m), Kokkos::ALL());
91
92 ConstUnmanagedViewType<ordinal_type_array> P(_piv.data() + offm * 4, m * 4);
93 ConstUnmanagedViewType<value_type_matrix> D(_diag.data() + offm * 2, m, 2);
94
95 Scale2x2_BlockInverseDiagonals<Side::Left, Algo::Internal>::invoke(member, P, D, tT);
96
97 if (n_m > 0) {
98 // update
99 const UnmanagedViewType<value_type_matrix> ATR(aptr, m, n_m); // aptr += m*n;
100 const UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
101 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, minus_one, ATR, bB, one, tT);
102 member.team_barrier();
103 }
104 Trsv<Uplo::Lower, Trans::Transpose, TrsvAlgoType>::invoke(member, Diag::Unit(), ATL, tT);
105
106 if (!s.do_not_apply_pivots) {
107 ConstUnmanagedViewType<ordinal_type_array> fpiv(P.data() + m, m);
108 ApplyPivots<PivotMode::Flame, Side::Left, Direct::Backward, Algo::Internal>
109 ::invoke(member, fpiv, tT);
110 }

241 {

274 const ordinal_type p = _pbeg + member.league_rank();
275 const ordinal_type sid = _level_sids(p);

277 if (p < _pend && mode == 1) {
278 using solve_tag_type = SolveTag<Var>;
279
280 const supernode_type &s = _supernodes(sid);
281 value_type *bptr = _buf.data() + _buf_ptr(member.league_rank());
