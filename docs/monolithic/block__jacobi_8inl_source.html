44template <typename ValueType>
45struct block_jacobi_presmooth_functor
46{
47 ValueType omega;
48
49 block_jacobi_presmooth_functor(ValueType omega) : omega(omega) {}
50
51 __host__ __device__
52 ValueType operator()(const ValueType& b, const ValueType& d) const
53 {
54 return omega * b / d;
55 }
56};
57
58template <typename ValueType>
59struct block_jacobi_postsmooth_functor
60{
61 ValueType omega;
62
63 block_jacobi_postsmooth_functor(ValueType omega) : omega(omega) {}
64
65 template <typename Tuple>
66 __host__ __device__
67 ValueType operator()(const Tuple& t)
68 {
69 const ValueType x = thrust::get<0>(t);
70 const ValueType d = thrust::get<1>(t);
71 const ValueType b = thrust::get<2>(t);
72 const ValueType y = thrust::get<3>(t);
73
74 return x + omega * (b - y) / d;
75 }
76};

130template <typename ValueType, typename MemorySpace>
131template<typename MatrixType, typename VectorType1, typename VectorType2>
132 void block_jacobi<ValueType,MemorySpace>
133 ::operator()(const MatrixType& A, const VectorType1& b, VectorType2& x, ValueType omega)
134 {
135 CUSP_PROFILE_SCOPED();
136
137 // temp <- A*x
138 cusp::multiply(A, x, temp);
139
140 for (int i = 0; i < x.num_cols; i++){
141 // x <- x + omega * D^-1 * (b - temp)
142 thrust::transform(thrust::make_zip_iterator(thrust::make_tuple(x.column(i).begin(), diagonal.begin(), b.column(i).begin(), temp.column(i).begin())),
143 thrust::make_zip_iterator(thrust::make_tuple(x.column(i).end(), diagonal.end(), b.column(i).end(), temp.column(i).end())),
144 x.column(i).begin(),
145 detail::block_jacobi_postsmooth_functor<ValueType>(omega));
146
147 }
148 }
