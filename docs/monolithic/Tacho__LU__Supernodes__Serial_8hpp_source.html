51 template <typename MemberType, typename SupernodeInfoType>
52 KOKKOS_INLINE_FUNCTION static int
53 factorize(MemberType &member, const SupernodeInfoType &info, const typename SupernodeInfoType::ordinal_type_array &P,
54 const typename SupernodeInfoType::value_type_matrix &ABR, const ordinal_type sid) {
55 using supernode_info_type = SupernodeInfoType;
56 using value_type = typename supernode_info_type::value_type;
57 using value_type_matrix = typename supernode_info_type::value_type_matrix;
58 using range_type = typename supernode_info_type::range_type;
59
60 using LU_AlgoType = typename LU_Algorithm::type;
61 using TrsmAlgoType = typename TrsmAlgorithm::type;
62 using GemmAlgoType = typename GemmAlgorithm::type;
63
64 // get current supernode
65 const auto &s = info.supernodes(sid);
66
67 // panel (s.m x s.n) is divided into ATL (m x m) and ATR (m x n)
68 const ordinal_type m = s.m, n = s.n, n_m = s.n - s.m;
69
70 // m and n are available, then factorize the supernode block
71 if (m > 0) {
73 UnmanagedViewType<value_type_matrix> AT(s.u_buf, m, n);
74
75 LU<LU_AlgoType>::invoke(member, AT, P);
76 LU<LU_AlgoType>::modify(member, m, P);
77
78 if (n > 0) {
79 const value_type one(1), minus_one(-1), zero(0);
80
81 UnmanagedViewType<value_type_matrix> ATL(s.u_buf, m, m);
82 UnmanagedViewType<value_type_matrix> AL(s.l_buf, n, m);
83 const auto ABL = Kokkos::subview(AL, range_type(m, n), Kokkos::ALL());
84
85 Trsm<Side::Right, Uplo::Upper, Trans::NoTranspose, TrsmAlgoType>::invoke(member, Diag::NonUnit(), one, ATL,
86 ABL);
87
88 TACHO_TEST_FOR_ABORT(static_cast<ordinal_type>(ABR.extent(0)) != n_m ||
89 static_cast<ordinal_type>(ABR.extent(1)) != n_m,
90 "ABR dimension does not match to supernodes");
91
92 UnmanagedViewType<value_type_matrix> ATR(s.u_buf + ATL.span(), m, n_m);
93 Gemm<Trans::NoTranspose, Trans::NoTranspose, GemmAlgoType>::invoke(member, minus_one, ABL, ATR, zero, ABR);
94 }
95 }
96 return 0;
97 }

99 template <typename MemberType, typename SupernodeInfoType>
100 KOKKOS_INLINE_FUNCTION static int solve_lower(MemberType &member, const SupernodeInfoType &info,
101 const typename SupernodeInfoType::ordinal_type_array &P,
102 const typename SupernodeInfoType::value_type_matrix &xB,
103 const ordinal_type sid) {
104 using supernode_info_type = SupernodeInfoType;
105 using value_type = typename supernode_info_type::value_type;
106 using value_type_matrix = typename supernode_info_type::value_type_matrix;
107 using ordinal_type_array = typename supernode_info_type::ordinal_type_array;
108 using range_type = typename supernode_info_type::range_type;
109
110 const auto &s = info.supernodes(sid);
111
112 using TrsvAlgoType = typename TrsvAlgorithm::type;
113 using GemvAlgoType = typename GemvAlgorithm::type;
114
115 // panel is divided into diagonal and interface block
116 const ordinal_type m = s.m, n = s.n, n_m = s.n - s.m; //, nrhs = info.x.extent(1);
117
118 // m and n are available, then factorize the supernode block
119 if (m > 0) {
120 const value_type minus_one(-1), zero(0);
121 const ordinal_type offm = s.row_begin;
122 UnmanagedViewType<value_type_matrix> ATL(s.u_buf, m, m);
123 const auto xT = Kokkos::subview(info.x, range_type(offm, offm + m), Kokkos::ALL());
124 const ConstUnmanagedViewType<ordinal_type_array> fpiv(P.data() + m, m);
125
126 ApplyPivots<PivotMode::Flame, Side::Left, Direct::Forward, Algo::Internal>
127 ::invoke(member, fpiv, xT);
128
129 Trsv<Uplo::Lower, Trans::NoTranspose, TrsvAlgoType>::invoke(member, Diag::Unit(), ATL, xT);
130
131 if (n_m > 0) {
132 UnmanagedViewType<value_type_matrix> AL(s.l_buf, n, m);
133 const auto ABL = Kokkos::subview(AL, range_type(m, n), Kokkos::ALL());
134 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, minus_one, ABL, xT, zero, xB);
135 }
136 }
137 return 0;
138 }

176 template <typename MemberType, typename SupernodeInfoType>
177 KOKKOS_INLINE_FUNCTION static int
178 factorize_recursive_serial(MemberType &member, const SupernodeInfoType &info, const ordinal_type sid,
179 const bool final, typename SupernodeInfoType::ordinal_type_array::pointer_type piv,
180 typename SupernodeInfoType::value_type_array::pointer_type buf, const size_type bufsize) {
181 using supernode_info_type = SupernodeInfoType;
182 using value_type = typename supernode_info_type::value_type;
183 using value_type_matrix = typename supernode_info_type::value_type_matrix;
184 using ordinal_type_array = typename supernode_info_type::ordinal_type_array;
185
186 const auto &s = info.supernodes(sid);
187 if (final) {
188 // serial recursion
189 for (ordinal_type i = 0; i < s.nchildren; ++i)
190 factorize_recursive_serial(member, info, s.children[i], final, piv, buf, bufsize);
191 }
192
193 {
194 const ordinal_type m = s.m;
195 const ordinal_type rbeg = s.row_begin;
196 UnmanagedViewType<ordinal_type_array> ipiv(piv + rbeg * 4, 4 * m);
197
198 const ordinal_type n = s.n - s.m;
199 const size_type bufsize_required = n * n * sizeof(value_type);
200 TACHO_TEST_FOR_ABORT(bufsize < bufsize_required, "bufsize is smaller than required");
201 value_type *bufptr = buf;
202 UnmanagedViewType<value_type_matrix> ABR(bufptr, n, n);
203
204 LU_Supernodes<Algo::Workflow::Serial>::factorize(member, info, ipiv, ABR, sid);
205
206 constexpr bool update_lower = true;
207 CholSupernodes<Algo::Workflow::Serial>::update(member, info, ABR, sid, bufsize - ABR.span() * sizeof(value_type),
208 (void *)((value_type *)buf + ABR.span()), update_lower);
209 }
210 return 0;
211 }
