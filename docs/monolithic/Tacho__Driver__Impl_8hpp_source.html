76template <typename VT, typename DT>
77void Driver<VT, DT>::setMatrixType(const int symmetric, // 0 - unsymmetric, 1 - structure sym, 2 - symmetric
78 const bool is_positive_definite) {
79 switch (symmetric) {
80 case 0: {
81 _method_setup = LU;
82 _method = LU;
83 break;
84 }
85 case 1: {
86 _method_setup = SymLU;
87 _method = SymLU;
88 break;
89 }
90 case 2: {
91 if (is_positive_definite) {
92 if (std::is_same<value_type, double>::value || std::is_same<value_type, float>::value ||
93 std::is_same<value_type, Kokkos::complex<float>>::value ||
94 std::is_same<value_type, Kokkos::complex<double>>::value) {
95 // real symmetric posdef
96 _method_setup = Cholesky;
97 _method = Cholesky;
98 }
99 } else { // real or complex symmetric indef
100 _method_setup = LDL;
101 _method = LDL;
102 }
103 break;
104 }
105 default: {
106 TACHO_TEST_FOR_EXCEPTION(true, std::logic_error, "symmetric argument is wrong");
107 }
108 }
109}

245// internal only
246template <typename VT, typename DT> int Driver<VT, DT>::analyze() {
247 int r_val(0);
248 if (_m <= _small_problem_thres) {
250 if (_verbose) {
251 printf("TachoSolver: Analyze (Small Problem)\n");
252 printf("====================================\n");
253 printf(" Linear system A\n");
254 printf(" number of equations: %10d\n", _m);
255 printf("\n");
256 printf(" A is a small problem ( < %d ) and LAPACK is used\n", _small_problem_thres);
257 printf("\n");
258 }
259 } else {
260 const bool use_condensed_graph = (_m_graph > 0 && _m_graph < _m);
261 if (use_condensed_graph) {
262 Graph graph(_m_graph, _nnz_graph, _h_ap_graph, _h_aj_graph);
263 graph_tools_type G(graph);
264#if defined(TACHO_HAVE_METIS)
265 if (_order_connected_graph_separately) {
266 idx_t one_i = 1;
267 G.setOption(METIS_OPTION_CCORDER, one_i);
268 }
269#endif
270 if (_graph_algo_type >= 0) {
271 G.setAlgorithm(_graph_algo_type);
272 }
273 G.reorder(_verbose);
274
275 _h_perm_graph = G.PermVector();
276 _h_peri_graph = G.InvPermVector();
277
278 r_val = analyze_condensed_graph();
279 } else {
280 const bool use_graph_partitioner = (_h_perm.extent(0) == 0 && _h_peri.extent(0) == 0);
281 if (use_graph_partitioner) {
282 Graph graph(_m, _nnz, _h_ap, _h_aj);
283 graph_tools_type G(graph);
284#if defined(TACHO_HAVE_METIS)
285 if (_order_connected_graph_separately) {
286 idx_t one_i = 1;
287 G.setOption(METIS_OPTION_CCORDER, one_i);
288 }
289#endif
290 if (_graph_algo_type >= 0) {
291 G.setAlgorithm(_graph_algo_type);
292 }
293 G.reorder(_verbose);
294
295 _h_perm = G.PermVector();
296 _h_peri = G.InvPermVector();
297
298 r_val = analyze_linear_system();
299 } else {
300 r_val = analyze_linear_system();
301 }
302 }
303 }
304 return r_val;
305}

307template <typename VT, typename DT> int Driver<VT, DT>::analyze_linear_system() {
308 if (_verbose) {
309 printf("TachoSolver: Analyze Linear System\n");
310 printf("==================================\n");
311 }
312
313 {
314 symbolic_tools_type S(_m, _h_ap, _h_aj, _h_perm, _h_peri);
315 S.symbolicFactorize(_verbose);
316
317 _nnz_u = S.NumNonzerosU();
318 _nsupernodes = S.NumSupernodes();
319 _stree_level = S.SupernodesTreeLevel();
320 _stree_roots = S.SupernodesTreeRoots();
321
322 _supernodes = Kokkos::create_mirror_view(exec_memory_space(), S.Supernodes());
323 _gid_super_panel_ptr = Kokkos::create_mirror_view(exec_memory_space(), S.gidSuperPanelPtr());
324 _gid_super_panel_colidx = Kokkos::create_mirror_view(exec_memory_space(), S.gidSuperPanelColIdx());
325 _sid_super_panel_ptr = Kokkos::create_mirror_view(exec_memory_space(), S.sidSuperPanelPtr());
326 _sid_super_panel_colidx = Kokkos::create_mirror_view(exec_memory_space(), S.sidSuperPanelColIdx());
327 _blk_super_panel_colidx = Kokkos::create_mirror_view(exec_memory_space(), S.blkSuperPanelColIdx());
328 _stree_parent = Kokkos::create_mirror_view(exec_memory_space(), S.SupernodesTreeParent());
329 _stree_ptr = Kokkos::create_mirror_view(exec_memory_space(), S.SupernodesTreePtr());
330 _stree_children = Kokkos::create_mirror_view(exec_memory_space(), S.SupernodesTreeChildren());
331
332 Kokkos::deep_copy(_supernodes, S.Supernodes());
333 Kokkos::deep_copy(_gid_super_panel_ptr, S.gidSuperPanelPtr());
334 Kokkos::deep_copy(_gid_super_panel_colidx, S.gidSuperPanelColIdx());
335 Kokkos::deep_copy(_sid_super_panel_ptr, S.sidSuperPanelPtr());
336 Kokkos::deep_copy(_sid_super_panel_colidx, S.sidSuperPanelColIdx());
337 Kokkos::deep_copy(_blk_super_panel_colidx, S.blkSuperPanelColIdx());
338 Kokkos::deep_copy(_stree_parent, S.SupernodesTreeParent());
339 Kokkos::deep_copy(_stree_ptr, S.SupernodesTreePtr());
340 Kokkos::deep_copy(_stree_children, S.SupernodesTreeChildren());
341
342 // perm and peri is updated during symbolic factorization
343 _perm = Kokkos::create_mirror_view(exec_memory_space(), _h_perm);
344 _peri = Kokkos::create_mirror_view(exec_memory_space(), _h_peri);
345
346 Kokkos::deep_copy(_perm, _h_perm);
347 Kokkos::deep_copy(_peri, _h_peri);
348 }
349 return 0;
350}

399 if (_verbose) {
400 printf("TachoSolver: Initialize(method = %d)\n",_method_setup);
401 printf("====================================\n");
402 }
403 if (_method != _method_setup) {
404 // Reset method
405 setFactorizationMethod(_method_setup);
406 }
407
411 if (_m <= _small_problem_thres) {
413 } else {
417 NumericToolsFactory<VT, DT> factory;

424
425 factory.createObject(_N);
426 }
427 return 0;

437 }
438 if (_verbose) {

440 case LDL_nopiv: {
441 printf("TachoSolver: Factorize LDL (no pivot)\n");
442 printf("=====================================\n");
443 break;
444 }
445 case Cholesky: {
446 printf("TachoSolver: Factorize Cholesky\n");
447 printf("===============================\n");
448 break;
449 }
450 case LDL: {
451 printf("TachoSolver: Factorize LDL\n");
452 printf("==========================\n");
453 break;
454 }
455 case SymLU: {
456 printf("TachoSolver: Factorize SymLU\n");
457 printf("============================\n");
458 break;

474template <typename VT, typename DT> int Driver<VT, DT>::factorize_small_host(const value_type_array &ax) {
475 double t_copy(0), t_factor(0);
476 {
477 Kokkos::Timer timer;
478
479 timer.reset();
480 _A = value_type_matrix_host("A", _m, _m);
481 auto h_ax = Kokkos::create_mirror_view_and_copy(host_memory_space(), ax);
482 for (ordinal_type i = 0; i < _m; ++i) {
483 const size_type jbeg = _h_ap(i), jend = _h_ap(i + 1);
484 for (size_type j = jbeg; j < jend; ++j) {
485 const ordinal_type col = _h_aj(j);
486 const bool flag = ((_method == LDL_nopiv && i <= col) ||
487 (_method == Cholesky && i <= col) ||
488 (_method == LDL && i >= col) ||
489 (_method == SymLU));
490 if (flag)
491 _A(i, col) = h_ax(j);
492 }
493 }
494 t_copy = timer.seconds();
495
496 timer.reset();
497 switch (_method) {
498 case LDL_nopiv:
499 case Cholesky : {
500 Tacho::Chol<Uplo::Upper, Algo::External>::invoke(_A);
501 break;
502 }
503 case LDL: {
504 _P = ordinal_type_array_host("P", 4 * _m);
505 _D = value_type_matrix_host("D", _m, 2);
506 auto W = value_type_array_host("W", 32 * _m);
507 Tacho::LDL<Uplo::Lower, Algo::External>::invoke(_A, _P, W);
508 Tacho::LDL<Uplo::Lower, Algo::External>::modify(_A, _P, _D);
509 break;
510 }
511 case SymLU: {
512 _P = ordinal_type_array_host("P", 4 * _m);
513 Tacho::LU<Algo::External>::invoke(_A, _P);
514 Tacho::LU<Algo::External>::modify(_m, _P);
515 break;
516 }
517 default: {
518 std::stringstream ss;
519 ss << "Error: the solution method (" << _method << ") is not supported, 0 - LDL no-pivot, 1 - Chol, 2 - LDL, 3 - SymLU";
520 TACHO_TEST_FOR_EXCEPTION(true, std::logic_error, ss.str().c_str());
521 }
522 }
523 t_factor = timer.seconds();
524 }
525
526 if (_verbose) {
527 switch (_method) {
528 case Cholesky: {
529 printf("TachoSolver: Factorize Cholesky (SmallDenseFactorization)\n");
530 printf("=========================================================\n");
531 break;
532 }
533 case LDL: {
534 printf("TachoSolver: Factorize LDL (SmallDenseFactorization)\n");
535 printf("====================================================\n");
536 break;
537 }
538 case SymLU: {
539 printf("TachoSolver: Factorize SymLU (SmallDenseFactorization)\n");
540 printf("======================================================\n");
541 break;
542 }
543 }
544 printf(" Time\n");
545 printf(" time for copying A into supernodes: %10.6f s\n", t_copy);
546 printf(" time for numeric factorization: %10.6f s\n", t_factor);
547 printf(" total time spent: %10.6f s\n", (t_copy + t_factor));
548 printf("\n");
549 }
550
551 return 0;
552}

554template <typename VT, typename DT>
555int Driver<VT, DT>::solve(const value_type_matrix &x, const value_type_matrix &b, const value_type_matrix &t) {
556 if (_verbose) {
557 switch (_method) {
558 case LDL_nopiv: {
559 printf("TachoSolver: Solve LDL (no pivot)\n");
560 printf("=================================\n");
561 break;
562 }
563 case Cholesky: {
564 printf("TachoSolver: Solve Cholesky\n");
565 printf("===========================\n");
566 break;
567 }
568 case LDL: {
569 printf("TachoSolver: Solve LDL\n");
570 printf("======================\n");
571 break;
572 }
573 case SymLU: {
574 printf("TachoSolver: Solve SymLU\n");
575 printf("========================\n");
576 break;
577 }
578 }
579 }
580
581 if (_m <= _small_problem_thres) {
582 solve_small_host(x, b, t);
583 } else {
584 TACHO_TEST_FOR_EXCEPTION(t.extent(0) < x.extent(0) || t.extent(1) < x.extent(1), std::logic_error,
585 "Temporary rhs vector t is smaller than x");
586 auto tt = Kokkos::subview(t, Kokkos::pair<ordinal_type, ordinal_type>(0, x.extent(0)),
587 Kokkos::pair<ordinal_type, ordinal_type>(0, x.extent(1)));
588 _N->solve(x, b, tt, _verbose);
589 }
590 return 0;
591}

593template <typename VT, typename DT>
594int Driver<VT, DT>::solve_small_host(const value_type_matrix &x, const value_type_matrix &b,
595 const value_type_matrix &t) {
596 Kokkos::Timer timer;
597 double t_copy(0), t_solve(0);
598 {
599 timer.reset();
600 Kokkos::deep_copy(x, b);
601 t_copy = timer.seconds();
602
603 timer.reset();
604 switch (_method) {
605 case LDL_nopiv:
606 case Cholesky : {
607 auto h_x = Kokkos::create_mirror_view_and_copy(host_memory_space(), x);
608 Trsm<Side::Left, Uplo::Upper, Trans::ConjTranspose, Algo::External>::invoke(Diag::NonUnit(), 1.0, _A, h_x);
609 Trsm<Side::Left, Uplo::Upper, Trans::NoTranspose, Algo::External>::invoke(Diag::NonUnit(), 1.0, _A, h_x);
610 Kokkos::deep_copy(x, h_x);
611 break;
612 }
613 case LDL: {
614 auto perm = ordinal_type_array_host(_P.data() + 2 * _m, _m);
615 auto peri = ordinal_type_array_host(_P.data() + 3 * _m, _m);
616 auto h_x = Kokkos::create_mirror_view_and_copy(host_memory_space(), x);
617 auto h_t = Kokkos::create_mirror_view(host_memory_space(), t);
618
619 ApplyPermutation<Side::Left, Trans::NoTranspose, Algo::Internal>::invoke(h_x, perm, h_t);

635 Kokkos::deep_copy(x, h_x);
636 break;
637 }
638 }
639 t_solve = timer.seconds();
640 }

650 }

671template <typename VT, typename DT> int Driver<VT, DT>::exportFactorsToCrsMatrix(crs_matrix_type &A) {
672 if (_m <= _small_problem_thres) {
673 typedef ArithTraits<value_type> ats;
674 const typename ats::mag_type zero(0);
675
677 const ordinal_type m = _m;
678 size_type_array_host h_ap("h_ap", m + 1);
679 for (ordinal_type i = 0; i < m; ++i)
680 for (ordinal_type j = 0; j < m; ++j)
681 h_ap(i + 1) += (ats::abs(_A(i, j)) > zero);
682
684 h_ap(0) = 0;
685 for (ordinal_type i = 0; i < m; ++i)
686 h_ap(i + 1) += h_ap(i);
687
689 const ordinal_type nnz = h_ap(m);
690 ordinal_type_array_host h_aj(do_not_initialize_tag("h_aj"), nnz);
691 value_type_array_host h_ax(do_not_initialize_tag("h_ax"), nnz);
692
693 for (ordinal_type i = 0, k = 0; i < m; ++i)
694 for (ordinal_type j = i; j < m; ++j)
695 if (ats::abs(_A(i, j)) > zero) {
696 h_aj(k) = j;
697 h_ax(k) = _A(i, j);
698 ++k;
699 }
700
701 crs_matrix_type_host h_A;
702 h_A.setExternalMatrix(m, m, nnz, h_ap, h_aj, h_ax);
704 A.clear();
705 A.createConfTo(h_A);
706 A.copy(h_A);
707 } else {
708 _N->exportFactorsToCrsMatrix(A, false);
709 }
710 return 0;
