155 template <typename Ordinal, typename PCEType>
156 class PCESerializationTraitsImp {
157 typedef typename Sacado::ValueType<PCEType>::type ValueT;
158 typedef Teuchos::SerializationTraits<Ordinal,ValueT> vSerT;
159 typedef Teuchos::SerializationTraits<Ordinal,int> iSerT;
160 typedef Teuchos::SerializationTraits<Ordinal,Ordinal> oSerT;
161
162 public:
163
165 static const bool supportsDirectSerialization = false;
166
168
169
171 static Ordinal fromCountToIndirectBytes(const Ordinal count,
172 const PCEType buffer[]) {
173 Ordinal bytes = 0;
174 for (Ordinal i=0; i<count; i++) {
175 int sz = buffer[i].size();
176 Ordinal b1 = iSerT::fromCountToIndirectBytes(1, &sz);
177 Ordinal b2 = vSerT::fromCountToIndirectBytes(sz, buffer[i].coeff());
178 Ordinal b3 = oSerT::fromCountToIndirectBytes(1, &b2);
179 bytes += b1+b2+b3;
180 }
181 return bytes;
182 }
183
185 static void serialize (const Ordinal count,
186 const PCEType buffer[],
187 const Ordinal bytes,
188 char charBuffer[]) {
189 for (Ordinal i=0; i<count; i++) {
190 // First serialize size
191 int sz = buffer[i].size();
192 Ordinal b1 = iSerT::fromCountToIndirectBytes(1, &sz);
193 iSerT::serialize(1, &sz, b1, charBuffer);
194 charBuffer += b1;
195
196 // Next serialize PCE coefficients
197 Ordinal b2 = vSerT::fromCountToIndirectBytes(sz, buffer[i].coeff());
198 Ordinal b3 = oSerT::fromCountToIndirectBytes(1, &b2);
199 oSerT::serialize(1, &b2, b3, charBuffer);
200 charBuffer += b3;
201 vSerT::serialize(sz, buffer[i].coeff(), b2, charBuffer);
202 charBuffer += b2;
203 }
204 }
205
207 static Ordinal fromIndirectBytesToCount(const Ordinal bytes,
208 const char charBuffer[]) {
209 Ordinal count = 0;
210 Ordinal bytes_used = 0;
211 while (bytes_used < bytes) {
212
213 // Bytes for size
214 Ordinal b1 = iSerT::fromCountToDirectBytes(1);
215 bytes_used += b1;
216 charBuffer += b1;
217
218 // Bytes for PCE coefficients
219 Ordinal b3 = oSerT::fromCountToDirectBytes(1);
220 const Ordinal *b2 = oSerT::convertFromCharPtr(charBuffer);
221 bytes_used += b3;
222 charBuffer += b3;
223 bytes_used += *b2;
224 charBuffer += *b2;
225
226 ++count;
227 }
228 return count;
229 }
230
232 static void deserialize (const Ordinal bytes,
233 const char charBuffer[],
234 const Ordinal count,
235 PCEType buffer[]) {
236 for (Ordinal i=0; i<count; i++) {
237
238 // Deserialize size
239 Ordinal b1 = iSerT::fromCountToDirectBytes(1);
240 const int *sz = iSerT::convertFromCharPtr(charBuffer);
241 charBuffer += b1;
242
243 // Make sure PCE object is ready to receive values
244 // We assume it has already been initialized with the proper
245 // cijk object
246 if (buffer[i].size() != *sz)
247 buffer[i].reset(buffer[i].cijk(), *sz);
248 buffer[i].copyForWrite();
249
250 // Deserialize PCE coefficients
251 Ordinal b3 = oSerT::fromCountToDirectBytes(1);
252 const Ordinal *b2 = oSerT::convertFromCharPtr(charBuffer);
253 charBuffer += b3;
254 vSerT::deserialize(*b2, charBuffer, *sz, buffer[i].coeff());
255 charBuffer += *b2;
256 }
257
258 }

307 Ordinal fromCountToIndirectBytes(const Ordinal count,
308 const PCEType buffer[]) const {
309 Ordinal bytes = 0;
310 PCEType *x = NULL;

312 for (Ordinal i=0; i<count; i++) {
313 int my_sz = buffer[i].size();
314 if (sz != my_sz) {
315 if (x == NULL)
316 x = new PCEType;
317 *x = buffer[i];
318 x->reset(cijk);
319 cx = x;
320 }
321 else
322 cx = &(buffer[i]);
323 Ordinal b1 = iSerT::fromCountToIndirectBytes(1, &sz);
324 Ordinal b2 = vs->fromCountToIndirectBytes(sz, cx->coeff());
325 Ordinal b3 = oSerT::fromCountToIndirectBytes(1, &b2);
326 bytes += b1+b2+b3;
327 }
328 if (x != NULL)
329 delete x;
330 return bytes;
