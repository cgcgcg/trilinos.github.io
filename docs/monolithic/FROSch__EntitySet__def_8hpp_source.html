84 template<class SC,class LO,class GO,class NO>
85 int EntitySet<SC,LO,GO,NO>::buildEntityMap(ConstXMapPtr localToGlobalNodesMap)
86 {
87 if (!EntityMapIsUpToDate_) {
88 LO localNumberEntities = getNumEntities();
89 LO globalNumberEntities = 0; // AH 10/13/2017: Can we stick with LO here
90 LO maxLocalNumberEntities = 0;
91 reduceAll(*localToGlobalNodesMap->getComm(),REDUCE_SUM,localNumberEntities,ptr(&globalNumberEntities));
92 reduceAll(*localToGlobalNodesMap->getComm(),REDUCE_MAX,localNumberEntities,ptr(&maxLocalNumberEntities));
93
94 GOVec localToGlobalVector(0);
95 const GO INVALID = Teuchos::OrdinalTraits<GO>::invalid();
96 if (globalNumberEntities>0) {
97 // Set the Unique iD
98 setUniqueIDToFirstGlobalNodeID();
99
100 GOVec entities(maxLocalNumberEntities);
101 for (UN i=0; i<getNumEntities(); i++) {
102 entities[i] = getEntity(i)->getUniqueID()+1;
103 getEntity(i)->setLocalID(i);
104 }
105 XMapPtr entityMapping = MapFactory<LO,GO,NO>::Build(localToGlobalNodesMap->lib(),INVALID,entities(),0,localToGlobalNodesMap->getComm());
106
107 GOVec allEntities(maxLocalNumberEntities*localToGlobalNodesMap->getComm()->getSize(),0);
108 //localToGlobalNodesMap->getComm().GatherAll(&(entities->at(0)),&(allEntities->at(0)),maxLocalNumberEntities);
109 gatherAll(*localToGlobalNodesMap->getComm(),maxLocalNumberEntities,entities.getRawPtr(),maxLocalNumberEntities*localToGlobalNodesMap->getComm()->getSize(),allEntities.getRawPtr());
110
111 allEntities.push_back(0); // Um sicherzugehen, dass der erste Eintrag nach sort_unique eine 0 ist.
112
113 sortunique(allEntities);
114
115 localToGlobalVector.resize(localNumberEntities);
116 int LocalID;
117 for (UN i=1; i<allEntities.size(); i++) { // Wir fangen bei 1 an, weil wir am Anfang 1 auf die ID addiert haben
118 LocalID = entityMapping->getLocalElement(allEntities[i]);
119 if ( LocalID != -1) {
120 localToGlobalVector[LocalID] = i-1;
121 }
122 }
123
124 }
125 EntityMap_ = MapFactory<LO,GO,NO>::Build(localToGlobalNodesMap->lib(),INVALID,localToGlobalVector(),0,localToGlobalNodesMap->getComm());
126 EntityMapIsUpToDate_ = true;
127 }
128 return 0;
129 }

141 typename EntitySet<SC,LO,GO,NO>::EntitySetPtr EntitySet<SC,LO,GO,NO>::findRoots()
142 {
143 EntitySetPtr Roots(new EntitySet<SC,LO,GO,NO>(DefaultType));
144 for (UN i=0; i<getNumEntities(); i++) {

153 }
154 }

258 template<class SC,class LO,class GO,class NO>
259 int EntitySet<SC,LO,GO,NO>::flagStraightEntities(UN dimension,
260 ConstXMultiVectorPtr &nodeList)
261 {
262 FROSCH_ASSERT(dimension==nodeList->getNumVectors(),"FROSch::EntitySet: Inconsistent Dimension.");
263
264 bool straight;
265 LO length,j;
266 SCVec pt1(dimension);
267 SCVec dir1(dimension);
268 SCVec dir2(dimension);
269
270 for (UN i=0; i<getNumEntities(); i++) {
271 straight = true;
272 length = getEntity(i)->getNumNodes();
273
274 j=2;
275
276 if (length>2) {
277 // Anfangssteigung berechnen
278 for (UN k=0; k<dimension; k++) {
279 pt1[k] = nodeList->getData(k)[getEntity(i)->getLocalNodeID(0)];
280 }
281
282 for (UN k=0; k<dimension; k++) {
283 dir1[k] = nodeList->getData(k)[getEntity(i)->getLocalNodeID(1)]-pt1[k];
284 }
285
286 while (j<length) {
287 // Steigung zum zweiten Punkt berechnen
288 for (UN k=0; k<dimension; k++) {
289 dir2[k] = nodeList->getData(k)[getEntity(i)->getLocalNodeID(j)]-pt1[k];
290 }
291
292 if (!ismultiple<SC,LO>(dir1(),dir2())) {
293 straight = false;
294 break;
295 }
296 j++;
297 }
298 if (straight) {
299 getEntity(i)->resetEntityFlag(StraightFlag);
300 }
301 }
302 }
303 return 0;
304 }

331 template<class SC,class LO,class GO,class NO>
332 int EntitySet<SC,LO,GO,NO>::removeNodesWithDofs(GOVecView dirichletBoundaryDofs)
333 {
334 UN dofsPerNode = 0;
335 if (getNumEntities()>0) dofsPerNode = EntityVector_[0]->getDofsPerNode();
336 for (UN i=0; i<getNumEntities(); i++) {
337 UN length = getEntity(i)->getNumNodes();
338 for (UN j=0; j<length; j++) {
339 UN itmp = length-1-j;
340 UN k = 0;
341 while (k<dofsPerNode) {
342 GO dofGlobal = getEntity(i)->getGlobalDofID(itmp,k);
343 if (binary_search(dirichletBoundaryDofs.begin(),dirichletBoundaryDofs.end(),dofGlobal)) {
344 getEntity(i)->removeNode(itmp);
345 break;
346 }
347 k++;
348 }
349 }
350 }
351 return 0;

540 template<class SC,class LO,class GO,class NO>
541 const typename EntitySet<SC,LO,GO,NO>::SCVecPtr EntitySet<SC,LO,GO,NO>::getDirection(UN dimension,
542 ConstXMultiVectorPtr &nodeList,
543 UN iD) const
544 {
545 FROSCH_ASSERT(iD<getNumEntities(),"FROSch::EntitySet: Cannot access Entity because iD>=getNumEntities().");
546
547 if (getEntity(iD)->getEntityFlag()==StraightFlag) {
548
549 LO length = getEntity(iD)->getNumNodes();
550 LO j=2;
551
552 FROSCH_ASSERT(length>2,"Edge is not a straight edge!");
553
554 bool straight=true;
555
556 SCVec pt1(dimension);
557 SCVec dir2(dimension);
558 SCVecPtr dir1(dimension);
559
560 // Anfangssteigung berechnen
561 for (UN k=0; k<dimension; k++) {
562 pt1[k] = nodeList->getData(k)[getEntity(iD)->getLocalNodeID(0)];
563 }
564 for (UN k=0; k<dimension; k++) {
565 dir1[k] = nodeList->getData(k)[getEntity(iD)->getLocalNodeID(1)]-pt1[k];
566 }
567
568 while (j<length) {
569 for (UN k=0; k<dimension; k++) {
570 dir2[k] = nodeList->getData(k)[getEntity(iD)->getLocalNodeID(j)]-pt1[k];
571 }
572
573 if (!ismultiple<SC,LO>(dir1(),dir2())) {
574 straight = false;
575 break;
576 }
577 j++;
578 }
579
580 FROSCH_ASSERT(straight,"FROSch::EntitySet: Edge is not straight!");
581
582 return dir1;
583
584 } else if (getEntity(iD)->getEntityFlag()==ShortFlag) {
585
586 int length = getEntity(iD)->getNumNodes();
587
588 FROSCH_ASSERT(length==2,"FROSch::EntitySet: Edge is not a short edge!");
589
590 SCVec pt1(dimension);
591 SCVecPtr dir1(dimension);
592
593 // Anfangssteigung berechnen
594 for (UN k=0; k<dimension; k++) {
595 pt1[k] = nodeList->getData(k)[getEntity(iD)->getLocalNodeID(0)];
596 }
597 for (UN k=0; k<dimension; k++) {
598 dir1[k] = nodeList->getData(k)[getEntity(iD)->getLocalNodeID(1)]-pt1[k];
599 }
600
601 return dir1;
602
603 } else {
604 FROSCH_ASSERT(false,"FROSch::EntitySet: There is a problem while computing the direction of an edge!");
605 }
606 }
