78 KOKKOS_INLINE_FUNCTION void solve_chol_var0(MemberType &member, const supernode_type &s, value_type *bptr) const {
79 using GemvAlgoType = typename GemvAlgorithm_Team::type;
80 using TrsvAlgoType = typename TrsvAlgorithm_Team::type;
81
82 const value_type minus_one(-1), zero(0);
83 {
84 const ordinal_type m = s.m, n = s.n, n_m = n - m;
85 if (m > 0) {
86 value_type *aptr = s.u_buf;
87 // solve with current block
88 UnmanagedViewType<value_type_matrix> ATL(aptr, m, m);
89 aptr += m * m;
90
91 const ordinal_type offm = s.row_begin;
92 auto tT = Kokkos::subview(_t, range_type(offm, offm + m), Kokkos::ALL());
93 Trsv<Uplo::Upper, Trans::ConjTranspose, TrsvAlgoType>::invoke(member, Diag::NonUnit(), ATL, tT);
94
95 if (n_m > 0) {
96 // update remaining
97 member.team_barrier();
98 UnmanagedViewType<value_type_matrix> ATR(aptr, m, n_m); // aptr += m*n;
99 UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
100 Gemv<Trans::ConjTranspose, GemvAlgoType>::invoke(member, minus_one, ATR, tT, zero, bB);
101 }
102 }

141 KOKKOS_INLINE_FUNCTION void update_var0(MemberType &member, const supernode_type &s, value_type *bptr) const {
142 {
143 const ordinal_type m = s.m, n = s.n, n_m = n - m;
144 if (n_m > 0) {
145 UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
146 // update
147 const ordinal_type sbeg = s.sid_col_begin + 1, send = s.sid_col_end - 1;
148 for (ordinal_type i = sbeg, ip = 0 /*is=0*/; i < send; ++i) {
149 const ordinal_type tbeg = _sid_block_colidx(i).second, tend = _sid_block_colidx(i + 1).second,
150 tcnt = tend - tbeg;
151
152 Kokkos::parallel_for(
153 Kokkos::TeamVectorRange(member, tcnt),
154 [&, tbeg,
155 ip](const ordinal_type &ii) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
156 const ordinal_type it = tbeg + ii;
157 const ordinal_type is = ip + ii;
158 // for (ordinal_type it=tbeg;it<tend;++it,++is) {
159 const ordinal_type row = _gid_colidx(s.gid_col_begin + it);
160 for (ordinal_type j = 0; j < _nrhs; ++j)

392 const ordinal_type p = _pbeg + member.league_rank();
393 const ordinal_type sid = _level_sids(p);
394 const ordinal_type mode = _compute_mode(sid);
395 if (p < _pend && mode == 1) {
396 typedef SolveTag<Var> solve_tag_type;
397 const supernode_type &s = _supernodes(sid);
398 value_type *bptr = _buf.data() + _buf_ptr(member.league_rank());
399 if (_ldl) {
400 if (solve_tag_type::variant == 0)
401 solve_ldl_var0(member, s, bptr);
402 else if (solve_tag_type::variant == 1)
403 solve_ldl_var1(member, s, bptr);
404 else if (solve_tag_type::variant == 2 || solve_tag_type::variant == 3)
405 solve_ldl_var2(member, s, bptr);
406 else
407 Kokkos::printf("Error: TeamFunctorSolveLowerChol::SolveTag, algorithm variant is not supported\n");
408 } else {
