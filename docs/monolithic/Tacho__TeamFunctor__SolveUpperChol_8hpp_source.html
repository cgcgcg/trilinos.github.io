76 template <typename MemberType>
77 KOKKOS_INLINE_FUNCTION void solve_var0(MemberType &member, const supernode_type &s, value_type *bptr) const {
78 using GemvAlgoType = typename GemvAlgorithm_Team::type;
79 using TrsvAlgoType = typename TrsvAlgorithm_Team::type;
80
81 const value_type minus_one(-1), one(1);
82 {
83 const ordinal_type m = s.m, n = s.n, n_m = n - m;
84 if (m > 0) {
85 value_type *aptr = s.u_buf;
86 // solve
87 const UnmanagedViewType<value_type_matrix> ATL(aptr, m, m);
88 aptr += m * m;
89 const ordinal_type offm = s.row_begin;
90 const auto tT = Kokkos::subview(_t, range_type(offm, offm + m), Kokkos::ALL());
91
92 if (n_m > 0) {
93 // update
94 const UnmanagedViewType<value_type_matrix> ATR(aptr, m, n_m); // aptr += m*n;
95 const UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
96 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, minus_one, ATR, bB, one, tT);
97 member.team_barrier();
98 }
99 if (_ldl) {
100 Trsv<Uplo::Upper, Trans::NoTranspose, TrsvAlgoType>::invoke(member, Diag::Unit(), ATL, tT);
101 } else {
102 Trsv<Uplo::Upper, Trans::NoTranspose, TrsvAlgoType>::invoke(member, Diag::NonUnit(), ATL, tT);
103 }
104 }
105 }

134 using GemvAlgoType = typename GemvAlgorithm::type;
135
136 const value_type minus_one(-1), one(1), zero(0);
137 {
138 const ordinal_type m = s.m, n = s.n, n_m = n - m;
139 if (m > 0) {
140 value_type *aptr = s.u_buf;
141
142 const UnmanagedViewType<value_type_matrix> ATL(aptr, m, m);
143 aptr += m * m;
144 const UnmanagedViewType<value_type_matrix> bT(bptr, m, _nrhs);
145 bptr += m * _nrhs;
146
147 const ordinal_type offm = s.row_begin;
148 const auto tT = Kokkos::subview(_t, range_type(offm, offm + m), Kokkos::ALL());
149
150 if (n_m > 0) {
151 // update
152 const UnmanagedViewType<value_type_matrix> ATR(aptr, m, n_m);
153 const UnmanagedViewType<value_type_matrix> bB(bptr, n_m, _nrhs);
154
155 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, minus_one, ATR, bB, one, tT);
156 member.team_barrier();
157 }
158 // solve
159 if (_ldl) {
160 Trmv<Uplo::Upper, Trans::NoTranspose, GemvAlgoType>::invoke(member, Diag::Unit(), one, ATL, tT, zero, bT);
161 } else {
162 Gemv<Trans::NoTranspose, GemvAlgoType>::invoke(member, one, ATL, tT, zero, bT);
163 }
164 member.team_barrier();
165 // copy to t (TODO: trmv could be in place, to avoid this copy)
166 Kokkos::parallel_for(
167 Kokkos::TeamVectorRange(member, m * _nrhs),
168 [&, m](const ordinal_type &k) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
169 const ordinal_type i = k % m, j = k / m;
170 tT(i, j) = bT(i, j);
171 });

227 KOKKOS_INLINE_FUNCTION void update_var2(MemberType &member, const supernode_type &s, value_type *bptr) const {
228 {
229 const ordinal_type m = s.m, n = s.n;
230 UnmanagedViewType<value_type_matrix> b(bptr, n, _nrhs);
231 if (n > 0) {
232 const ordinal_type offm = s.row_begin;
233 const ordinal_type goffset = s.gid_col_begin + s.m;
234 Kokkos::parallel_for(
235 Kokkos::TeamVectorRange(member, n),
236 [&, m, goffset,
237 offm](const ordinal_type &i) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
238 for (ordinal_type j = 0; j < _nrhs; ++j) {
239 if (i < m) {
240 b(i, j) = _t(offm + i, j);
241 } else {
242 const ordinal_type row = _gid_colidx(i - m + goffset);
243 b(i, j) = _t(row, j);
244 }
245 }
246 });
247 }

261 const ordinal_type p = _pbeg + member.league_rank();
262 const ordinal_type sid = _level_sids(p);
263 const ordinal_type mode = _compute_mode(sid);
264 if (p < _pend && mode == 1) {
265 typedef SolveTag<Var> solve_tag_type;
266 const supernode_type &s = _supernodes(sid);
267 value_type *bptr = _buf.data() + _buf_ptr(member.league_rank());
268 if (solve_tag_type::variant == 0)
269 solve_var0(member, s, bptr);
270 else if (solve_tag_type::variant == 1)
271 solve_var1(member, s, bptr);
272 else if (solve_tag_type::variant == 2 || solve_tag_type::variant == 3)
273 solve_var2(member, s, bptr);
274 else
275 Kokkos::printf("Error: TeamFunctorSolveUpperChol::SolveTag, algorithm variant is not supported\n");
276 } else if (mode == -1) {
277 Kokkos::printf("Error: TeamFunctorSolveUpperChol::SolveTag, computing mode is not determined\n");
278 } else {
279 // skip
