63 template <class SC,class LO,class GO,class NO>
64 void MultiplicativeOperator<SC,LO,GO,NO>::apply(const XMultiVector &x,
65 XMultiVector &y,
66 bool usePreconditionerOnly,
67 ETransp mode,
68 SC alpha,
69 SC beta) const
70 {
71 FROSCH_TIMER_START_LEVELID(applyTime,"MultiplicativeOperator::apply");
72 FROSCH_ASSERT(usePreconditionerOnly,"MultiplicativeOperator can only be used as a preconditioner.");
73 FROSCH_ASSERT(this->OperatorVector_.size()==2,"Should be a Two-Level Operator.");
74
75
76 if (XTmp_.is_null()) XTmp_ = MultiVectorFactory<SC,LO,GO,NO>::Build(x.getMap(),x.getNumVectors());
77 *XTmp_ = x; // Need this for the case when x aliases y
78
79 if (YTmp_.is_null()) YTmp_ = MultiVectorFactory<SC,LO,GO,NO>::Build(y.getMap(),y.getNumVectors());
80 *YTmp_ = y; // for the second apply
81
82 this->OperatorVector_[0]->apply(*XTmp_,*YTmp_,true);
83
84 this->K_->apply(*YTmp_,*XTmp_);
85
86 this->OperatorVector_[1]->apply(*XTmp_,*XTmp_,true);
87
88 YTmp_->update(ScalarTraits<SC>::one(),*XTmp_,-ScalarTraits<SC>::one());
89 y.update(alpha,*YTmp_,beta);
90 }

165 ret -= 10;
166 }
167 //FROSCH_ASSERT(op->OperatorDomainMap().SameAs(OperatorVector_.at(0)->OperatorDomainMap()),"The DomainMaps of the operators are not identical.");
168 //FROSCH_ASSERT(op->OperatorRangeMap().SameAs(OperatorVector_.at(0)->OperatorRangeMap()),"The RangeMaps of the operators are not identical.");
169 }
170 OperatorVector_.push_back(op);
