246 template <typename MemberType>
247 KOKKOS_INLINE_FUNCTION void update(MemberType &member, const supernode_type &cur,
248 const value_type_matrix &ABR) const {
249
250 static constexpr bool runOnHost = run_tacho_on_host_v<typename value_type_matrix::execution_space>;
251
252 const auto info = _info;
253 value_type *buf = ABR.data() + ABR.span();
254 const ordinal_type sbeg = cur.sid_col_begin + 1, send = cur.sid_col_end - 1;
255
256 const ordinal_type srcbeg = info.sid_block_colidx(sbeg).second, srcend = info.sid_block_colidx(send).second,
257 srcsize = srcend - srcbeg;
258
259 // short cut to direct update
260 if ((send - sbeg) == 1) {
261 const auto &s = info.supernodes(info.sid_block_colidx(sbeg).first);
262 const ordinal_type tgtbeg = info.sid_block_colidx(s.sid_col_begin).second,
263 tgtend = info.sid_block_colidx(s.sid_col_end - 1).second, tgtsize = tgtend - tgtbeg;
264
265 if (srcsize == tgtsize) {
266 /* */ value_type *tgt = s.u_buf;
267 const value_type *src = (value_type *)ABR.data();
268
269 Kokkos::parallel_for(
270 Kokkos::TeamThreadRange(member, srcsize),
271 [&, srcsize, src,
272 tgt](const ordinal_type &j) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
273 const value_type *KOKKOS_RESTRICT ss = src + j * srcsize;
274 /* */ value_type *KOKKOS_RESTRICT tt = tgt + j * srcsize;
275 Kokkos::parallel_for(Kokkos::ThreadVectorRange(member, srcsize),
276 [&](const ordinal_type &i) { Kokkos::atomic_add(&tt[i], ss[i]); });
277 });
278 return;
279 }
280 }
281
282 const ordinal_type *s_colidx = sbeg < send ? &info.gid_colidx(cur.gid_col_begin + srcbeg) : NULL;
283
284 // loop over target
285 // const size_type s2tsize = srcsize*sizeof(ordinal_type)*member.team_size();
286 if constexpr(runOnHost) {
287 ordinal_type *s2t = (ordinal_type *)buf;
288 //const size_type s2tsize = srcsize * sizeof(ordinal_type);
289 //TACHO_TEST_FOR_ABORT(bufsize < s2tsize, "bufsize is smaller than required s2t workspace");
290
291 for (ordinal_type i = sbeg; i < send; ++i) {
292 const ordinal_type tgtsid = info.sid_block_colidx(i).first;
293 const auto &s = info.supernodes(tgtsid);
294 {
295 const ordinal_type tgtbeg = info.sid_block_colidx(s.sid_col_begin).second,
296 tgtend = info.sid_block_colidx(s.sid_col_end - 1).second, tgtsize = tgtend - tgtbeg;
297
298 const ordinal_type *t_colidx = &info.gid_colidx(s.gid_col_begin + tgtbeg);
299 for (ordinal_type k = 0, l = 0; k < srcsize; ++k) {
300 s2t[k] = -1;
301 for (; l < tgtsize && t_colidx[l] <= s_colidx[k]; ++l)
302 if (s_colidx[k] == t_colidx[l]) {
303 s2t[k] = l;
304 break;
305 }
306 }
307 }
308
309 {
310 UnmanagedViewType<value_type_matrix> U(s.u_buf, s.m, s.n);
311 UnmanagedViewType<value_type_matrix> Lp(s.l_buf, s.n, s.m);
312 const auto L = Kokkos::subview(Lp, range_type(s.m, s.n), Kokkos::ALL());
313
314 ordinal_type ijbeg = 0;
315 for (; s2t[ijbeg] == -1; ++ijbeg)
316 ;
317
318 // lock
319 while (Kokkos::atomic_compare_exchange(&s.lock, 0, 1))
320 TACHO_IMPL_PAUSE;
321 Kokkos::store_fence();
322
323 for (ordinal_type jj = ijbeg; jj < srcsize; ++jj) {
324 const ordinal_type col = s2t[jj];
325 for (ordinal_type ii = ijbeg; ii < srcsize; ++ii) {
326 const ordinal_type row = s2t[ii];
327 if (row < s.m) {
328 U(row, col) += ABR(ii, jj);
329 //Kokkos::atomic_add(&U(row, col), ABR(ii, jj));
330 if (col >= s.m) {
331 L(col - s.m, row) += ABR(jj, ii);
332 //Kokkos::atomic_add(&L(col - s.m, row), ABR(jj, ii));
333 }
334 } else
335 break;
336 }
337 }
338
339 // unlock
340 s.lock = 0;
341 Kokkos::load_fence();
342 }
343 }
344 } else {
345 Kokkos::parallel_for(
346 Kokkos::TeamThreadRange(member, sbeg, send),
347 [&, buf,
348 srcsize](const ordinal_type &i) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
349 ordinal_type *s2t = ((ordinal_type *)(buf)) + member.team_rank() * srcsize;
350 const auto &s = info.supernodes(info.sid_block_colidx(i).first);
351 {
352 const ordinal_type tgtbeg = info.sid_block_colidx(s.sid_col_begin).second,
353 tgtend = info.sid_block_colidx(s.sid_col_end - 1).second, tgtsize = tgtend - tgtbeg;
354
355 const ordinal_type *t_colidx = &info.gid_colidx(s.gid_col_begin + tgtbeg);
356 Kokkos::parallel_for(
357 Kokkos::ThreadVectorRange(member, srcsize),
358 [&, t_colidx, s_colidx, tgtsize](
359 const ordinal_type &k) { // Value capture is a workaround for cuda + gcc-7.2 compiler bug w/c++14
360 s2t[k] = -1;
361 auto found = lower_bound(&t_colidx[0], &t_colidx[tgtsize - 1], s_colidx[k],
362 [](ordinal_type left, ordinal_type right) { return left < right; });
363 if (s_colidx[k] == *found) {
364 s2t[k] = found - t_colidx;
365 }
366 });
367 }
368 {
369 UnmanagedViewType<value_type_matrix> U(s.u_buf, s.m, s.n);
370 UnmanagedViewType<value_type_matrix> Lp(s.l_buf, s.n, s.m);
371 const auto L = Kokkos::subview(Lp, range_type(s.m, s.n), Kokkos::ALL());
372
373 ordinal_type ijbeg = 0;
374 for (; s2t[ijbeg] == -1; ++ijbeg)
375 ;
376
377 Kokkos::parallel_for(Kokkos::ThreadVectorRange(member, ijbeg, srcsize), [&](const ordinal_type &ii) {
378 const ordinal_type row = s2t[ii];
379 if (row < s.m) {
380 for (ordinal_type jj = ijbeg; jj < srcsize; ++jj) {
381 const ordinal_type col = s2t[jj];
382 Kokkos::atomic_add(&U(row, col), ABR(ii, jj));
383 if (col >= s.m) {
384 Kokkos::atomic_add(&L(col - s.m, row), ABR(jj, ii));
385 }
386 }
387 }
388 });
389 }
390 });
391 }
392 return;
393 }

416 template <typename MemberType, int Var>
417 KOKKOS_INLINE_FUNCTION void operator()(const FactorizeTag<Var> &, const MemberType &member) const {
418 const ordinal_type lid = member.league_rank();
419 const ordinal_type p = _pbeg + lid;
420 const ordinal_type sid = _level_sids(p);
421 const ordinal_type mode = _compute_mode(sid);
422 if (p < _pend && mode == 1) {
423 using factorize_tag_type = FactorizeTag<Var>;
424
425 const auto &s = _info.supernodes(sid);
426 const ordinal_type m = s.m, n = s.n, n_m = n - m;
427 const ordinal_type offm = s.row_begin;
428
429 UnmanagedViewType<ordinal_type_array> P(_piv.data() + offm * 4, m * 4);
430
431 const auto bufptr = _buf.data() + _buf_ptr(lid);
432 if (factorize_tag_type::variant == 0) { // temporary to push to trilinos
433 UnmanagedViewType<value_type_matrix> ABR(bufptr, n_m, n_m);
434 factorize_var0(member, s, P, ABR);
435 } else if (factorize_tag_type::variant == 1) {
436 UnmanagedViewType<value_type_matrix> ABR(bufptr, n_m, n_m);
437 UnmanagedViewType<value_type_matrix> T(bufptr, m, m);
438 factorize_var1(member, s, P, T, ABR);
439 } else if (factorize_tag_type::variant == 2 || factorize_tag_type::variant == 3) {
440 UnmanagedViewType<value_type_matrix> ABR(bufptr, n_m, n_m);
441 UnmanagedViewType<value_type_matrix> T(bufptr + ABR.span(), m, m);
442 factorize_var2(member, s, P, T, ABR);
443 }
444 } else if (mode == -1) {
445 Kokkos::printf("Error: TeamFunctorFactorizeChol, computing mode is not determined\n");
446 } else {
447 // skip
448 }
