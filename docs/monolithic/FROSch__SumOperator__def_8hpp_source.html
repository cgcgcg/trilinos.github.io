78 // Y = alpha * A^mode * X + beta * Y
79 template <class SC,class LO,class GO,class NO>
80 void SumOperator<SC,LO,GO,NO>::apply(const XMultiVector &x,
81 XMultiVector &y,
82 bool usePreconditionerOnly,
83 ETransp mode,
84 SC alpha,
85 SC beta) const
86 {
87 FROSCH_TIMER_START_LEVELID(applyTime,"SumOperator::apply");
88 if (OperatorVector_.size()>0) {
89 if (XTmp_.is_null() || XTmp_->getNumVectors() != x.getNumVectors()) {
90 XTmp_ = MultiVectorFactory<SC,LO,GO,NO>::Build(x.getMap(),x.getNumVectors());
91 }
92 *XTmp_ = x; // Das brauche ich für den Fall das x=y
93 UN itmp = 0;
94 for (UN i=0; i<OperatorVector_.size(); i++) {
95 if (EnableOperators_[i]) {
96 OperatorVector_[i]->apply(*XTmp_,y,usePreconditionerOnly,mode,alpha,beta);
97 if (itmp==0) beta = ScalarTraits<SC>::one();
98 itmp++;
99 }
100 }
101 } else {
102 y.update(alpha,x,beta);
103 }
104 }

143 int ret = 0;
144 if (OperatorVector_.size()>0) {
145 if (!op->getDomainMap()->isSameAs(*OperatorVector_[0]->getDomainMap())) {

150 if (this->Verbose_) cerr << "SumOperator<SC,LO,GO,NO>::addOperator(SchwarzOperatorPtr op)\t\t!op->getRangeMap().isSameAs(OperatorVector_[0]->getRangeMap())\n";
151 ret -= 10;
152 }
153 //FROSCH_ASSERT(op->OperatorDomainMap().SameAs(OperatorVector_[0]->OperatorDomainMap()),"The DomainMaps of the operators are not identical.");
154 //FROSCH_ASSERT(op->OperatorRangeMap().SameAs(OperatorVector_[0]->OperatorRangeMap()),"The RangeMaps of the operators are not identical.");
155 }
156 OperatorVector_.push_back(op);
157 EnableOperators_.push_back(true);
