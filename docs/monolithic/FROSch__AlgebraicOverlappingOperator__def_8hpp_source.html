117 template <class SC,class LO,class GO,class NO>
118 int AlgebraicOverlappingOperator<SC,LO,GO,NO>::buildOverlappingMatrices(int overlap,
119 ConstXMapPtr repeatedMap)
120 {
121 FROSCH_DETAILTIMER_START_LEVELID(buildOverlappingMatricesTime,"AlgebraicOverlappingOperator::buildOverlappingMatrices");
122 // ====================================================================================
123 // AH 08/09/2019: This is just temporary. Implement this properly in all the classes
124 Verbosity verbosity = All;
125 if (!this->ParameterList_->get("Verbosity","All").compare("None")) {
126 verbosity = None;
127 } else if (!this->ParameterList_->get("Verbosity","All").compare("All")) {
128 verbosity = All;
129 } else {
130 FROSCH_ASSERT(false,"FROSch::AlgebraicOverlappingOperator: Specify a valid verbosity level.");
131 }
132 // ====================================================================================
133
134 this->OverlappingMap_ = repeatedMap;
135 this->OverlappingMatrix_ = this->K_;
136
137 GO global = 0, sum = 0;
138 LO local,minVal,maxVal;
139 SC avg;
140 if (verbosity==All) {
141 FROSCH_DETAILTIMER_START_LEVELID(printStatisticsTime,"print statistics");
142
143 global = this->OverlappingMap_->getMaxAllGlobalIndex();
144 if (this->OverlappingMap_->getGlobalNumElements()>0) {
145 global += 1;
146 }
147
148 local = (LO) max((LO) this->OverlappingMap_->getLocalNumElements(),(LO) 0);
149 reduceAll(*this->MpiComm_,REDUCE_SUM,GO(local),ptr(&sum));
150 avg = max(sum/double(this->MpiComm_->getSize()),0.0);
151 reduceAll(*this->MpiComm_,REDUCE_MIN,local,ptr(&minVal));
152 reduceAll(*this->MpiComm_,REDUCE_MAX,local,ptr(&maxVal));
153
154 if (this->Verbose_) {
155 cout
156 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
157 << setw(89) << "-----------------------------------------------------------------------------------------"
158 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
159 << "| "
160 << left << setw(74) << "> Overlapping Subdomains Statistics " << right << setw(8) << "(Level " << setw(2) << this->LevelID_ << ")"
161 << " |"
162 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
163 << setw(89) << "========================================================================================="
164 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
165 << "| " << left << setw(20) << " " << right
166 << " | " << setw(10) << "total"
167 << " | " << setw(10) << "avg"
168 << " | " << setw(10) << "min"
169 << " | " << setw(10) << "max"
170 << " | " << setw(10) << "global sum"
171 << " |"
172 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
173 << setw(89) << "-----------------------------------------------------------------------------------------"
174 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
175 << "| " << left << setw(20) << "Layer 0" << right
176 << " | " << setw(10) << global
177 << " | " << setw(10) << setprecision(5) << avg
178 << " | " << setw(10) << minVal
179 << " | " << setw(10) << maxVal
180 << " | " << setw(10) << sum
181 << " |";
182 }
183 }
184
185 // Adding Layers of Elements to the overlapping subdomains
186 ConstXCrsGraphPtr overlappingGraph = this->OverlappingMatrix_->getCrsGraph();
187 for (int i=0; i<overlap; i++) {
188 switch (AddingLayersStrategy_) {
189 case LayersFromGraph:
190 ExtendOverlapByOneLayer(overlappingGraph,this->OverlappingMap_,overlappingGraph,this->OverlappingMap_);
191 break;
192
193 case LayersFromMatrix:
194 ExtendOverlapByOneLayer(this->OverlappingMatrix_,this->OverlappingMap_,this->OverlappingMatrix_,this->OverlappingMap_);
195 break;
196
197 case LayersOld:
198 ExtendOverlapByOneLayer_Old(this->OverlappingMatrix_,this->OverlappingMap_,this->OverlappingMatrix_,this->OverlappingMap_);
199 break;
200
201 default:
202 FROSCH_ASSERT(false,"FROSch::AlgebraicOverlappingOperator: Specify a valid strategy for adding layers.");
203 }
204 if (verbosity==All) {
205 FROSCH_DETAILTIMER_START_LEVELID(printStatisticsTime,"print statistics");
206 local = (LO) max((LO) this->OverlappingMap_->getLocalNumElements(),(LO) 0);
207 reduceAll(*this->MpiComm_,REDUCE_SUM,GO(local),ptr(&sum));
208 avg = max(sum/double(this->MpiComm_->getSize()),0.0);
209 reduceAll(*this->MpiComm_,REDUCE_MIN,local,ptr(&minVal));
210 reduceAll(*this->MpiComm_,REDUCE_MAX,local,ptr(&maxVal));
211
212 if (this->Verbose_) {
213 cout
214 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
215 << "| " << left << "Layer " << setw(14) << i+1 << right
216 << " | " << setw(10) << global
217 << " | " << setw(10) << setprecision(5) << avg
218 << " | " << setw(10) << minVal
219 << " | " << setw(10) << maxVal
220 << " | " << setw(10) << sum
221 << " |";
222 }
223 }
224 }
225
226 if (verbosity==All) {
227 FROSCH_DETAILTIMER_START_LEVELID(printStatisticsTime,"print statistics");
228 if (this->Verbose_) {
229 cout
230 << "\n" << setw(FROSCH_OUTPUT_INDENT) << " "
231 << setw(89) << "-----------------------------------------------------------------------------------------"
232 << endl;
233 }
234 }
235
236 // AH 08/28/2019 TODO: It seems that ExtendOverlapByOneLayer_Old is currently the fastest method because the map is sorted. This seems to be better for the direct solver. (At least Klu)
237 if (this->ParameterList_->get("Sort Overlapping Map",true)) {
238 switch (AddingLayersStrategy_) {
239 case LayersFromGraph:
240 this->OverlappingMap_ = SortMapByGlobalIndex(this->OverlappingMap_);
241 break;
242
243 case LayersFromMatrix:
244 this->OverlappingMap_ = SortMapByGlobalIndex(this->OverlappingMap_);
245 break;
246
247 case LayersOld:
248 if (this->Verbose_) cout << "FROSch::AlgebraicOverlappingOperator : The overlapping map is already sorted" << endl;
249 break;
250
251 default:
252 FROSCH_ASSERT(false,"FROSch::AlgebraicOverlappingOperator: Specify a valid strategy for adding layers.");
253 }
254 }
255
256 return 0;
257 }

259 template <class SC,class LO,class GO,class NO>
260 int AlgebraicOverlappingOperator<SC,LO,GO,NO>::updateLocalOverlappingMatrices()
261 {
262 FROSCH_DETAILTIMER_START_LEVELID(updateLocalOverlappingMatricesTime,"AlgebraicOverlappingOperator::updateLocalOverlappingMatrices");
263 if (this->ExtractLocalSubdomainMatrix_Symbolic_Done_) {
264 // using original K_ as input
265 ExtractLocalSubdomainMatrix_Compute(this->subdomainScatter_, this->K_, this->subdomainMatrix_, this->localSubdomainMatrix_);
266 this->OverlappingMatrix_ = this->localSubdomainMatrix_.getConst();
267 } else {
268 if (this->IsComputed_) {
269 // already computed once and we want to recycle the information. That is why we reset OverlappingMatrix_ to K_, because K_ has been reset at this point
270 this->OverlappingMatrix_ = this->K_;
271 }
272 this->OverlappingMatrix_ = ExtractLocalSubdomainMatrix(this->OverlappingMatrix_, this->OverlappingMap_);
273 }
274 return 0;
275 }

286 template <class SC,class LO,class GO,class NO>
287 void AlgebraicOverlappingOperator<SC,LO,GO,NO>::extractLocalSubdomainMatrix_Symbolic()
288 {
289 if (this->OverlappingMap_->lib() == UseTpetra) {
290 FROSCH_DETAILTIMER_START_LEVELID(AlgebraicOverlappin_extractLocalSubdomainMatrix_SymbolicTime,"AlgebraicOverlappinOperator::extractLocalSubdomainMatrix_Symbolic");
291 // buid sudomain matrix
292 this->subdomainMatrix_ = MatrixFactory<SC,LO,GO,NO>::Build(this->OverlappingMap_, this->OverlappingMatrix_->getGlobalMaxNumRowEntries());
293 RCP<Import<LO,GO,NO> > scatter = ImportFactory<LO,GO,NO>::Build(this->OverlappingMatrix_->getRowMap(), this->OverlappingMap_);
294 this->subdomainMatrix_->doImport(*(this->OverlappingMatrix_), *scatter, ADD);
295
296 // Used to Map original K_ to overlapping suubdomainMatrix
297 this->subdomainScatter_ = ImportFactory<LO,GO,NO>::Build(this->K_->getRowMap(), this->OverlappingMap_);
298
299 // build local subdomain matrix
300 RCP<const Comm<LO> > SerialComm = rcp(new MpiComm<LO>(MPI_COMM_SELF));
301 RCP<Map<LO,GO,NO> > localSubdomainMap = MapFactory<LO,GO,NO>::Build(this->OverlappingMap_->lib(), this->OverlappingMap_->getLocalNumElements(), 0, SerialComm);
302 this->localSubdomainMatrix_ = MatrixFactory<SC,LO,GO,NO>::Build(localSubdomainMap, localSubdomainMap, this->OverlappingMatrix_->getGlobalMaxNumRowEntries());
303
304 // fill in column indexes
305 ExtractLocalSubdomainMatrix_Symbolic(this->subdomainMatrix_, // input
306 this->localSubdomainMatrix_); // output
307
308 // turn flag on
309 this->ExtractLocalSubdomainMatrix_Symbolic_Done_ = true;
310 }
