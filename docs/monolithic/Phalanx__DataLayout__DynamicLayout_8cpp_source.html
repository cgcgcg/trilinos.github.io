24bool PHX::Layout::operator==(const PHX::DataLayout& src) const
25{
26 const auto* src_cast = dynamic_cast<const PHX::Layout*>(&src);
27 if (src_cast == nullptr)
28 return false;
29
30 // Check identifier
31 if (m_identifier != src_cast->m_identifier)
32 return false;
33
34 // Check rank
35 if (m_extents.size() != src_cast->m_extents.size())
36 return false;
37
38 // Check dimensions
39 for (size_t i=0; i < m_extents.size(); ++i)
40 if (m_extents[i] != src_cast->m_extents[i])
41 return false;
42
43 return true;
44}
