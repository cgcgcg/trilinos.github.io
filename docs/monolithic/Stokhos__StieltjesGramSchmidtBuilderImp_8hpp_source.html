14template <typename ordinal_type, typename value_type>
15Stokhos::StieltjesGramSchmidtBuilder<ordinal_type,value_type>::
16StieltjesGramSchmidtBuilder(
17 const Teuchos::RCP<const Stokhos::Quadrature<ordinal_type, value_type> >& quad_,
18 const Teuchos::Array< Stokhos::OrthogPolyApprox<ordinal_type, value_type> >& pces,
19 ordinal_type new_order, bool use_pce_qp, bool normalize) :
20 quad(quad_)
21{
22 // Create array to store new coordinate bases
23 ordinal_type new_dim = pces.size();
24 Teuchos::Array< Teuchos::RCP<const OneDOrthogPolyBasis<ordinal_type, value_type> > > new_coordinate_bases(new_dim);
25
26 // Create Stieltjes basis for each pce
27 for (ordinal_type k=0; k<new_dim; k++) {
28 new_coordinate_bases[k] = Teuchos::rcp(
29 new StieltjesPCEBasis<ordinal_type,value_type>(
30 new_order, Teuchos::rcp(&(pces[k]),false), quad, use_pce_qp,
31 normalize)
32 );
33 }
34
35 // Create tensor product basis from coordinate bases
36 tensor_basis = Teuchos::rcp(
37 new CompletePolynomialBasis<ordinal_type,value_type>(new_coordinate_bases)
38 );
39
40 // Use Gram-Schmidt to orthogonalize tensor product bases
41 const Teuchos::Array<value_type>& weights = quad->getQuadWeights();
42 const Teuchos::Array< Teuchos::Array<value_type> >& points =
43 quad->getQuadPoints();
44 ordinal_type nqp = points.size();
45 Teuchos::RCP< Teuchos::Array< Teuchos::Array<value_type> > > new_points =
46 Teuchos::rcp(new Teuchos::Array< Teuchos::Array<value_type> >(nqp));
47 Teuchos::RCP< Teuchos::Array<value_type> > new_weights =
48 Teuchos::rcp(new Teuchos::Array<value_type>(weights));
49 for (ordinal_type i=0; i<nqp; i++)
50 (*new_points)[i].resize(new_dim);
51 for (ordinal_type k=0; k<new_dim; k++) {
52 Teuchos::Array<value_type> st_points;
53 Teuchos::Array<value_type> st_weights;
54 Teuchos::Array< Teuchos::Array<value_type> > st_values;
55 new_coordinate_bases[k]->getQuadPoints(new_order+1, st_points, st_weights,
56 st_values);
57
58 for (ordinal_type i=0; i<nqp; i++)
59 (*new_points)[i][k] = st_points[i];
60 }
61 gs_basis = Teuchos::rcp(
62 new GramSchmidtBasis<ordinal_type,value_type>(tensor_basis,
63 *new_points,
64 *new_weights,
65 1e-15)
66 );
67
68 // Create new quadrature object
69 Teuchos::RCP<const OrthogPolyBasis<ordinal_type,value_type> > new_basis =
70 gs_basis;
71 gs_quad = Teuchos::rcp(
72 new UserDefinedQuadrature<ordinal_type,value_type>(new_basis,
73 new_points,
74 new_weights)
75 );
76
77}

95template <typename ordinal_type, typename value_type>
96void
97Stokhos::StieltjesGramSchmidtBuilder<ordinal_type,value_type>::
98computeReducedPCEs(
99 const Teuchos::Array< Stokhos::OrthogPolyApprox<ordinal_type, value_type> >& pces,
100 Teuchos::Array< Stokhos::OrthogPolyApprox<ordinal_type, value_type> >& new_pces)
101{
102 // Map pce coefficients to tensor basis to Gram-Schmidt basis
103 ordinal_type dim = pces.size();
104 if (new_pces.size() != pces.size())
105 new_pces.resize(dim);
106 for (ordinal_type k=0; k<dim; k++) {
107 OrthogPolyApprox<ordinal_type,value_type> p_tensor(tensor_basis);
108 p_tensor.term(k, 0) = pces[k].mean();
109 p_tensor.term(k, 1) = 1.0;
110 new_pces[k].reset(gs_basis);
111 gs_basis->transformCoeffs(p_tensor.coeff(), new_pces[k].coeff());
112 }
113}
