56 // Permutes -- unwraps the torus-wrap for the solution
57 // using the communication buffer
58 template<class HandleType, class ZDView>
59 inline
60 void perm1_(HandleType& ahandle, ZDView& ZV) {
61
62 MPI_Comm comm = ahandle.get_comm();
63 int me = ahandle.get_myrank();
64 int my_rhs_ = ahandle.get_my_rhs();
65 int my_rows = ahandle.get_my_rows();
66 int nprocs_row = ahandle.get_nprocs_row();
67 int nprocs_col = ahandle.get_nprocs_col();
68 int nrows_matrix = ahandle.get_nrows_matrix();
69 int ncols_matrix = ahandle.get_ncols_matrix();
70 int my_first_row = ahandle.get_my_first_row();
71 int my_first_col = ahandle.get_my_first_col();
72
73 int i;
74
75 int bytes;
76 int dest;
77 int type;
78
79 int global_index;
80 int local_index;
81
82
83 int col_offset, row_offset;
84 int ncols_proc1, ncols_proc2, nprocs_row1;
85 int nrows_proc1, nrows_proc2, nprocs_col1;
86
87 int change_nosend;
88 int change_send;
89 int next, next_s;
90 int inc;
91
92 #ifdef GET_TIMING
93 double t2;
94 double totalpermtime;
95 #endif
96
97 MPI_Request msgrequest;
98 MPI_Status msgstatus;
99
100 int ptr1_idx;
101
102 #ifdef GET_TIMING
103 t2 = MPI_Wtime();
104 #endif
105
106 typedef typename ZDView::value_type value_type;
107 typedef typename ZDView::device_type::execution_space execution_space;
108 typedef typename ZDView::device_type::memory_space memory_space;
109 typedef Kokkos::View<value_type*, Kokkos::LayoutLeft, memory_space> ViewVectorType;
110
111 ViewVectorType temp_s ( "temp_s", (my_rhs_ + 1) );
112 ViewVectorType rhs_temp ( "rhs_temp", (my_rows*(my_rhs_+1)) );
113 ViewVectorType my_rhs_temp( "my_rhs_temp", (my_rows*(my_rhs_+1)) );
114
115 if (my_rhs_ > 0) {
116 ncols_proc1 = ncols_matrix/nprocs_row;
117 ncols_proc2 = ncols_proc1;
118 if (ncols_matrix%nprocs_row > 0) ncols_proc1++;
119 nprocs_row1 = (ncols_matrix%nprocs_row);
120 row_offset = ncols_proc1 * nprocs_row1;
121
122 nrows_proc1 = nrows_matrix/nprocs_col;
123 nrows_proc2 = nrows_proc1;
124 if (nrows_matrix%nprocs_col > 0) nrows_proc1++;
125 nprocs_col1 = (nrows_matrix%nprocs_col);
126 col_offset = nrows_proc1 * nprocs_col1;
127
128 ptr1_idx = 0;
129 //change_count = 0;
130 change_nosend = 0;
131 next = 0;
132 change_send = 0;
133 next_s = 0;
134
135 #ifdef PRINT_STATUS
136 printf("Rank %i -- perm1_() Begin permutation, execution_space %s, memory_space %s\n",me,typeid(execution_space).name(),typeid(memory_space).name());
137 #endif
138
139 for (i=0; i<my_rows; i++) {
140 global_index = my_first_row + i*nprocs_col;
141 // break down global index using torus wrap in row direction
142 dest = global_index%nprocs_row;
143 local_index = global_index/nprocs_row;
144
145 // rebuild global index using block in row direction
146 if (dest < nprocs_row1) {
147 global_index = dest*ncols_proc1 + local_index;
148 } else {
149 global_index = row_offset + (dest-nprocs_row1)*ncols_proc2
150 + local_index;
151 }
152
153 // break down global index using blocks in the column direction
154 if (global_index < col_offset) {
155 dest = global_index/nrows_proc1;
156 local_index = global_index%nrows_proc1;
157 } else {
158 dest = (global_index - col_offset)/nrows_proc2 + nprocs_col1;
159 local_index = (global_index - col_offset)%nrows_proc2;
160 }
161
162 dest = dest*nprocs_row + my_first_col;
163
164
165 if ((local_index != i) || (dest != me)) {
166
167 // Check if I need to send the data or just change position
168
169 if( dest == me ) {
170
171 auto sub_ZV = subview(ZV, ptr1_idx, Kokkos::ALL());
172 auto sub_my_rhs_temp = subview(my_rhs_temp, Kokkos::make_pair(next, next+(my_rhs_ + 1)));
173 zcopy_wr_local_index(my_rhs_, sub_ZV, sub_my_rhs_temp, local_index);
174
175 change_nosend++;
176
177 next = change_nosend * (my_rhs_ + 1);
178
179 }
180
181 if( dest != me ) {
182
183 bytes = (my_rhs_ + 1)*sizeof(ADELUS_DATA_TYPE);
184
185 MPI_Irecv( (char *)(reinterpret_cast<ADELUS_DATA_TYPE *>(rhs_temp.data())+next_s),bytes,MPI_CHAR,MPI_ANY_SOURCE,
186 MPI_ANY_TAG,comm,&msgrequest);
187
188 auto sub_ZV = subview(ZV, ptr1_idx, Kokkos::ALL());
189 zcopy_wr_local_index(my_rhs_, sub_ZV, temp_s, local_index);
190
191 type = PERMTYPE+change_send;
192 MPI_Send((char *)(reinterpret_cast<ADELUS_DATA_TYPE *>(temp_s.data())),bytes,MPI_CHAR,dest,
193 type,comm);
194 change_send++;
195
196 next_s = change_send * (my_rhs_+1);
197
198 MPI_Wait(&msgrequest,&msgstatus);
199
200 }
201
202 }
203 ptr1_idx++;
204 }
205
206 // Unpack changes from other processors
207
208 next_s = 0;
209 inc = 0;
210 for (i = 0; i < change_send; i++) {
211 auto sub_rhs_temp = subview(rhs_temp, Kokkos::make_pair(next_s, next_s+(my_rhs_ + 1)));
212 zcopy_ld_local_index(my_rhs_, sub_rhs_temp, ZV);
213
214 inc++;
215 next_s = inc * (my_rhs_+1);
216 }
217
218 // Unpack my changes
219 next = 0;
220 inc = 0;
221 for (i = 0; i < change_nosend; i++) {
222 auto sub_my_rhs_temp = subview(my_rhs_temp, Kokkos::make_pair(next, next+(my_rhs_ + 1)));
223 zcopy_ld_local_index(my_rhs_, sub_my_rhs_temp, ZV);
224 inc++;
225 next = inc * (my_rhs_+1);
226 }
227
228 }
229
230 #ifdef GET_TIMING
231 totalpermtime = MPI_Wtime() - t2;
232 #endif
233 #ifdef GET_TIMING
234 showtime(ahandle.get_comm_id(), comm, me, ahandle.get_nprocs_cube(), "Total time in perm", &totalpermtime);
235 #endif
236 }

252 void perm1_(HandleType& ahandle, ZDView& ZV) {
253
254 MPI_Comm col_comm = ahandle.get_col_comm();
255 int myrow = ahandle.get_myrow();
256 int my_rhs_ = ahandle.get_my_rhs();
257 int my_rows = ahandle.get_my_rows();
258 int nprocs_row = ahandle.get_nprocs_row();
259 int nprocs_col = ahandle.get_nprocs_col();

261 int ncols_matrix = ahandle.get_ncols_matrix();
262 int my_first_row = ahandle.get_my_first_row();
263
264 int i;
265
266 int dest, global_index, local_index;
267
268 int row_offset;
269 int ncols_proc1, ncols_proc2, nprocs_row1;
270 int ptr1_idx, myfirstrow;
271
272 #ifdef GET_TIMING
273 #if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
274 double t1, copyhostpinnedtime;

283
284 typedef typename ZDView::value_type value_type;
285 #ifdef PRINT_STATUS
286 typedef typename ZDView::device_type::execution_space execution_space;
287 #endif
288 typedef typename ZDView::device_type::memory_space memory_space;
289 typedef Kokkos::View<value_type**, Kokkos::LayoutLeft, memory_space> ViewMatrixType;
290
291 #ifdef ADELUS_HOST_PINNED_MEM_MPI
292 #if defined(KOKKOS_ENABLE_CUDA)
293 typedef Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::CudaHostPinnedSpace> View2DHostPinnType;//CudaHostPinnedSpace
294 #elif defined(KOKKOS_ENABLE_HIP)
295 typedef Kokkos::View<value_type**, Kokkos::LayoutLeft, Kokkos::HIPHostPinnedSpace> View2DHostPinnType;//HIPHostPinnedSpace
296 #endif
297 #endif
298
299 if (my_rhs_ > 0) {
300
301 ViewMatrixType rhs_temp ( "rhs_temp", nrows_matrix, my_rhs_ );//allocate full-size RHS vectors
302 #if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
303 View2DHostPinnType h_rhs_temp( "h_rhs_temp", nrows_matrix, my_rhs_ );
304 #endif
305
306 Kokkos::deep_copy(rhs_temp, 0);//initialize with 0s
307
308 ncols_proc1 = ncols_matrix/nprocs_row;
309 ncols_proc2 = ncols_proc1;
310 if (ncols_matrix%nprocs_row > 0) ncols_proc1++;
311 nprocs_row1 = (ncols_matrix%nprocs_row);
312 row_offset = ncols_proc1 * nprocs_row1;
313
314 myfirstrow = myrow * (nrows_matrix / nprocs_col) + 1;
315 myfirstrow = ( myrow > (nrows_matrix%nprocs_col) ) ? myfirstrow + (nrows_matrix%nprocs_col) :
316 myfirstrow + myrow;
317
318 ptr1_idx = 0;
319
320 #ifdef PRINT_STATUS
321 printf("Rank %i -- perm1_() Begin permutation, execution_space %s, memory_space %s\n",ahandle.get_myrank(),typeid(execution_space).name(),typeid(memory_space).name());
322 #endif
323
324 for (i=0; i<my_rows; i++) {
325 global_index = my_first_row + i*nprocs_col;
326
327 /* break down global index using torus wrap in row direction */
328 dest = global_index%nprocs_row;
329 local_index = global_index/nprocs_row;
330
331 /* rebuild global index using block in row direction */
332 if (dest < nprocs_row1) {
333 global_index = dest*ncols_proc1 + local_index;
334 } else {
335 global_index = row_offset + (dest-nprocs_row1)*ncols_proc2 + local_index;
336 }
337
338 auto sub_ZV = subview(ZV, ptr1_idx, Kokkos::ALL());
339 zcopy_wr_global_index(my_rhs_, sub_ZV, rhs_temp, global_index);
340
341 ptr1_idx++;
342 }
343
344 //Workaround for GPU-aware MPI issue on HIP: fence here to make sure "zcopy_wr_global_index"
345 // completes before MPI_Allreduce
346 #if defined(KOKKOS_ENABLE_HIP) && !defined(ADELUS_HOST_PINNED_MEM_MPI)
347 Kokkos::fence();
348 #endif
349
350 #if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
351 #ifdef GET_TIMING
352 t1 = MPI_Wtime();
353 #endif
354 Kokkos::deep_copy(h_rhs_temp,rhs_temp);
355 #ifdef GET_TIMING
356 copyhostpinnedtime = (MPI_Wtime()-t1);
357 #endif
358
359 MPI_Allreduce( MPI_IN_PLACE, h_rhs_temp.data(), nrows_matrix*my_rhs_, ADELUS_MPI_DATA_TYPE, MPI_SUM, col_comm);
360
361 Kokkos::deep_copy( subview(ZV, Kokkos::ALL(), Kokkos::make_pair(0, my_rhs_)),
362 subview(h_rhs_temp, Kokkos::make_pair(myfirstrow-1, myfirstrow-1+my_rows), Kokkos::ALL()) );
363 #else //GPU-aware MPI
364 MPI_Allreduce( MPI_IN_PLACE, rhs_temp.data(), nrows_matrix*my_rhs_, ADELUS_MPI_DATA_TYPE, MPI_SUM, col_comm);
365
366 Kokkos::deep_copy( subview(ZV, Kokkos::ALL(), Kokkos::make_pair(0, my_rhs_)),
367 subview(rhs_temp, Kokkos::make_pair(myfirstrow-1, myfirstrow-1+my_rows), Kokkos::ALL()) );
368 #endif
369 }
370
371 #ifdef GET_TIMING
372 totalpermtime = MPI_Wtime() - t2;
373 #endif
374
375 #ifdef GET_TIMING
376 #if defined(ADELUS_HOST_PINNED_MEM_MPI) && (defined(KOKKOS_ENABLE_CUDA) || defined(KOKKOS_ENABLE_HIP))
377 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
378 "Time to copy dev mem --> host pinned mem", &copyhostpinnedtime);
379 #endif
380 showtime(ahandle.get_comm_id(), ahandle.get_comm(), ahandle.get_myrank(), ahandle.get_nprocs_cube(),
381 "Total time in perm", &totalpermtime);
382 #endif
383 }
